Identifiants de type.

Un identifiant de type est une valeur qui désigne un type. Étant donné deux identifiants de type, ils peuvent être testés avec Type.Id.provably_equal pour prouver qu'ils désignent le même type. Noter que :

-Des identifiants inégaux n'impliquent pas des types inégaux : un type donné peut être désigné par plus d'un identifiant.

-Les identifiants de type peuvent être marshalisés, mais ils obtiennent une nouvelle identité distincte lors du démarshalisage, donc les égalités sont perdues.

Voir un Type.Id.example d'utilisation.

Identifiants de type

type !'a t

Le type des identifiants pour le type 'a .

val make : unit -> 'a t

make () est un nouvel identifiant de type.

val uid : 'a t -> int

uid id est un identifiant unique en runtime pour id .

val provably_equal : 'a t -> 'b t -> ('a, 'b) Type.eq option

provably_equal i0 i1 est Some Equal si l'identifiant i0 est égal à i1 et None sinon.

Exemple

Ce qui suit montre comment les identifiants de type peuvent être utilisés pour implémenter un simple dictionnaire de clés-valeurs hétérogène. Contrairement aux valeurs Map dont les clés sont mappées sur un unique type homogène de valeurs, ce dictionnaire peut associer un type de valeur différent à chaque clé.

(** Dictionnaires hétérogènes. *)
module Dict : sig
  type t
  (** Le type des dictionnaires. *)
  type 'a key
  (** Le type des clés liant des valeurs de type ['a]. *)
  val key : unit -> 'a key
  (** [key ()] est une nouvelle clé de dictionnaire. *)
  val empty : t
  (** [empty] est le dictionnaire vide. *)
  val add : 'a key -> 'a -> t -> t
  (** [add k v d] est [d] avec [k] lié à [v]. *)
  val remove : 'a key -> t -> t
  (** [remove k d] est [d] avec la dernière liaison de [k] supprimée. *)
  val find : 'a key -> t -> 'a option
  (** [find k d] est la liaison de [k] dans [d], le cas échéant. *)
end = struct
  type 'a key = 'a Type.Id.t
  type binding = B : 'a key * 'a -> binding
  type t = (int * binding) list
  let key () = Type.Id.make ()
  let empty = []
  let add k v d = (Type.Id.uid k, B (k, v)) :: d
  let remove k d = List.remove_assoc (Type.Id.uid k) d
  let find : type a. a key -> t -> a option = fun k d ->
    match List.assoc_opt (Type.Id.uid k) d with
    | None -> None
    | Some (B (k', v)) ->
        match Type.Id.provably_equal k k' with
        | Some Type.Equal -> Some v
        | None -> assert false
end
