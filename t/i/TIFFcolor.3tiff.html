NOM

TIFFcolor - routines de conversion de couleur

SYNOPSIS

#include <tiffio.h>
int TIFFYCbCrToRGBInit(TIFFYCbCrToRGB *ycbcr, float *luma, float *refBlackWhite)
void TIFFYCbCrtoRGB(TIFFYCbCrToRGB *ycbcr, uint32_t Y, int32_t Cb, int32_t Cr, uint32_t *R, uint32_t *G, uint32_t *B)
int TIFFCIELabToRGBInit(TIFFCIELabToRGB *cielab, const TIFFDisplay *displayw, float *refWhite)
void TIFFCIELabToXYZ(TIFFCIELabToRGB *cielab, uint32_t L, int32_t a, int32_t b, float *X, float *Y, float *Z)
void TIFFXYZToRGB(TIFFCIELabToRGB *cielab, float X, float Y, float Z, uint32_t *R, uint32_t *G, uint32_t *B)

DESCRIPTION

TIFF prend en charge plusieurs espaces colorimétriques pour les images stockées dans ce format. Il y a généralement un problème pour que l'application gère correctement les données et convertisse entre différents espaces colorimétriques à des fins d'affichage et d'impression. Pour simplifier cette tâche, libtiff implémente elle-même plusieurs routines de conversion de couleur. En particulier, ces routines sont utilisées dans l'interface TIFFRGBAImage.

TIFFYCbCrToRGBInit() est utilisé pour initialiser l'état de conversion de YCbCr en RGB. L'allocation et la libération de la structure YCbCr incombent au programmeur. TIFFYCbCrToRGB est défini dans tiffio.h comme suit :

typedef struct {                /* Support YCbCr->RGB */
        TIFFRGBValue* clamptab; /* table de plafonnement de plage */
        int*          Cr_r_tab;
        int*          Cb_b_tab;
        int32_t*      Cr_g_tab;
        int32_t*      Cb_g_tab;
        int32_t*      Y_tab;
} TIFFYCbCrToRGB;

luma est un tableau flottant de trois valeurs représentant les proportions de rouge, vert et bleu dans la luminance, Y (voir la section 21 de la spécification TIFF 6.0, où les images YCbCr sont discutées). TIFFTAG_YCBCRCOEFFICIENTS contient ces valeurs dans le fichier TIFF. refBlackWhite est un tableau flottant de 6 valeurs qui spécifie une paire de valeurs de headroom et footroom des données de l'image (codes) pour chaque composante de l'image (voir la section 20 de la spécification TIFF 6.0 où les champs de colorimétrie sont discutés). TIFFTAG_REFERENCEBLACKWHITE est responsable du stockage de ces valeurs dans le fichier TIFF. L'extrait de code suivant devrait aider à comprendre la technique :

float *luma, *refBlackWhite;
uint16_t hs, vs;
/* Initialisation des structures */
ycbcr = (TIFFYCbCrToRGB*)
    _TIFFmalloc(TIFFroundup(sizeof(TIFFYCbCrToRGB), sizeof(long))
        + 4*256*sizeof(TIFFRGBValue)
        + 2*256*sizeof(int)
        + 3*256*sizeof(int32_t));
if (ycbcr == NULL) {
    TIFFError("YCbCr->RGB",
    "Pas d'espace pour l'état de conversion YCbCr->RGB");
    exit(0);
}
TIFFGetFieldDefaulted(tif, TIFFTAG_YCBCRCOEFFICIENTS, &luma);
TIFFGetFieldDefaulted(tif, TIFFTAG_REFERENCEBLACKWHITE, &refBlackWhite);
if (TIFFYCbCrToRGBInit(ycbcr, luma, refBlackWhite) < 0) {
    exit(0);
}
/* Début de la conversion */
uint32_t r, g, b;
uint32_t Y;
int32_t Cb, Cr;
pour chaque pixel de l'image {
    TIFFYCbCrtoRGB(img->ycbcr, Y, Cb, Cr, &r, &g, &b);
}
/* Libération de la structure d'état */
_TIFFfree(ycbcr);

TIFFCIELabToRGBInit() initialise l'état de conversion de CIE L*a*b* 1976 en RGB. TIFFCIELabToRGB est défini comme suit :

#define CIELABTORGB_TABLE_RANGE 1500
typedef struct {                                /* Support CIE Lab 1976->RGB */
    int      range;                             /* Taille de la table de conversion */
    float    rstep, gstep, bstep;
    float    X0, Y0, Z0;                        /* Point blanc de référence */
    TIFFDisplay display;
    float    Yr2r[CIELABTORGB_TABLE_RANGE + 1]; /* Conversion de Yr en r */
    float    Yg2g[CIELABTORGB_TABLE_RANGE + 1]; /* Conversion de Yg en g */
    float    Yb2b[CIELABTORGB_TABLE_RANGE + 1]; /* Conversion de Yb en b */
} TIFFCIELabToRGB;

display est une description du périphérique d'affichage, déclarée comme suit :

typedef struct {
    float d_mat[3][3]; /* Matrice XYZ -> luminance */
    float d_YCR;       /* Sortie lumineuse pour le blanc de référence */
    float d_YCG;
    float d_YCB;
    uint32_t d_Vrwr;   /* Valeurs de pixel pour le blanc de référence */
    uint32_t d_Vrwg;
    uint32_t d_Vrwb;
    float d_Y0R;       /* Lumière résiduelle pour un pixel noir */
    float d_Y0G;
    float d_Y0B;
    float d_gammaR;    /* Valeurs Gamma pour les trois canons */
    float d_gammaG;
    float d_gammaB;
} TIFFDisplay;

Par exemple, on peut utiliser le périphérique sRGB, qui a les paramètres suivants :

TIFFDisplay display_sRGB = {
    {   /* Matrice XYZ -> luminance */
        {  3.2410F, -1.5374F, -0.4986F },
        {  -0.9692F, 1.8760F, 0.0416F },
        {  0.0556F, -0.2040F, 1.0570F }
    },
    100.0F, 100.0F, 100.0F, /* Sortie lumineuse pour le blanc de référence */
    255, 255, 255,          /* Valeurs de pixel pour le blanc de référence */
    1.0F, 1.0F, 1.0F,       /* Sortie lumineuse résiduelle pour un pixel noir */
    2.4F, 2.4F, 2.4F,       /* Valeurs Gamma pour les trois canons */
};

refWhite est la température de couleur du blanc de référence. La balise TIFFTAG_WHITEPOINT contient la chromaticité du point blanc de l'image à partir de laquelle le blanc de référence peut être calculé en utilisant les formules suivantes :

refWhite_Y = 100.0
refWhite_X = whitePoint_x / whitePoint_y * refWhite_Y
refWhite_Z = (1.0 - whitePoint_x - whitePoint_y) / whitePoint_y * refWhite_X

La conversion elle-même s'effectue en deux étapes : la première étape convertit CIE L*a*b* 1976 en CIE XYZ en utilisant la routine TIFFCIELabToXYZ(), et la seconde étape convertit CIE XYZ en RGB en utilisant TIFFXYZToRGB(). Regardez l'exemple de code ci-dessous :

float   *whitePoint;
float   refWhite[3];
/* Initialisation des structures */
img->cielab = (TIFFCIELabToRGB *)
    _TIFFmalloc(sizeof(TIFFCIELabToRGB));
if (!cielab) {
    TIFFError("CIE L*a*b*->RGB",
        "Pas d'espace pour l'état de conversion CIE L*a*b*->RGB.");
    exit(0);
}
TIFFGetFieldDefaulted(tif, TIFFTAG_WHITEPOINT, &whitePoint);
refWhite[1] = 100.0F;
refWhite[0] = whitePoint[0] / whitePoint[1] * refWhite[1];
refWhite[2] = (1.0F - whitePoint[0] - whitePoint[1])
    / whitePoint[1] * refWhite[1];
if (TIFFCIELabToRGBInit(cielab, &display_sRGB, refWhite) < 0) {
    TIFFError("CIE L*a*b*->RGB",
        "Échec de l'initialisation de l'état de conversion CIE L*a*b*->RGB.");
    _TIFFfree(cielab);
    exit(0);
}
/* Nous pouvons maintenant commencer la conversion */
uint32_t r, g, b;
uint32_t L;
int32_t a, b;
float X, Y, Z;
pour chaque pixel de l'image {
    TIFFCIELabToXYZ(cielab, L, a, b, &X, &Y, &Z);
    TIFFXYZToRGB(cielab, X, Y, Z, &r, &g, &b);
}
/* N'oubliez pas de libérer la structure d'état */
_TIFFfree(cielab);
