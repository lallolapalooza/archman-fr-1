DESCRIPTION

TCGETS
Équivalent à tcgetattr(fd, argp).
Obtenir les paramètres actuels du port série.
TCSETS
Équivalent à tcsetattr(fd, TCSANOW, argp).
Définir les paramètres actuels du port série.
TCSETSW
Équivalent à tcsetattr(fd, TCSADRAIN, argp).
Permettre à la mémoire tampon de sortie de se vider, et définir les paramètres actuels du port série.
TCSETSF
Équivalent à tcsetattr(fd, TCSAFLUSH, argp).
Permettre à la mémoire tampon de sortie de se vider, rejeter les entrées en attente, et définir les paramètres actuels du port série.

Les quatre ioctls suivants sont identiques à TCGETS, TCSETS, TCSETSW, TCSETSF, sauf qu'ils prennent un struct termios2 * au lieu d'un struct termios *. Si le membre de structure c_cflag contient le flag BOTHER, alors le débit binaire est stocké dans les membres de structure c_ispeed et c_ospeed sous forme de valeurs entières. Ces ioctls ne sont pas supportés sur toutes les architectures.

VALEUR RETOURnée

En cas de succès, 0 est retourné. En cas d'erreur, -1 est retourné et errno est défini pour indiquer l'erreur.

AVERTISSEMENTS

Le struct termios de <asm/termbits.h> est différent et incompatible avec le struct termios de <termios.h>. Ces appels ioctl nécessitent le struct termios de <asm/termbits.h>.

EXEMPLES

Obtenir ou définir un débit binaire arbitraire sur le port série.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER n'est pas supporté\n");
    /* Le programme peut se rabattre sur TCGETS/TCSETS avec les constantes Bnnn */
    exit(EXIT_FAILURE);
#else
    /* Déclarer la structure tio, son type dépend de l'ioctl supporté */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "Usage: %s device [sortie [entrée] ]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* Obtenir les paramètres actuels du port série via l'ioctl supporté */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* Changer le débit binaire lorsque des arguments supplémentaires sont fournis */
    if (argc == 3 || argc == 4) {
        /* Effacer le débit binaire de sortie actuel et y insérer une nouvelle valeur */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* Effacer le débit binaire d'entrée actuel et y insérer une nouvelle valeur */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* Lorsque le 4ème argument n'est pas fourni, réutiliser le débit binaire de sortie */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* Définir les nouveaux paramètres du port série via l'ioctl supporté */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* Et obtenir les nouvelles valeurs qui ont été réellement configurées */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("débit binaire de sortie : %u\n", tio.c_ospeed);
    printf("débit binaire d'entrée : %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}
