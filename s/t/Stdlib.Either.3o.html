Module Either
: (module Stdlib__Either)

type ('a, 'b) t =
| Left of 'a
| Right of 'b

Une valeur de ('a, 'b) Either.t contient soit une valeur de type 'a soit une valeur de type 'b

val left : 'a -> ('a, 'b) t

left v vaut Left v.

val right : 'b -> ('a, 'b) t

right v vaut Right v.

val is_left : ('a, 'b) t -> bool

is_left (Left v) vaut true, is_left (Right v) vaut false.

val is_right : ('a, 'b) t -> bool

is_right (Left v) vaut false, is_right (Right v) vaut true.

val get_left : ('a, 'b) t -> 'a

get_left e vaut v si e est Left v et lève une exception sinon.

Depuis 5.4

Lève Invalid_argument si e est Right _.

val get_right : ('a, 'b) t -> 'b

get_right e vaut v si e est Right v et lève une exception sinon.

Depuis 5.4

Lève Invalid_argument si e est Left _.

val find_left : ('a, 'b) t -> 'a option

find_left (Left v) vaut Some v, find_left (Right _) vaut None

val find_right : ('a, 'b) t -> 'b option

find_right (Right v) vaut Some v, find_right (Left _) vaut None

val map_left : ('a1 -> 'a2) -> ('a1, 'b) t -> ('a2, 'b) t

map_left f e vaut Left (f v) si e est Left v et vaut e si e est Right _.

val map_right : ('b1 -> 'b2) -> ('a, 'b1) t -> ('a, 'b2) t

map_right f e vaut Right (f v) si e est Right v et vaut e si e est Left _.

val map : left:('a1 -> 'a2) -> right:('b1 -> 'b2) -> ('a1, 'b1) t -> ('a2, 'b2) t

map ~left ~right (Left v) vaut Left (left v), map ~left ~right (Right v) vaut Right (right v).

val fold : left:('a -> 'c) -> right:('b -> 'c) -> ('a, 'b) t -> 'c

fold ~left ~right (Left v) vaut left v, et fold ~left ~right (Right v) vaut right v.

val retract : ('a, 'a) t -> 'a

retract (Left v) vaut v, et retract (Right v) vaut v.

Depuis 5.4

val iter : left:('a -> unit) -> right:('b -> unit) -> ('a, 'b) t -> unit

iter ~left ~right (Left v) vaut left v, et iter ~left ~right (Right v) vaut right v.

val for_all : left:('a -> bool) -> right:('b -> bool) -> ('a, 'b) t -> bool

for_all ~left ~right (Left v) vaut left v, et for_all ~left ~right (Right v) vaut right v.

val equal : left:('a -> 'a -> bool) -> right:('b -> 'b -> bool) -> ('a, 'b) t -> ('a, 'b) t -> bool

equal ~left ~right e0 e1 teste l'égalité de e0 et e1 en utilisant left et right pour comparer respectivement les valeurs encapsulées par Left _ et Right _.

val compare : left:('a -> 'a -> int) -> right:('b -> 'b -> int) -> ('a, 'b) t -> ('a, 'b) t -> int

compare ~left ~right e0 e1 ordonne totalement e0 et e1 en utilisant left et right pour comparer respectivement les valeurs encapsulées par Left _ et Right _. Les valeurs Left _ sont inférieures aux valeurs Right _.
