NAME

SoCoordinateElement — Element for storing coordinate data

SYNOPSIS

#include <Coin/SbVec.h>
#include <Coin/SoNode.h>
#include <Coin/SoReplacedElement.h>

class SoCoordinateElement : public SoReplacedElement
{
public:
  // For Coin 2.0 compatibility:
  static void initClass(void);

  // Creates and returns a new instance of this type of element.
  static SoType classTypeId(void);

  // Returns the 3D coordinate array.
  static const SbVec3f * getArrayPtr3 (void);

  // Returns the 4D coordinate array.
  static const SbVec4f * getArrayPtr4 (void);

  // Returns TRUE if the coordinates are 3D.
  static SbBool is3D(void);

  // Returns TRUE if the coordinates are 4D.
  static SbBool is4D(void);

protected:
  // Constructor.
  SoCoordinateElement(void);

  // Destructor.
  ~SoCoordinateElement();

  // Reimplemented from SoReplacedElement.
  virtual void print(FILE * file) const;

  // Sets the coordinate array.
  static void set(SoState * state, const SbVec3f * array, int32_t numcoords);
  static void set(SoState * state, const SbVec4f * array, int32_t numcoords);

  // Gets the coordinate array.
  static const SbVec3f * get(SoState * state);
  static const SbVec4f * get(SoState * state);

  // Gets the number of coordinates.
  static int32_t getNum(SoState * state);

  // Gets the dimensionality.
  static int32_t getDimensionality(SoState * state);

  // Reimplemented from SoReplacedElement.
  virtual void push(SoState * state);

private:
  // Sets the value for the specific field.
  static void setPtr(SoState * const state, const void * const ptr,
                     const SbBool is4D, const int32_t numcoords);

  // Returns the value for the specific field.
  static const void * getPtr(SoState * const state, const SbBool is4D);

  // Gets the number of coordinates.
  static int32_t getNum(SoState * const state, const SbBool is4D);

  // Gets the dimensionality.
  static int32_t getDimensionality(SoState * const state, const SbBool is4D);

  // Sets the value for the specific field.
  static void setDimensionality(SoState * const state,
                                const SbBool is4D);

  // Pointer to the coordinate array.
  const void * ptr;

  // Number of coordinates.
  int32_t numcoords;

  // Whether the coordinates are 4D.
  SbBool is4D;
};

Detailed Description

This is the element used for storing coordinate data for scene graph nodes like SoCube, SoSphere, etc. This class is part of the Coin API and not part of the original SGI Open Inventor v2.1 API.

The element can store either 3D or 4D coordinates.
