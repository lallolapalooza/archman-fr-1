XcmsColor(3) FONCTIONS XLIB XcmsColor(3)

NOM

XcmsColor, XcmsRGB, XcmsRGBi, XcmsCIEXYZ, XcmsCIEuvY, XcmsCIExyY, XcmsCIELab, XcmsCIELuv, XcmsTekHVC, XcmsPad - structure de couleur Xcms

STRUCTURES

La structure XcmsColor contient :

typedef unsigned long XcmsColorFormat;			/* Format de spécification de couleur */
typedef struct {
	union {
		XcmsRGB RGB;
		XcmsRGBi RGBi;
		XcmsCIEXYZ CIEXYZ;
		XcmsCIEuvY CIEuvY;
		XcmsCIExyY CIExyY;
		XcmsCIELab CIELab;
		XcmsCIELuv CIELuv;
		XcmsTekHVC TekHVC;
		XcmsPad Pad;
	} spec;
	unsigned long pixel;
	XcmsColorFormat format;
} XcmsColor;			/* Structure de couleur Xcms */
typedef double XcmsFloat;
typedef struct {
	unsigned short red;	/* 0x0000 à 0xffff */
	unsigned short green;	/* 0x0000 à 0xffff */
	unsigned short blue;	/* 0x0000 à 0xffff */
} XcmsRGB;		/* Appareil RGB */
typedef struct {
	XcmsFloat red;	/* 0.0 à 1.0 */
	XcmsFloat green;	/* 0.0 à 1.0 */
	XcmsFloat blue;	/* 0.0 à 1.0 */
} XcmsRGBi;		/* Intensité RGB */
typedef struct {
	XcmsFloat X;
	XcmsFloat Y;	/* 0.0 à 1.0 */
	XcmsFloat Z;
} XcmsCIEXYZ;		/* CIE XYZ */
typedef struct {
	XcmsFloat u_prime;	/* 0.0 à ~0.6 */
	XcmsFloat v_prime;	/* 0.0 à ~0.6 */
	XcmsFloat Y; 	/* 0.0 à 1.0 */
} XcmsCIEuvY;		/* CIE u'v'Y */
typedef struct {
	XcmsFloat x; 	/* 0.0 à ~.75 */
	XcmsFloat y; 	/* 0.0 à ~.85 */
	XcmsFloat Y; 	/* 0.0 à 1.0 */
} XcmsCIExyY;		/* CIE xyY */
typedef struct {
	XcmsFloat L_star; 	/* 0.0 à 100.0 */
	XcmsFloat a_star;
	XcmsFloat b_star;
} XcmsCIELab;		/* CIE L*a*b* */
typedef struct {
	XcmsFloat L_star; 	/* 0.0 à 100.0 */
	XcmsFloat u_star;
	XcmsFloat v_star;
} XcmsCIELuv;		/* CIE L*u*v* */
typedef struct {
	XcmsFloat H; 	/* 0.0 à 360.0 */
	XcmsFloat V; 	/* 0.0 à 100.0 */
	XcmsFloat C; 	/* 0.0 à 100.0 */
} XcmsTekHVC;		/* TekHVC */
typedef struct {
	XcmsFloat pad0;
	XcmsFloat pad1;
	XcmsFloat pad2;
	XcmsFloat pad3;
} XcmsPad;		/* quatre doubles */

DESCRIPTION

La structure XcmsColor contient une union de sous-structures, chacune supportant l'encodage de la spécification de couleur pour un espace de couleur particulier.
