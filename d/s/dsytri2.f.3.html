subroutine dsytri2 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer lwork, integer info)

DSYTRI2

Objectif :

!>
!> DSYTRI2 calcule l'inverse d'une matrice symétrique indéfinie en DOUBLE
!> PRECISION A en utilisant la factorisation A = U*D*U**T ou A = L*D*L**T
!> calculée par DSYTRF. DSYTRI2 définit la dimension principale de l'espace
!> de travail avant d'appeler DSYTRI2X qui calcule réellement l'inverse.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme est A = U*D*U**T ;
!>          = 'L':  Triangulaire inférieure, forme est A = L*D*L**T.
!>

N

!>          N est un ENTIER
!>          Ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, de dimension (LDA,N)
!>          En entrée, la matrice diagonale par blocs D et les multiplicateurs
!>          utilisés pour obtenir le facteur U ou L calculés par DSYTRF.
!>
!>          En sortie, si INFO = 0, l'inverse (symétrique) de la matrice
!>          originale.  Si UPLO = 'U', la partie triangulaire supérieure de
!>          l'inverse est formée et la partie de A sous la diagonale n'est pas
!>          référencée ; si UPLO = 'L', la partie triangulaire inférieure de
!>          l'inverse est formée et la partie de A au-dessus de la diagonale
!>          n'est pas référencée.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau ENTIER, de dimension (N)
!>          Détails des échanges et de la structure par blocs de D
!>          tels que déterminés par DSYTRF.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, de dimension (MAX(1,LWORK))
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK.
!>          Si N = 0, LWORK >= 1, sinon LWORK >= (N+NB+1)*(NB+3).
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ;
!>          la routine calcule :
!>              - la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK,
!>              - et aucun message d'erreur lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est un ENTIER
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0: si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!>
