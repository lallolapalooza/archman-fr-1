subroutine dsygv_2stage (itype, jobz, uplo, n, a, lda, b, ldb, w, work, lwork, info)
DSYGV_2STAGE

Documentation de la fonction/sous-routine

sous-routine dsygv_2stage (integer itype, character jobz, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) w, double precision, dimension( * ) work, integer lwork, integer info)

DSYGV_2STAGE

Objectif :

!>
!> DSYGV_2STAGE calcule toutes les valeurs propres, et optionnellement, les vecteurs propres
!> d'un problème propre généralisé symétrique-défini réel, de la forme
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  ou B*A*x=(lambda)*x.
!> Ici, A et B sont supposés symétriques et B est également
!> définie positive.
!> Cette routine utilise la technique 2stage pour la réduction à une forme tridiagonale
!> qui a montré des performances supérieures sur les architectures récentes et pour de grandes
!> tailles N>2000.
!>

Paramètres

ITYPE 
!>          ITYPE est INTEGER
!>          Spécifie le type de problème à résoudre :
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ est CHARACTER*1
!>          = 'N':  Calculer uniquement les valeurs propres ;
!>          = 'V':  Calculer les valeurs propres et les vecteurs propres.
!>                  Non disponible dans cette version.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Les triangles supérieurs de A et B sont stockés ;
!>          = 'L':  Les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (LDA, N)
!>          À l'entrée, la matrice symétrique A. Si UPLO = 'U', la
!>          partie triangulaire supérieure N-par-N principale de A contient
!>          la partie triangulaire supérieure de la matrice A. Si UPLO = 'L',
!>          la partie triangulaire inférieure N-par-N principale de A contient
!>          la partie triangulaire inférieure de la matrice A.
!>
!>          À la sortie, si JOBZ = 'V', alors si INFO = 0, A contient la
!>          matrice Z des vecteurs propres. Les vecteurs propres sont normalisés
!>          comme suit :
!>          si ITYPE = 1 ou 2, Z**T*B*Z = I ;
!>          si ITYPE = 3, Z**T*inv(B)*Z = I.
!>          Si JOBZ = 'N', alors à la sortie, le triangle supérieur (si UPLO='U')
!>          ou le triangle inférieur (si UPLO='L') de A, y compris la
!>          diagonale, est détruit.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB, N)
!>          À l'entrée, la matrice symétrique définie positive B.
!>          Si UPLO = 'U', la partie triangulaire supérieure N-par-N principale de B
!>          contient la partie triangulaire supérieure de la matrice B.
!>          Si UPLO = 'L', la partie triangulaire inférieure N-par-N principale de B
!>          contient la partie triangulaire inférieure de la matrice B.
!>
!>          À la sortie, si INFO <= N, la partie de B contenant la matrice est
!>          écrasée par le facteur triangulaire U ou L de la factorisation de Cholesky
!>          B = U**T*U ou B = L*L**T.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

W

!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          Si INFO = 0, les valeurs propres par ordre croissant.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur du tableau WORK. LWORK >= 1, lorsque N <= 1 ;
!>          sinon
!>          Si JOBZ = 'N' et N > 1, LWORK doit être interrogé.
!>                                   LWORK = MAX(1, dimension) où
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + 2*N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + 2*N
!>                                   où KD est la taille du bloc de la réduction,
!>                                   FACTOPTNB est le bloc utilisé par l'algorithme QR ou LQ
!>                                   (généralement FACTOPTNB=128 est un bon choix)
!>                                   NTHREADS est le nombre de threads utilisés lorsque
!>                                   la compilation openMP est activée, sinon =1.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être interrogé. Pas encore disponible.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur dans la première entrée du tableau WORK, et aucun message d'erreur
!>          relatif à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  Exécution réussie
!>          < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  DPOTRF ou DSYEV ont retourné un code d'erreur :
!>             <= N:  Si INFO = i, DSYEV n'a pas convergé ;
!>                    i éléments hors diagonale d'une forme tridiagonale intermédiaire
!>                    n'ont pas convergé vers zéro ;
!>             > N:   Si INFO = N + i, pour 1 <= i <= N, alors le mineur principal
!>                    principal d'ordre i de B n'est pas positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ou vecteur propre n'a été calculé.
!>
