DSBGVX

But:

!>
!> DSBGVX calcule les valeurs propres sélectionnées, et optionnellement, les vecteurs propres
!> d'un problème aux valeurs propres généralisé symétrique-défini et bande réel,
!> de la forme A*x=(lambda)*B*x. Ici A et B sont supposés symétriques et bande,
!> et B est aussi défini positif. Les valeurs propres et les vecteurs propres peuvent être
!> sélectionnés en spécifiant soit toutes les valeurs propres,
!> une plage de valeurs ou une plage d'indices pour les valeurs propres désirées.
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N':  Calcul des valeurs propres uniquement ;
!>          = 'V':  Calcul des valeurs propres et des vecteurs propres.
!>

RANGE

!>          RANGE est CHARACTER*1
!>          = 'A': toutes les valeurs propres seront trouvées.
!>          = 'V': toutes les valeurs propres dans l'intervalle ouvert à droite (VL,VU]
!>                 seront trouvées.
!>          = 'I': les valeurs propres de rang IL à IU seront trouvées.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Les triangles supérieurs de A et B sont stockés ;
!>          = 'L':  Les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est un ENTIER
!>          L'ordre des matrices A et B.  N >= 0.
!>

KA

!>          KA est un ENTIER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de subdiagonales si UPLO = 'L'.  KA >= 0.
!>

KB

!>          KB est un ENTIER
!>          Le nombre de superdiagonales de la matrice B si UPLO = 'U',
!>          ou le nombre de subdiagonales si UPLO = 'L'.  KB >= 0.
!>

AB

!>          AB est un tableau de DOUBLE PRECISION, de dimension (LDAB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice bande symétrique
!>          A, stocké dans les ka+1 premières lignes du tableau. La
!>          j-ème colonne de A est stockée dans la j-ème colonne du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) pour max(1,j-ka) <= i <= j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j <= i <= min(n,j+ka).
!>
!>          À la sortie, le contenu de AB est détruit.
!>

LDAB

!>          LDAB est un ENTIER
!>          La dimension principale du tableau AB.  LDAB >= KA+1.
!>

BB

!>          BB est un tableau de DOUBLE PRECISION, de dimension (LDBB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice bande symétrique
!>          B, stocké dans les kb+1 premières lignes du tableau. La
!>          j-ème colonne de B est stockée dans la j-ème colonne du tableau BB
!>          comme suit :
!>          si UPLO = 'U', BB(kb+1+i-j,j) = B(i,j) pour max(1,j-kb) <= i <= j ;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) pour j <= i <= min(n,j+kb).
!>
!>          À la sortie, le facteur S de la factorisation de Cholesky éclatée
!>          B = S**T*S, tel que retourné par DPBSTF.
!>

LDBB

!>          LDBB est un ENTIER
!>          La dimension principale du tableau BB.  LDBB >= KB+1.
!>

Q

!>          Q est un tableau de DOUBLE PRECISION, de dimension (LDQ, N)
!>          Si JOBZ = 'V', la matrice n x n utilisée dans la réduction de
!>          A*x = (lambda)*B*x à la forme standard, c'est-à-dire C*x = (lambda)*x,
!>          et par conséquent de C à la forme tridiagonale.
!>          Si JOBZ = 'N', le tableau Q n'est pas référencé.
!>

LDQ

!>          LDQ est un ENTIER
!>          La dimension principale du tableau Q.  Si JOBZ = 'N',
!>          LDQ >= 1. Si JOBZ = 'V', LDQ >= max(1,N).
!>

VL

!>          VL est un DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne inférieure de l'intervalle à
!>          explorer pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

VU

!>          VU est un DOUBLE PRECISION
!>
!>          Si RANGE='V', la borne supérieure de l'intervalle à
!>          explorer pour les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

IL

!>          IL est un ENTIER
!>
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

IU

!>          IU est un ENTIER
!>
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

ABSTOL

!>          ABSTOL est un DOUBLE PRECISION
!>          La tolérance d'erreur absolue pour les valeurs propres.
!>          Une valeur propre approximative est acceptée comme convergée
!>          lorsqu'elle est déterminée comme étant dans un intervalle [a,b]
!>          de largeur inférieure ou égale à
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          où EPS est la précision machine. Si ABSTOL est inférieur ou
!>          égal à zéro, alors EPS*|T| sera utilisé à sa place,
!>          où |T| est la norme 1 de la matrice tridiagonale obtenue
!>          en réduisant A à la forme tridiagonale.
!>
!>          Les valeurs propres seront calculées de manière la plus précise lorsque ABSTOL
!>          est réglé sur deux fois le seuil de sous-dépassement 2*DLAMCH('S'), et non zéro.
!>          Si cette routine retourne avec INFO>0, indiquant que certains
!>          vecteurs propres n'ont pas convergé, essayez de régler ABSTOL sur
!>          2*DLAMCH('S').
!>

M

!>          M est un ENTIER
!>          Le nombre total de valeurs propres trouvées.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, et si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W est un tableau de DOUBLE PRECISION, dimension (N)
!>          Si INFO = 0, les valeurs propres par ordre croissant.
!>

Z

!>          Z est un tableau de DOUBLE PRECISION, de dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres, avec la i-ème colonne de Z contenant le
!>          vecteur propre associé à W(i). Les vecteurs propres sont
!>          normalisés de sorte que Z**T*B*Z = I.
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est un ENTIER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau de DOUBLE PRECISION, de dimension (MAX(1,LRWORK))
!>          Espace de travail utilisé par la routine. LRWORK est définie ci-dessous.
!>

IWORK

!>          IWORK est un tableau d'ENTIERS, de dimension (MAX(1,N)).
!>          Espace de travail utilisé par la routine.
!>

IFAIl

!>          IFAIL est un tableau d'ENTIERS, de dimension (N)
!>          Si JOBZ = 'V', et si INFO = 0, puis IFAIL contient les indices
!>          des vecteurs propres qui n'ont pas convergé.
!>          Si JOBZ = 'N', alors IFAIl n'est pas référencé.
!>

INFO

!>          INFO est un ENTIER
!>          = 0:  Succès.
!>          < 0:  Si INFO = -i, la i-ème argument non valide a été rencontré.
!>                Aucun calcul n'a été effectué.
!>          > 0:  Si INFO = i, et 1 <= i <= N, alors la routine a convergé mais
!>                les i derniers vecteurs propres n'ont pas été calculés.
!>                Cela peut se produire si ABSTOL est trop petit.
!>                Se sont les vecteurs propres correspondant aux i plus grandes valeurs propres.
!>                Pour plus de détails, voir le paramètre IFAIL.
!>          > N:  Si INFO = N+1, la factorisation de Cholesky de B a échoué.
!>                B n'est pas défini positif.
!>          Si INFO = N+2, la réduction à la forme tridiagonale a échoué.
!>          Si INFO = N+3, le calcul des valeurs propres de la matrice tridiagonale
!>                a échoué.
!>          Si INFO = N+4, une précision insuffisante a été rencontrée lors du calcul
!>                des vecteurs propres.
!>

EXEMPLE UTILISATION

La structure suivante est un exemple d'appel de la routine dsbgvx.

      PROGRAM test_dsbgvx
C
C     # Used for testing DSBGVX
C
      IMPLICIT NONE
C
      INTEGER           COLDIM, INFO, J, KA, KB, KMAX, KMIN, LDAB, LDBB, LDQ, LDZ
      INTEGER           M, N
      PARAMETER        ( N = 8 )
      PARAMETER        ( KA = 2 )
      PARAMETER        ( KB = 1 )
      PARAMETER        ( KMAX = KA )
      PARAMETER        ( KMIN = KB )
      PARAMETER        ( LDAB = KMAX + 1 )
      PARAMETER        ( LDBB = KMIN + 1 )
      PARAMETER        ( COL DIM = N )
      PARAMETER        ( LDQ = N )
      PARAMETER        ( LDZ = N )
C
      DOUBLE PRECISION  AB(LDAB,N), BB(LDBB,N), Q(LDQ,N), W(N), Z(LDZ,N)
      DOUBLE PRECISION  VL, VU
      DOUBLE PRECISION  WORK(4*N)
      INTEGER           IFAIL(N), IWORK(N)
      CHARACTER*1       JOBZ, UPLO, RANGE
C
      INTEGER           IL, IU
      DOUBLE PRECISION  ABSTOL
C
C     # Define the problem dimensions and parameters
C
      UPLO = 'U'
      JOBZ = 'V'
      RANGE = 'A'
C
C     # Call DSBGVX to compute all eigenvalues and eigenvectors
C
      CALL DSBGVX( JOBZ, RANGE, UPLO, N, KA, KB, AB, LDAB, BB, LDBB, Q, LDQ,
     +             VL, VU, IL, IU, ABSTOL, M, W, Z, LDZ, WORK, IWORK, IFAIL, INFO )
C
C     # Print the results
C
      IF ( INFO .NE. 0 ) THEN
        PRINT *, 'DSBGVX returned with INFO = ', INFO
      ELSE
        PRINT *, 'Eigenvalues W:'
        DO J = 1, N
          PRINT '(I4, 2X, D12.5)', J, W(J)
        END DO
C
        IF ( JOBZ .EQ. 'V' ) THEN
          PRINT *,'Eigenvectors Z:'
          DO J = 1, N
            PRINT '(I4, 2X, D12.5)', J, Z(1,J)
          END DO
        END IF
      END IF
C
C     # End of example
C
      END
