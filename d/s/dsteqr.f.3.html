subroutine dsteqr (character compz, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer info)

DSTEQR

But :

!>
!> DSTEQR calcule toutes les valeurs propres et, éventuellement, les vecteurs propres d'une
!> matrice triangulaire symétrique en utilisant la méthode implicite QL ou QR.
!> Les vecteurs propres d'une matrice symétrique complète ou bande peuvent également être trouvés
!> si DSYTRD ou DSPTRD ou DSBTRD a été utilisé pour réduire cette matrice
!> à la forme triangulaire.
!>

Paramètres

COMPZ 
!>          COMPZ est CHARACTER*1
!>          = 'N' : Calculer uniquement les valeurs propres.
!>          = 'V' : Calculer les valeurs propres et les vecteurs propres de la
!>                  matrice symétrique d'origine. À l'entrée, Z doit contenir la
!>                  matrice orthogonale utilisée pour réduire la matrice d'origine
!>                  à la forme triangulaire.
!>          = 'I' : Calculer les valeurs propres et les vecteurs propres de la
!>                  matrice triangulaire. Z est initialisée à la matrice identité.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice. N >= 0.
!>

D

!>          D est un tableau DOUBLE PRECISION, de dimension (N)
!>          À l'entrée, les éléments diagonaux de la matrice triangulaire.
!>          À la sortie, si INFO = 0, les valeurs propres en ordre croissant.
!>

E

!>          E est un tableau DOUBLE PRECISION, de dimension (N-1)
!>          À l'entrée, les (n-1) éléments sous-diagonaux de la matrice
!>          triangulaire.
!>          À la sortie, E a été détruit.
!>

Z

!>          Z est un tableau DOUBLE PRECISION, de dimension (LDZ, N)
!>          À l'entrée, si COMPZ = 'V', alors Z contient la matrice
!>          orthogonale utilisée dans la réduction à la forme triangulaire.
!>          À la sortie, si INFO = 0, alors si COMPZ = 'V', Z contient les
!>          vecteurs propres orthonormés de la matrice symétrique d'origine,
!>          et si COMPZ = 'I', Z contient les vecteurs propres
!>          orthonormés de la matrice triangulaire symétrique.
!>          Si COMPZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1, et si
!>          des vecteurs propres sont souhaités, alors LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, de dimension (max(1,2*N-2))
!>          Si COMPZ = 'N', alors WORK n'est pas référencé.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : l'algorithme n'a pas réussi à trouver toutes les valeurs propres en
!>                un total de 30*N itérations ; si INFO = i, alors i
!>                éléments de E n'ont pas convergé vers zéro ; à la sortie, D
!>                et E contiennent les éléments d'une matrice triangulaire symétrique
!>                qui est orthogonalement similaire à la matrice d'origine.
!>
