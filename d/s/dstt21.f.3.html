subroutine dstt21 (n, kband, ad, ae, sd, se, u, ldu, work, result)
DSTT21

Documentation de la fonction/sous-programme

sous-programme dstt21 (integer n, integer kband, double precision, dimension( * ) ad, double precision, dimension( * ) ae, double precision, dimension( * ) sd, double precision, dimension( * ) se, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( * ) work, double precision, dimension( 2 ) result)

DSTT21

Objectif :

!>
!> DSTT21 vérifie une décomposition de la forme
!>
!>    A = U S U'
!>
!> où ' signifie transposée, A est symétrique tridiagonale, U est orthogonale,
!> et S est diagonale (si KBAND=0) ou symétrique tridiagonale (si KBAND=1).
!> Deux tests sont effectués :
!>
!>    RESULT(1) = | A - U S U' | / ( |A| n ulp )
!>
!>    RESULT(2) = | I - UU' | / ( n ulp )
!>

Paramètres

N 
!>          N est un entier (INTEGER)
!>          La taille de la matrice. Si elle est nulle, DSTT21 ne fait rien.
!>          Elle doit être supérieure ou égale à zéro.
!>

KBAND

!>          KBAND est un entier (INTEGER)
!>          La largeur de bande de la matrice S. Elle ne peut être que zéro ou un.
!>          Si elle est nulle, alors S est diagonale, et SE n'est pas référencé. Si
!>          elle est un, alors S est symétrique tri-diagonale.
!>

AD

!>          AD est un tableau de double précision (DOUBLE PRECISION), dimension (N)
!>          La diagonale de la matrice originale (non factorisée) A. A est
!>          supposée être symétrique tridiagonale.
!>

AE

!>          AE est un tableau de double précision (DOUBLE PRECISION), dimension (N-1)
!>          La diagonale hors-axe de la matrice originale (non factorisée) A. A est
!>          supposée être symétrique tridiagonale. AE(1) est l'élément (1,2)
!>          et (2,1), AE(2) est l'élément (2,3) et (3,2), etc.
!>

SD

!>          SD est un tableau de double précision (DOUBLE PRECISION), dimension (N)
!>          La diagonale de la matrice diagonale (symétrique tri-) S.
!>

SE

!>          SE est un tableau de double précision (DOUBLE PRECISION), dimension (N-1)
!>          La diagonale hors-axe de la matrice diagonale (symétrique tri-) S.
!>          Non référencé si KBSND=0. Si KBAND=1, alors AE(1) est l'élément
!>          (1,2) et (2,1), SE(2) est l'élément (2,3) et (3,2), etc.
!>

U

!>          U est un tableau de double précision (DOUBLE PRECISION), dimension (LDU, N)
!>          La matrice orthogonale dans la décomposition.
!>

LDU

!>          LDU est un entier (INTEGER)
!>          La dimension principale de U. LDU doit être au moins N.
!>

WORK

!>          WORK est un tableau de double précision (DOUBLE PRECISION), dimension (N*(N+1))
!>

RESULT

!>          RESULT est un tableau de double précision (DOUBLE PRECISION), dimension (2)
!>          Les valeurs calculées par les deux tests décrits ci-dessus. Les
!>          valeurs sont actuellement limitées à 1/ulp, pour éviter les dépassements.
!>          RESULT(1) est toujours modifié.
!>
