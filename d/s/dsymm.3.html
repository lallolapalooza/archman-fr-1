Documentation de la fonction/sous-programme

subroutine dsymm (character side, character uplo, integer m, integer n, double precision alpha, double precision, dimension(lda,*) a, integer lda, double precision, dimension(ldb,*) b, integer ldb, double precision beta, double precision, dimension(ldc,*) c, integer ldc)

DSYMM

But :

!>
!> DSYMM  effectue l'une des opérations matricielles :
!>
!>    C := alpha*A*B + beta*C,
!>
!> ou
!>
!>    C := alpha*B*A + beta*C,
!>
!> où alpha et beta sont des scalaires, A est une matrice symétrique et B et
!> C sont des matrices de taille m par n.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>           À l'entrée, SIDE spécifie si la matrice symétrique A apparaît
!>           à gauche ou à droite dans l'opération comme suit :
!>
!>              SIDE = 'L' ou 'l'   C := alpha*A*B + beta*C,
!>
!>              SIDE = 'R' ou 'r'   C := alpha*B*A + beta*C,
!>

UPLO

!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la partie triangulaire supérieure ou
!>           inférieure de la matrice symétrique A est à référencer comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de
!>                                  la matrice symétrique est à référencer.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de
!>                                  la matrice symétrique est à référencer.
!>

M

!>          M est INTEGER
!>           À l'entrée, M spécifie le nombre de lignes de la matrice C.
!>           M doit être supérieur ou égal à zéro.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie le nombre de colonnes de la matrice C.
!>           N doit être supérieur ou égal à zéro.
!>

ALPHA

!>          ALPHA est DOUBLE PRECISION.
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau DOUBLE PRECISION, de dimension ( LDA, ka ), où ka est
!>           m si SIDE = 'L' ou 'l', et n sinon.
!>           Avant l'entrée avec SIDE = 'L' ou 'l', la partie m par m du
!>           tableau A doit contenir la matrice symétrique, de sorte que
!>           lorsque UPLO = 'U' ou 'u', la partie supérieure de m par m du
!>           tableau A doit contenir la partie triangulaire supérieure de
!>           la matrice symétrique et la partie strictement inférieure de
!>           A n'est pas référencée, et lorsque UPLO = 'L' ou 'l',
!>           la partie inférieure de m par m du tableau A
!>           doit contenir la partie triangulaire inférieure de la matrice
!>           symétrique et la partie strictement supérieure de A n'est pas
!>           référencée.
!>           Avant l'entrée avec SIDE = 'R' ou 'r', la partie n par n du
!>           tableau A doit contenir la matrice symétrique, de sorte que
!>           lorsque UPLO = 'U' ou 'u', la partie supérieure de n par n du
!>           tableau A doit contenir la partie triangulaire supérieure de
!>           la matrice symétrique et la partie strictement inférieure de
!>           A n'est pas référencée, et lorsque UPLO = 'L' ou 'l',
!>           la partie inférieure de n par n du tableau A
!>           doit contenir la partie triangulaire inférieure de la matrice
!>           symétrique et la partie strictement supérieure de A n'est pas
!>           référencée.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant. Lorsque SIDE = 'L' ou 'l',
!>           LDA doit être au moins max( 1, m ), sinon LDA doit être
!>           au moins max( 1, n ).
!>

B

!>          B est un tableau DOUBLE PRECISION, de dimension ( LDB, N )
!>           Avant l'entrée, la partie m par n du tableau B doit
!>           contenir la matrice B.
!>

LDB

!>          LDB est INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que déclarée
!>           dans le programme appelant. LDB doit être au moins
!>           max( 1, m ).
!>

BETA

!>          BETA est DOUBLE PRECISION.
!>           À l'entrée, BETA spécifie le scalaire beta. Lorsque BETA
!>           est fourni comme zéro, C n'a pas besoin d'être défini à l'entrée.
!>

C

!>          C est un tableau DOUBLE PRECISION, de dimension ( LDC, N )
!>           Avant l'entrée, la partie m par n du tableau C doit
!>           contenir la matrice C, sauf lorsque beta est zéro, auquel cas
!>           C n'a pas besoin d'être défini à l'entrée.
!>           À la sortie, le tableau C est écrasé par la matrice mise à jour de taille m par n.
!>

LDC

!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que déclarée
!>           dans le programme appelant. LDC doit être au moins
!>           max( 1, m ).
!>
