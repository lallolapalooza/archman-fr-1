sous-routine dsbtrd (character vect, character uplo, integer n, integer kd, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) work, integer info)

DSBTRD

Objectif :

!>
!> DSBTRD réduit une matrice bande symétrique réelle A en forme symétrique
!> tridiagonale T par une transformation orthogonale de similarité :
!> Q**T * A * Q = T.
!>

Paramètres

VECT 
!>          VECT est CHARACTER*1
!>          = 'N':  ne pas former Q ;
!>          = 'V':  former Q ;
!>          = 'U':  mettre à jour une matrice X, en formant X*Q.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB est un tableau de DOUBLE PRECISION, de dimension (LDAB,N)
!>          En entrée, le triangle supérieur ou inférieur de la matrice bande
!>          symétrique A, stocké dans les KD+1 premières lignes du tableau. La
!>          colonne j de A est stockée dans la colonne j du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>          En sortie, les éléments diagonaux de AB sont écrasés par les
!>          éléments diagonaux de la matrice tridiagonale T ; si KD > 0, les
!>          éléments sur la première super-diagonale (si UPLO = 'U') ou la
!>          première sous-diagonale (si UPLO = 'L') sont écrasés par les
!>          éléments hors diagonaux de T ; le reste de AB est écrasé par
!>          les valeurs générées lors de la réduction.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

D

!>          D est un tableau de DOUBLE PRECISION, de dimension (N)
!>          Les éléments diagonaux de la matrice tridiagonale T.
!>

E

!>          E est un tableau de DOUBLE PRECISION, de dimension (N-1)
!>          Les éléments hors diagonaux de la matrice tridiagonale T :
!>          E(i) = T(i,i+1) si UPLO = 'U' ; E(i) = T(i+1,i) si UPLO = 'L'.
!>

Q

!>          Q est un tableau de DOUBLE PRECISION, de dimension (LDQ,N)
!>          En entrée, si VECT = 'U', alors Q doit contenir une matrice X de taille N x N ;
!>          si VECT = 'N' ou 'V', le tableau Q n'a pas besoin d'être défini.
!>
!>          En sortie :
!>          si VECT = 'V', Q contient la matrice orthogonale Q de taille N x N ;
!>          si VECT = 'U', Q contient le produit X*Q ;
!>          si VECT = 'N', le tableau Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= 1, et LDQ >= N si VECT = 'V' ou 'U'.
!>

WORK

!>          WORK est un tableau de DOUBLE PRECISION, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : succès de l'exécution
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
