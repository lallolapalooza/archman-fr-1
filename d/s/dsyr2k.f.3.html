subroutine dsyr2k (character uplo, character trans, integer n, integer k, double precision alpha, double precision, dimension(lda,*) a, integer lda, double precision, dimension(ldb,*) b, integer ldb, double precision beta, double precision, dimension(ldc,*) c, integer

DSYR2

K

But

: 
!>
!> DSYR2K effectue l'une des opérations de rang 2k symétriques
!>
!>    C := alpha*A*B**T + alpha*B*A**T + beta*C,
!>
!> ou
!>
!>    C := alpha*A**T*B + alpha*B**T*A + beta*C,
!>
!> où alpha et beta sont des scalaires, C est une matrice symétrique
!> n par n, et A et B sont des matrices n par k dans le premier cas et
!> k par n dans le second cas.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO indique si la partie triangulaire supérieure
!>           ou inférieure du tableau C doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire
!>                                  supérieure de C doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire
!>                                  inférieure de C doit être référencée.
!
>

TRAN

S 
!>          TRANS est CHARACTER*1
!>           À l'entrée, TRANS spécifie l'opération à effectuer comme
!>           suit :
!>
!>              TRANS = 'N' ou 'n'   C := alpha*A*B**T + alpha*B*A**T +
!>                                        beta*C.
!>
!>              TRANS = 'T' ou 't'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!>
!>              TRANS = 'C' ou 'c'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!
>

N 
!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice C. N doit être
!>           au moins égal à zéro.
!
>

K 
!>          K est INTEGER
!>           À l'entrée avec TRANS = 'N' ou 'n', K spécifie le nombre de
!>           colonnes des matrices A et B, et à l'entrée avec
!>           TRANS = 'T' ou 't' ou 'C' ou 'c', K spécifie le nombre de
!>           lignes des matrices A et B. K doit être au moins égal à zéro.
!
>

ALPH

A 
!>          ALPHA est DOUBLE PRECISION.
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!
>

A 
!>          A est un tableau DOUBLE PRECISION, dimension ( LDA, ka ), où ka est
!>           k quand TRANS = 'N' ou 'n', et n sinon.
!>           Avant l'entrée avec TRANS = 'N' ou 'n', la partie
!>           principale n par k du tableau A doit contenir la matrice A,
!>           sinon la partie principale k par n du tableau A doit
!>           contenir la matrice A.
!
>

LD

A 
!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que
!>           déclarée dans le (sous-)programme appelant. Quand
!>           TRANS = 'N' ou 'n', LDA doit être au moins max( 1, n ),
!>           sinon LDA doit être au moins max( 1, k ).
!
>

B 
!>          B est un tableau DOUBLE PRECISION, dimension ( LDB, kb ), où kb est
!>           k quand TRANS = 'N' ou 'n', et n sinon.
!>           Avant l'entrée avec TRANS = 'N' ou 'n', la partie
!>           principale n par k du tableau B doit contenir la matrice B,
!>           sinon la partie principale k par n du tableau B doit
!>           contenir la matrice B.
!
>

LD

B 
!>          LDB est INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que
!>           déclarée dans le (sous-)programme appelant. Quand
!>           TRANS = 'N' ou 'n', LDB doit être au moins max( 1, n ),
!>           sinon LDB doit être au moins max( 1, k ).
!
>

BET

A 
!>          BETA est DOUBLE PRECISION.
!>           À l'entrée, BETA spécifie le scalaire beta.
!
>

C 
!>          C est un tableau DOUBLE PRECISION, dimension ( LDC, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie
!>           triangulaire supérieure principale n par n du tableau C doit
!>           contenir la partie triangulaire supérieure de la matrice
!>           symétrique et la partie strictement triangulaire inférieure
!>           de C n'est pas référencée. À la sortie, la partie
!>           triangulaire supérieure du tableau C est écrasée par la
!>           partie triangulaire supérieure de la matrice mise à jour.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie
!>           triangulaire inférieure principale n par n du tableau C doit
!>           contenir la partie triangulaire inférieure de la matrice
!>           symétrique et la partie strictement triangulaire supérieure
!>           de C n'est pas référencée. À la sortie, la partie
!>           triangulaire inférieure du tableau C est écrasée par la
!>           partie triangulaire inférieure de la matrice mise à jour.
!
>

LD

C 
!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que
!>           déclarée dans le (sous-)programme appelant. LDC doit être au
!>           moins max( 1, n ).
!
>
