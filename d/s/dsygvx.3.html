subroutine dsygvx (integer itype, character jobz, character range, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

DSYGVX

But :

!>
!> DSYGVX calcule les valeurs propres sélectionnées et, si demandé, les vecteurs propres
!> d'un problème de valeurs propres généralisé symétrique-défini positif, de la forme
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  ou B*A*x=(lambda)*x.  Ici A
!> et B sont supposés symétriques et B est également défini positif.
!> Les valeurs propres et les vecteurs propres peuvent être sélectionnés en spécifiant soit une
!> plage de valeurs, soit une plage d'indices pour les valeurs propres souhaitées.
!>

Paramètres

ITYPE 
!>          ITYPE est INTEGER
!>          Spécifie le type de problème à résoudre :
!>          = 1 :  A*x = (lambda)*B*x
!>          = 2 :  A*B*x = (lambda)*x
!>          = 3 :  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ est CHARACTER*1
!>          = 'N' :  calcule uniquement les valeurs propres ;
!>          = 'V' :  calcule les valeurs propres et les vecteurs propres.
!>

RANGE

!>          RANGE est CHARACTER*1
!>          = 'A' : toutes les valeurs propres seront trouvées.
!>          = 'V' : toutes les valeurs propres dans l'intervalle semi-ouvert (VL,VU]
!>                 seront trouvées.
!>          = 'I' : les valeurs propres de la ième à la IU-ième seront trouvées.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A et B est stocké ;
!>          = 'L' : le triangle inférieur de A et B est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre du crayon de matrice (A,B).  N >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (LDA, N)
!>          À l'entrée, la matrice symétrique A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure de A de taille N par N contient la
!>          partie triangulaire supérieure de la matrice A.  Si UPLO = 'L',
!>          la partie triangulaire inférieure de A de taille N par N contient
!>          la partie triangulaire inférieure de la matrice A.
!>
!>          À la sortie, le triangle inférieur (si UPLO='L') ou le triangle
!>          supérieur (si UPLO='U') de A, y compris la diagonale, est
!>          détruit.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB, N)
!>          À l'entrée, la matrice symétrique B.  Si UPLO = 'U', la
!>          partie triangulaire supérieure de B de taille N par N contient la
!>          partie triangulaire supérieure de la matrice B.  Si UPLO = 'L',
!>          la partie triangulaire inférieure de B de taille N par N contient
!>          la partie triangulaire inférieure de la matrice B.
!>
!>          À la sortie, si INFO <= N, la partie de B contenant la matrice est
!>          écrasée par le facteur triangulaire U ou L de la factorisation
!>          de Cholesky B = U**T*U ou B = L*L**T.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

VL

!>          VL est DOUBLE PRECISION
!>          Si RANGE='V', la borne inférieure de l'intervalle dans
!>          lequel rechercher les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

VU

!>          VU est DOUBLE PRECISION
!>          Si RANGE='V', la borne supérieure de l'intervalle dans
!>          lequel rechercher les valeurs propres. VL < VU.
!>          Non référencé si RANGE = 'A' ou 'I'.
!>

IL

!>          IL est INTEGER
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

IU

!>          IU est INTEGER
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur propre à retourner.
!>          1 <= IL <= IU <= N, si N > 0 ; IL = 1 et IU = 0 si N = 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!>

ABSTOL

!>          ABSTOL est DOUBLE PRECISION
!>          La tolérance d'erreur absolue pour les valeurs propres.
!>          Une valeur propre approchée est acceptée comme convergée
!>          lorsqu'il est déterminé qu'elle se situe dans un intervalle [a,b]
!>          de largeur inférieure ou égale à
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          où EPS est la précision machine.  Si ABSTOL est inférieur ou
!>          égal à zéro, alors  EPS*|T|  sera utilisé à la place,
!>          où |T| est la 1-norme de la matrice tridiagonale obtenue
!>          en réduisant C à une forme tridiagonale, où C est la matrice
!>          symétrique du problème symétrique standard auquel le
!>          problème généralisé est transformé.
!>
!>          Les valeurs propres seront calculées avec le plus d'exactitude lorsque ABSTOL est
!>          défini à deux fois le seuil de débordement inférieur 2*DLAMCH('S'), et non à zéro.
!>          Si cette routine se termine avec INFO>0, indiquant que certains
!>          vecteurs propres n'ont pas convergé, essayez de définir ABSTOL à
!>          2*DLAMCH('S').
!>

M

!>          M est INTEGER
!>          Le nombre total de valeurs propres trouvées.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, et si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          À la sortie normale, les M premiers éléments contiennent les valeurs
!>          propres sélectionnées dans l'ordre croissant.
!>

Z

!>          Z est un tableau DOUBLE PRECISION, dimension (LDZ, max(1,M))
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>          Si JOBZ = 'V', alors si INFO = 0, les M premières colonnes de Z
!>          contiennent les vecteurs propres orthonormaux de la matrice A
!>          correspondant aux valeurs propres sélectionnées, avec la i-ième
!>          colonne de Z contenant le vecteur propre associé à W(i).
!>          Les vecteurs propres sont normalisés comme suit :
!>          si ITYPE = 1 ou 2, Z**T*B*Z = I ;
!>          si ITYPE = 3, Z**T*inv(B)*Z = I.
!>
!>          Si un vecteur propre ne converge pas, alors cette colonne de Z
!>          contient la dernière approximation du vecteur propre, et l'indice
!>          du vecteur propre est retourné dans IFAIL.
!>          Remarque : l'utilisateur doit s'assurer qu'au moins max(1,M) colonnes sont
!>          fournies dans le tableau Z ; si RANGE = 'V', la valeur exacte de M
!>          n'est pas connue à l'avance et une borne supérieure doit être utilisée.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!>

LWORK

!>          LWORK est un entier
!>          La longueur du tableau WORK.  LWORK >= max(1,8*N).
!>          Pour une efficacité optimale, LWORK >= (NB+3)*N,
!>          où NB est la taille de bloc pour DSYTRD retournée par ILAENV.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau d'entiers, dimension (5*N)
!>

IFAIL

!>          IFAIL est un tableau d'entiers, dimension (N)
!>          Si JOBZ = 'V', alors si INFO = 0, les premiers M éléments de
!>          IFAIL sont nuls.  Si INFO > 0, alors IFAIL contient les
!>          indices des vecteurs propres qui n'ont pas convergé.
!>          Si JOBZ = 'N', IFAIL n'est pas référencé.
!>

INFO

!>          INFO est un entier
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  DPOTRF ou DSYEVX a renvoyé un code d'erreur :
!>             <= N:  si INFO = i, DSYEVX n'a pas convergé ;
!>                    i vecteurs propres n'ont pas convergé.  Leurs indices
!>                    sont stockés dans le tableau IFAIL.
!>             > N:   si INFO = N + i, pour 1 <= i <= N, alors le mineur
!>                    principal de tête d'ordre i de B n'est pas positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ni vecteur propre n'a été calculé.
!>
