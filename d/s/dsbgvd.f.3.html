subroutine dsbgvd (character jobz, character uplo, integer n, integer ka, integer kb, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldbb, * ) bb, integer ldbb, double precision, dimension( * ) w, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer

info)

DSBGV

D

But

: 
!>
!> DSBGVD calcule toutes les valeurs propres, et éventuellement les vecteurs propres
!> d'un problème de valeurs propres généralisé symétrique-défini positif réel, de la
!> forme A*x=(lambda)*B*x. Ici, A et B sont supposés symétriques et
!> par bandes, et B est également défini positif. Si les vecteurs propres sont
!> souhaités, un algorithme de type « divide and conquer » est utilisé.
!>
!
>

Paramètre

s
JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' : calcule uniquement les valeurs propres ;
!>          = 'V' : calcule les valeurs propres et les vecteurs propres.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' : les triangles supérieurs de A et B sont stockés ;
!>          = 'L' : les triangles inférieurs de A et B sont stockés.
!
>

N 
!>          N est INTEGER
!>          L'ordre des matrices A et B. N >= 0.
!
>

K

A 
!>          KA est INTEGER
!>          Le nombre de surdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KA >= 0.
!
>

K

B 
!>          KB est INTEGER
!>          Le nombre de surdiagonales de la matrice B si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KB >= 0.
!
>

A

B 
!>          AB est un tableau DOUBLE PRECISION, dimension (LDAB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          symétrique par bandes A, stocké dans les premières ka+1 lignes du tableau. La
!>          j-ième colonne de A est stockée dans la j-ième colonne du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) pour max(1,j-ka)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+ka).
!>
!>          À la sortie, le contenu de AB est détruit.
!
>

LDA

B 
!>          LDAB est INTEGER
!>          La dimension principale du tableau AB. LDAB >= KA+1.
!
>

B

B 
!>          BB est un tableau DOUBLE PRECISION, dimension (LDBB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          symétrique par bandes B, stocké dans les premières kb+1 lignes du tableau. La
!>          j-ième colonne de B est stockée dans la j-ième colonne du tableau BB
!>          comme suit :
!>          si UPLO = 'U', BB(ka+1+i-j,j) = B(i,j) pour max(1,j-kb)<=i<=j ;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) pour j<=i<=min(n,j+kb).
!>
!>          À la sortie, le facteur S de la factorisation de Cholesky scindée
!>          B = S**T*S, tel que retourné par DPBSTF.
!
>

LDB

B 
!>          LDBB est INTEGER
!>          La dimension principale du tableau BB. LDBB >= KB+1.
!
>

W 
!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!
>

Z 
!>          Z est un tableau DOUBLE PRECISION, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres, avec la i-ième colonne de Z contenant le
!>          vecteur propre associé à W(i). Les vecteurs propres sont
!>          normalisés de sorte que Z**T*B*Z = I.
!>          Si JOBZ = 'N', Z n'est pas référencé.
!
>

LD

Z 
!>          LDZ est INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!
>

WOR

K 
!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) retourne la valeur optimale de LWORK.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LWORK >= 2*N.
!>          Si JOBZ = 'V' et N > 1, LWORK >= 1 + 5*N + 2*N**2.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement les tailles optimales des tableaux WORK et IWORK,
!>          retourne ces valeurs comme premières entrées des tableaux WORK
!>          et IWORK, et aucun message d'erreur lié à LWORK ou
!>          LIWORK n'est émis par XERBLA.
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si LIWORK > 0, IWORK(1) retourne la valeur optimale de LIWORK.
!
>

LIWOR

K 
!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ  = 'N' ou N <= 1, LIWORK >= 1.
!>          Si JOBZ  = 'V' et N > 1, LIWORK >= 3 + 5*N.
!>
!>          Si LIWORK = -1, une interrogation d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK et
!>          IWORK, retourne ces valeurs comme premières entrées des
!>          tableaux WORK et IWORK, et aucun message d'erreur lié à
!>          LWORK ou LIWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, et i est :
!>             <= N :  l'algorithme n'a pas convergé :
!>                    i éléments hors diagonale d'une forme
!>                    tridiagonale intermédiaire n'ont pas convergé vers zéro ;
!>             > N :   si INFO = N + i, pour 1 <= i <= N, alors DPBSTF
!>                    a retourné INFO = i : B n'est pas défini positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ni vecteur propre n'a été calculé.
!
>
