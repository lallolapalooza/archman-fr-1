subroutine dlatm4 (integer itype, integer n, integer nz1, integer nz2, integer isign, double precision amagn, double precision rcond, double precision triang, integer idist, integer, dimension( 4 ) iseed, double precision, dimension( lda, * ) a, integer lda)

DLATM4

But :

!>
!> DLATM4 génère des matrices carrées de base, qui peuvent ensuite être
!> multipliées par d'autres pour produire des matrices de test. Elle est
!> principalement destinée à tester les routines de valeurs propres généralisées.
!>
!> Elle génère d'abord la diagonale et (éventuellement) la sous-diagonale,
!> selon les valeurs de ITYPE, NZ1, NZ2, ISIGN, AMAGN et RCOND.
!> Elle remplit ensuite le triangle supérieur avec des nombres aléatoires, si TRIANG est
!> non nul.
!>

Paramètres

ITYPE 
!>          ITYPE est un entier (INTEGER)
!>          Le type de matrice sur la diagonale et la sous-diagonale.
!>          Si ITYPE < 0, alors le type abs(ITYPE) est généré puis
!>             inversé bout à bout (A(I,J) := A'(N-J,N-I).) Voir aussi
!>             la description de AMAGN et ISIGN.
!>
!>          Types spéciaux :
!>          = 0 : la matrice nulle.
!>          = 1 : l'identité.
!>          = 2 : un bloc de Jordan transposé.
!>          = 3 : Si N est impair, alors un bloc de Jordan transposé k+1 x k+1
!>                suivi d'un bloc identité k x k, où k=(N-1)/2.
!>                Si N est pair, alors k=(N-2)/2, et une entrée diagonale nulle
!>                est ajoutée à la fin.
!>
!>          Types diagonaux. La diagonale consiste en NZ1 zéros, puis
!>             k=N-NZ1-NZ2 éléments non nuls. La sous-diagonale est nulle. ITYPE
!>             spécifie les entrées diagonales non nulles comme suit :
!>          = 4 : 1, ..., k
!>          = 5 : 1, RCOND, ..., RCOND
!>          = 6 : 1, ..., 1, RCOND
!>          = 7 : 1, a, a^2, ..., a^(k-1)=RCOND
!>          = 8 : 1, 1-d, 1-2*d, ..., 1-(k-1)*d=RCOND
!>          = 9 : nombres aléatoires choisis dans (RCOND,1)
!>          = 10 : nombres aléatoires avec la distribution IDIST (voir DLARND.)
!>

N

!>          N est un entier (INTEGER)
!>          L'ordre de la matrice.
!>

NZ1

!>          NZ1 est un entier (INTEGER)
!>          Si abs(ITYPE) > 3, alors les NZ1 premières entrées diagonales seront
!>          nulles.
!>

NZ2

!>          NZ2 est un entier (INTEGER)
!>          Si abs(ITYPE) > 3, alors les NZ2 dernières entrées diagonales seront
!>          nulles.
!>

ISIGN

!>          ISIGN est un entier (INTEGER)
!>          = 0 : Le signe des entrées diagonales et sous-diagonales sera
!>               laissé inchangé.
!>          = 1 : Le signe des entrées diagonales et sous-diagonales sera
!>               changé aléatoirement.
!>          = 2 : Si ITYPE est 2 ou 3, alors c'est la même chose que ISIGN=1.
!>               Sinon, avec une probabilité de 0.5, des paires impaires-paires
!>               d'entrées diagonales A(2*j-1,2*j-1), A(2*j,2*j) seront
!>               converties en un bloc 2x2 en pré- et post-multipliant
!>               par des rotations orthogonales aléatoires distinctes. Les autres
!>               entrées diagonales verront leur signe changé aléatoirement.
!>

AMAGN

!>          AMAGN est un double précision (DOUBLE PRECISION)
!>          Les entrées diagonales et sous-diagonales seront multipliées par
!>          AMAGN.
!>

RCOND

!>          RCOND est un double précision (DOUBLE PRECISION)
!>          Si abs(ITYPE) > 4, alors la plus petite entrée diagonale sera
!>          l'entrée sera RCOND. RCOND doit être compris entre 0 et 1.
!>

TRIANG

!>          TRIANG est un double précision (DOUBLE PRECISION)
!>          Les entrées au-dessus de la diagonale seront des nombres aléatoires
!>          de magnitude bornée par TRIANG (c'est-à-dire des nombres aléatoires
!>          multipliés par TRIANG.)
!>

IDIST

!>          IDIST est un entier (INTEGER)
!>          Spécifie le type de distribution à utiliser pour générer une
!>          matrice aléatoire.
!>          = 1 : UNIFORME( 0, 1 )
!>          = 2 : UNIFORME( -1, 1 )
!>          = 3 : NORMALE ( 0, 1 )
!>

ISEED

!>          ISEED est un tableau d'entiers (INTEGER array), dimension (4)
!>          À l'entrée, ISEED spécifie la graine du générateur de nombres
!>          aléatoires. Les valeurs de ISEED sont modifiées à la sortie, et peuvent
!>          être utilisées dans le prochain appel à DLATM4 pour continuer la même
!>          séquence de nombres aléatoires.
!>          Note : ISEED(4) doit être impair, pour le générateur de nombres
!>          aléatoires utilisé actuellement.
!>

A

!>          A est un tableau de double précision (DOUBLE PRECISION array), dimension (LDA, N)
!>          Tableau à calculer.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          Dimension principale de A. Doit être au moins 1 et au moins N.
!>
