subroutine dlaqgb (integer m, integer n, integer kl, integer ku, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, character equed)

DLAQGB met à l'échelle une matrice bande générale, en utilisant les facteurs de mise à l'échelle de ligne et de colonne calculés par sgbequ.

Objectif :

!>
!> DLAQGB équilibre une matrice bande générale M par N de bande avec KL
!> sous-diagonales et KU super-diagonales en utilisant les facteurs de mise
!> à l'échelle de ligne et de colonne dans les vecteurs R et C.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

KL

!>          KL est un ENTIER
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!>

KU

!>          KU est un ENTIER
!>          Le nombre de super-diagonales dans la bande de A.  KU >= 0.
!>

AB

!>          AB est un tableau DOUBLE PRECISION, dimension (LDAB,N)
!>          À l'entrée, la matrice A en stockage bande, dans les lignes 1 à KL+KU+1.
!>          La j-ième colonne de A est stockée dans la j-ième colonne du
!>          tableau AB comme suit :
!>          AB(ku+1+i-j,j) = A(i,j) pour max(1,j-ku)<=i<=min(m,j+kl)
!>
!>          À la sortie, la matrice équilibrée, dans le même format de stockage
!>          que A. Voir EQUED pour la forme de la matrice équilibrée.
!>

LDAB

!>          LDAB est un ENTIER
!>          La dimension principale du tableau AB.  LDA >= KL+KU+1.
!>

R

!>          R est un tableau DOUBLE PRECISION, dimension (M)
!>          Les facteurs d'échelle de ligne pour A.
!>

C

!>          C est un tableau DOUBLE PRECISION, dimension (N)
!>          Les facteurs d'échelle de colonne pour A.
!>

ROWCND

!>          ROWCND est un DOUBLE PRECISION
!>          Ratio du plus petit R(i) au plus grand R(i).
!>

COLCND

!>          COLCND est un DOUBLE PRECISION
!>          Ratio du plus petit C(i) au plus grand C(i).
!>

AMAX

!>          AMAX est un DOUBLE PRECISION
!>          Valeur absolue du plus grand élément de la matrice.
!>

EQUED

!>          EQUED est un CARACTERE*1
!>          Spécifie la forme d'équilibrage qui a été effectuée.
!>          = 'N' : Pas d'équilibrage
!>          = 'R' : Équilibrage par ligne, c.-à-d. A a été pré-multiplié par
!>                  diag(R).
!>          = 'C' : Équilibrage par colonne, c.-à-d. A a été post-multiplié
!>                  par diag(C).
!>          = 'B' : Équilibrage par ligne et par colonne, c.-à-d. A a été
!>                  remplacé par diag(R) * A * diag(C).
!>

Paramètres internes :

!>  THRESH est une valeur seuil utilisée pour décider si une mise à l'échelle
!>  par ligne ou par colonne doit être effectuée en fonction du ratio des
!>  facteurs d'échelle de ligne ou de colonne.  Si ROWCND < THRESH, une mise
!>  à l'échelle par ligne est effectuée, et si COLCND < THRESH, une mise à
!>  l'échelle par colonne est effectuée.
!>
!>  LARGE et SMALL sont des valeurs seuils utilisées pour décider si une mise
!>  à l'échelle par ligne doit être effectuée en fonction de la taille
!>  absolue du plus grand élément de la matrice.  Si AMAX > LARGE ou AMAX
!>  < SMALL, une mise à l'échelle par ligne est effectuée.
!>
