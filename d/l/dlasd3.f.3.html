subroutine dlasd3 (integer nl, integer nr, integer sqre, integer k, double precision, dimension( * ) d, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) dsigma, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldu2, * ) u2, integer ldu2, double precision, dimension( ldvt, * ) vt, integer ldvt, double precision, dimension( ldvt2, * ) vt2, integer ldvt2, integer, dimension( * ) idxc, integer, dimension( * ) ctot, double precision, dimension( * ) z, integer info)

DLASD3 trouve toutes les racines carrées des racines de l'équation séculaire, telles que définies par les valeurs dans D et Z, puis met à jour les vecteurs singuliers par multiplication matricielle. Utilisé par sbdsdc.

Objectif :

!>
!> DLASD3 trouve toutes les racines carrées des racines de l'équation
!> séculaire, telles que définies par les valeurs dans D et Z. Elle effectue
!> les appels appropriés à DLASD4, puis met à jour les vecteurs
!> singuliers par multiplication matricielle.
!>
!> DLASD3 est appelée depuis DLASD1.
!>

Paramètres

NL 
!>          NL est INTEGER
!>         La dimension de ligne du bloc supérieur. NL >= 1.
!>

NR

!>          NR est INTEGER
!>         La dimension de ligne du bloc inférieur. NR >= 1.
!>

SQRE

!>          SQRE est INTEGER
!>         = 0 : le bloc inférieur est une matrice carrée NRxNR.
!>         = 1 : le bloc inférieur est une matrice rectangulaire NRx(NR+1).
!>
!>         La matrice bidiagonale a N = NL + NR + 1 lignes et
!>         M = N + SQRE >= N colonnes.
!>

K

!>          K est INTEGER
!>         La taille de l'équation séculaire, 1 <= K <= N.
!>

D

!>          D est un tableau de type DOUBLE PRECISION, dimension(K)
!>         À la sortie, contient les racines carrées des racines de l'équation
!>         séculaire, dans l'ordre croissant.
!>

Q

!>          Q est un tableau de type DOUBLE PRECISION, dimension (LDQ,K)
!>

LDQ

!>          LDQ est INTEGER
!>         La dimension principale du tableau Q. LDQ >= K.
!>

DSIGMA

!>          DSIGMA est un tableau de type DOUBLE PRECISION, dimension(K)
!>         Les K premiers éléments de ce tableau contiennent les anciennes racines
!>         du problème de mise à jour déflaté. Ce sont les pôles
!>         de l'équation séculaire.
!>

U

!>          U est un tableau de type DOUBLE PRECISION, dimension (LDU, N)
!>         Les N - K dernières colonnes de cette matrice contiennent les vecteurs
!>         singuliers gauches déflatés.
!>

LDU

!>          LDU est INTEGER
!>         La dimension principale du tableau U. LDU >= N.
!>

U2

!>          U2 est un tableau de type DOUBLE PRECISION, dimension (LDU2, N)
!>         Les K premières colonnes de cette matrice contiennent les vecteurs
!>         singuliers gauches non déflatés pour le problème divisé.
!>

LDU2

!>          LDU2 est INTEGER
!>         La dimension principale du tableau U2. LDU2 >= N.
!>

VT

!>          VT est un tableau de type DOUBLE PRECISION, dimension (LDVT, M)
!>         Les M - K dernières colonnes de VT**T contiennent les vecteurs
!>         singuliers droits déflatés.
!>

LDVT

!>          LDVT est INTEGER
!>         La dimension principale du tableau VT. LDVT >= N.
!>

VT2

!>          VT2 est un tableau de type DOUBLE PRECISION, dimension (LDVT2, N)
!>         Les K premières colonnes de VT2**T contiennent les vecteurs
!>         singuliers droits non déflatés pour le problème divisé.
!>

LDVT2

!>          LDVT2 est INTEGER
!>         La dimension principale du tableau VT2. LDVT2 >= N.
!>

IDXC

!>          IDXC est un tableau de type INTEGER, dimension ( N )
!>         La permutation utilisée pour organiser les colonnes de U (et les lignes
!>         de VT) en trois groupes : le premier groupe contient des entrées non nulles
!>         uniquement aux positions supérieures (ou avant) NL + 1 ; le second
!>         contient des entrées non nulles uniquement aux positions inférieures (ou
!>         après) NL + 2 ; et le troisième est dense. La première colonne de U et
!>         la ligne de VT sont traitées séparément.
!>
!>         Les lignes des vecteurs singuliers trouvés par DLASD4
!>         doivent être permutées de manière similaire avant que les multiplications
!>         matricielles puissent avoir lieu.
!>

CTOT

!>          CTOT est un tableau de type INTEGER, dimension ( 4 )
!>         Un compte du nombre total des différents types de colonnes
!>         dans U (ou lignes dans VT), comme décrit dans IDXC. Le quatrième type de
!>         colonne concerne toute colonne qui a été déflatée.
!>

Z

!>          Z est un tableau de type DOUBLE PRECISION, dimension (K)
!>         Les K premiers éléments de ce tableau contiennent les composantes
!>         du vecteur ligne de mise à jour ajusté par déflation.
!>

INFO

!>          INFO est INTEGER
!>         = 0 : sortie réussie.
!>         < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>         > 0 : si INFO = 1, une valeur singulière n'a pas convergé.
!>
