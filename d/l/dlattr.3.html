subroutine dlattr (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) b, double precision, dimension( * ) work, integer info)

DLATTR

Objectif

!>
!> DLATTR génère une matrice de test triangulaire.
!> IMAT et UPLO spécifient de manière unique les propriétés de la matrice de test,
!> qui est retournée dans le tableau A.
!>

Paramètres

IMAT 
!>          IMAT est INTEGER
!>          Une clé entière décrivant quelle matrice générer pour ce
!>          chemin.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A sera triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie si la matrice ou sa transposée sera utilisée.
!>          = 'N':  Pas de transposée
!>          = 'T':  Transposée
!>          = 'C':  Transposée conjuguée (= Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire unitaire ou non.
!>          = 'N':  Triangulaire non unitaire
!>          = 'U':  Triangulaire unitaire
!>

ISEED

!>          ISEED est un tableau INTEGER, de dimension (4)
!>          Le vecteur de départ pour le générateur de nombres aléatoires (utilisé dans
!>          DLATMS). Modifié à la sortie.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice à générer.
!>

A

!>          A est un tableau DOUBLE PRECISION, de dimension (LDA,N)
!>          La matrice triangulaire A. Si UPLO = 'U', la partie supérieure
!>          triangulaire de n par n de la matrice A contient la matrice
!>          triangulaire supérieure, et la partie strictement inférieure
!>          triangulaire de A n'est pas référencée. Si UPLO = 'L', la partie
!>          inférieure triangulaire de n par n de la matrice A contient la matrice
!>          triangulaire inférieure, et la partie strictement supérieure
!>          triangulaire de A n'est pas référencée. Si DIAG = 'U', les éléments diagonaux de A sont
!>          définis de telle sorte que A(k,k) = k pour 1 <= k <= n.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

B

!>          B est un tableau DOUBLE PRECISION, de dimension (N)
!>          Le vecteur de membre de droite, si IMAT > 10.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, de dimension (3*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  succès
!>          < 0: si INFO = -k, le k-ième argument avait une valeur illégale
!>
