subroutine dlaord (character job, integer n, double precision, dimension( * ) x, integer incx)

DLAORD

Objectif:

!>
!> DLAORD trie les éléments d'un vecteur x par ordre croissant ou décroissant.
!>

Paramètres

JOB 
!>          JOB est CHARACTER
!>          = 'I':  Trier par ordre croissant
!>          = 'D':  Trier par ordre décroissant
!>

N

!>          N est INTEGER
!>          La longueur du vecteur X.
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension
!>                         (1+(N-1)*INCX)
!>          À l'entrée, le vecteur de longueur n à trier.
!>          À la sortie, le vecteur x est trié dans l'ordre prescrit.
!>

INCX

!>          INCX est INTEGER
!>          L'espacement entre les éléments successifs de X.  INCX >= 0.
!>
