subroutine dlarrb (n, d, lld, ifirst, ilast, rtol1, rtol2, offset, w, wgap, werr, work, iwork, pivmin, spdiam, twist, info)
DLARRB fournit une bissection limitée pour localiser les valeurs propres avec plus de précision.

subroutine dlarrb (integer n, double precision, dimension( * ) d, double precision, dimension( * ) lld, integer ifirst, integer ilast, double precision rtol1, double precision rtol2, integer offset, double precision, dimension( * ) w, double precision, dimension( * ) wgap, double precision, dimension( * ) werr, double precision, dimension( * ) work, integer, dimension( * ) iwork, double precision pivmin, double precision spdiam, integer twist, integer

info)

DLARRB fournit une bissection limitée pour localiser les valeurs propres avec plus de

précision.

But

: 
!>
!> Étant donné la représentation robuste relative (RRR) L D L^T, DLARRB
!> effectue une bissection pour affiner les valeurs propres de L D L^T,
!> W( IFIRST-OFFSET ) à W( ILAST-OFFSET ), avec plus de précision. Les
!> valeurs initiales de ces valeurs propres sont fournies en entrée dans W,
!> l'estimation correspondante de l'erreur sur ces valeurs et leurs écarts
!> sont fournies en entrée dans WERR et WGAP, respectivement. Pendant la
!> bissection, les intervalles [left, right] sont maintenus en stockant
!> leurs points médians et leurs demi-largeurs dans les tableaux W et WERR
!> respectivement.
!
>

Paramètre

s
N 
!>          N est INTEGER
!>          L'ordre de la matrice.
!
>

D 
!>          D est un tableau DOUBLE PRECISION, dimension (N)
!>          Les N éléments diagonaux de la matrice diagonale D.
!
>

LL

D 
!>          LLD est un tableau DOUBLE PRECISION, dimension (N-1)
!>          Les (N-1) éléments L(i)*L(i)*D(i).
!
>

IFIRS

T 
!>          IFIRST est INTEGER
!>          L'indice de la première valeur propre à calculer.
!
>

ILAS

T 
!>          ILAST est INTEGER
!>          L'indice de la dernière valeur propre à calculer.
!
>

RTOL

1 
!>          RTOL1 est DOUBLE PRECISION
!
>

RTOL

2 
!>          RTOL2 est DOUBLE PRECISION
!>          Tolérance pour la convergence des intervalles de bissection.
!>          Un intervalle [LEFT,RIGHT] a convergé si
!>          RIGHT-LEFT < MAX( RTOL1*GAP, RTOL2*MAX(|LEFT|,|RIGHT|) )
!>          où GAP est la distance (estimée) à la valeur propre la plus
!>          proche.
!
>

OFFSE

T 
!>          OFFSET est INTEGER
!>          Décalage pour les tableaux W, WGAP et WERR, c'est-à-dire que les
!>          éléments de IFIRST-OFFSET à ILAST-OFFSET de ces tableaux doivent
!>          être utilisés.
!
>

W 
!>          W est un tableau DOUBLE PRECISION, dimension (N)
!>          En entrée, W( IFIRST-OFFSET ) à W( ILAST-OFFSET ) sont
!>          des estimations des valeurs propres de L D L^T indexées de IFIRST
!>          à ILAST.
!>          En sortie, ces estimations sont affinées.
!
>

WGA

P 
!>          WGAP est un tableau DOUBLE PRECISION, dimension (N-1)
!>          En entrée, les écarts (estimés) entre les valeurs propres
!>          consécutives de L D L^T, c'est-à-dire que WGAP(I-OFFSET) est
!>          l'écart entre les valeurs propres I et I+1. Notez que si
!>          IFIRST = ILAST, alors WGAP(IFIRST-OFFSET) doit être mis à ZERO.
!>          En sortie, ces écarts sont affinés.
!
>

WER

R 
!>          WERR est un tableau DOUBLE PRECISION, dimension (N)
!>          En entrée, WERR( IFIRST-OFFSET ) à WERR( ILAST-OFFSET ) sont
!>          les erreurs dans les estimations des éléments correspondants dans W.
!>          En sortie, ces erreurs sont affinées.
!
>

WOR

K 
!>          WORK est un tableau DOUBLE PRECISION, dimension (2*N)
!>          Espace de travail.
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (2*N)
!>          Espace de travail.
!
>

PIVMI

N 
!>          PIVMIN est DOUBLE PRECISION
!>          Le pivot minimum dans la séquence de Sturm.
!
>

SPDIA

M 
!>          SPDIAM est DOUBLE PRECISION
!>          Le diamètre spectral de la matrice.
!
>

TWIS

T 
!>          TWIST est INTEGER
!>          L'indice de torsion pour la factorisation tordue qui est utilisée
!>          pour le negcount.
!>          TWIST = N : Calculer negcount à partir de L D L^T - LAMBDA I = L+ D+ L+^T
!>          TWIST = 1 : Calculer negcount à partir de L D L^T - LAMBDA I = U- D- U-^T
!>          TWIST = R : Calculer negcount à partir de L D L^T - LAMBDA I = N(r) D(r) N(r)
!
>

INF

O 
!>          INFO est INTEGER
!>          Drapeau d'erreur.
!
>
