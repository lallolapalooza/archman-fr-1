subroutine dbdsdc (character uplo, character compq, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldvt, * ) vt, integer ldvt, double precision, dimension( * ) q, integer, dimension( * ) iq, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer

DBDSD

C

But

: 
!>
!> DBDSDC calcule la décomposition en valeurs singulières (SVD) d'une
!> matrice bidiagonale réelle N x N (supérieure ou inférieure) B :  B = U * S * VT,
!> en utilisant une méthode de diviser pour mieux régner, où S est une matrice diagonale
!> dont les éléments diagonaux sont non négatifs (les valeurs singulières de B), et
!> U et VT sont des matrices orthogonales contenant respectivement les vecteurs
!> singuliers gauches et droits. DBDSDC peut être utilisé pour calculer toutes les valeurs
!> singulières et, de manière facultative, les vecteurs singuliers ou les vecteurs
!> singuliers sous forme compacte.
!>
!> Le code appelle actuellement DLASDQ si seules les valeurs singulières sont souhaitées.
!> Cependant, il peut être légèrement modifié pour calculer les valeurs singulières
!> en utilisant la méthode de diviser pour mieux régner.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' :  B est bidiagonale supérieure.
!>          = 'L' :  B est bidiagonale inférieure.
!
>

COMP

Q 
!>          COMPQ est CHARACTER*1
!>          Indique si les vecteurs singuliers doivent être calculés
!>          comme suit :
!>          = 'N' :  Calculer uniquement les valeurs singulières ;
!>          = 'P' :  Calculer les valeurs singulières et calculer les vecteurs
!>                  singuliers sous forme compacte ;
!>          = 'I' :  Calculer les valeurs singulières et les vecteurs singuliers.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice B.  N >= 0.
!
>

D 
!>          D est un tableau DOUBLE PRECISION, dimension (N)
!>          À l'entrée, les n éléments diagonaux de la matrice bidiagonale B.
!>          À la sortie, si INFO=0, les valeurs singulières de B.
!
>

E 
!>          E est un tableau DOUBLE PRECISION, dimension (N-1)
!>          À l'entrée, les éléments de E contiennent les éléments hors diagonale
!>          de la matrice bidiagonale dont la SVD est souhaitée.
!>          À la sortie, E a été détruit.
!
>

U 
!>          U est un tableau DOUBLE PRECISION, dimension (LDU,N)
!>          Si  COMPQ = 'I', alors :
!>             À la sortie, si INFO = 0, U contient les vecteurs singuliers gauches
!>             de la matrice bidiagonale.
!>          Pour les autres valeurs de COMPQ, U n'est pas référencé.
!
>

LD

U 
!>          LDU est INTEGER
!>          La dimension principale du tableau U.  LDU >= 1.
!>          Si les vecteurs singuliers sont souhaités, alors LDU >= max( 1, N ).
!
>

V

T 
!>          VT est un tableau DOUBLE PRECISION, dimension (LDVT,N)
!>          Si  COMPQ = 'I', alors :
!>             À la sortie, si INFO = 0, VT**T contient les vecteurs singuliers
!>             droits de la matrice bidiagonale.
!>          Pour les autres valeurs de COMPQ, VT n'est pas référencé.
!
>

LDV

T 
!>          LDVT est INTEGER
!>          La dimension principale du tableau VT.  LDVT >= 1.
!>          Si les vecteurs singuliers sont souhaités, alors LDVT >= max( 1, N ).
!
>

Q 
!>          Q est un tableau DOUBLE PRECISION, dimension (LDQ)
!>          Si  COMPQ = 'P', alors :
!>             À la sortie, si INFO = 0, Q et IQ contiennent les vecteurs singuliers
!>             gauches et droits sous une forme compacte,
!>             nécessitant O(N log N) espace au lieu de 2*N**2.
!>             En particulier, Q contient toutes les données DOUBLE PRECISION dans
!>             LDQ >= N*(11 + 2*SMLSIZ + 8*INT(LOG_2(N/(SMLSIZ+1))))
!>             mots de mémoire, où SMLSIZ est renvoyé par ILAENV et
!>             est égal à la taille maximale des sous-problèmes au
!>             bas de l'arbre de calcul (généralement environ 25).
!>          Pour les autres valeurs de COMPQ, Q n'est pas référencé.
!
>

I

Q 
!>          IQ est un tableau INTEGER, dimension (LDIQ)
!>          Si  COMPQ = 'P', alors :
!>             À la sortie, si INFO = 0, Q et IQ contiennent les vecteurs singuliers
!>             gauches et droits sous une forme compacte,
!>             nécessitant O(N log N) espace au lieu de 2*N**2.
!>             En particulier, IQ contient toutes les données INTEGER dans
!>             LDIQ >= N*(3 + 3*INT(LOG_2(N/(SMLSIZ+1))))
!>             mots de mémoire, où SMLSIZ est renvoyé par ILAENV et
!>             est égal à la taille maximale des sous-problèmes au
!>             bas de l'arbre de calcul (généralement environ 25).
!>          Pour les autres valeurs de COMPQ, IQ n'est pas référencé.
!
>

WOR

K 
!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          Si COMPQ = 'N' alors LWORK >= (4 * N).
!>          Si COMPQ = 'P' alors LWORK >= (6 * N).
!>          Si COMPQ = 'I' alors LWORK >= (3 * N**2 + 4 * N).
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (8*N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie.
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 :  L'algorithme n'a pas réussi à calculer une valeur singulière.
!>                Le processus de mise à jour de diviser pour mieux régner a échoué.
!
>
