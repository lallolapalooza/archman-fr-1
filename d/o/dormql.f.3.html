subroutine dormql (side, trans, m, n, k, a, lda, tau, c, ldc, work, lwork, info)
DORMQL

Documentation de la fonction/sous-programme

sous-programme dormql (caractère side, caractère trans, entier m, entier n, entier k, double précision, dimension( lda, * ) a, entier lda, double précision, dimension( * ) tau, double précision, dimension( ldc, * ) c, entier ldc, double précision, dimension( * ) work, entier lwork, entier info)

DORMQL

Objectif :

!>
!> DORMQL écrase la matrice réelle générale M par N, C, avec
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> où Q est une matrice orthogonale réelle définie comme le produit de k
!> réflecteurs élémentaires
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> telle que renvoyée par DGEQLF. Q est de taille M si SIDE = 'L' et de taille N
!> si SIDE = 'R'.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : appliquer Q ou Q**T à gauche ;
!>          = 'R' : appliquer Q ou Q**T à droite.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N' : Pas de transposée, appliquer Q ;
!>          = 'T' : Transposée, appliquer Q**T.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q.
!>          Si SIDE = 'L', M >= K >= 0 ;
!>          si SIDE = 'R', N >= K >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (LDA,K)
!>          La i-ème colonne doit contenir le vecteur qui définit le
!>          réflecteur élémentaire H(i), pour i = 1,2,...,k, tel que renvoyé par
!>          DGEQLF dans les k dernières colonnes de son argument de tableau A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.
!>          Si SIDE = 'L', LDA >= max(1,M) ;
!>          si SIDE = 'R', LDA >= max(1,N).
!>

TAU

!>          TAU est un tableau DOUBLE PRECISION, dimension (K)
!>          TAU(i) doit contenir le facteur scalaire du réflecteur
!>          élémentaire H(i), tel que renvoyé par DGEQLF.
!>

C

!>          C est un tableau DOUBLE PRECISION, dimension (LDC,N)
!>          À l'entrée, la matrice M par N, C.
!>          À la sortie, C est écrasé par Q*C ou Q**T*C ou C*Q**T ou C*Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si SIDE = 'L', LWORK >= max(1,N) ;
!>          si SIDE = 'R', LWORK >= max(1,M).
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur dans la première entrée du tableau WORK, et aucun message
!>          d'erreur relatif à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
