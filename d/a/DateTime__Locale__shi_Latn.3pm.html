%c (%a %b %e %H:%M:%S %Y) - format date heure

 1995-12-22T09:05:02 = asim duj 22 09:05:02 1995
-0010-09-15T04:44:23 = asiḍ cut 15 04:44:23 -10

-0010-09-15T04:44:23 = 09/15/10
