subroutine dhgeqz (character job, character compq, character compz, integer n, integer ilo, integer ihi, double precision, dimension( ldh, * ) h, integer ldh, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) alphar, double precision, dimension( * ) alphai, double precision, dimension( * ) beta, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer lwork, integer info)

DHGEQZ

But :

!>
!> DHGEQZ calcule les valeurs propres d'une paire de matrices réelles (H,T),
!> où H est une matrice de Hessenberg supérieure et T est une matrice
!> triangulaire supérieure, en utilisant la méthode QZ à double décalage.
!> Les paires de matrices de ce type sont produites par la réduction à la
!> forme de Hessenberg supérieure généralisée d'une paire de matrices
!> réelles (A,B) :
!>
!>    A = Q1*H*Z1**T,  B = Q1*T*Z1**T,
!>
!> telle que calculée par DGGHRD.
!>
!> Si JOB='S', la paire Hessenberg-triangulaire (H,T) est également
!> réduite à la forme de Schur généralisée,
!>
!>    H = Q*S*Z**T,  T = Q*P*Z**T,
!>
!> où Q et Z sont des matrices orthogonales, P est une matrice
!> triangulaire supérieure, et S est une matrice quasi-triangulaire avec
!> des blocs diagonaux 1-par-1 et 2-par-2.
!>
!> Les blocs 1-par-1 correspondent aux valeurs propres réelles de la paire
!> de matrices (H,T) et les blocs 2-par-2 correspondent aux paires de
!> valeurs propres conjuguées complexes.
!>
!> De plus, les blocs diagonaux triangulaires supérieurs 2-par-2 de P
!> correspondant aux blocs 2-par-2 de S sont réduits à une forme
!> diagonale positive, c'est-à-dire que si S(j+1,j) est non nul, alors
!> P(j+1,j) = P(j,j+1) = 0, P(j,j) > 0, et P(j+1,j+1) > 0.
!>
!> En option, la matrice orthogonale Q issue de la factorisation de Schur
!> généralisée peut être post-multipliée par une matrice d'entrée Q1, et
!> la matrice orthogonale Z peut être post-multipliée par une matrice
!> d'entrée Z1.
!> Si Q1 et Z1 sont les matrices orthogonales issues de DGGHRD qui ont
!> réduit la paire de matrices (A,B) à la forme de Hessenberg supérieure
!> généralisée, alors les matrices de sortie Q1*Q et Z1*Z sont les
!> facteurs orthogonaux de la factorisation de Schur généralisée de (A,B) :
!>
!>    A = (Q1*Q)*S*(Z1*Z)**T,  B = (Q1*Q)*P*(Z1*Z)**T.
!>
!> Pour éviter le débordement, les valeurs propres de la paire de matrices
!> (H,T) (équivalent, de (A,B)) sont calculées sous la forme d'une paire
!> de valeurs (alpha,beta), où alpha est complexe et beta est réel.
!> Si beta est non nul, lambda = alpha / beta est une valeur propre du
!> problème de valeurs propres non symétrique généralisé (GNEP)
!>    A*x = lambda*B*x
!> et si alpha est non nul, mu = beta / alpha est une valeur propre de
!> la forme alternative du GNEP
!>    mu*A*y = B*y.
!> Les valeurs propres réelles peuvent être lues directement depuis la
!> forme de Schur généralisée :
!>   alpha = S(i,i), beta = P(i,i).
!>
!> Réf : C.B. Moler & G.W. Stewart, , SIAM J. Numer. Anal., 10(1973),
!>      pp. 241--256.
!>

Paramètres

JOB 
!>          JOB est CHARACTER*1
!>          = 'E' : Calcule les valeurs propres uniquement ;
!>          = 'S' : Calcule les valeurs propres et la forme de Schur.
!>

COMPQ

!>          COMPQ est CHARACTER*1
!>          = 'N' : Les vecteurs de Schur gauches (Q) ne sont pas calculés ;
!>          = 'I' : Q est initialisée à la matrice unité et la matrice Q
!>                  des vecteurs de Schur gauches de (H,T) est retournée ;
!>          = 'V' : Q doit contenir une matrice orthogonale Q1 à l'entrée et
!>                  le produit Q1*Q est retourné.
!>

COMPZ

!>          COMPZ est CHARACTER*1
!>          = 'N' : Les vecteurs de Schur droits (Z) ne sont pas calculés ;
!>          = 'I' : Z est initialisée à la matrice unité et la matrice Z
!>                  des vecteurs de Schur droits de (H,T) est retournée ;
!>          = 'V' : Z doit contenir une matrice orthogonale Z1 à l'entrée et
!>                  le produit Z1*Z est retourné.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices H, T, Q et Z.  N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>          ILO et IHI marquent les lignes et les colonnes de H qui sont
!>          sous forme de Hessenberg.  On suppose que A est déjà triangulaire
!>          supérieure dans les lignes et les colonnes 1:ILO-1 et IHI+1:N.
!>          Si N > 0, 1 <= ILO <= IHI <= N ; si N = 0, ILO=1 et IHI=0.
!>

H

!>          H est un tableau DOUBLE PRECISION, dimension (LDH, N)
!>          À l'entrée, la matrice de Hessenberg supérieure H de taille N-par-N.
!>          À la sortie, si JOB = 'S', H contient la matrice quasi-triangulaire
!>          supérieure S issue de la factorisation de Schur généralisée.
!>          Si JOB = 'E', les blocs diagonaux de H correspondent à ceux de S,
!>          mais le reste de H est indéfini.
!>

LDH

!>          LDH est INTEGER
!>          La dimension principale du tableau H.  LDH >= max( 1, N ).
!>

T

!>          T est un tableau DOUBLE PRECISION, dimension (LDT, N)
!>          À l'entrée, la matrice triangulaire supérieure T de taille N-par-N.
!>          À la sortie, si JOB = 'S', T contient la matrice triangulaire
!>          supérieure P issue de la factorisation de Schur généralisée ;
!>          les blocs diagonaux 2-par-2 de P correspondant aux blocs 2-par-2
!>          de S sont réduits à une forme diagonale positive, c'est-à-dire
!>          que si H(j+1,j) est non nul, alors T(j+1,j) = T(j,j+1) = 0,
!>          T(j,j) > 0, et T(j+1,j+1) > 0.
!>          Si JOB = 'E', les blocs diagonaux de T correspondent à ceux de P,
!>          mais le reste de T est indéfini.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T.  LDT >= max( 1, N ).
!>

ALPHAR

!>          ALPHAR est un tableau DOUBLE PRECISION, dimension (N)
!>          Les parties réelles de chaque scalaire alpha définissant une
!>          valeur propre du GNEP.
!>

ALPHAI

!>          ALPHAI est un tableau DOUBLE PRECISION, dimension (N)
!>          Les parties imaginaires de chaque scalaire alpha définissant une
!>          valeur propre du GNEP.
!>          Si ALPHAI(j) est nul, alors la j-ième valeur propre est réelle ;
!>          si positive, alors les j-ième et (j+1)-ème valeurs propres forment
!>          une paire conjuguée complexe, avec ALPHAI(j+1) = -ALPHAI(j).
!>

BETA

!>          BETA est un tableau DOUBLE PRECISION, dimension (N)
!>          Les scalaires beta qui définissent les valeurs propres du GNEP.
!>          Ensemble, les quantités alpha = (ALPHAR(j),ALPHAI(j)) et
!>          beta = BETA(j) représentent la j-ième valeur propre de la paire
!>          de matrices (A,B), sous l'une des formes lambda = alpha/beta ou
!>          mu = beta/alpha.  Puisque lambda ou mu peut déborder, ils ne
!>          devraient pas, en général, être calculés.
!>

Q

!>          Q est un tableau DOUBLE PRECISION, dimension (LDQ, N)
!>          À l'entrée, si COMPQ = 'V', la matrice orthogonale Q1 utilisée
!>          dans la réduction de (A,B) à la forme de Hessenberg généralisée.
!>          À la sortie, si COMPQ = 'I', la matrice orthogonale des vecteurs
!>          de Schur gauches de (H,T), et si COMPQ = 'V', la matrice
!>          orthogonale des vecteurs de Schur gauches de (A,B).
!>          Non référencée si COMPQ = 'N'.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.  LDQ >= 1.
!>          Si COMPQ='V' ou 'I', alors LDQ &gt;= N.
!>

Z

!>          Z est un tableau DOUBLE PRECISION, dimension (LDZ, N)
!>          À l'entrée, si COMPZ = 'V', la matrice orthogonale Z1 utilisée dans la
!>          réduction de (A,B) à la forme de Hessenberg généralisée.
!>          À la sortie, si COMPZ = 'I', la matrice orthogonale des
!>          vecteurs de Schur droits de (H,T), et si COMPZ = 'V', la
!>          matrice orthogonale des vecteurs de Schur droits de (A,B).
!>          Non référencée si COMPZ = 'N'.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1.
!>          Si COMPZ='V' ou 'I', alors LDZ >= N.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO >= 0, WORK(1) renvoie la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.  LWORK >= max(1,N).
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message
!>          d'erreur lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          = 1,...,N : l'itération QZ n'a pas convergé.  (H,T) n'est pas
!>                     en forme de Schur, mais ALPHAR(i), ALPHAI(i), et
!>                     BETA(i), i=INFO+1,...,N devraient être corrects.
!>          = N+1,...,2*N : le calcul du décalage a échoué.  (H,T) n'est pas
!>                     en forme de Schur, mais ALPHAR(i), ALPHAI(i), et
!>                     BETA(i), i=INFO-N+1,...,N devraient être corrects.
!>
