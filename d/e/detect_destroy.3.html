detect_destroy(3)

Paquet: libchardet

Licences: MPL

detect_destroy(3) Manuel des fonctions de bibliothèque detect_destroy(3)

NOM

detect_destroy - libérer la structure Detector

SYNTAXE

#include <chardet.h>

void detect_destroy (Detect ** handle);

DESCRIPTION

Libère les ressources allouées par l'API detect_init.

VALEUR DE RETOUR

void

EXEMPLE

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "L'initialisation du handle chardet a échoué\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	detect_reset (&d);
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "Allocation mémoire échouée\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect_handledata (&d, "안녕하세요", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "Lors du traitement, une erreur de manque de mémoire s'est produite\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"Le 3ème argument de chardet_handledata() doit être alloué avec l'API detect_obj_init"
					"\n");
			return CHARDET_NULL_OBJECT;
	}
	# vérifiez le support de l'obj->bom avec la constante CHARDET_BOM_CHECK
	printf ("encodage: %s, confiance: %f, bom existant: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
	detect_destroy (&d);
    return 0;
}
