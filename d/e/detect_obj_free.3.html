detect_obj_free(3) Manuel des fonctions de bibliothèque detect_obj_free(3)

NOM

chardet_obj_free - libérer la structure DetectObject

SYNTAXE

#include <chardet.h>
void chardet_obj_free (DetectObj **);

DESCRIPTION

Libère les ressources de la structure DetectObject allouées par l'API chardet_obj_init.

VALEUR DE RETOUR

void

EXEMPLE

#include <chardet.h>
int main (void) {
	DetectObj *obj;
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "Échec de l'allocation mémoire\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect ("¾È³çÇÏ¼¼¿ä", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "Lors du traitement de la requête, une erreur de manque de mémoire s'est produite\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"Le 2ème argument de chardet() doit être alloué avec l'API detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# vérifie le support de obj->bom avec la constante CHARDET_BOM_CHECK
	printf ("encodage : %s, confiance : %f, bom existant : %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
    return 0;
}
