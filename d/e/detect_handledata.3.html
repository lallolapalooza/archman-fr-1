detect_handledata(3) Manuel des fonctions de bibliothèque detect_handledata(3)

NOM

detect_handledata, detect_handledata_r - Détection de l'encodage de caractères et mesure de la précision de l'encodage

SYNOPIS

#include <chardet.h>

short chardet_handledata (Detect ** handle, const char * inbuf, DetectObj ** outbuf);

short chardet_handledata_r (Detect ** handle, const char * inbuf, size_t inlen, DetectObj ** outbuf);

DESCRIPTION

Stocke l'encodage de caractères et la précision de inbuf dans outbuf.

L'API detect_handledata est obsolète car elle n'est pas binaire sûre. Utilisez ou remplacez par l'API detect_handledata_r.

Arguments :

handle

Ressource de manipulation Detect allouée par l'API detect_init.
inbuf

Chaîne d'entrée pour la détection.
inlen

Longueur de la chaîne d'entrée pour la détection.
outbuf

Stocke les informations de inbuf. La structure de outbuf est la suivante : 
typedef struct DetectObject {
	char * encoding;
	float confidence;
} DetectObj;

La variable outbuf doit être initialisée par l'API detect_obj_init avant d'appeler cette API de détection.

VALEURS DE RETOUR

Retourne la condition suivante au cas par cas.

CHARDET_SUCCESS

Détection réussie.
CHARDET_NO_RESULT

Échec de la détection.
CHARDET_NULL_OBJECT

outbuf non initialisé avec chardet_obj_init.
CHARDET_OUT_OF_MEMORY

Erreur de type mémoire insuffisante dans une API interne.

EXEMPLE

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	int i, arrayNum;
	char *str[] = {
		"this is ascii",
		"이건 euc-kr 입니다."
	};
	arrayNum = sizeof (str) / sizeof (str[0]);
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "chardet handle initialize failed\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	for ( i=0; i<arrayNum; i++ ) {
		detect_reset (&d);
		if ( (obj = detect_obj_init ()) == NULL ) {
			fprintf (stderr, "Memory Allocation failed\n");
			return CHARDET_MEM_ALLOCATED_FAIL;
		}
		//switch (detect_handledata (&d, str[i], &obj))
		switch (detect_handledata (&d, str[i], strlen(str[i]), &obj))
		{
			case CHARDET_OUT_OF_MEMORY :
				fprintf (stderr, "On handle processing, occured out of memory\n");
				detect_obj_free (&obj);
				return CHARDET_OUT_OF_MEMORY;
			case CHARDET_NULL_OBJECT :
				fprintf (stderr,
						"2st argument of chardet() is must memory allocation "
						"with detect_obj_init API\n");
				return CHARDET_NULL_OBJECT;
		}
		# check support obj->bom with CHARDET_BOM_CHECK constant
		printf ("encoding: %s, confidence: %f, exists bom: %d\n", obj->encoding, obj->confidence, obj->bom);
		detect_obj_free (&obj);
	}
	detect_destroy (&d);
    return 0;
}
