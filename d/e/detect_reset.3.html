detect_reset(3)

Paquet: libchardet

Licences: MPL

detect_reset(3) Manuel des fonctions de bibliothèque detect_reset(3)

NOM

detect_reset - réinitialiser la structure Detect

SYNTAXE

#include <chardet.h>
void chardet_reset (Detect **handle);

DESCRIPTION

Réinitialise la structure Detect.

VALEUR DE RETOUR

void

EXEMPLE

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	int i, arrayNum;
	char *str[] = {
		"this is ascii",
		"ÀÌ°Ç euc-kr ÀÔ´Ï´Ù."
	};
	arrayNum = sizeof (str) / sizeof (str[0]);
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "Échec de l'initialisation du gestionnaire chardet\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	for ( i=0; i<arrayNum; i++ ) {
		detect_reset (&d);
		if ( (obj = detect_obj_init ()) == NULL ) {
			fprintf (stderr, "Allocation mémoire échouée\n");
			return CHARDET_MEM_ALLOCATED_FAIL;
		}
		switch (detect_handledata (&d, "¾È³çÇÏ¼¼¿ä", &obj)) {
			case CHARDET_OUT_OF_MEMORY :
				fprintf (stderr, "Lors du traitement du gestionnaire, une erreur de manque de mémoire s'est produite\n");
				detect_obj_free (&obj);
				return CHARDET_OUT_OF_MEMORY;
			case CHARDET_NULL_OBJECT :
				fprintf (stderr,
						"Le 2e argument de chardet() doit être alloué en mémoire "
						"avec l'API detect_obj_init\n");
				return CHARDET_NULL_OBJECT;
		}
		# vérifier le support obj->bom avec la constante CHARDET_BOM_CHECK
		printf ("encodage : %s, confiance : %f, bom existant : %d\n", obj->encoding, obj->confidence, obj->bom);
		detect_obj_free (&obj);
	}
	detect_destroy (&d);
    return 0;
}
