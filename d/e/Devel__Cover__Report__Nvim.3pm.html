Exemple de configuration pour le thème solarized dans init.lua :
-- Définir les couleurs avant de charger la couverture
vim.g.devel_cover_fg = "#859900"        -- vert solarized
vim.g.devel_cover_error_fg = "#dc322f"  -- rouge solarized
vim.g.devel_cover_valid_bg = "#073642"  -- base02 solarized
vim.g.devel_cover_old_bg = "#342a2a"    -- arrière-plan plus sombre teinté de rouge
vim.g.devel_cover_cterm = "NONE"
vim.g.devel_cover_gui = "NONE"
-- Caractères de signe personnalisés (optionnel)
-- Remarque : Le même caractère est utilisé pour couvert/non couvert
-- La différence est montrée par les couleurs fg/bg définies ci-dessus
vim.g.devel_cover_signs = {
  pod = "P ",
  subroutine = "R ",
  statement = "· ",  -- Point subtil pour les instructions (le plus courant)
  branch = "B ",
  condition = "C "
}
-- Ou pour un aspect minimaliste reposant uniquement sur les couleurs
vim.g.devel_cover_bg = "#002b36"        -- sombre pour les lignes couvertes
vim.g.devel_cover_error_bg = "#2d1616"  -- rouge sombre pour les lignes non couvertes
vim.g.devel_cover_signs = {
  pod = "  ",
  subroutine = "  ",
  statement = "│ ",  -- Barre verticale fine (discrète)
  branch = "  ",
  condition = "  "
}
-- Ou utiliser des caractères Unicode subtils
vim.g.devel_cover_signs = {
  pod = "¶ ",
  subroutine = "ƒ ",
  statement = "• ",  -- Petit puce (discrète)
  branch = "» ",
  condition = "? "
}
-- Pour une colonne de couverture dédiée (recommandé)
-- Cela garde les signes de couverture séparés des autres signes comme les diagnostics
vim.g.devel_cover_sign_group = "DevelCover"
