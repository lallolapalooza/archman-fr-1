subroutine dtpt01 (character uplo, character diag, integer n, double precision, dimension( * ) ap, double precision, dimension( * ) ainvp, double precision rcond, double precision, dimension( * ) work, double precision resid)

DTPT01

Objectif :

!>
!> DTPT01 calcule le résidu pour une matrice triangulaire A multipliée par son
!> inverse lorsque A est stockée au format compacté :
!>    RESID = norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS ),
!> où EPS est le epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U' : Triangulaire supérieure
!>          = 'L' : Triangulaire inférieure
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire ou non.
!>          = 'N' : Triangulaire non unitaire
!>          = 'U' : Triangulaire unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

AP

!>          AP est un tableau DOUBLE PRECISION, de dimension (N*(N+1)/2)
!>          La matrice triangulaire d'origine A, stockée compactée
!>          par colonnes dans un tableau linéaire. La j-ème colonne de A est
!>          stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) pour j<=i<=n.
!>

AINVP

!>          AINVP est un tableau DOUBLE PRECISION, de dimension (N*(N+1)/2)
!>          À l'entrée, l'inverse (triangulaire) de la matrice A, stocké
!>          compacté par colonnes dans un tableau linéaire comme dans AP.
!>          À la sortie, le contenu de AINVP est détruit.
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>          Le nombre de condition réciproque de A, calculé comme
!>          1/(norm(A) * norm(AINV)).
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, de dimension (N)
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS )
!>
