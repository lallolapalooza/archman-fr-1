subroutine dtpttr (character uplo, integer n, double precision, dimension( * ) ap, double precision, dimension( lda, * ) a, integer lda, integer info)

DTPTTR copie une matrice triangulaire du format standard compact (TP) au format standard complet (TR).

Objectif :

!>
!> DTPTTR copie une matrice triangulaire A du format standard compact (TP)
!> au format standard complet (TR).
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' :  A est triangulaire supérieure.
!>          = 'L' :  A est triangulaire inférieure.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

AP

!>          AP est un tableau DOUBLE PRECISION, de dimension ( N*(N+1)/2 ),
!>          À l'entrée, la matrice triangulaire supérieure ou inférieure A,
!>          compactée par colonne dans un tableau linéaire. La j-ième colonne
!>          de A est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) pour j<=i<=n.
!>

A

!>          A est un tableau DOUBLE PRECISION, de dimension ( LDA, N )
!>          À la sortie, la matrice triangulaire A.  Si UPLO = 'U', la partie
!>          supérieure triangulaire principale N par N de A contient la partie
!>          triangulaire supérieure de la matrice A, et la partie strictement
!>          inférieure triangulaire de A n'est pas référencée. Si UPLO = 'L',
!>          la partie inférieure triangulaire principale N par N de A contient
!>          la partie triangulaire inférieure de la matrice A, et la partie
!>          strictement supérieure triangulaire de A n'est pas référencée.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>
