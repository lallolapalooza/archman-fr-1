sous-programme dtrt02 (caractère uplo, caractère trans, caractère diag, entier n, entier nrhs, double précision, dimension( lda, * ) a, entier lda, double précision, dimension( ldx, * ) x, entier ldx, double précision, dimension( ldb, * ) b, entier ldb, double précision, dimension( * ) work, double précision resid)

DTRT02

Objectif :

!>
!> DTRT02 calcule le résidu pour la solution calculée d'un système
!> linéaire triangulaire op(A)*X = B, où A est une matrice triangulaire.
!> Le ratio de test est le maximum sur
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS ),
!> où op(A) = A ou A**T, b est la colonne de B, x est le vecteur solution,
!> et EPS est l'epsilon machine.
!> La norme utilisée est la 1-norme.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U' : Triangulaire supérieure
!>          = 'L' : Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie l'opération appliquée à A.
!>          = 'N' : A    * X = B  (Pas de transposée)
!>          = 'T' : A**T * X = B  (Transposée)
!>          = 'C' : A**H * X = B  (Conjuguée transposée = Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire unitaire ou non.
!>          = 'N' : Triangulaire non unitaire
!>          = 'U' : Triangulaire unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.  NRHS >= 0.
!>

A

!>          A est un tableau DOUBLE PRÉCISION, de dimension (LDA,N)
!>          La matrice triangulaire A.  Si UPLO = 'U', la partie supérieure
!>          triangulaire gauche n par n du tableau A contient la matrice
!>          triangulaire supérieure, et la partie strictement inférieure
!>          triangulaire de A n'est pas référencée.  Si UPLO = 'L', la partie
!>          inférieure triangulaire gauche n par n du tableau A contient la
!>          matrice triangulaire inférieure, et la partie strictement
!>          supérieure triangulaire de A n'est pas référencée.  Si DIAG = 'U',
!>          les éléments diagonaux de A ne sont pas non plus référencés et
!>          sont supposés être 1.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

X

!>          X est un tableau DOUBLE PRÉCISION, de dimension (LDX,NRHS)
!>          Les vecteurs solution calculés pour le système d'équations
!>          linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est un tableau DOUBLE PRÉCISION, de dimension (LDB,NRHS)
!>          Les vecteurs du second membre pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau DOUBLE PRÉCISION, de dimension (N)
!>

RESID

!>          RESID est DOUBLE PRÉCISION
!>          Le maximum sur le nombre de seconds membres de
!>          norm(op(A)*X - B) / ( norm(op(A)) * norm(X) * EPS ).
!>
