sous-routine dtpt02 (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision resid)

DTPT02

Objectif :

!>
!> DTPT02 calcule le résidu pour la solution calculée d'un système
!> d'équations linéaires triangulaires op(A)*X = B, lorsque la matrice
!> triangulaire A est stockée au format compacté. Le ratio de test est
!> le maximum sur
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS ),
!> où op(A) = A ou A**T, b est la colonne de B, x est le vecteur solution,
!> et EPS est l'epsilon machine.
!> La norme utilisée est la 1-norme.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U' : Triangulaire supérieure
!>          = 'L' : Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie l'opération appliquée à A.
!>          = 'N' : A    * X = B  (Pas de transposée)
!>          = 'T' : A**T * X = B  (Transposée)
!>          = 'C' : A**H * X = B  (Transposée conjuguée = Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire triangulaire ou non.
!>          = 'N' : Non unitaire triangulaire
!>          = 'U' : Unitaire triangulaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.  NRHS >= 0.
!>

AP

!>          AP est un tableau de DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          La matrice triangulaire supérieure ou inférieure A, compactée par
!>          colonnes dans un tableau linéaire. La j-ème colonne de A est
!>          stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) pour 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) pour j<=i<=n.
!>

X

!>          X est un tableau de DOUBLE PRECISION, dimension (LDX,NRHS)
!>          Les vecteurs solution calculée pour le système d'équations
!>          linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est un tableau de DOUBLE PRECISION, dimension (LDB,NRHS)
!>          Les vecteurs de membre de droite pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau de DOUBLE PRECISION, dimension (N)
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          Le maximum sur le nombre de membres de droite de
!>          norm(op(A)*X - B) / ( norm(op(A)) * norm(X) * EPS ).
!>
