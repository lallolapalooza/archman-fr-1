subroutine dtptrs (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DTPTRS

Objectif:

!>
!> DTPTRS résout un système triangulaire de la forme
!>
!>    A * X = B  ou  A**T * X = B,
!>
!> où A est une matrice triangulaire d'ordre N stockée en format compacté, et B est une matrice N-par-NRHS.
!>
!> Ce sous-programme vérifie que A est non singulière, mais les appelants doivent noter que seule la singularité exacte est détectée. Il est concevable qu'un ou plusieurs éléments diagonaux de A soient des nombres très petits sans que ce sous-programme ne signale d'erreur.
!>
!> Si une perte possible de précision numérique due à des matrices quasi-singulières est une préoccupation, l'appelant doit vérifier que A est non singulière dans une certaine tolérance avant d'appeler ce sous-programme.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  A est triangulaire supérieure ;
!>          = 'L':  A est triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B  (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée = Transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          = 'N':  A est triangulaire non unitaire ;
!>          = 'U':  A est triangulaire unitaire.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

AP

!>          AP est un tableau DOUBLE PRECISION, de dimension (N*(N+1)/2)
!>          La matrice triangulaire A, supérieure ou inférieure, compactée par colonnes dans
!>          un tableau linéaire.  La j-ième colonne de A est stockée dans le tableau
!>          AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>

B

!>          B est un tableau DOUBLE PRECISION, de dimension (LDB,NRHS)
!>          À l'entrée, la matrice des seconds membres B.
!>          À la sortie, si INFO = 0, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  si INFO = i, le i-ème élément diagonal de A est exactement zéro,
!>                indiquant que la matrice est singulière et que les
!>                solutions X n'ont pas été calculées.
!>
