subroutine dtgsen (integer ijob, logical wantq, logical wantz, logical, dimension( * ) select, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) alphar, double precision, dimension( * ) alphai, double precision, dimension( * ) beta, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( ldz, * ) z, integer ldz, integer m, double precision pl, double precision pr, double precision, dimension( * ) dif, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

DTGSEN

But :

!>
!> DTGSEN réordonne la décomposition de Schur généralisée réelle d'une paire
!> de matrices réelles (A, B) (en termes d'une transformation d'équivalence
!> orthonormée Q**T * (A, B) * Z), de sorte qu'un cluster sélectionné de valeurs
!> propres apparaisse dans les blocs diagonaux principaux de la matrice
!> quasi-triangulaire supérieure A et de la matrice triangulaire supérieure B.
!> Les colonnes principales de Q et Z forment des bases orthonormées des
!> sous-espaces propres (sous-espaces déflationnants) correspondants.
!> (A, B) doit être dans une forme canonique de Schur généralisée réelle
!> (telle que retournée par DGGES), c'est-à-dire que A est bloc-triangulaire
!> supérieure avec des blocs diagonaux de 1x1 et 2x2. B est triangulaire
!> supérieure.
!>
!> DTGSEN calcule également les valeurs propres généralisées
!>
!>             w(j) = (ALPHAR(j) + i*ALPHAI(j))/BETA(j)
!>
!> de la paire de matrices réordonnée (A, B).
!>
!> Optionnellement, DTGSEN calcule les estimations des réciproques des
!> nombres de condition pour les valeurs propres et les sous-espaces propres.
!> Ce sont Difu[(A11,B11), (A22,B22)] et Difl[(A11,B11), (A22,B22)],
!> c'est-à-dire les séparations (entre la paire de matrices (A11, B11) et
!> (A22,B22) qui correspondent au cluster sélectionné et aux valeurs propres
!> en dehors du cluster, respectivement, et les normes de Q et Z sur les
!> sous-espaces propres gauches et droits par rapport au cluster sélectionné
!> dans le bloc (1,1).
!>

Paramètres

IJOB 
!>          IJOB est INTEGER
!>          Spécifie si les nombres de condition sont requis pour le
!>          cluster de valeurs propres (PL et PR) ou les sous-espaces
!>          déflationnants (Difu et Difl):
!>           =0: Seulement réordonner par rapport à SELECT. Pas d'extras.
!>           =1: Réciproques des normes de Q et Z sur les sous-espaces
!>               propres gauches et droits par rapport au cluster
!>               sélectionné (PL et PR).
!>           =2: Bornes supérieures pour Difu et Difl. Estimation basée sur
!>               la norme F (DIF(1:2)).
!>           =3: Estimation de Difu et Difl. Estimation basée sur la norme
!>               1 (DIF(1:2)).
!>               Environ 5 fois plus coûteux que IJOB = 2.
!>           =4: Calculer PL, PR et DIF (c'est-à-dire 0, 1 et 2 ci-dessus) :
!>               Version économique pour tout obtenir.
!>           =5: Calculer PL, PR et DIF (c'est-à-dire 0, 1 et 3 ci-dessus)
!>

WANTQ

!>          WANTQ est LOGICAL
!>          .TRUE. : mettre à jour la matrice de transformation gauche Q;
!>          .FALSE.: ne pas mettre à jour Q.
!>

WANTZ

!>          WANTZ est LOGICAL
!>          .TRUE. : mettre à jour la matrice de transformation droite Z;
!>          .FALSE.: ne pas mettre à jour Z.
!>

SELECT

!>          SELECT est un tableau LOGICAL, dimension (N)
!>          SELECT spécifie les valeurs propres dans le cluster sélectionné.
!>          Pour sélectionner une valeur propre réelle w(j), SELECT(j) doit être
!>          mis à .TRUE.. Pour sélectionner une paire de valeurs propres
!>          conjuguées complexes w(j) et w(j+1), correspondant à un bloc diagonal
!>          de 2x2, soit SELECT(j) soit SELECT(j+1) ou les deux doivent être
!>          mis à .TRUE. ; une paire de valeurs propres conjuguées complexes
!>          doit être soit incluse dans le cluster, soit exclue.
!>

N

!>          N est INTEGER
!>          L'ordre des matrices A et B. N >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension(LDA,N)
!>          En entrée, la matrice quasi-triangulaire supérieure A, avec (A, B)
!>          dans la forme canonique de Schur généralisée réelle.
!>          En sortie, A est écrasée par la matrice réordonnée A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension(LDB,N)
!>          En entrée, la matrice triangulaire supérieure B, avec (A, B)
!>          dans la forme canonique de Schur généralisée réelle.
!>          En sortie, B est écrasée par la matrice réordonnée B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR est un tableau DOUBLE PRECISION, dimension (N)
!>

ALPHAI

!>          ALPHAI est un tableau DOUBLE PRECISION, dimension (N)
!>

BETA

!>          BETA est un tableau DOUBLE PRECISION, dimension (N)
!>
!>          En sortie, (ALPHAR(j) + ALPHAI(j)*i)/BETA(j), j=1,...,N, seront
!>          les valeurs propres généralisées. ALPHAR(j) + ALPHAI(j)*i et
!>          BETA(j), j=1,...,N sont les diagonales de la forme de Schur
!>          complexe (S,T) qui résulterait si les blocs diagonaux de 2x2
!>          de la forme de Schur généralisée réelle de (A,B) étaient
!>          ulterieurement réduits en forme triangulaire en utilisant des
!>          transformations unitaires complexes.
!>          Si ALPHAI(j) est zéro, alors la j-ème valeur propre est réelle ;
!>          si positive, alors la j-ème et (j+1)-ème valeurs propres sont une
!>          paire conjuguée complexe, avec ALPHAI(j+1) négatif.
!>

Q

!>          Q est un tableau DOUBLE PRECISION, dimension (LDQ,N)
!>          En entrée, si WANTQ = .TRUE., Q est une matrice N x N.
!>          En sortie, Q a été post-multipliée par la matrice de transformation
!>          orthogonale gauche qui réordonne (A, B) ; les M colonnes principales
!>          de Q forment des bases orthonormées pour la paire spécifiée de
!>          sous-espaces propres gauches (sous-espaces déflationnants).
!>          Si WANTQ = .FALSE., Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q. LDQ >= 1 ;
!>          et si WANTQ = .TRUE., LDQ >= N.
!>

Z

!>          Z est un tableau DOUBLE PRECISION, dimension (LDZ,N)
!>          En entrée, si WANTZ = .TRUE., Z est une matrice N x N.
!>          En sortie, Z a été post-multipliée par la matrice de transformation
!>          orthogonale gauche qui réordonne (A, B) ; les M colonnes principales
!>          de Z forment des bases orthonormées pour la paire spécifiée de
!>          sous-espaces propres gauches (sous-espaces déflationnants).
!>          Si WANTZ = .FALSE., Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1 ;
!>          Si WANTZ = .TRUE., LDZ >= N.
!>

M

!>          M est INTEGER
!>          La dimension de la paire spécifiée de sous-espaces propres
!>          gauches et droits (sous-espaces déflationnants). 0 <= M <= N.
!>
re>

PL

!>          PL est de type DOUBLE PRECISION
!>

PR

!>          PR est de type DOUBLE PRECISION
!>
!>          Si IJOB = 1, 4 ou 5, PL, PR sont des bornes inférieures sur
!>          la réciproque de la norme des sous-espaces propres gauche et droit
!>          par rapport au cluster sélectionné.
!>          0 < PL, PR <= 1.
!>          Si M = 0 ou M = N, PL = PR = 1.
!>          Si IJOB = 0, 2 ou 3, PL et PR ne sont pas référencés.
!>

DIF

!>          DIF est un tableau de type DOUBLE PRECISION, de dimension (2).
!>          Si IJOB >= 2, DIF(1:2) stockent les estimations de Difu et Difl.
!>          Si IJOB = 2 ou 4, DIF(1:2) sont des bornes supérieures basées sur la
!>          norme F de Difu et Difl. Si IJOB = 3 ou 5, DIF(1:2) sont des estimations
!>          basées sur la norme 1 de Difu et Difl.
!>          Si M = 0 ou N, DIF(1:2) = F-norm([A, B]).
!>          Si IJOB = 0 ou 1, DIF n'est pas référencé.
!>

WORK

!>          WORK est un tableau de type DOUBLE PRECISION, de dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) renvoie la taille optimale de LWORK.
!>

LWORK

!>          LWORK est de type INTEGER
!>          La dimension du tableau WORK. LWORK >=  4*N+16.
!>          Si IJOB = 1, 2 ou 4, LWORK >= MAX(4*N+16, 2*M*(N-M)).
!>          Si IJOB = 3 ou 5, LWORK >= MAX(4*N+16, 4*M*(N-M)).
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur dans la première entrée du tableau WORK, et aucun message d'erreur
!>          relatif à LWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau de type INTEGER, de dimension (MAX(1,LIWORK))
!>          En sortie, si INFO = 0, IWORK(1) renvoie la taille optimale de LIWORK.
!>

LIWORK

!>          LIWORK est de type INTEGER
!>          La dimension du tableau IWORK. LIWORK >= 1.
!>          Si IJOB = 1, 2 ou 4, LIWORK >=  N+6.
!>          Si IJOB = 3 ou 5, LIWORK >= MAX(2*M*(N-M), N+6).
!>
!>          Si LIWORK = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule uniquement la taille optimale du tableau IWORK,
!>          renvoie cette valeur dans la première entrée du tableau IWORK, et
!>          aucun message d'erreur relatif à LIWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est de type INTEGER
!>            =0: Exécution réussie.
!>            <0: Si INFO = -i, le i-ème argument avait une valeur illégale.
!>            =1: Le réordonnancement de (A, B) a échoué car la matrice
!>                transformée (A, B) serait trop éloignée de la forme de Schur généralisée ;
!>                le problème est très mal conditionné.
!>                (A, B) a pu être partiellement réordonné.
!>                Si demandé, 0 est retourné dans DIF(*), PL et PR.
!>

Détails supplémentaires :

!>
!>  DTGSEN collecte d'abord les valeurs propres sélectionnées en calculant
!>  des matrices orthogonales U et W qui les déplacent en haut à gauche de (A, B).
!>  Autrement dit, les valeurs propres sélectionnées sont les valeurs propres de
!>  (A11, B11) dans :
!>
!>              U**T*(A, B)*W = (A11 A12) (B11 B12) n1
!>                              ( 0  A22),( 0  B22) n2
!>                                n1  n2    n1  n2
!>
!>  où N = n1+n2 et U**T désigne la transposée de U. Les n1 premières colonnes
!>  de U et W s'étendent sur les paires de sous-espaces propres gauche et droit
!>  (sous-espaces déflateurs) de (A, B).
!>
!>  Si (A, B) a été obtenu à partir de la décomposition de Schur réelle généralisée
!>  d'une paire de matrices (C, D) = Q*(A, B)*Z**T, alors la forme de Schur réelle généralisée
!>  réordonnée de (C, D) est donnée par
!>
!>           (C, D) = (Q*U)*(U**T*(A, B)*W)*(Z*W)**T,
!>
!>  et les n1 premières colonnes de Q*U et Z*W s'étendent sur les sous-espaces
!>  déflateurs correspondants de (C, D) (Q et Z stockent Q*U et Z*W, respectivement).
!>
!>  Notez que si la valeur propre sélectionnée est suffisamment mal conditionnée,
!>  sa valeur peut différer significativement de sa valeur avant réordonnancement.
!>
!>  La réciproque des nombres de condition des sous-espaces propres gauche et droit
!>  s'étendant sur les n1 premières colonnes de U et W (ou Q*U et Z*W) peut
!>  être retournée dans DIF(1:2), correspondant respectivement à Difu et Difl.
!>
!>  Difu et Difl sont définis comme suit :
!>
!>       Difu[(A11, B11), (A22, B22)] = sigma-min( Zu )
!>  et
!>       Difl[(A11, B11), (A22, B22)] = Difu[(A22, B22), (A11, B11)],
!>
!>  où sigma-min(Zu) est la plus petite valeur singulière de la matrice
!>  (2*n1*n2)-par-(2*n1*n2)
!>
!>       Zu = [ kron(In2, A11)  -kron(A22**T, In1) ]
!>            [ kron(In2, B11)  -kron(B22**T, In1) ] .
!>
!>  Ici, Inx est la matrice identité de taille nx et A22**T est la
!>  transposée de A22. kron(X, Y) est le produit de Kronecker entre
!>  les matrices X et Y.
!>
!>  Lorsque DIF(2) est petit, de petites modifications de (A, B) peuvent entraîner de grands changements
!>  dans le sous-espace déflateur. Une borne approximative (asymptotique) sur l'erreur
!>  angulaire maximale dans les sous-espaces déflateurs calculés est
!>
!>       EPS * norm((A, B)) / DIF(2),
!>
!>  où EPS est la précision machine.
!>
!>  La réciproque de la norme des projecteurs sur les sous-espaces propres gauche et droit
!>  associés à (A11, B11) peut être retournée dans PL et PR.
!>  Ils sont calculés comme suit. D'abord, nous calculons L et R tels que
!>  P*(A, B)*Q soit diagonal par blocs, où
!>
!>       P = ( I -L ) n1           Q = ( I R ) n1
!>           ( 0  I ) n2    et        ( 0 I ) n2
!>             n1 n2                    n1 n2
!>
!>  et (L, R) est la solution de l'équation de Sylvester généralisée
!>
!>       A11*R - L*A22 = -A12
!>       B11*R - L*B22 = -B12
!>
!>  Alors PL = (F-norm(L)**2+1)**(-1/2) et PR = (F-norm(R)**2+1)**(-1/2).
!>  Une borne approximative (asymptotique) sur l'erreur absolue moyenne des
!>  valeurs propres sélectionnées est
!>
!>       EPS * norm((A, B)) / PL.
!>
!>  Il existe également des bornes d'erreur globales valides pour des perturbations
!>  jusqu'à une certaine restriction : une borne inférieure (x) sur la plus petite
!>  F-norm(E,F) pour laquelle une valeur propre de (A11, B11) peut se déplacer et
!>  fusionner avec une valeur propre de (A22, B22) sous perturbation (E,F),
!>  (c'est-à-dire (A + E, B + F)), est
!>
!>   x = min(Difu,Difl)/((1/(PL*PL)+1/(PR*PR))**(1/2)+2*max(1/PL,1/PR)).
!>
!>  Une borne approximative de x peut être calculée à partir de DIF(1:2), PL et PR.
!>
!>  Si y = ( F-norm(E,F) / x) <= 1, les angles entre les sous-espaces déflateurs
!>  perturbés (L', R') et non perturbés (L, R) gauche et droit
!>  associés au cluster sélectionné dans les blocs (1,1) peuvent être bornés par
!>
!>   max-angle(L, L') <= arctan( y * PL / (1 - y * (1 - PL * PL)**(1/2))
!>   max-angle(R, R') <= arctan( y * PR / (1 - y * (1 - PR * PR)**(1/2))
!>
!>  Voir la section 4.11 du manuel LAPACK ou les références suivantes
!>  pour plus d'informations.
!>
!>  Notez que si la méthode par défaut pour calculer l'estimation DIF basée sur la norme de Frobenius
!>  n'est pas souhaitée (voir DLATDF), alors le paramètre
!>  IDIFJB (voir ci-dessous) doit être changé de 3 à 4 (la routine DLATDF
!>  (IJOB = 2 sera utilisée)). Voir DTGSYL pour plus de détails.
!>
