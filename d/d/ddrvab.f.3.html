sous-routine ddrvab (logique, dimension( * ) dotype, entier nm, entier, dimension( * ) mval, entier nns, entier, dimension( * ) nsval, double précision thresh, entier nmax, double précision, dimension( * ) a, double précision, dimension( * ) afac, double précision, dimension( * ) b, double précision, dimension( * ) x, double précision, dimension( * ) work, double précision, dimension( * ) rwork, réel, dimension(*) swork, entier, dimension( * ) iwork, entier nout)

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, de dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI. ; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NM

!>          NM est un ENTIER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau ENTIER, de dimension (NM)
!>          Les valeurs de la dimension de ligne M de la matrice.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau ENTIER, de dimension (NNS)
!>          Les valeurs du nombre de membres du second membre NRHS.
!>

THRESH

!>          THRESH est un DOUBLE PRÉCISION
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utilisez THRESH = 0.
!>

NMAX

!>          NMAX est un ENTIER
!>          La valeur maximale autorisée pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!>

A

!>          A est un tableau DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

B

!>          B est un tableau DOUBLE PRÉCISION, de dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée de NSVAL.
!>

X

!>          X est un tableau DOUBLE PRÉCISION, de dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau DOUBLE PRÉCISION, de dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau DOUBLE PRÉCISION, de dimension
!>                      (max(2*NMAX,2*NSMAX+NWORK))
!>

SWORK

!>          SWORK est un tableau RÉEL, de dimension
!>                      (NMAX*(NSMAX+NMAX))
!>

IWORK

!>          IWORK est un tableau ENTIER, de dimension
!>                      NMAX
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité pour la sortie.
!>
