sous-routine ddrvls (logical, dimension( * ) dotype, integer nm, integer, dimension( * ) mval, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, integer nnb, integer, dimension( * ) nbval, integer, dimension( * ) nxval, double precision thresh, logical tsterr, double precision, dimension( * ) a, double precision, dimension( * ) copya, double precision, dimension( * ) b, double precision, dimension( * ) copyb, double precision, dimension( * ) c, double precision, dimension( * ) s, double precision, dimension( * ) copys, integer nout)

DDRVLS

But :

!>
!> DDRVLS teste les routines pilotes des moindres carrés DGELS, DGELST,
!> DGETSLS, DGELSS, DGELSY, et DGELSD.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGICAL, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE. ; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>          La matrice de type j est générée comme suit :
!>          j=1 : A = U*D*V où U et V sont des matrices orthogonales aléatoires
!>               et D a des entrées aléatoires (> 0.1) tirées d'une distribution
!>               uniforme (0,1). A est de rang complet.
!>          j=2 : Idem que 1, mais A est mise à l'échelle vers le haut.
!>          j=3 : Idem que 1, mais A est mise à l'échelle vers le bas.
!>          j=4 : A = U*D*V où U et V sont des matrices orthogonales aléatoires
!>               et D a 3*min(M,N)/4 entrées aléatoires (> 0.1) tirées
!>               d'une distribution uniforme (0,1) et les entrées restantes
!>               sont mises à 0. A est de rang déficient.
!>          j=5 : Idem que 4, mais A est mise à l'échelle vers le haut.
!>          j=6 : Idem que 5, mais A est mise à l'échelle vers le bas.
!>

NM

!>          NM est INTEGER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau INTEGER, dimension (NM)
!>          Les valeurs de la dimension de lignes de la matrice M.
!>

NN

!>          NN est INTEGER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau INTEGER, dimension (NN)
!>          Les valeurs de la dimension de colonnes de la matrice N.
!>

NNS

!>          NNS est INTEGER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau INTEGER, dimension (NNS)
!>          Les valeurs du nombre de seconds membres NRHS.
!>

NNB

!>          NNB est INTEGER
!>          Le nombre de valeurs de NB et NX contenues dans les
!>          vecteurs NBVAL et NXVAL. Les paramètres de blocage sont utilisés
!>          par paires (NB,NX).
!>

NBVAL

!>          NBVAL est un tableau INTEGER, dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NXVAL

!>          NXVAL est un tableau INTEGER, dimension (NNB)
!>          Les valeurs du point de bascule NX.
!>

THRESH

!>          THRESH est DOUBLE PRECISION
!>          La valeur de seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGICAL
!>          Drapeau qui indique si les sorties d'erreur doivent être testées.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (MMAX*NMAX)
!>          où MMAX est la valeur maximale de M dans MVAL et NMAX est la
!>          valeur maximale de N dans NVAL.
!>

COPYA

!>          COPYA est un tableau DOUBLE PRECISION, dimension (MMAX*NMAX)
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (MMAX*NSMAX)
!>          où MMAX est la valeur maximale de M dans MVAL et NSMAX est la
!>          valeur maximale de NRHS dans NSVAL.
!>

COPYB

!>          COPYB est un tableau DOUBLE PRECISION, dimension (MMAX*NSMAX)
!>

C

!>          C est un tableau DOUBLE PRECISION, dimension (MMAX*NSMAX)
!>

S

!>          S est un tableau DOUBLE PRECISION, dimension
!>                      (min(MMAX,NMAX))
!>

COPYS

!>          COPYS est un tableau DOUBLE PRECISION, dimension
!>                      (min(MMAX,NMAX))
!>

NOUT

!>          NOUT est INTEGER
!>          Le numéro d'unité pour la sortie.
!>
