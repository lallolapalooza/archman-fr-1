subroutine dckcsd (integer nm, integer, dimension( * ) mval, integer, dimension( * ) pval, integer, dimension( * ) qval, integer nmats, integer, dimension( 4 ) iseed, double precision thresh, integer mmax, double precision, dimension( * ) x, double precision, dimension( * ) xf, double precision, dimension( * ) u1, double precision, dimension( * ) u2, double precision, dimension( * ) v1t, double precision, dimension( * ) v2t, double precision, dimension( * ) theta, integer, dimension( * ) iwork, double precision, dimension( * ) work, double precision, dimension( * ) rwork, integer nin, integer nout, integer info)

DCKCSD

But :

!>
!> DCKCSD teste DORCSD :
!>        la CSD pour une matrice orthogonale M par M X partitionnée comme
!>        [ X11 X12; X21 X22 ]. X11 est P par Q.
!>

Paramètres

NM 
!>          NM est INTEGER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau INTEGER, dimension (NM)
!>          Les valeurs de la dimension de ligne M de la matrice.
!>

PVAL

!>          PVAL est un tableau INTEGER, dimension (NM)
!>          Les valeurs de la dimension de ligne P de la matrice.
!>

QVAL

!>          QVAL est un tableau INTEGER, dimension (NM)
!>          Les valeurs de la dimension de colonne Q de la matrice.
!>

NMATS

!>          NMATS est INTEGER
!>          Le nombre de types de matrices à tester pour chaque combinaison
!>          de dimensions de matrice. Si NMATS >= NTYPES (le nombre
!>          maximal de types de matrices), alors tous les types différents sont
!>          générés pour le test. Si NMATS < NTYPES, une autre ligne d'entrée
!>          est lue pour obtenir les numéros des types de matrices à utiliser.
!>

ISEED

!>          ISEED est un tableau INTEGER, dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires. Les éléments
!>          du tableau doivent être compris entre 0 et 4095, sinon ils seront
!>          réduits modulo 4096, et ISEED(4) doit être impair.
!>          À la sortie, la graine suivante dans la séquence de nombres aléatoires après
!>          que toutes les matrices de test ont été générées.
!>

THRESH

!>          THRESH est DOUBLE PRECISION
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

MMAX

!>          MMAX est INTEGER
!>          La valeur maximale autorisée pour M, utilisée pour dimensionner les
!>          tableaux de travail.
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension (MMAX*MMAX)
!>

XF

!>          XF est un tableau DOUBLE PRECISION, dimension (MMAX*MMAX)
!>

U1

!>          U1 est un tableau DOUBLE PRECISION, dimension (MMAX*MMAX)
!>

U2

!>          U2 est un tableau DOUBLE PRECISION, dimension (MMAX*MMAX)
!>

V1T

!>          V1T est un tableau DOUBLE PRECISION, dimension (MMAX*MMAX)
!>

V2T

!>          V2T est un tableau DOUBLE PRECISION, dimension (MMAX*MMAX)
!>

THETA

!>          THETA est un tableau DOUBLE PRECISION, dimension (MMAX)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (MMAX)
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION
!>

NIN

!>          NIN est INTEGER
!>          Le numéro d'unité pour l'entrée.
!>

NOUT

!>          NOUT est INTEGER
!>          Le numéro d'unité pour la sortie.
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          > 0 :  Si DLAROR retourne un code d'erreur, la valeur absolue
!>                 de celui-ci est retournée.
!>
