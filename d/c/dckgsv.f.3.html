!>
!> DCKGSV teste DGGSVD :
!>        la GSVD pour la matrice A de taille M par N et la matrice B de taille P par N.
!
>

Paramètre

s
NM 
!>          NM est un INTEGER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!
>

MVA

L 
!>          MVAL est un tableau INTEGER, dimension (NM)
!>          Les valeurs de la dimension de lignes M de la matrice.
!
>

PVA

L 
!>          PVAL est un tableau INTEGER, dimension (NP)
!>          Les valeurs de la dimension de lignes P de la matrice.
!
>

NVA

L 
!>          NVAL est un tableau INTEGER, dimension (NN)
!>          Les valeurs de la dimension de colonnes N de la matrice.
!
>

NMAT

S 
!>          NMATS est un INTEGER
!>          Le nombre de types de matrices à tester pour chaque combinaison
!>          de dimensions de matrices. Si NMATS >= NTYPES (le nombre
!>          maximal de types de matrices), alors tous les types différents sont
!>          générés pour le test. Si NMATS < NTYPES, une autre ligne d'entrée
!>          est lue pour obtenir les numéros des types de matrices à utiliser.
!
>

ISEE

D 
!>          ISEED est un tableau INTEGER, dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires. Les éléments
!>          du tableau doivent être compris entre 0 et 4095, sinon ils seront
!>          réduits modulo 4096, et ISEED(4) doit être impair.
!>          À la sortie, la graine suivante dans la séquence de nombres aléatoires après
!>          que toutes les matrices de test ont été générées.
!
>

THRES

H 
!>          THRESH est un DOUBLE PRECISION
!>          La valeur de seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!
>

NMA

X 
!>          NMAX est un INTEGER
!>          La valeur maximale autorisée pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!
>

A 
!>          A est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

A

F 
!>          AF est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

B 
!>          B est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

B

F 
!>          BF est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

U 
!>          U est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

V 
!>          V est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

Q 
!>          Q est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

ALPH

A 
!>          ALPHA est un tableau DOUBLE PRECISION, dimension (NMAX)
!
>

BET

A 
!>          BETA est un tableau DOUBLE PRECISION, dimension (NMAX)
!
>

R 
!>          R est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (NMAX)
!
>

WOR

K 
!>          WORK est un tableau DOUBLE PRECISION, dimension (NMAX*NMAX)

K
