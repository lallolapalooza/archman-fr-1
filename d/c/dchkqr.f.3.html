sous-routine dchkqr (logique, dimension( * ) dotype, entier nm, entier, dimension( * ) mval, entier nn, entier, dimension( * ) nval, entier nnb, entier, dimension( * ) nbval, entier, dimension( * ) nxval, entier nrhs, double précision thresh, logique tsterr, entier nmax, double précision, dimension( * ) a, double précision, dimension( * ) af, double précision, dimension( * ) aq, double précision, dimension( * ) ar, double précision, dimension( * ) ac, double précision, dimension( * ) b, double précision, dimension( * ) x, double précision, dimension( * ) xact, double précision, dimension( * ) tau, double précision, dimension( * ) work, double précision, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

DCHKQR

Objectif:

!>
!> DCHKQR teste DGEQRF, DORGQR et DORMQR.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, de dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI.; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NM

!>          NM est un ENTIER
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau d'ENTIERS, de dimension (NM)
!>          Les valeurs de la dimension ligne M de la matrice.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'ENTIERS, de dimension (NN)
!>          Les valeurs de la dimension colonne N de la matrice.
!>

NNB

!>          NNB est un ENTIER
!>          Le nombre de valeurs de NB et NX contenues dans les
!>          vecteurs NBVAL et NXVAL. Les paramètres de bloc sont utilisés
!>          par paires (NB,NX).
!>

NBVAL

!>          NBVAL est un tableau d'ENTIERS, de dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NXVAL

!>          NXVAL est un tableau d'ENTIERS, de dimension (NNB)
!>          Les valeurs du point de croisement NX.
!>

NRHS

!>          NRHS est un ENTIER
!>          Le nombre de vecteurs de second membre à générer pour
!>          chaque système linéaire.
!>

THRESH

!>          THRESH est une DOUBLE PRÉCISION
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utiliser THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGIQUE
!>          Indicateur qui détermine si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un ENTIER
!>          La valeur maximale autorisée pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!>

A

!>          A est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

AF

!>          AF est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

AQ

!>          AQ est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

AR

!>          AR est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

AC

!>          AC est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

B

!>          B est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NRHS)
!>

X

!>          X est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NRHS)
!>

XACT

!>          XACT est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NRHS)
!>

TAU

!>          TAU est un tableau de DOUBLE PRÉCISION, de dimension (NMAX)
!>

WORK

!>          WORK est un tableau de DOUBLE PRÉCISION, de dimension (NMAX*NMAX)
!>

RWORK

!>          RWORK est un tableau de DOUBLE PRÉCISION, de dimension (NMAX)
!>

IWORK

!>          IWORK est un tableau d'ENTIERS, de dimension (NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité de sortie.
!>
