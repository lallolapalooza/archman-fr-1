sous-programme dcsdts (entier m, entier p, entier q, double précision, dimension( ldx, * ) x, double précision, dimension( ldx, * ) xf, entier ldx, double précision, dimension( ldu1, * ) u1, entier ldu1, double précision, dimension( ldu2, * ) u2, entier ldu2, double précision, dimension( ldv1t, * ) v1t, entier ldv1t, double précision, dimension( ldv2t, * ) v2t, entier ldv2t, double précision, dimension( * ) theta, entier, dimension( * ) iwork, double précision, dimension( lwork ) work, entier lwork, double précision, dimension( * ) rwork, double précision, dimension( 15 ) result)

DCSDTS

Objectif:

!>
!> DCSDTS teste DORCSD, qui, étant donnée une matrice orthogonale partitionnée M-par-M
!> X,
!>              Q  M-Q
!>       X = [ X11 X12 ] P   ,
!>           [ X21 X22 ] M-P
!>
!> calcule la CSD
!>
!>       [ U1    ]**T * [ X11 X12 ] * [ V1    ]
!>       [    U2 ]      [ X21 X22 ]   [    V2 ]
!>
!>                             [  I  0  0 |  0  0  0 ]
!>                             [  0  C  0 |  0 -S  0 ]
!>                             [  0  0  0 |  0  0 -I ]
!>                           = [---------------------] = [ D11 D12 ] ,
!>                             [  0  0  0 |  I  0  0 ]   [ D21 D22 ]
!>                             [  0  S  0 |  0  C  0 ]
!>                             [  0  0  I |  0  0  0 ]
!>
!> et aussi DORCSD2BY1, qui, étant donnée
!>          Q
!>       [ X11 ] P   ,
!>       [ X21 ] M-P
!>
!> calcule la CSD 2-par-1
!>
!>                                     [  I  0  0 ]
!>                                     [  0  C  0 ]
!>                                     [  0  0  0 ]
!>       [ U1    ]**T * [ X11 ] * V1 = [----------] = [ D11 ] ,
!>       [    U2 ]      [ X21 ]        [  0  0  0 ]   [ D21 ]
!>                                     [  0  S  0 ]
!>                                     [  0  0  I ]
!>

Paramètres

M 
!>          M est ENTIER
!>          Le nombre de lignes de la matrice X.  M >= 0.
!>

P

!>          P est ENTIER
!>          Le nombre de lignes de la matrice X11.  P >= 0.
!>

Q

!>          Q est ENTIER
!>          Le nombre de colonnes de la matrice X11.  Q >= 0.
!>

X

!>          X est un tableau DOUBLE PRÉCISION, dimension (LDX,M)
!>          La matrice X de taille M-par-M.
!>

XF

!>          XF est un tableau DOUBLE PRÉCISION, dimension (LDX,M)
!>          Détails de la CSD de X, comme retournée par DORCSD;
!>          voir DORCSD pour plus de détails.
!>

LDX

!>          LDX est ENTIER
!>          La dimension principale des tableaux X et XF.
!>          LDX >= max( 1,M ).
!>

U1

!>          U1 est un tableau DOUBLE PRÉCISION, dimension(LDU1,P)
!>          La matrice orthogonale U1 de taille P-par-P.
!>

LDU1

!>          LDU1 est ENTIER
!>          La dimension principale du tableau U1. LDU >= max(1,P).
!>

U2

!>          U2 est un tableau DOUBLE PRÉCISION, dimension(LDU2,M-P)
!>          La matrice orthogonale U2 de taille (M-P)-par-(M-P).
!>

LDU2

!>          LDU2 est ENTIER
!>          La dimension principale du tableau U2. LDU >= max(1,M-P).
!>

V1T

!>          V1T est un tableau DOUBLE PRÉCISION, dimension(LDV1T,Q)
!>          La matrice orthogonale V1T de taille Q-par-Q.
!>

LDV1T

!>          LDV1T est ENTIER
!>          La dimension principale du tableau V1T. LDV1T >=
!>          max(1,Q).
!>

V2T

!>          V2T est un tableau DOUBLE PRÉCISION, dimension(LDV2T,M-Q)
!>          La matrice orthogonale V2T de taille (M-Q)-par-(M-Q).
!>

LDV2T

!>          LDV2T est ENTIER
!>          La dimension principale du tableau V2T. LDV2T >=
!>          max(1,M-Q).
!>

THETA

!>          THETA est un tableau DOUBLE PRÉCISION, dimension MIN(P,M-P,Q,M-Q)
!>          Les valeurs CS de X; les matrices diagonales essentielles C et
!>          S sont construites à partir de THETA; voir la sous-routine DORCSD pour
!>          les détails.
!>

IWORK

!>          IWORK est un tableau ENTIER, dimension (M)
!>

WORK

!>          WORK est un tableau DOUBLE PRÉCISION, dimension (LWORK)
!>

LWORK

!>          LWORK est ENTIER
!>          La dimension du tableau WORK
!>

RWORK

!>          RWORK est un tableau DOUBLE PRÉCISION
!>

RESULT

!>          RESULT est un tableau DOUBLE PRÉCISION, dimension (15)
!>          Les rapports de test :
!>          Premièrement, la CSD 2-par-2 :
!>          RESULT(1) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(2) = norm( U1'*X12*V2 - D12 ) / ( MAX(1,P,M-Q)*EPS2 )
!>          RESULT(3) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(4) = norm( U2'*X22*V2 - D22 ) / ( MAX(1,M-P,M-Q)*EPS2 )
!>          RESULT(5) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(6) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(7) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(8) = norm( I - V2T'*V2T ) / ( MAX(1,M-Q)*ULP )
!>          RESULT(9) = 0        si THETA est dans l'ordre croissant et
!>                               tous les angles sont dans [0,pi/2] 
!>                    = ULPINV   sinon.
!>          Ensuite, la CSD 2-par-1 :
!>          RESULT(10) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(11) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(12) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(13) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(14) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(15) = 0        si THETA est dans l'ordre croissant et
!>                               tous les angles sont dans [0,pi/2] 
!>                     = ULPINV   sinon.
!>          ( EPS2 = MAX( norm( I - X'*X ) / M, ULP ). )
!>
