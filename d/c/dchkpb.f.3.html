subroutine dchkpb (dotype, nn, nval, nnb, nbval, nns, nsval, thresh, tsterr, nmax, a, afac, ainv, b, x, xact, work, rwork, iwork, nout)

Documentation de fonction/sous-programme

sous-programme dchkpb (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nnb, entier, dimension( * ) nbval, entier nns, entier, dimension( * ) nsval, double précision thresh, logique tsterr, entier nmax, double précision, dimension( * ) a, double précision, dimension( * ) afac, double précision, dimension( * ) ainv, double précision, dimension( * ) b, double précision, dimension( * ) x, double précision, dimension( * ) xact, double précision, dimension( * ) work, double précision, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

DCHKPB

Objectif:

!>
!> DCHKPB teste DPBTRF, -TRS, -RFS et -CON.
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau logique, de dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .VRAI. ; si DOTYPE(j) = .FAUX., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un entier
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'entiers, de dimension (NN)
!>          Les valeurs de la dimension de matrice N.
!>

NNB

!>          NNB est un entier
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau d'entiers, de dimension (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NNS

!>          NNS est un entier
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'entiers, de dimension (NNS)
!>          Les valeurs du nombre de seconds membres NRHS.
!>

THRESH

!>          THRESH est une double précision
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour avoir
!>          chaque ratio de test imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est une logique
!>          Indicateur qui détermine si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un entier
!>          La valeur maximale autorisée pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau de double précision, de dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau de double précision, de dimension (NMAX*NMAX)
!>

AINV

!>          AINV est un tableau de double précision, de dimension (NMAX*NMAX)
!>

B

!>          B est un tableau de double précision, de dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau de double précision, de dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau de double précision, de dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau de double précision, de dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau de double précision, de dimension
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK est un tableau d'entiers, de dimension (NMAX)
!>

NOUT

!>          NOUT est un entier
!>          Le numéro d'unité de sortie.
!>
