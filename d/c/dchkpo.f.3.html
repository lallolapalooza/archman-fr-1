!>
!> DCHKPO teste DPOTRF, -TRI, -TRS, -RFS, et -CON

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGICAL, dimensionné (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE. ; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un INTEGER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau INTEGER, dimensionné (NN)
!>          Les valeurs de la dimension de la matrice N.
!>

NNB

!>          NNB est un INTEGER
!>          Le nombre de valeurs de NB contenues dans le vecteur NBVAL.
!>

NBVAL

!>          NBVAL est un tableau INTEGER, dimensionné (NNB)
!>          Les valeurs de la taille de bloc NB.
!>

NNS

!>          NNS est un INTEGER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau INTEGER, dimensionné (NNS)
!>          Les valeurs du nombre de membres de droite NRHS.
!>

THRESH

!>          THRESH est un DOUBLE PRECISION
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGICAL
!>          Indicateur qui détermine si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un INTEGER
!>          La valeur maximale autorisée pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimensionné (NMAX*NMAX)
!>

AFAC

!>          AFAC est un tableau DOUBLE PRECISION, dimensionné (NMAX*NMAX)
!>

AINV

!>          AINV est un tableau DOUBLE PRECISION, dimensionné (NMAX*NMAX)
!>

B

!>          B est un tableau DOUBLE PRECISION, dimensionné (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée de NSVAL.
!>

X

!>          X est un tableau DOUBLE PRECISION, dimensionné (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau DOUBLE PRECISION, dimensionné (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimensionné
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimensionné
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK est un tableau INTEGER, dimensionné (NMAX)
!>

NOUT

!>          NOUT est un INTEGER
!>          Le numéro d'unité pour la sortie.
!>
