Documentation Fonction/Sous routine

subroutine dgbt02 (character trans, integer m, integer n, integer kl, integer ku, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

DGBT02

Purpose:

!>
!> DGBT02 calcule le résidu pour une solution d'un système d'équations
!> à bande op(A)*X = B :
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> où op(A) = A ou A**T, en fonction de TRANS, et EPS est le
!> epsilon machine.
!> La norme utilisée est la 1-norme.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N' :  A    * X = B  (Pas de transposée)
!>          = 'T' :  A**T * X = B  (Transposée)
!>          = 'C' :  A**H * X = B  (Conjuguée transposée = Transposée)
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

KL

!>          KL est INTEGER
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!>

KU

!>          KU est INTEGER
!>          Le nombre de super-diagonales dans la bande de A.  KU >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes de B.  NRHS >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (LDA,N)
!>          La matrice originale A en stockage bande, stockée dans les
!>          lignes 1 à KL+KU+1.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,KL+KU+1).
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>          Les vecteurs solutions calculés pour le système d'équations
!>          linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  Si TRANS = 'N',
!>          LDX >= max(1,N) ; si TRANS = 'T' ou 'C', LDX >= max(1,M).
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          En entrée, les vecteurs du second membre pour le système
!>          d'équations linéaires.
!>          En sortie, B est écrasé par la différence B - A*X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  Si TRANS = 'N',
!>          LDB >= max(1,M) ; si TRANS = 'T' ou 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LRWORK)),
!>          où LRWORK >= M lorsque TRANS = 'T' ou 'C' ; sinon, RWORK
!>          n'est pas référencé.
!>

RESID

!>          RESID est DOUBLE PRECISION
!>          Le maximum sur le nombre de seconds membres de
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
