sous-routine dgbbrd (caractère vect, entier m, entier n, entier ncc, entier kl, entier ku, double précision, dimension( ldab, * ) ab, entier ldab, double précision, dimension( * ) d, double précision, dimension( * ) e, double précision, dimension( ldq, * ) q, entier ldq, double précision, dimension( ldpt, * ) pt, entier ldpt, double précision, dimension( ldc, * ) c, entier ldc, double précision, dimension( * ) work, entier info)

DGBBRD

But :

!>
!> DGBBRD réduit une matrice bande générale réelle m-par-n A à sa forme
!> bidiagonale supérieure B par une transformation orthogonale : Q**T * A * P = B.
!>
!> La routine calcule B, et éventuellement forme Q ou P**T, ou calcule
!> Q**T*C pour une matrice C donnée.
!>

Paramètres :

VECT 
!>          VECT est CHARACTER*1
!>          Spécifie si les matrices Q et P**T doivent être formées ou non.
!>          = 'N' : ne pas former Q ou P**T ;
!>          = 'Q' : former Q seulement ;
!>          = 'P' : former P**T seulement ;
!>          = 'B' : former les deux.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

NCC

!>          NCC est INTEGER
!>          Le nombre de colonnes de la matrice C.  NCC >= 0.
!>

KL

!>          KL est INTEGER
!>          Le nombre de sous-diagonales de la matrice A. KL >= 0.
!>

KU

!>          KU est INTEGER
!>          Le nombre de super-diagonales de la matrice A. KU >= 0.
!>

AB

!>          AB est un tableau de DOUBLE PRECISION, dimension (LDAB,N)
!>          À l'entrée, la matrice bande m-par-n A, stockée dans les lignes 1 à
!>          KL+KU+1. La j-ème colonne de A est stockée dans la j-ème colonne du
!>          tableau AB comme suit :
!>          AB(ku+1+i-j,j) = A(i,j) pour max(1,j-ku)<=i<=min(m,j+kl).
!>          À la sortie, A est écrasée par les valeurs générées pendant la
!>          réduction.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau A. LDAB >= KL+KU+1.
!>

D

!>          D est un tableau de DOUBLE PRECISION, dimension (min(M,N))
!>          Les éléments diagonaux de la matrice bidiagonale B.
!>

E

!>          E est un tableau de DOUBLE PRECISION, dimension (min(M,N)-1)
!>          Les éléments superdiagonaux de la matrice bidiagonale B.
!>

Q

!>          Q est un tableau de DOUBLE PRECISION, dimension (LDQ,M)
!>          Si VECT = 'Q' ou 'B', la matrice orthogonale m-par-m Q.
!>          Si VECT = 'N' ou 'P', le tableau Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= max(1,M) si VECT = 'Q' ou 'B' ; LDQ >= 1 sinon.
!>

PT

!>          PT est un tableau de DOUBLE PRECISION, dimension (LDPT,N)
!>          Si VECT = 'P' ou 'B', la matrice orthogonale n-par-n P'.
!>          Si VECT = 'N' ou 'Q', le tableau PT n'est pas référencé.
!>

LDPT

!>          LDPT est INTEGER
!>          La dimension principale du tableau PT.
!>          LDPT >= max(1,N) si VECT = 'P' ou 'B' ; LDPT >= 1 sinon.
!>

C

!>          C est un tableau de DOUBLE PRECISION, dimension (LDC,NCC)
!>          À l'entrée, une matrice C m-par-ncc.
!>          À la sortie, C est écrasée par Q**T*C.
!>          C n'est pas référencée si NCC = 0.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C.
!>          LDC >= max(1,M) si NCC > 0 ; LDC >= 1 si NCC = 0.
!>

WORK

!>          WORK est un tableau de DOUBLE PRECISION, dimension (2*max(M,N))
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie.
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>
