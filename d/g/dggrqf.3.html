subroutine dggrqf (integer m, integer p, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) taua, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) taub, double precision, dimension( * ) work, integer lwork, integer

DGGRQ

F

But

: 
!>
!> DGGRQF calcule une factorisation RQ généralisée d'une matrice A de
!> dimensions M par N et d'une matrice B de dimensions P par N :
!>
!>             A = R*Q,        B = Z*T*Q,
!>
!> où Q est une matrice orthogonale de dimensions N par N, Z est une
!> matrice orthogonale de dimensions P par P, et R et T prennent l'une
!> des formes suivantes :
!>
!> si M <= N,  R = ( 0  R12 ) M,   ou si M > N,  R = ( R11 ) M-N,
!>                  N-M  M                           ( R21 ) N
!>                                                      N
!>
!> où R12 ou R21 est triangulaire supérieur, et
!>
!> si P >= N,  T = ( T11 ) N  ,   ou si P < N,  T = ( T11  T12 ) P,
!>                 (  0  ) P-N                         P   N-P
!>                    N
!>
!> où T11 est triangulaire supérieur.
!>
!> En particulier, si B est carrée et non singulière, la factorisation
!> GRQ de A et B donne implicitement la factorisation RQ de A*inv(B) :
!>
!>              A*inv(B) = (R*inv(T))*Z**T
!>
!> où inv(B) désigne l'inverse de la matrice B, et Z**T désigne la
!> transposée de la matrice Z.
!
>

Paramètre

s
M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!
>

P 
!>          P est INTEGER
!>          Le nombre de lignes de la matrice B.  P >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes des matrices A et B. N >= 0.
!
>

A 
!>          A est un tableau DOUBLE PRECISION, dimension (LDA,N)
!>          À l'entrée, la matrice A de dimensions M par N.
!>          À la sortie, si M <= N, le triangle supérieur du sous-tableau
!>          A(1:M,N-M+1:N) contient la matrice triangulaire supérieure R de
!>          dimensions M par M ;
!>          si M > N, les éléments sur et au-dessus de la (M-N)-ième
!>          sous-diagonale contiennent la matrice trapézoïdale supérieure R
!>          de dimensions M par N ; les éléments restants, avec le tableau
!>          TAUA, représentent la matrice orthogonale Q sous la forme d'un
!>          produit de réflecteurs élémentaires (voir Détails
!>          supplémentaires).
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!
>

A 
!>          TAUA est un tableau DOUBLE PRECISION, dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires qui
!>          représentent la matrice orthogonale Q (voir Détails
!>          supplémentaires).
!
>

B 
!>          B est un tableau DOUBLE PRECISION, dimension (LDB,N)
!>          À l'entrée, la matrice B de dimensions P par N.
!>          À la sortie, les éléments sur et au-dessus de la diagonale du
!>          tableau contiennent la matrice trapézoïdale supérieure T de
!>          dimensions min(P,N) par N (T est triangulaire supérieure si
!>          P >= N) ; les éléments sous la diagonale, avec le tableau TAUB,
!>          représentent la matrice orthogonale Z sous la forme d'un
!>          produit de réflecteurs élémentaires (voir Détails
!>          supplémentaires).
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,P).
!
>

TAU

B

!>          Les facteurs scalaires des réflecteurs élémentaires qui
!>          représentent la matrice orthogonale Z (voir Détails
!>          supplémentaires).
!
>

WOR

K 
!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= max(1,N,M,P).
!>          Pour des performances optimales, LWORK >= max(N,M,P)*max(NB1,NB2,NB3),
!>          où NB1 est la taille de bloc optimale pour la factorisation RQ
!>          d'une matrice de dimensions M par N, NB2 est la taille de bloc
!>          optimale pour la factorisation QR d'une matrice de dimensions P
!>          par N, et NB3 est la taille de bloc optimale pour un appel à
!>          DORMRQ.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille optimale du
!>          tableau WORK, renvoie cette valeur comme première entrée du
!>          tableau WORK, et aucun message d'erreur lié à LWORK n'est émis
!>          par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO= -i, le i-ème argument avait une valeur illégale.
!

Détails supplémentaires

: 
!>
!>  La matrice Q est représentée comme un produit de réflecteurs élémentaires
!>
!>     Q = H(1) H(2) . . . H(k), où k = min(m,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - taua * v * v**T
!>
!>  où taua est un scalaire réel, et v est un vecteur réel avec
!>  v(n-k+i+1:n) = 0 et v(n-k+i) = 1 ; v(1:n-k+i-1) est stocké à la sortie dans
!>  A(m-k+i,1:n-k+i-1), et taua dans TAUA(i).
!>  Pour former Q explicitement, utilisez la sous-routine LAPACK DORGRQ.
!>  Pour utiliser Q afin de mettre à jour une autre matrice, utilisez la sous-routine LAPACK DORMRQ.
!>
!>  La matrice Z est représentée comme un produit de réflecteurs élémentaires
!>
!>     Z = H(1) H(2) . . . H(k), où k = min(p,n).
!>
!>  Chaque H(i) a la forme
!>
!>     H(i) = I - taub * v * v**T
!>
!>  où taub est un scalaire réel, et v est un vecteur réel avec
!>  v(1:i-1) = 0 et v(i) = 1 ; v(i+1:p) est stocké à la sortie dans B(i+1:p,i),
!>  et taub dans TAUB(i).
!>  Pour former Z explicitement, utilisez la sous-routine LAPACK DORGQR.
!>  Pour utiliser Z afin de mettre à jour une autre matrice, utilisez la sous-routine LAPACK DORMQR.
!
>
