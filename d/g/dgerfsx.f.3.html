subroutine dgerfsx (character trans, character equed, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx , * ) x, integer ldx, double precision rcond, double precision, dimension( * ) berr, integer n_err_bnds, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, integer nparams, double precision, dimension( * ) params, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGERFSX

But :

!>
!>    DGERFSX améliore la solution calculée d'un système d'équations
!>    linéaires et fournit des bornes d'erreur et des estimations
!>    d'erreur arrière pour la solution. En plus de la borne d'erreur
!>    au sens de la norme, le code fournit, si possible, la borne
!>    d'erreur maximale composante par composante. Voir les commentaires
!>    pour ERR_BNDS_NORM et ERR_BNDS_COMP pour les détails des bornes
!>    d'erreur.
!>
!>    Le système d'équations linéaires d'origine peut avoir été
!>    équilibré avant l'appel de cette routine, comme décrit par les
!>    arguments EQUED, R et C ci-dessous. Dans ce cas, la solution et
!>    les bornes d'erreur renvoyées concernent le système d'origine non
!>    équilibré.
!> 
!>     Certains paramètres optionnels sont regroupés dans le tableau
!>     PARAMS. Ces réglages déterminent la manière dont le raffinement
!>     est effectué, mais souvent les valeurs par défaut sont
!>     acceptables. Si les valeurs par défaut sont acceptables, les
!>     utilisateurs peuvent passer NPARAMS = 0, ce qui empêche le code
!>     source d'accéder à l'argument PARAMS.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>     Spécifie la forme du système d'équations :
!>       = 'N':  A * X = B     (Pas de transposition)
!>       = 'T':  A**T * X = B  (Transposition)
!>       = 'C':  A**H * X = B  (Transposition conjuguée = Transposition)
!>

EQUED

!>          EQUED est CHARACTER*1
!>     Spécifie la forme d'équilibrage qui a été appliquée à A
!>     avant l'appel de cette routine. Ceci est nécessaire pour
!>     calculer correctement la solution et les bornes d'erreur.
!>       = 'N':  Pas d'équilibrage
!>       = 'R':  Équilibrage par les lignes, c'est-à-dire que A a été
!>               prémultiplié par diag(R).
!>       = 'C':  Équilibrage par les colonnes, c'est-à-dire que A a été
!>               postmultiplié par diag(C).
!>       = 'B':  Équilibrage à la fois par les lignes et par les
!>               colonnes, c'est-à-dire que A a été remplacé par
!>               diag(R) * A * diag(C).
!>               Le membre de droite B a été modifié en conséquence.
!>

N

!>          N est INTEGER
!>     L'ordre de la matrice A. N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>     Le nombre de membres de droite, c'est-à-dire le nombre de
!>     colonnes des matrices B et X. NRHS >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (LDA,N)
!>     La matrice A d'origine de taille N par N.
!>

LDA

!>          LDA est INTEGER
!>     La dimension principale du tableau A. LDA >= max(1,N).
!>

AF

!>          AF est un tableau DOUBLE PRECISION, dimension (LDAF,N)
!>     Les facteurs L et U de la factorisation A = P*L*U
!>     tels que calculés par DGETRF.
!>

LDAF

!>          LDAF est INTEGER
!>     La dimension principale du tableau AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>     Les indices de pivot de DGETRF ; pour 1<=i<=N, la ligne i de la
!>     matrice a été échangée avec la ligne IPIV(i).
!>

R

!>          R est un tableau DOUBLE PRECISION, dimension (N)
!>     Les facteurs d'échelle par ligne pour A. Si EQUED = 'R' ou 'B', A est
!>     multiplié à gauche par diag(R) ; si EQUED = 'N' ou 'C', R
!>     n'est pas accédé.
!>     Si R est accédé, chaque élément de R devrait être une puissance
!>     de la base afin de garantir une solution et des estimations
!>     d'erreur fiables. La mise à l'échelle par des puissances de la
!>     base ne provoque pas d'erreurs d'arrondi à moins que le résultat
!>     ne déborde ou ne soit sous le seuil de représentabilité. Les
!>     erreurs d'arrondi pendant la mise à l'échelle conduisent à un
!>     raffinement avec une matrice qui n'est pas équivalente à la
!>     matrice d'entrée, produisant des estimations d'erreur qui
!>     peuvent ne pas être fiables.
!>

C

!>          C est un tableau DOUBLE PRECISION, dimension (N)
!>     Les facteurs d'échelle par colonne pour A. Si EQUED = 'C' ou 'B', A est
!>     multiplié à droite par diag(C) ; si EQUED = 'N' ou 'R', C
!>     n'est pas accédé.
!>     Si C est accédé, chaque élément de C devrait être une puissance
!>     de la base afin de garantir une solution et des estimations
!>     d'erreur fiables. La mise à l'échelle par des puissances de la
!>     base ne provoque pas d'erreurs d'arrondi à moins que le résultat
!>     ne déborde ou ne soit sous le seuil de représentabilité. Les
!>     erreurs d'arrondi pendant la mise à l'échelle conduisent à un
!>     raffinement avec une matrice qui n'est pas équivalente à la
!>     matrice d'entrée, produisant des estimations d'erreur qui
!>     peuvent ne pas être fiables.
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>     La matrice membre de droite B.
!>

LDB

!>          LDB est INTEGER
!>     La dimension principale du tableau B. LDB >= max(1,N).
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>     À l'entrée, la matrice solution X, telle que calculée par DGETRS.
!>     À la sortie, la matrice solution X améliorée.
!>

LDX

!>          LDX est INTEGER
!>     La dimension principale du tableau X. LDX >= max(1,N).
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>     Nombre de condition réciproque mis à l'échelle. C'est une
!>     estimation du réciproque du nombre de condition de Skeel de la
!>     matrice A après équilibrage (si effectué). Si celui-ci est
!>     inférieur à la précision machine (en particulier, s'il est
!>     nul), la matrice est singulière à la précision de travail.
!>     Notez que l'erreur peut être petite même si ce nombre est très
!>     petit et que la matrice semble mal conditionnée.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>     Erreur arrière relative composante par composante. C'est
!>     l'erreur arrière relative composante par composante de chaque
!>     vecteur solution X(j) (c'est-à-dire le plus petit changement
!>     relatif dans n'importe quel élément de A ou B qui rend X(j) une
!>     solution exacte).
!>

N_ERR_BNDS

!>          N_ERR_BNDS est INTEGER
!>     Nombre de bornes d'erreur à renvoyer pour chaque membre de
!>     droite et chaque type (au sens de la norme ou composante par
!>     composante). Voir ERR_BNDS_NORM et ERR_BNDS_COMP ci-dessous.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM est un tableau DOUBLE PRECISION, de dimension (NRHS, N_ERR_BNDS)
!>     Pour chaque membre de droite, ce tableau contient des informations sur
!>     diverses bornes d'erreur et nombres de condition correspondant à
!>     l'erreur relative en norme, qui est définie comme suit :
!>
!>     Erreur relative en norme dans le i-ème vecteur solution :
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     Le tableau est indexé par le type d'information d'erreur comme décrit
!>     ci-dessous. Il y a actuellement jusqu'à trois informations
!>     retournées.
!>
!>     Le premier indice dans ERR_BNDS_NORM(i,:) correspond au i-ème
!>     membre de droite.
!>
!>     Le deuxième indice dans ERR_BNDS_NORM(:,err) contient les trois
!>     champs suivants :
!>     err = 1  booléen. Faire confiance à la réponse si le
!>              nombre de condition réciproque est inférieur au seuil
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  borne d'erreur : L'erreur avant estimée, qui est
!>              presque certainement dans un facteur 10 de l'erreur réelle
!>              tant que l'entrée suivante est supérieure au seuil
!>              sqrt(n) * dlamch('Epsilon'). Cette borne d'erreur ne doit
!>              être considérée fiable que si le booléen précédent est vrai.
!>
!>     err = 3  Nombre de condition réciproque : Nombre de condition
!>              réciproque en norme estimé. Comparé avec le seuil
!>              sqrt(n) * dlamch('Epsilon') pour déterminer si l'estimation
!>              d'erreur est . Ces nombres de condition
!>              réciproques sont 1 / (norm(Z^{-1},inf) * norm(Z,inf)) pour une
!>              matrice Z correctement mise à l'échelle.
!>              Soit Z = S*A, où S met chaque ligne à l'échelle par une puissance
!>              de la base de sorte que toutes les sommes absolues de lignes de Z
!>              soient approximativement égales à 1.
!>
!>     Voir Lapack Working Note 165 pour plus de détails et de
!>     précautions supplémentaires.
!>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP est un tableau DOUBLE PRECISION, de dimension (NRHS, N_ERR_BNDS)
!>     Pour chaque membre de droite, ce tableau contient des informations sur
!>     diverses bornes d'erreur et nombres de condition correspondant à
!>     l'erreur relative composante par composante, qui est définie comme suit :
!>
!>     Erreur relative composante par composante dans le i-ème vecteur solution :
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     Le tableau est indexé par le membre de droite i (dont dépend
!>     l'erreur relative composante par composante), et le type d'information
!>     d'erreur comme décrit ci-dessous. Il y a actuellement jusqu'à trois
!>     informations retournées pour chaque membre de droite. Si la
!>     précision composante par composante n'est pas demandée (PARAMS(3) = 0.0), alors
!>     ERR_BNDS_COMP n'est pas accédé. Si N_ERR_BNDS < 3, alors au plus
!>     les premières entrées (:,N_ERR_BNDS) sont retournées.
!>
!>     Le premier indice dans ERR_BNDS_COMP(i,:) correspond au i-ème
!>     membre de droite.
!>
!>     Le deuxième indice dans ERR_BNDS_COMP(:,err) contient les trois
!>     champs suivants :
!>     err = 1  booléen. Faire confiance à la réponse si le
!>              nombre de condition réciproque est inférieur au seuil
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  borne d'erreur : L'erreur avant estimée, qui est
!>              presque certainement dans un facteur 10 de l'erreur réelle
!>              tant que l'entrée suivante est supérieure au seuil
!>              sqrt(n) * dlamch('Epsilon'). Cette borne d'erreur ne doit
!>              être considérée fiable que si le booléen précédent est vrai.
!>
!>     err = 3  Nombre de condition réciproque : Nombre de condition
!>              réciproque composante par composante estimé. Comparé avec le seuil
!>              sqrt(n) * dlamch('Epsilon') pour déterminer si l'estimation
!>              d'erreur est . Ces nombres de condition
!>              réciproques sont 1 / (norm(Z^{-1},inf) * norm(Z,inf)) pour une
!>              matrice Z correctement mise à l'échelle.
!>              Soit Z = S*(A*diag(x)), où x est la solution pour le
!>              membre de droite courant et S met chaque ligne de
!>              A*diag(x) à l'échelle par une puissance de la base de sorte que
!>              toutes les sommes absolues de lignes de Z
!>              soient approximativement égales à 1.
!>
!>     Voir Lapack Working Note 165 pour plus de détails et de
!>     précautions supplémentaires.
!>

NPARAMS

!>          NPARAMS est INTEGER
!>     Spécifie le nombre de paramètres définis dans PARAMS. Si <= 0, le
!>     tableau PARAMS n'est jamais référencé et les valeurs par défaut sont utilisées.
!>

PARAMS

!>          PARAMS est un tableau DOUBLE PRECISION, de dimension (NPARAMS)
!>     Spécifie les paramètres de l'algorithme. Si une entrée est < 0.0, alors
!>     cette entrée sera remplie avec la valeur par défaut utilisée pour ce
!>     paramètre. Seules les positions jusqu'à NPARAMS sont accédées ; les valeurs
!>     par défaut sont utilisées pour les paramètres de numéro supérieur.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : S'il faut effectuer le raffinement
!>            itératif ou non.
!>         Par défaut : 1.0D+0
!>            = 0.0 :  Aucun raffinement n'est effectué, et aucune borne d'erreur n'est
!>                    calculée.
!>            = 1.0 :  Utiliser l'algorithme de raffinement en double précision,
!>                    éventuellement avec des calculs single-double si
!>                    l'environnement de compilation ne supporte pas DOUBLE
!>                    PRECISION.
!>              (les autres valeurs sont réservées pour une utilisation future)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : Nombre maximal de calculs de résidu
!>            autorisés pour le raffinement.
!>         Par défaut : 10
!>         Agressif : Définir à 100 pour permettre la convergence en utilisant des
!>                     factorisations approchées ou des factorisations autres que LU. Si la
!>                     factorisation utilise une technique autre que
!>                     l'élimination de Gauss, les garanties dans
!>                     err_bnds_norm et err_bnds_comp peuvent ne plus être
!>                     dignes de confiance.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3 : Drapeau déterminant si le code
!>            essaiera de trouver une solution avec une petite erreur relative
!>            composante par composante dans l'algorithme en double précision. Positif
!>            est vrai, 0.0 est faux.
!>         Par défaut : 1.0 (tenter la convergence composante par composante)
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, de dimension (4*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>       = 0 :  Sortie réussie. La solution à chaque membre de droite est
!>         garantie.
!>       < 0 :  Si INFO = -i, le i-ème argument avait une valeur illégale
!>       > 0 et <= N :  U(INFO,INFO) est exactement zéro. La factorisation
!>         a été complétée, mais le facteur U est exactement singulier, donc
!>         la solution et les bornes d'erreur n'ont pas pu être calculées. RCOND = 0
!>         est retourné.
!>       = N+J : La solution correspondant au J-ème membre de droite n'est
!>         pas garantie. Les solutions correspondant aux autres
!>         membres de droite K avec K > J peuvent ne pas être garanties non plus, mais
!>         seul le premier tel membre de droite est rapporté. Si une petite
!>         erreur composante par composante n'est pas demandée (PARAMS(3) = 0.0) alors
!>         le J-ème membre de droite est le premier avec une borne d'erreur
!>        en norme qui n'est pas garantie (le plus petit J tel
!>         que ERR_BNDS_NORM(J,1) = 0.0). Par défaut (PARAMS(3) = 1.0)
!>         le J-ème membre de droite est le premier avec soit une borne d'erreur
!>         en norme soit une borne d'erreur composante par composante qui n'est pas garantie (le plus petit
!>         J tel que soit ERR_BNDS_NORM(J,1) = 0.0 soit
!>         ERR_BNDS_COMP(J,1) = 0.0). Voir la définition de
!>         ERR_BNDS_NORM(:,1) et ERR_BNDS_COMP(:,1). Pour obtenir des informations
!>         sur tous les membres de droite, vérifier ERR_BNDS_NORM ou
!>         ERR_BNDS_COMP.
!>
