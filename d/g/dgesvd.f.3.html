subroutine dgesvd (character jobu, character jobvt, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldvt, * ) vt, integer ldvt, double precision, dimension( * ) work, integer lwork, integer

DGESVD calcule la décomposition en valeurs singulières (SVD) des matrices

GE

But

: 
!>
!> DGESVD calcule la décomposition en valeurs singulières (SVD) d'une matrice
!> réelle A de dimensions M par N, en calculant éventuellement les vecteurs
!> singuliers gauche et/ou droit. La SVD est écrite
!>
!>      A = U * SIGMA * transpose(V)
!>
!> où SIGMA est une matrice de dimensions M par N qui est nulle à l'exception
!> de ses min(m,n) éléments diagonaux, U est une matrice orthogonale de
!> dimensions M par M, et V est une matrice orthogonale de dimensions N par N.
!> Les éléments diagonaux de SIGMA sont les valeurs singulières de A ; elles
!> sont réelles et non négatives, et sont renvoyées par ordre décroissant. Les
!> premières min(m,n) colonnes de U et V sont les vecteurs singuliers gauche
!> et droit de A.
!>
!> Notez que la routine renvoie V**T, et non V.
!
>

Paramètre

s
JOBU 
!>          JOBU est de type CHARACTER*1
!>          Spécifie les options pour le calcul de toute ou partie de la matrice U :
!>          = 'A' :  les M colonnes de U sont renvoyées dans le tableau U ;
!>          = 'S' :  les premières min(m,n) colonnes de U (les vecteurs
!>                   singuliers gauche) sont renvoyées dans le tableau U ;
!>          = 'O' :  les premières min(m,n) colonnes de U (les vecteurs
!>                   singuliers gauche) écrasent le tableau A ;
!>          = 'N' :  aucune colonne de U (aucun vecteur singulier gauche)
!>                   n'est calculée.
!
>

JOBV

T 
!>          JOBVT est de type CHARACTER*1
!>          Spécifie les options pour le calcul de toute ou partie de la matrice
!>          V**T :
!>          = 'A' :  les N lignes de V**T sont renvoyées dans le tableau VT ;
!>          = 'S' :  les premières min(m,n) lignes de V**T (les vecteurs
!>                   singuliers droit) sont renvoyées dans le tableau VT ;
!>          = 'O' :  les premières min(m,n) lignes de V**T (les vecteurs
!>                   singuliers droit) écrasent le tableau A ;
!>          = 'N' :  aucune ligne de V**T (aucun vecteur singulier droit)
!>                   n'est calculée.
!>
!>          JOBVT et JOBU ne peuvent pas être tous les deux 'O'.
!
>

M 
!>          M est de type INTEGER
!>          Le nombre de lignes de la matrice d'entrée A.  M >= 0.
!
>

N 
!>          N est de type INTEGER
!>          Le nombre de colonnes de la matrice d'entrée A.  N >= 0.
!
>

A 
!>          A est un tableau de type DOUBLE PRECISION, dimension (LDA,N)
!>          À l'entrée, la matrice A de dimensions M par N.
!>          À la sortie,
!>          si JOBU = 'O',  A est écrasée par les premières min(m,n)
!>                          colonnes de U (les vecteurs singuliers gauche,
!>                          stockés par colonnes) ;
!>          si JOBVT = 'O', A est écrasée par les premières min(m,n)
!>                          lignes de V**T (les vecteurs singuliers droit,
!>                          stockés par lignes) ;
!>          si JOBU .ne. 'O' et JOBVT .ne. 'O', le contenu de A
!>                          est détruit.
!
>

LD

A 
!>          LDA est de type INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!
>

S 
!>          S est un tableau de type DOUBLE PRECISION, dimension (min(M,N))
!>          Les valeurs singulières de A, triées de sorte que S(i) >= S(i+1).
!
>

U 
!>          U est un tableau de type DOUBLE PRECISION, dimension (LDU,UCOL)
!>          (LDU,M) si JOBU = 'A' ou (LDU,min(M,N)) si JOBU = 'S'.
!>          Si JOBU = 'A', U contient la matrice orthogonale U de dimensions M par M ;
!>          si JOBU = 'S', U contient les premières min(m,n) colonnes de U
!>          (les vecteurs singuliers gauche, stockés par colonnes) ;
!>          si JOBU = 'N' ou 'O', U n'est pas référencée.
!
>

LD

U 
!>          LDU est de type INTEGER
!>          La dimension principale du tableau U.  LDU >= 1 ; si
!>          JOBU = 'S' ou 'A', LDU >= M.
!
>

V

T 
!>          VT est un tableau de type DOUBLE PRECISION, dimension (LDVT,N)
!>          Si JOBVT = 'A', VT contient la matrice orthogonale V**T
!>          de dimensions N par N ;
!>          si JOBVT = 'S', VT contient les premières min(m,n) lignes de
!>          V**T (les vecteurs singuliers droit, stockés par lignes) ;
!>          si JOBVT = 'N' ou 'O', VT n'est pas référencée.
!
>

LDV

T 
!>          LDVT est de type INTEGER
!>          La dimension principale du tableau VT.  LDVT >= 1 ; si
!>          JOBVT = 'A', LDVT >= N ; si JOBVT = 'S', LDVT >= min(M,N).
!
>

WOR

K 
!>          WORK est un tableau de type DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie la valeur optimale de LWORK ;
!>          si INFO > 0, WORK(2:MIN(M,N)) contient les éléments superdiagonaux
!>          non convergés d'une matrice bidiagonale supérieure B dont la
!>          diagonale est dans S (pas nécessairement triée). B
!>          satisfait A = U * B * VT, elle a donc les mêmes valeurs singulières
!>          que A, et des vecteurs singuliers liés par U et VT.
!
>

LWOR

K 
!>          LWORK est de type INTEGER
!>          La dimension du tableau WORK.
!>          LWORK >= MAX(1,5*MIN(M,N)) pour les chemins (voir les commentaires dans le code) :
!>             - CHEMIN 1  (M beaucoup plus grand que N, JOBU='N')
!>             - CHEMIN 1t (N beaucoup plus grand que M, JOBVT='N')
!>          LWORK >= MAX(1,3*MIN(M,N) + MAX(M,N),5*MIN(M,N)) pour les autres chemins.
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message
!>          d'erreur lié à LWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est de type INTEGER
!>          = 0 :  sortie réussie.
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 :  si DBDSQR n'a pas convergé, INFO indique combien de
!>                superdiagonales d'une forme bidiagonale intermédiaire B
!>                n'ont pas convergé vers zéro. Voir la description de WORK
!>                ci-dessus pour plus de détails.
!
>
