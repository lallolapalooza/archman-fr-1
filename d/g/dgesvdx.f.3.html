subroutine dgesvdx (character jobu, character jobvt, character range, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision vl, double precision vu, integer il, integer iu, integer ns, double precision, dimension( * ) s, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldvt, * ) vt, integer ldvt, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer

info)

DGESVDX calcule la décomposition en valeurs singulières (SVD) pour les matrices

GE

But

: 
!>
!>  DGESVDX calcule la décomposition en valeurs singulières (SVD) d'une matrice
!>  réelle A de dimensions M par N, en calculant éventuellement les vecteurs
!>  singuliers gauche et/ou droit. La SVD est écrite
!>
!>      A = U * SIGMA * transpose(V)
!>
!>  où SIGMA est une matrice de dimensions M par N qui est nulle sauf pour ses
!>  min(m,n) éléments diagonaux, U est une matrice orthogonale de dimensions
!>  M par M, et V est une matrice orthogonale de dimensions N par N. Les
!>  éléments diagonaux de SIGMA sont les valeurs singulières de A ; elles sont
!>  réelles et non négatives, et sont renvoyées par ordre décroissant. Les
!>  premières min(m,n) colonnes de U et V sont les vecteurs singuliers gauche
!>  et droit de A.
!>
!>  DGESVDX utilise un problème de valeurs propres pour obtenir la SVD, ce qui
!>  permet de calculer un sous-ensemble de valeurs singulières et de vecteurs.
!>  Voir DBDSVDX pour plus de détails.
!>
!>  Notez que la routine renvoie V**T, et non V.
!
>

Paramètre

s
JOBU 
!>          JOBU est CHARACTER*1
!>          Spécifie les options pour calculer toute ou partie de la matrice U :
!>          = 'V' : les premières min(m,n) colonnes de U (les vecteurs
!>                  singuliers gauche) ou telles que spécifiées par RANGE sont
!>                  renvoyées dans le tableau U ;
!>          = 'N' : aucune colonne de U (aucun vecteur singulier gauche) n'est
!>                  calculée.
!
>

JOBV

T 
!>          JOBVT est CHARACTER*1
!>           Spécifie les options pour calculer toute ou partie de la matrice
!>           V**T :
!>           = 'V' : les premières min(m,n) lignes de V**T (les vecteurs
!>                   singuliers droit) ou telles que spécifiées par RANGE sont
!>                   renvoyées dans le tableau VT ;
!>           = 'N' : aucune ligne de V**T (aucun vecteur singulier droit) n'est
!>                   calculée.
!
>

RANG

E 
!>          RANGE est CHARACTER*1
!>          = 'A' : toutes les valeurs singulières seront trouvées.
!>          = 'V' : toutes les valeurs singulières dans l'intervalle semi-ouvert
!>                 (VL,VU] seront trouvées.
!>          = 'I' : les valeurs singulières de la ième à la IU-ième seront
!>                 trouvées.
!
>

M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice d'entrée A.  M >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice d'entrée A.  N >= 0.
!
>

A 
!>          A est un tableau DOUBLE PRECISION, dimension (LDA,N)
!>          À l'entrée, la matrice A de dimensions M par N.
!>          À la sortie, le contenu de A est détruit.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!
>

V

L 
!>          VL est DOUBLE PRECISION
!>          Si RANGE='V', la borne inférieure de l'intervalle dans lequel
!>          rechercher les valeurs singulières. VU > VL.
!>          Non référencé si RANGE = 'A' ou 'I'.
!
>

V

U 
!>          VU est DOUBLE PRECISION
!>          Si RANGE='V', la borne supérieure de l'intervalle dans lequel
!>          rechercher les valeurs singulières. VU > VL.
!>          Non référencé si RANGE = 'A' ou 'I'.
!
>

I

L 
!>          IL est INTEGER
!>          Si RANGE='I', l'indice de la
!>          plus petite valeur singulière à renvoyer.
!>          1 <= IL <= IU <= min(M,N), si min(M,N) > 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!
>

I

U 
!>          IU est INTEGER
!>          Si RANGE='I', l'indice de la
!>          plus grande valeur singulière à renvoyer.
!>          1 <= IL <= IU <= min(M,N), si min(M,N) > 0.
!>          Non référencé si RANGE = 'A' ou 'V'.
!
>

N

S 
!>          NS est INTEGER
!>          Le nombre total de valeurs singulières trouvées,
!>          0 <= NS <= min(M,N).
!>          Si RANGE = 'A', NS = min(M,N) ; si RANGE = 'I', NS = IU-IL+1.
!
>

S 
!>          S est un tableau DOUBLE PRECISION, dimension (min(M,N))
!>          Les valeurs singulières de A, triées de sorte que S(i) >= S(i+1).
!
>

U 
!>          U est un tableau DOUBLE PRECISION, dimension (LDU,UCOL)
!>          Si JOBU = 'V', U contient les colonnes de U (les vecteurs
!>          singuliers gauche, stockés par colonnes) telles que spécifiées par
!>          RANGE ; si JOBU = 'N', U n'est pas référencé.
!>          Note : L'utilisateur doit s'assurer que UCOL >= NS ; si RANGE = 'V',
!>          la valeur exacte de NS n'est pas connue à l'avance et une borne
!>          supérieure doit être utilisée.
!
>

LD

U 
!>          LDU est INTEGER
!>          La dimension principale du tableau U.  LDU >= 1 ; si
!>          JOBU = 'V', LDU >= M.
!
>

V

T 
!>          VT est un tableau DOUBLE PRECISION, dimension (LDVT,N)
!>          Si JOBVT = 'V', VT contient les lignes de V**T (les vecteurs
!>          singuliers droit, stockés par lignes) telles que spécifiées par
!>          RANGE ; si JOBVT = 'N', VT n'est pas référencé.
!>          Note : L'utilisateur doit s'assurer que LDVT >= NS ; si RANGE = 'V',
!>          la valeur exacte de NS n'est pas connue à l'avance et une borne
!>          supérieure doit être utilisée.
!
>

LDV

T 
!>          LDVT est INTEGER
!>          La dimension principale du tableau VT.  LDVT >= 1 ; si
!>          JOBVT = 'V', LDVT >= NS (voir ci-dessus).
!
>

WOR

K 
!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie la valeur optimale de LWORK ;
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          LWORK >= MAX(1,MIN(M,N)*(MIN(M,N)+4)) pour les chemins (voir
!>          les commentaires à l'intérieur du code) :
!>             - CHEMIN 1  (M beaucoup plus grand que N)
!>             - CHEMIN 1t (N beaucoup plus grand que M)
!>          LWORK >= MAX(1,MIN(M,N)*2+MAX(M,N)) pour les autres chemins.
!>          Pour de bonnes performances, LWORK devrait généralement être plus
!>          grand.
!>
!>          Si LWORK = -1, une interrogation de l'espace de travail est supposée ;
!>          la routine ne calcule que la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message
!>          d'erreur lié à LWORK n'est émis par XERBLA.
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (12*MIN(M,N))
!>          Si INFO = 0, les premiers NS éléments de IWORK sont nuls. Si INFO > 0,
!>          alors IWORK contient les indices des vecteurs propres qui n'ont pas
!>          convergé dans DBDSVDX/DSTEVX.
!
>

INF

O 
!>     INFO est INTEGER
!>           = 0 :  sortie réussie
!>           < 0 :  si INFO = -i, le ième argument avait une valeur illégale
!>           > 0 :  si INFO = i, alors i vecteurs propres n'ont pas convergée
!>                 in DBDSVDX/DSTEVX.
!>                 si INFO = N*2 + 1, une erreur interne est survenue dans
!>                 DBDSVDX
!
>
