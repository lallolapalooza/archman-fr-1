subroutine dgebak (character job, character side, integer n, integer ilo, integer ihi, double precision, dimension( * ) scale, integer m, double precision, dimension( ldv, * ) v, integer ldv, integer info)

DGEBAK

But :

!>
!> DGEBAK forme les vecteurs propres droits ou gauches d'une matrice générale
!> réelle en appliquant une transformation inverse sur les vecteurs propres calculés
!> de la matrice équilibrée produite par DGEBAL.
!>

Paramètres

JOB 
!>          JOB est CHARACTER*1
!>          Spécifie le type de transformation inverse requise :
!>          = 'N' : ne rien faire, retourner immédiatement ;
!>          = 'P' : effectuer la transformation inverse pour la permutation uniquement ;
!>          = 'S' : effectuer la transformation inverse pour la mise à l'échelle uniquement ;
!>          = 'B' : effectuer les transformations inverses pour la permutation et
!>                 la mise à l'échelle.
!>          JOB doit être le même que l'argument JOB fourni à DGEBAL.
!>

SIDE

!>          SIDE est CHARACTER*1
!>          = 'R' : V contient les vecteurs propres droits ;
!>          = 'L' : V contient les vecteurs propres gauches.
!>

N

!>          N est INTEGER
!>          Le nombre de lignes de la matrice V. N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>          Les entiers ILO et IHI déterminés par DGEBAL.
!>          1 <= ILO <= IHI <= N, si N > 0 ; ILO=1 et IHI=0, si N=0.
!>

SCALE

!>          SCALE est un tableau DOUBLE PRECISION, de dimension (N)
!>          Détails de la permutation et des facteurs de mise à l'échelle, tels que retournés
!>          par DGEBAL.
!>

M

!>          M est INTEGER
!>          Le nombre de colonnes de la matrice V. M >= 0.
!>

V

!>          V est un tableau DOUBLE PRECISION, de dimension (LDV,M)
!>          À l'entrée, la matrice des vecteurs propres droits ou gauches à
!>          transformer, telle que retournée par DHSEIN ou DTREVC.
!>          À la sortie, V est écrasé par les vecteurs propres transformés.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V. LDV >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>
