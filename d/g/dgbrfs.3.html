subroutine dgbrfs (caractère trans, entier n, entier kl, entier ku, entier nrhs, double précision, dimension( ldab, * ) ab, entier ldab, double précision, dimension( ldafb, * ) afb, entier ldafb, entier, dimension( * ) ipiv, double précision, dimension( ldb, * ) b, entier ldb, double précision, dimension( ldx, * ) x, entier ldx, double précision, dimension( * ) ferr, double précision, dimension( * ) berr, double précision, dimension( * ) work, entier, dimension( * ) iwork, entier info)

DGBRFS

But:

!>
!> DGBRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est bande, et fournit des bornes d'erreur
!> et des estimations d'erreur arrière pour la solution.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Transposée conjuguée = Transposée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KL

!>          KL est INTEGER
!>          Le nombre de sous-diagonales dans la bande de A.  KL >= 0.
!>

KU

!>          KU est INTEGER
!>          Le nombre de super-diagonales dans la bande de A.  KU >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres droits, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AB

!>          AB est un tableau de DOUBLE PRÉCISION, de dimension (LDAB,N)
!>          La matrice bande originale A, stockée dans les lignes 1 à KL+KU+1.
!>          La j-ème colonne de A est stockée dans la j-ème colonne du
!>          tableau AB comme suit :
!>          AB(ku+1+i-j,j) = A(i,j) pour max(1,j-ku)<=i<=min(n,j+kl).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB est un tableau de DOUBLE PRÉCISION, de dimension (LDAFB,N)
!>          Détails de la factorisation LU de la matrice bande A, tels que
!>          calculés par DGBTRF.  U est stocké comme une matrice bande triangulaire
!>          supérieure avec KL+KU super-diagonales dans les lignes 1 à KL+KU+1, et
!>          les multiplicateurs utilisés lors de la factorisation sont stockés dans
!>          les lignes KL+KU+2 à 2*KL+KU+1.
!>

LDAFB

!>          LDAFB est INTEGER
!>          La dimension principale du tableau AFB.  LDAFB >= 2*KL*KU+1.
!>

IPIV

!>          IPIV est un tableau de INTEGER, de dimension (N)
!>          Les indices de pivot de DGBTRF ; pour 1<=i<=N, la ligne i de la
!>          matrice a été échangée avec la ligne IPIV(i).
!>

B

!>          B est un tableau de DOUBLE PRÉCISION, de dimension (LDB,NRHS)
!>          La matrice membre droit B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau de DOUBLE PRÉCISION, de dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par DGBTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau de DOUBLE PRÉCISION, de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          majoration de l'erreur réelle.
!>

BERR

!>          BERR est un tableau de DOUBLE PRÉCISION, de dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur
!>          solution X(j) (c'est-à-dire le plus petit changement relatif dans
!>          tout élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau de DOUBLE PRÉCISION, de dimension (3*N)
!>

IWORK

!>          IWORK est un tableau de INTEGER, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes:

!>  ITMAX est le nombre maximal d'étapes de raffinement itératif.
!>
