sous-programme dget10 (entier m, entier n, double précision, dimension( lda, * ) a, entier lda, double précision, dimension( ldb, * ) b, entier ldb, double précision, dimension( * ) work, double précision result)

DGET10

But :

!>
!> DGET10 compare deux matrices A et B et calcule le rapport
!> RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>

Paramètres

M 
!>          M est ENTIER
!>          Le nombre de lignes des matrices A et B.
!>

N

!>          N est ENTIER
!>          Le nombre de colonnes des matrices A et B.
!>

A

!>          A est un tableau DOUBLE PRÉCISION, dimension (LDA,N)
!>          La matrice A de dimension m x n.
!>

LDA

!>          LDA est ENTIER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

B

!>          B est un tableau DOUBLE PRÉCISION, dimension (LDB,N)
!>          La matrice B de dimension m x n.
!>

LDB

!>          LDB est ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>

WORK

!>          WORK est un tableau DOUBLE PRÉCISION, dimension (M)
!>

RESULT

!>          RESULT est DOUBLE PRÉCISION
!>          RESULT = norm( A - B ) / ( norm(A) * M * EPS )
!>
