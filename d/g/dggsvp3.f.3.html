sous-routine dggsvp3 (character jobu, character jobv, character jobq, integer m, integer p, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision tola, double precision tolb, integer k, integer l, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) iwork, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DGGSVP3

But :

!>
!> DGGSVP3 calcule les matrices orthogonales U, V et Q telles que
!>
!>                    N-K-L  K    L
!>  U**T*A*Q =     K ( 0    A12  A13 )  si M-K-L >= 0;
!>                 L ( 0     0   A23 )
!>             M-K-L ( 0     0    0  )
!>
!>                  N-K-L  K    L
!>         =     K ( 0    A12  A13 )  si M-K-L < 0;
!>             M-K ( 0     0   A23 )
!>
!>                  N-K-L  K    L
!>  V**T*B*Q =   L ( 0     0   B13 )
!>             P-L ( 0     0    0  )
!>
!> où la matrice KxK A12 et la matrice LxL B13 sont des matrices triangulaires
!> supérieures non singulières ; A23 est une matrice triangulaire supérieure LxL
!> si M-K-L >= 0, sinon A23 est une matrice trapézoïdale supérieure (M-K)xL. K+L est
!> le rang numérique effectif de la matrice (M+P)xN (A**T,B**T)**T.
!>
!> Cette décomposition est l'étape de pré-traitement pour calculer la
!> Décomposition en Valeurs Singulières Généralisées (GSVD), voir la sous-routine
!> DGGSVD3.
!>

Paramètres

JOBU 
!>          JOBU est CHARACTER*1
!>          = 'U':  La matrice orthogonale U est calculée ;
!>          = 'N':  U n'est pas calculée.
!>

JOBV

!>          JOBV est CHARACTER*1
!>          = 'V':  La matrice orthogonale V est calculée ;
!>          = 'N':  V n'est pas calculée.
!>

JOBQ

!>          JOBQ est CHARACTER*1
!>          = 'Q':  La matrice orthogonale Q est calculée ;
!>          = 'N':  Q n'est pas calculée.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A. M >= 0.
!>

P

!>          P est INTEGER
!>          Le nombre de lignes de la matrice B. P >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes des matrices A et B. N >= 0.
!>

A

!>          A est un tableau DOUBLE PRECISION, dimension (LDA,N)
!>          À l'entrée, A contient la matrice A M-by-N.
!>          À la sortie, A contient la matrice triangulaire (ou trapézoïdale)
!>          décrite dans la section But.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,N)
!>          À l'entrée, B contient la matrice P-by-N.
!>          À la sortie, B contient la matrice triangulaire décrite dans
!>          la section But.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,P).
!>

TOLA

!>          TOLA est DOUBLE PRECISION
!>

TOLB

!>          TOLB est DOUBLE PRECISION
!>
!>          TOLA et TOLB sont les seuils pour déterminer le rang numérique
!>          effectif de la matrice B et d'un sous-bloc de A. Généralement,
!>          ils sont définis par
!>             TOLA = MAX(M,N)*norm(A)*MACHEPS,
!>             TOLB = MAX(P,N)*norm(B)*MACHEPS.
!>          La valeur de TOLA et TOLB peut affecter la taille des erreurs arrière
!>          de la décomposition.
!>

K

!>          K est INTEGER
!>

L

!>          L est INTEGER
!>
!>          À la sortie, K et L spécifient la dimension des sous-blocs
!>          décrits dans la section But.
!>          K + L = rang numérique effectif de (A**T,B**T)**T.
!>

U

!>          U est un tableau DOUBLE PRECISION, dimension (LDU,M)
!>          Si JOBU = 'U', U contient la matrice orthogonale U.
!>          Si JOBU = 'N', U n'est pas référencé.
!>

LDU

!>          LDU est INTEGER
!>          La dimension principale du tableau U. LDU >= max(1,M) si
!>          JOBU = 'U' ; sinon LDU >= 1.
!>

V

!>          V est un tableau DOUBLE PRECISION, dimension (LDV,P)
!>          Si JOBV = 'V', V contient la matrice orthogonale V.
!>          Si JOBV = 'N', V n'est pas référencé.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V. LDV >= max(1,P) si
!>          JOBV = 'V' ; sinon LDV >= 1.
!>

Q

!>          Q est un tableau DOUBLE PRECISION, dimension (LDQ,N)
!>          Si JOBQ = 'Q', Q contient la matrice orthogonale Q.
!>          Si JOBQ = 'N', Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q. LDQ >= max(1,N) si
!>          JOBQ = 'Q' ; sinon LDQ >= 1.
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (N)
!>

TAU

!>          TAU est un tableau DOUBLE PRECISION, dimension (N)
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= 1.
!>
!>          Si LWORK = -1, alors une requête de mémoire est supposée ; la routine
!>          calcule seulement la taille optimale du tableau WORK, retourne
!>          cette valeur comme première entrée du tableau WORK, et aucun message
!>          d'erreur relatif à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  exécution réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!>
