subroutine dgtrfs (character trans, integer n, integer nrhs, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) dlf, double precision, dimension( * ) df, double precision, dimension( * ) duf, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGTRFS

But :

!>
!> DGTRFS améliore la solution calculée d'un système d'équations
!> linéaires lorsque la matrice des coefficients est tridiagonale, et
!> fournit des bornes d'erreur et des estimations d'erreur arrière
!> pour la solution.
!>

Paramètres

TRANS 
!>          TRANS est de type CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Transposée conjuguée = Transposée)
!>

N

!>          N est de type INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est de type INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de
!>          colonnes de la matrice B.  NRHS >= 0.
!>

DL

!>          DL est un tableau de type DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux de A.
!>

D

!>          D est un tableau de type DOUBLE PRECISION, dimension (N)
!>          Les éléments diagonaux de A.
!>

DU

!>          DU est un tableau de type DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) éléments sur-diagonaux de A.
!>

DLF

!>          DLF est un tableau de type DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L à
!>          partir de la factorisation LU de A telle que calculée par
!>          DGTTRF.
!>

DF

!>          DF est un tableau de type DOUBLE PRECISION, dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire
!>          supérieure U issue de la factorisation LU de A.
!>

DUF

!>          DUF est un tableau de type DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) éléments de la première sur-diagonale de U.
!>

DU2

!>          DU2 est un tableau de type DOUBLE PRECISION, dimension (N-2)
!>          Les (n-2) éléments de la deuxième sur-diagonale de U.
!>

IPIV

!>          IPIV est un tableau de type INTEGER, dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la
!>          matrice a été interchangée avec la ligne IPIV(i).
!>          IPIV(i) sera toujours soit i soit i+1 ; IPIV(i) = i indique
!>          qu'un échange de lignes n'était pas requis.
!>

B

!>          B est un tableau de type DOUBLE PRECISION, dimension (LDB,NRHS)
!>          La matrice du second membre B.
!>

LDB

!>          LDB est de type INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau de type DOUBLE PRECISION, dimension (LDX,NRHS)
!>          À l'entrée, la matrice solution X, telle que calculée par
!>          DGTTRS.  À la sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est de type INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau de type DOUBLE PRECISION, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur
!>          solution X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la solution réelle correspondant à X(j),
!>          FERR(j) est une borne supérieure estimée pour la magnitude
!>          du plus grand élément de (X(j) - XTRUE) divisée par la
!>          magnitude du plus grand élément de X(j).  L'estimation est
!>          aussi fiable que celle pour RCOND, et est presque toujours
!>          une légère surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau de type DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de
!>          chaque vecteur solution X(j) (c'est-à-dire le plus petit
!>          changement relatif dans n'importe quel élément de A ou B
!>          qui rend X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau de type DOUBLE PRECISION, dimension (3*N)
!>

IWORK

!>          IWORK est un tableau de type INTEGER, dimension (N)
!>

INFO

!>          INFO est de type INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur
!>          illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximal d'étapes de raffinement itératif.
!>
