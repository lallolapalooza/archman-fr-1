subroutine dgttrs (character trans, integer n, integer nrhs, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DGTTRS

But:

!>
!> DGTTRS résout un des systèmes d'équations
!>    A*X = B  ou  A**T*X = B,
!> avec une matrice tridiagonale A en utilisant la factorisation LU calculée
!> par DGTTRF.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations.
!>          = 'N':  A * X = B  (Pas de transposée)
!>          = 'T':  A**T* X = B  (Transposée)
!>          = 'C':  A**T* X = B  (Conjuguée transposée = Transposée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

DL

!>          DL est tableau DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L de la
!>          factorisation LU de A.
!>

D

!>          D est tableau DOUBLE PRECISION, dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire supérieure U de
!>          la factorisation LU de A.
!>

DU

!>          DU est tableau DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) éléments de la première super-diagonale de U.
!>

DU2

!>          DU2 est tableau DOUBLE PRECISION, dimension (N-2)
!>          Les (n-2) éléments de la deuxième super-diagonale de U.
!>

IPIV

!>          IPIV est tableau INTEGER, dimension (N)
!>          Les indices de pivot; pour 1 <= i <= n, la ligne i de la matrice a été
!>          échangée avec la ligne IPIV(i).  IPIV(i) sera toujours soit
!>          i soit i+1; IPIV(i) = i indique qu'un échange de ligne n'était pas
!>          requis.
!>

B

!>          B est tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          À l'entrée, la matrice des vecteurs de côté droit B.
!>          À la sortie, B est écrasé par les vecteurs solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  terminaison réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
