subroutine dgtcon (character norm, integer n, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGTCON

Purpose:

!>
!> DGTCON estime la réciproque du nombre de condition d'une matrice
!> tridiagonale réelle A en utilisant la factorisation LU calculée par
!> DGTTRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et la réciproque du
!> nombre de condition est calculée comme RCOND = 1 / (ANORM * norm(inv(A))).
!>

Paramètres

NORM 
!>          NORM est CHARACTER*1
!>          Spécifie si le nombre de condition en norme 1 ou en norme
!>          infini est requis :
!>          = '1' ou 'O':  norme 1;
!>          = 'I':         norme infini.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

DL

!>          DL est un tableau DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L de la
!>          factorisation LU de A calculée par DGTTRF.
!>

D

!>          D est un tableau DOUBLE PRECISION, dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire supérieure U
!>          issue de la factorisation LU de A.
!>

DU

!>          DU est un tableau DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) éléments de la première superdiagonale de U.
!>

DU2

!>          DU2 est un tableau DOUBLE PRECISION, dimension (N-2)
!>          Les (n-2) éléments de la deuxième superdiagonale de U.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la matrice
!>          a été échangée avec la ligne IPIV(i). IPIV(i) sera toujours
!>          soit i soit i+1 ; IPIV(i) = i indique qu'aucun échange de ligne
!>          n'était requis.
!>

ANORM

!>          ANORM est DOUBLE PRECISION
!>          Si NORM = '1' ou 'O', la norme 1 de la matrice originale A.
!>          Si NORM = 'I', la norme infini de la matrice originale A.
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>          La réciproque du nombre de condition de la matrice A,
!>          calculée comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée par cette routine.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (2*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
