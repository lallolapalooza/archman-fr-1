subroutine drotm (integer n, double precision, dimension(*) dx, integer incx, double precision, dimension(*) dy, integer incy, double precision, dimension(5) dparam)

DROTM

But :

!>
!>    APPLIQUE LA TRANSFORMATION DE GIVENS MODIFIÉE, H, À LA MATRICE 2 PAR N
!>
!>    (DX**T) , OÙ **T INDIQUE TRANSPOSE. LES ÉLÉMENTS DE DX SONT DANS
!>    (DY**T)
!>
!>    DX(LX+I*INCX), I = 0 À N-1, OÙ LX = 1 SI INCX .GE. 0, SINON
!>    LX = (-INCX)*N, ET DE MANIÈRE SIMILAIRE POUR SY EN UTILISANT LY ET INCY.
!>    AVEC DPARAM(1)=DFLAG, H A UNE DES FORMES SUIVANTES..
!>
!>    DFLAG=-1.D0     DFLAG=0.D0        DFLAG=1.D0     DFLAG=-2.D0
!>
!>      (DH11  DH12)    (1.D0  DH12)    (DH11  1.D0)    (1.D0  0.D0)
!>    H=(          )    (          )    (          )    (          )
!>      (DH21  DH22),   (DH21  1.D0),   (-1.D0 DH22),   (0.D0  1.D0).
!>    VOIR DROTMG POUR UNE DESCRIPTION DU STOCKAGE DES DONNÉES DANS DPARAM.
!>

Paramètres

N 
!>          N est un ENTIER
!>         nombre d'éléments dans le(s) vecteur(s) d'entrée
!>

DX

!>          DX est un tableau de DOUBLE PRÉCISION, dimension ( 1 + ( N - 1 )*abs( INCX ) )
!>

INCX

!>          INCX est un ENTIER
!>         espacement de stockage entre les éléments de DX
!>

DY

!>          DY est un tableau de DOUBLE PRÉCISION, dimension ( 1 + ( N - 1 )*abs( INCY ) )
!>

INCY

!>          INCY est un ENTIER
!>         espacement de stockage entre les éléments de DY
!>

DPARAM

!>          DPARAM est un tableau de DOUBLE PRÉCISION, dimension (5)
!>     DPARAM(1)=DFLAG
!>     DPARAM(2)=DH11
!>     DPARAM(3)=DH21
!>     DPARAM(4)=DH12
!>     DPARAM(5)=DH22
!>
