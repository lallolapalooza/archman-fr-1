sous-programme dpbt05 (caractere uplo, entier n, entier kd, entier nrhs, double precision, dimension( ldab, * ) ab, entier ldab, double precision, dimension( ldb, * ) b, entier ldb, double precision, dimension( ldx, * ) x, entier ldx, double precision, dimension( ldxact, * ) xact, entier ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

DPBT05

But:

!>
!> DPBT05 teste les bornes d'erreur de l'affinement itératif pour la
!> solution calculée d'un système d'équations A*X = B, où A est une
!> matrice bande symétrique.
!>
!> RESLTS(1) = test de la borne d'erreur
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> Une grande valeur est retournée si ce ratio n'est pas inférieur à un.
!>
!> RESLTS(2) = résidu de la routine d'affinement itératif
!>           = le maximum de BERR / ( NZ*EPS + (*) ), où
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             et NZ = nombre max. de non-zéros dans une ligne de A, plus 1
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice symétrique A est stockée.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

N

!>          N est INTEGER
!>          Le nombre de lignes des matrices X, B, et XACT, et l'ordre de la
!>          matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes des matrices X, B, et XACT.
!>          NRHS >= 0.
!>

AB

!>          AB est un tableau DOUBLE PRECISION, dimension (LDAB,N)
!>          Le triangle supérieur ou inférieur de la matrice bande symétrique A,
!>          stocké dans les KD+1 premières lignes du tableau.  La j-ème colonne
!>          de A est stockée dans la j-ème colonne du tableau AB comme suit:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          Les vecteurs du second membre pour le système d'équations linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>          Les vecteurs solution calculés.  Chaque vecteur est stocké comme une
!>          colonne de la matrice X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

XACT

!>          XACT est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>          Les vecteurs solution exacts.  Chaque vecteur est stocké comme une
!>          colonne de la matrice XACT.
!>

LDXACT

!>          LDXACT est INTEGER
!>          La dimension principale du tableau XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          Les bornes d'erreur avant estimées pour chaque vecteur solution
!>          X.  Si XTRUE est la vraie solution, FERR borne la magnitude de
!>          la plus grande entrée de (X - XTRUE) divisée par la magnitude de
!>          la plus grande entrée de X.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur relative composante par composante de chaque vecteur solution
!>          (c'est-à-dire, la plus petite modification relative de toute entrée de A
!>          ou B qui fait de X une solution exacte).
!>

RESLTS

!>          RESLTS est un tableau DOUBLE PRECISION, dimension (2)
!>          Le maximum sur les vecteurs solutions NRHS des ratios:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
