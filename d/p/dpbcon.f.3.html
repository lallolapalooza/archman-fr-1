subroutine dpbcon (character uplo, integer n, integer kd, double precision, dimension( ldab, * ) ab, integer ldab, double precision anorm, double precision rcond, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPBCON

Objectif :

!>
!> DPBCON estime la réciproque du nombre de conditionnement (en norme
!> 1) d'une matrice bande symétrique définie positive réelle, calculée par la
!> factorisation de Cholesky A = U**T*U ou A = L*L**T de DPBTRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et la réciproque du nombre
!> de conditionnement est calculée comme RCOND = 1 / (ANORM * norm(inv(A))).
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  facteur triangulaire supérieur stocké dans AB ;
!>          = 'L':  facteur triangulaire inférieur stocké dans AB.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB est un tableau DOUBLE PRECISION, de dimension (LDAB,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T de la matrice bande A, stocké dans les
!>          KD+1 premières lignes du tableau. La j-ième colonne de U ou L
!>          est stockée dans la j-ième colonne du tableau AB comme suit :
!>          si UPLO ='U', AB(kd+1+i-j,j) = U(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO ='L', AB(1+i-j,j)    = L(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

ANORM

!>          ANORM est DOUBLE PRECISION
!>          La norme 1 (ou norme infinie) de la matrice bande symétrique A.
!>

RCOND

!>          RCOND est DOUBLE PRECISION
!>          La réciproque du nombre de conditionnement de la matrice A,
!>          calculée comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée dans cette routine.
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, de dimension (3*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
