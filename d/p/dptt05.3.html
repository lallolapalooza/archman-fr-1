subroutine dptt05 (n, nrhs, d, e, b, ldb, x, ldx, xact, ldxact, ferr, berr, reslts)
DPTT05

Documentation de fonction/sous-programme

sous-programme dptt05 (integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

DPTT05

Objectif

!>
!> DPTT05 teste les bornes d'erreur de l'affinement itératif pour la
!> solution calculée d'un système d'équations A*X = B, où A est une
!> matrice tridiagonale symétrique d'ordre n.
!>
!> RESLTS(1) = test de la borne d'erreur
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> Une grande valeur est retournée si ce ratio n'est pas inférieur à un.
!>
!> RESLTS(2) = résidu de la routine d'affinement itératif
!>           = le maximum de BERR / ( NZ*EPS + (*) ), où
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             et NZ = nombre max. de non-zéros dans une ligne de A, plus 1
!>

Paramètres

N 
!>          N est INTEGER
!>          Le nombre de lignes des matrices X, B et XACT, et l'ordre de
!>          la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de colonnes des matrices X, B et XACT.
!>          NRHS >= 0.
!>

D

!>          D est un tableau DOUBLE PRECISION, dimension (N)
!>          Les n éléments diagonaux de la matrice tridiagonale A.
!>

E

!>          E est un tableau DOUBLE PRECISION, dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux de la matrice tridiagonale A.
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          Les vecteurs du second membre pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>          Les vecteurs solutions calculés. Chaque vecteur est stocké comme
!>          une colonne de la matrice X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

XACT

!>          XACT est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>          Les vecteurs solutions exacts. Chaque vecteur est stocké comme
!>          une colonne de la matrice XACT.
!>

LDXACT

!>          LDXACT est INTEGER
!>          La dimension principale du tableau XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          Les bornes d'erreur avant estimées pour chaque vecteur solution
!>          X. Si XTRUE est la vraie solution, FERR borne la magnitude
!>          de la plus grande entrée dans (X - XTRUE) divisée par la
!>          magnitude de la plus grande entrée dans X.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur relative composante par composante de la solution
!>          (c'est-à-dire, le plus petit changement relatif dans une entrée
!>          de A ou B qui rend X une solution exacte).
!>

RESLTS

!>          RESLTS est un tableau DOUBLE PRECISION, dimension (2)
!>          Le maximum sur les vecteurs solutions NRHS des ratios :
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
