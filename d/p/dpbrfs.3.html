sous-routine dpbrfs (caractère uplo, entier n, entier kd, entier nrhs, double précision, dimension( ldab, * ) ab, entier ldab, double précision, dimension( ldafb, * ) afb, entier ldafb, double précision, dimension( ldb, * ) b, entier ldb, double précision, dimension( ldx, * ) x, entier ldx, double précision, dimension( * ) ferr, double précision, dimension( * ) berr, double précision, dimension( * ) work, entier, dimension( * ) iwork, entier info)

DPBRFS

Objectif :

!>
!> DPBRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est symétrique définie positive
!> et bande, et fournit des bornes d'erreur et des estimations d'erreur à
!> posteriori pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CARACTERE*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est ENTIER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est ENTIER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS est ENTIER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AB

!>          AB est un tableau DOUBLE PRECISION, dimension (LDAB,N)
!>          Le triangle supérieur ou inférieur de la matrice bande symétrique A,
!>          stocké dans les KD+1 premières lignes du tableau. La j-ième colonne
!>          de A est stockée dans la j-ième colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est ENTIER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

AFB

!>          AFB est un tableau DOUBLE PRECISION, dimension (LDAFB,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T de la matrice bande A calculée par
!>          DPBTRF, dans le même format de stockage que A (voir AB).
!>

LDAFB

!>          LDAFB est ENTIER
!>          La dimension principale du tableau AFB.  LDAFB >= KD+1.
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          La matrice des membres de droite B.
!>

LDB

!>          LDB est ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par DPBTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est ENTIER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée de la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur relative par composante à posteriori de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui rend X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (3*N)
!>

IWORK

!>          IWORK est un tableau ENTIER, dimension (N)
!>

INFO

!>          INFO est ENTIER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
