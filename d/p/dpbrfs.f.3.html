sous-routine dpbrfs (character uplo, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldafb, * ) afb, integer ldafb, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPBRFS

But :

!>
!> DPBRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est symétrique définie positive
!> et à bande, et fournit des bornes d'erreur et des estimations d'erreur
!> arrière pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : Le triangle supérieur de A est stocké ;
!>          = 'L' : Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!>

AB

!>          AB est un tableau DOUBLE PRECISION, dimension (LDAB,N)
!>          Le triangle supérieur ou inférieur de la matrice bande symétrique A,
!>          stocké dans les KD+1 premières lignes du tableau. La j-ème colonne
!>          de A est stockée dans la j-ème colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB. LDAB >= KD+1.
!>

AFB

!>          AFB est un tableau DOUBLE PRECISION, dimension (LDAFB,N)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T de la matrice bande A, telle que calculée par
!>          DPBTRF, dans le même format de stockage que A (voir AB).
!>

LDAFB

!>          LDAFB est INTEGER
!>          La dimension principale du tableau AFB. LDAFB >= KD+1.
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          La matrice des côtés droits B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

X

!>          X est un tableau DOUBLE PRECISION, dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par DPBTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X. LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude de l'élément le plus grand
!>          de (X(j) - XTRUE) divisée par la magnitude de l'élément le plus grand
!>          de X(j). L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau DOUBLE PRECISION, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          tout élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau DOUBLE PRECISION, dimension (3*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
