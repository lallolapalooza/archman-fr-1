subroutine dpbsvx (character fact, character uplo, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldafb, * ) afb, integer ldafb, character equed, double precision, dimension( * ) s, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPBSVX calcule la solution du système d'équations linéaires A * X = B pour les matrices OTHER

But :

!>
!> DPBSVX utilise la factorisation de Cholesky A = U**T*U ou A = L*L**T pour
!> calculer la solution d'un système d'équations linéaires réelles
!>    A * X = B,
!> où A est une matrice bande symétrique définie positive de dimension N×N et X
!> et B sont des matrices de dimension N×NRHS.
!>
!> Des bornes d'erreur sur la solution et une estimation de condition sont
!> également fournies.
!>

Description :

!>
!> Les étapes suivantes sont effectuées :
!>
!> 1. Si FACT = 'E', des facteurs d'échelle réels sont calculés pour
!>    équilibrer le système :
!>       diag(S) * A * diag(S) * inv(diag(S)) * X = diag(S) * B
!>    Que le système soit équilibré ou non dépend de la mise à l'échelle
!>    de la matrice A, mais si l'équilibrage est utilisé, A est
!>    remplacée par diag(S)*A*diag(S) et B par diag(S)*B.
!>
!> 2. Si FACT = 'N' ou 'E', la décomposition de Cholesky est utilisée pour
!>    factoriser la matrice A (après équilibrage si FACT = 'E') comme
!>       A = U**T * U,  si UPLO = 'U', ou
!>       A = L * L**T,  si UPLO = 'L',
!>    où U est une matrice bande triangulaire supérieure, et L est une
!>    matrice bande triangulaire inférieure.
!>
!> 3. Si le mineur principal principal d'ordre i n'est pas positif,
!>    la routine retourne avec INFO = i. Sinon, la forme factorisée
!>    de A est utilisée pour estimer le conditionnement de la matrice
!>    A. Si le réciproque du conditionnement est inférieur à la
!>    précision machine, INFO = N+1 est retourné comme avertissement, mais la
!>    routine continue quand même pour résoudre X et calculer les bornes
!>    d'erreur comme décrit ci-dessous.
!>
!> 4. Le système d'équations est résolu pour X en utilisant la forme
!>    factorisée de A.
!>
!> 5. Un raffinement itératif est appliqué pour améliorer la matrice
!>    solution calculée et calculer les bornes d'erreur et les
!>    estimations d'erreur arrière pour celle-ci.
!>
!> 6. Si l'équilibrage a été utilisé, la matrice X est multipliée à
!>    gauche par diag(S) afin qu'elle résolve le système original
!>    avant équilibrage.
!>

Paramètres

FACT 
!>          FACT est CHARACTER*1
!>          Spécifie si la forme factorisée de la matrice A est
!>          fournée à l'entrée, et si ce n'est pas le cas, si la matrice A
!>          doit être équilibrée avant d'être factorisée.
!>          = 'F' : À l'entrée, AFB contient la forme factorisée de A.
!>                  Si EQUED = 'Y', la matrice A a été équilibrée
!>                  avec les facteurs d'échelle donnés par S. AB et AFB ne seront
!>                  pas modifiés.
!>          = 'N' : La matrice A sera copiée dans AFB et factorisée.
!>          = 'E' : La matrice A sera équilibrée si nécessaire, puis
!>                  copiée dans AFB et factorisée.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U' : Le triangle supérieur de A est stocké ;
!>          = 'L' : Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          Le nombre d'équations linéaires, c'est-à-dire l'ordre de la
!>          matrice A. N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!>

AB

!>          AB est un tableau DOUBLE PRECISION, dimension (LDAB,N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          bande symétrique A, stocké dans les premières KD+1 lignes du
!>          tableau, sauf si FACT = 'F' et EQUED = 'Y', auquel cas A doit
!>          contenir la matrice équilibrée diag(S)*A*diag(S). La j-ième
!>          colonne de A est stockée dans la j-ième colonne du tableau AB comme
!>          suit :
!>          si UPLO = 'U', AB(KD+1+i-j,j) = A(i,j) pour max(1,j-KD)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(N,j+KD).
!>          Voir ci-dessous pour plus de détails.
!>
!>          À la sortie, si FACT = 'E' et EQUED = 'Y', A est remplacée par
!>          diag(S)*A*diag(S).
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau A. LDAB >= KD+1.
!>

AFB

!>          AFB est un tableau DOUBLE PRECISION, dimension (LDAFB,N)
!>          Si FACT = 'F', alors AFB est un argument d'entrée et à l'entrée
!>          contient le facteur triangulaire U ou L de la factorisation
!>          de Cholesky A = U**T*U ou A = L*L**T de la matrice bande
!>          A, dans le même format de stockage que A (voir AB). Si EQUED = 'Y',
!>          alors AFB est la forme factorisée de la matrice équilibrée A.
!>
!>          Si FACT = 'N', alors AFB est un argument de sortie et à la sortie
!>          retourne le facteur triangulaire U ou L de la factorisation
!>          de Cholesky A = U**T*U ou A = L*L**T.
!>
!>          Si FACT = 'E', alors AFB est un argument de sortie et à la sortie
!>          retourne le facteur triangulaire U ou L de la factorisation
!>          de Cholesky A = U**T*U ou A = L*L**T de la matrice
!>          équilibrée A (voir la description de A pour la forme de la
!>          matrice équilibrée).
!>

LDAFB

!>          LDAFB est INTEGER
!>          La dimension principale du tableau AFB. LDAFB >= KD+1.
!>

EQUED

!>          EQUED est CHARACTER*1
!>          Spécifie la forme d'équilibrage qui a été effectuée.
!>          = 'N' : Pas d'équilibrage (toujours vrai si FACT = 'N').
!>          = 'Y' : Un équilibrage a été effectué, c'est-à-dire que A a été
!>                  remplacée par diag(S) * A * diag(S).
!>          EQUED est un argument d'entrée si FACT = 'F' ; sinon, c'est un
!>          argument de sortie.
!>

S

!>          S est un tableau DOUBLE PRECISION, dimension (N)
!>          Les facteurs d'échelle pour A ; non accédé si EQUED = 'N'. S est
!>          un argument d'entrée si FACT = 'F' ; sinon, S est un argument
!>          de sortie. Si FACT = 'F' et EQUED = 'Y', chaque élément de S
!>          doit être positif.
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          À l'entrée, la matrice des seconds membres B de dimension N×NRHS.
!>          À la sortie, si EQUED = 'N', B n'est pas modifiée ; si EQUED = 'Y',
!>          B est remplacée par diag(S) * B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!> re>

X


!>          X est un tableau de type DOUBLE PRECISION, de dimension (LDX,NRHS)
!>          Si INFO = 0 ou INFO = N+1, la matrice de solution X de dimension N par NRHS du
!>          système d'équations original.  Notez que si EQUED = 'Y',
!>          A et B sont modifiés à la sortie, et la solution du
!>          système équilibré est inv(diag(S))*X.
!> 


LDX


!>          LDX est de type INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!> 


RCOND


!>          RCOND est de type DOUBLE PRECISION
!>          L'estimation du nombre de condition réciproque de la matrice
!>          A après équilibrage (si effectué).  Si RCOND est inférieur à la
!>          précision machine (en particulier, si RCOND = 0), la matrice
!>          est singulière à la précision de travail.  Cette condition est
!>          indiquée par un code de retour INFO > 0.
!> 


FERR


!>          FERR est un tableau de type DOUBLE PRECISION, de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice de solution X).
!>          Si XTRUE est la solution réelle correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément dans (X(j) - XTRUE) divisée par la magnitude du
!>          plus grand élément dans X(j).  L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!> 


BERR


!>          BERR est un tableau de type DOUBLE PRECISION, de dimension (NRHS)
!>          L'erreur relative arrière composante par composante de chaque vecteur
!>          solution X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui rend X(j) une solution exacte).
!> 


WORK


!>          WORK est un tableau de type DOUBLE PRECISION, de dimension (3*N)
!> 


IWORK


!>          IWORK est un tableau de type INTEGER, de dimension (N)
!> 


INFO


!>          INFO est de type INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  si INFO = i, et i est
!>                <= N:  le mineur principal de tête d'ordre i de A
!>                       n'est pas positif, donc la factorisation n'a pas pu
!>                       être complétée, et la solution n'a pas été
!>                       calculée. RCOND = 0 est retourné.
!>                = N+1: U est non singulier, mais RCOND est inférieur à la précision
!>                       machine, ce qui signifie que la matrice est singulière
!>                       à la précision de travail.  Néanmoins, la
!>                       solution et les bornes d'erreur sont calculées car
!>                       il existe un certain nombre de situations où la
!>                       solution calculée peut être plus précise que la
!>                       valeur de RCOND ne le suggérerait.
!>

!>  N = 6, KD = 2, et UPLO = 'U' :
!>
!>  Stockage bidimensionnel de la matrice symétrique A :
!>
!>     a11  a12  a13
!>          a22  a23  a24
!>               a33  a34  a35
!>                    a44  a45  a46
!>                         a55  a56
!>     (aij=conjg(aji))         a66
!>
!>  Stockage en bande du triangle supérieur de A :
!>
!>      *    *   a13  a24  a35  a46
!>      *   a12  a23  a34  a45  a56
!>     a11  a22  a33  a44  a55  a66
!>
!>  De même, si UPLO = 'L', le format de A est le suivant :
!>
!>     a11  a22  a33  a44  a55  a66
!>     a21  a32  a43  a54  a65   *
!>     a31  a42  a53  a64   *    *
!>
!>  Les éléments de tableau marqués * ne sont pas utilisés par la routine.
!>
