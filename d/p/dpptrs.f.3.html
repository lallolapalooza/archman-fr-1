subroutine dpptrs (character uplo, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DPPTRS

But :

!>
!> DPPTRS résout un système d'équations linéaires A*X = B avec une matrice A
!> définie positive symétrique en stockage compact en utilisant la
!> factorisation de Cholesky A = U**T*U ou A = L*L**T calculée par DPPTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : Le triangle supérieur de A est stocké ;
!>          = 'L' : Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

AP

!>          AP est un tableau DOUBLE PRECISION, dimension (N*(N+1)/2)
!>          Le facteur triangulaire U ou L de la factorisation de Cholesky
!>          A = U**T*U ou A = L*L**T, stocké en colonne dans un tableau
!>          linéaire. La j-ième colonne de U ou L est stockée dans le
!>          tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = U(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = L(i,j) pour j<=i<=n.
!>

B

!>          B est un tableau DOUBLE PRECISION, dimension (LDB,NRHS)
!>          À l'entrée, la matrice B des seconds membres.
!>          À la sortie, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0 : succès
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
