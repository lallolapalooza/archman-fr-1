DESCRIPTION

Ces fonctions servent à étendre et/ou modifier le comportement des fonctions printf(3) de la famille.

registro_de_especificadores_de_conversión

Cette fonction enregistre un spécificateur de conversion personnalisé pour les fonctions printf(3) de la famille.

espec
Le caractère qui sera utilisé comme spécificateur de conversion dans la chaîne de format.
func
La fonction de rappel qui sera exécutée par les fonctions printf(3) de la famille pour formater les arguments d'entrée dans le flux de sortie stream.
stream
Le flux de sortie où le résultat du formatage sera imprimé. Ce flux est représenté de manière transparente même dans les cas de fonctions qui écrivent dans une chaîne.
info
La structure contenant les informations de contexte, y compris les modificateurs spécifiés dans la chaîne de format. Elle contient les mêmes informations que arginfo.
args
Tableau de pointeurs vers les arguments qui sont passés à la fonction de type printf(3).
arginfo
La fonction de rappel qui sera exécutée par les fonctions printf(3) de la famille pour savoir combien d'arguments doivent être analysés pour le spécificateur de conversion personnalisé et aussi leurs types.
info
La structure contenant les informations de contexte, y compris les modificateurs spécifiés dans la chaîne de format. Elle contient les mêmes informations que func.
n
Le nombre d'arguments restants à analyser.
argtypes
Ce tableau doit être défini pour spécifier le type de chacun des arguments à analyser. Chaque élément du tableau représente l'un des arguments à analyser, dans le même ordre qu'ils sont passés à la fonction de type printf(3). Chaque élément doit être défini à un type de base (PA_) de l'énumération ci-dessus, ou un type personnalisé, et optionnellement combiné avec un modificateur de longueur (PA_FLAG_*).

Le type est déterminé en utilisant l'une des constantes suivantes :

PA_INT
int.
PA_CHAR
int, converti en char.
PA_WCHAR
wchar_t.
PA_STRING
const char *, une chaîne de char terminée par \0.
PA_WSTRING
const wchar_t *, une chaîne de wchar_t terminée par \0.
PA_POINTER
void *.
PA_FLOAT
float.
PA_DOUBLE
double.
PA_LAST
TODO.
size
Pour les types définis par l'utilisateur, la taille du type (en octets) doit également être spécifiée via ce tableau. Sinon, laissez ce tableau inutilisé.

arginfo est appelé avant func, préparant certaines informations nécessaires pour appeler func.

Callbacks

La fonction de rappel de type función_de_tamaño_de_arginfo doit retourner le nombre d'arguments à analyser pour ce spécificateur. Elle passe également des informations sur le type de ces arguments à l'appel via argtypes. En cas d'erreur, elle doit retourner -1.

HISTORY

register_printf_function(3) est une ancienne fonction similaire à register_printf_specifier(), et est maintenant dépréciée. Cette fonction ne peut pas gérer les types définis par l'utilisateur.

register_printf_specifier() remplace register_printf_function(3).

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* Rembourrage avec ' ' (aligné à droite) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Préfixe "0b"/"0B" */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Rembourrage avec '0' */
    if (pad.ch == '0') {
        tmp = b_pad_zeros(stream, info, min_len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Afficher le nombre (y compris les zéros de tête pour remplir la précision) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    /* Rembourrage avec ' ' (aligné à gauche) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* short et char sont tous deux promus en int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* Afficher les zéros de tête pour remplir la précision */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* Afficher le nombre */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
