type core_type_desc =
| Ptyp_any (* _


*)
| Ptyp_var of string
(* Variables de type telles que 'a


*)
| Ptyp_arrow of Asttypes.arg_label * core_type * core_type
(* Ptyp_arrow(lbl, T1, T2) représente :

- T1 -> T2 quand lbl est Asttypes.arg_label.Nolabel ,

- ~l:T1 -> T2 quand lbl est Asttypes.arg_label.Labelled ,

- ?l:T1 -> T2 quand lbl est Asttypes.arg_label.Optional .


*)
| Ptyp_tuple of (string option * core_type) list
(* Ptyp_tuple(tl) représente un type produit :

- T1 * ... * Tn quand tl est (None, T1); ...; (None, Tn)

- L1:T1 * ... * Ln:Tn quand tl est (Some L1, T1); ...; (Some Ln, Tn)

- Un mélange, par exemple L1:T1 * T2 quand tl est (Some L1, T1); (None, T2)

Invariance : n >= 2 .
*)
| Ptyp_constr of Longident.t Asttypes.loc * core_type list
(* Ptyp_constr(lident, l) représente :

- tconstr quand l=[] ,

- T tconstr quand l=[T] ,

- (T1, ..., Tn) tconstr quand l=[T1 ; ... ; Tn] .


*)
| Ptyp_object of list of object_field * Asttypes.closed_flag
(* Ptyp_object([ l1:T1; ...; ln:Tn ], flag) représente :

- < l1:T1; ...; ln:Tn > quand flag est Asttypes.closed_flag.Closed ,

- < l1:T1; ...; ln:Tn; .. > quand flag est Asttypes.closed_flag.Open .


*)
| Ptyp_class of Longident.t Asttypes.loc * core_type list
(* Ptyp_class(tconstr, l) représente :

- #tconstr quand l=[] ,

- T #tconstr quand l=[T] ,

- (T1, ..., Tn) #tconstr quand l=[T1 ; ... ; Tn] .


*)
| Ptyp_alias of core_type * string Asttypes.loc
(* T as 'a .
*)
| Ptyp_variant of list of variant_field * Asttypes.closed_flag * label list
(* Ptyp_variant([`A;`B], flag, labels) représente :

- [ `A|`B ] quand flag est Asttypes.closed_flag.Closed , et labels est None ,

- [> `A|`B ] quand flag est Asttypes.closed_flag.Open , et labels est None ,

- [< `A|`B ] quand flag est Asttypes.closed_flag.Closed , et labels est Some [] ,

- [< `A|`B > `X `Y ] quand flag est Asttypes.closed_flag.Closed , et labels est Some ["X";"Y"] .


*)
| Ptyp_poly of string Asttypes.loc list * core_type
(* 'a1 ... 'an. T

Peut apparaître uniquement dans le contexte suivant :

- Comme Parsetree.core_type d'un nœud Parsetree.pattern_desc.Ppat_constraint correspondant à une contrainte sur un 'let' lié :

let x : 'a1 ... 'an. T = e ...

- Comme Parsetree.class_field_kind.Cfk_virtual pour les méthodes (pas les valeurs).

- Comme Parsetree.core_type d'un nœud Parsetree.class_type_field_desc.Pctf_method .

- Comme Parsetree.value_description.pval_type d'un nœud Parsetree.value_description .

type pattern_desc =
| Ppat_any (* Le motif _ .
*)
| Ppat_var de string Asttypes.loc
(* Un motif variable tel que x


*)
| Ppat_alias de pattern * string Asttypes.loc
(* Un motif alias tel que P comme 'a


*)
| Ppat_constant de constant
(* Motifs tels que 1 , 'a' , "vrai" , 1.0 , 1l , 1L , 1n


*)
| Ppat_interval de constant * constant
(* Motifs tels que 'a'..'z' .

D'autres formes d'intervalles sont reconnues par l'analyseur syntaxique mais rejetées par le vérificateur de types.
*)
| Ppat_tuple de (string option * pattern) list * Asttypes.closed_flag
(* Ppat_tuple(pl, Closed) représente

- (P1, ..., Pn) quand pl est (None, P1); ...; (None, Pn)

- (~L1:P1, ..., ~Ln:Pn) quand pl est (Some L1, P1); ...; (Some Ln, Pn)

-Un mélange, par exemple (~L1:P1, P2) quand pl est (Some L1, P1); (None, P2)

Ppat_tuple(pl, Open) est similaire, mais indique que le motif se termine en outre par un .. .

Invariant :

-Si Fermé, n >= 2 .

-Si Ouvert, n >= 1 .


*)
| Ppat_construct de Longident.t Asttypes.loc * (string Asttypes.loc list * pattern) option
(* Ppat_construct(C, args) représente :

- C quand args est None ,

- C P quand args est Some ([], P)

- C (P1, ..., Pn) quand args est Some ([], Ppat_tuple [P1; ...; Pn])

- C (type a b) P quand args est Some ([a; b], P)


*)
| Ppat_variant de Asttypes.label * pattern option
(* Ppat_variant(`A, pat) représente :

- `A quand pat est None ,

- `A P quand pat est Some P


*)
| Ppat_record de (Longident.t Asttypes.loc * pattern) list * Asttypes.closed_flag
(* Ppat_record([(l1, P1) ; ... ; (ln, Pn)], flag) représente :

- { l1=P1; ...; ln=Pn } quand flag est Asttypes.closed_flag.Closed

- { l1=P1; ...; ln=Pn; _} quand flag est Asttypes.closed_flag.Open

Invariant : n > 0


*)
| Ppat_array de pattern list
(* Motif [| P1; ...; Pn |]


*)
| Ppat_or de pattern * pattern
(* Motif P1 | P2


*)
| Ppat_constraint de pattern * core_type
(* Motif (P : T)


*)
| Ppat_type de Longident.t Asttypes.loc
(* Motif #tconst


*)
| Ppat_lazy de pattern
(* Motif lazy P


*)
| Ppat_unpack de string option Asttypes.loc
(* Ppat_unpack(s) représente :

- (module P) quand s est Some "P"

- (module _) quand s est None

Note : (module P : S) est représenté comme Ppat_constraint(Ppat_unpack(Some "P"), Ptyp_package S)


*)
| Ppat_exception de pattern
(* Motif exception P


*)
| Ppat_effect de pattern * pattern
| Ppat_extension de extension
(* Motif [%id]


*)
| Ppat_open de Longident.t Asttypes.loc * pattern
(* Motif M.(P)


*)

type expression_desc =
| Pexp_ident de Longident.t Asttypes.loc
(* Identificateurs tels que x et M.x


*)
| Pexp_constant de constant
(* Expressions constantes telles que 1 , 'a' , "vrai" , 1.0 , 1l , 1L , 1n


*)
| Pexp_let de Asttypes.rec_flag * value_binding list * expression
(* Pexp_let(flag, [(P1,E1) ; ... ; (Pn,En)], E) représente :

- let P1 = E1 et ... et Pn = EN dans E quand flag est Asttypes.rec_flag.Nonrecursive ,

- let rec P1 = E1 et ... et Pn = EN dans E quand flag est Asttypes.rec_flag.Recursive .


*)
| Pexp_function de function_param list * type_constraint option * function_body
(* Pexp_function ([P1; ...; Pn], C, body) représente tout construct impliquant fun ou function , y compris :

- fun P1 ... Pn -> E quand body = Pfunction_body E

- fun P1 ... Pn -> function p1 -> e1 | ... | pm -> em quand body = Pfunction_cases [ p1 -> e1; ...; pm -> em ]

C représente une contrainte de type ou une coercition placée immédiatement avant la flèche, par exemple fun P1 ... Pn : ty -> ... quand C = Some (Pconstraint ty) .

Une fonction doit avoir des paramètres : dans Pexp_function (params, _, body) , si params ne contient pas de Pparam_val _ , body doit être Pfunction_cases _ .
*)
| Pexp_apply de expression * (Asttypes.arg_label * expression) list
(* Pexp_apply(E0, [(l1, E1) ; ... ; (ln, En)]) représente E0 ~l1:E1 ... ~ln:En

li peut être Asttypes.arg_label.Nolabel (argument non étiqueté), Asttypes.arg_label.Labelled (arguments étiquetés) ou Asttypes.arg_label.Optional (argument optionnel).

Invariant : n > 0


*)
| Pexp_match de expression * case list
(* match E0 avec P1 -> E1 | ... | Pn -> En


*)
| Pexp_try de expression * case list
(* try E0 avec P1 -> E1 | ... | Pn -> En


*)
| Pexp_tuple de (string option * expression) list
(* Pexp_tuple(el) représente

- (E1, ..., En) quand el est (None, E1); ...; (None, En)

- (~L1:E1, ..., ~Ln:En) quand el est (Some L1, E1); ...; (Some Ln, En)

-Un mélange, par exemple (~L1:E1, E2) quand el est (Some L1, E1); (None, E2)

Invariant : n >= 2


*)
| Pexp_construct de Longident.t Asttypes.loc * expression option
(* Pexp_construct(C, exp) représente :

- C quand exp est None ,

- C E quand exp est Some E ,

- C (E1, ..., En) quand exp est Some (Pexp_tuple[E1;...;En])


*)
| Pexp_variant de Asttypes.label * expression option
(* Pexp_variant(`A, exp) représente

- `A quand exp est None

- `A E quand exp est Some E


*)
| Pexp_record de (Longident.t Asttypes.loc * expression) list * expression option
(* Pexp_record([(l1,P1) ; ... ; (ln,Pn)], exp0) représente

- { l1=P1; ...; ln=Pn } quand exp0 est None

- { E0 avec l1=P1; ...; ln=Pn } quand exp0 est Some E0

Invariant : n > 0


*)
| Pexp_field de expression * Longident.t Asttypes.loc
(* E.l

| Pexp_setfield de expression * Longident.t Asttypes.loc * expression
(* E1.l <- E2


*)
| Pexp_array de expression list
(* [| E1; ...; En |]


*)
| Pexp_ifthenelse de expression * expression * expression option
(* if E1 then E2 else E3


*)
| Pexp_sequence de expression * expression
(* E1; E2
