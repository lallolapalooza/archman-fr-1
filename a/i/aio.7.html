DESCRIPTIO

N

L'interface d'E/S asynchrone POSIX (AIO) permet aux applications de lancer une ou plusieurs opérations d'E/S qui sont effectuées de manière asynchrone (c'est-à-dire en arrière-plan). L'application peut choisir d'être notifiée de l'achèvement de l'opération d'E/S de plusieurs manières : par la livraison d'un signal, par l'instanciation d'un thread, ou sans aucune

notification.

L'interface POSIX AIO se compose des fonctions suivantes

:
aio_read(
3)
Mettre en file d'attente une requête de lecture. C'est l'équivalent asynchrone de read(
2).
aio_write(
3)
Mettre en file d'attente une requête d'écriture. C'est l'équivalent asynchrone de write(
2).
aio_fsync(
3)
Mettre en file d'attente une requête de synchronisation pour les opérations d'E/S sur un descripteur de fichier. C'est l'équivalent asynchrone de fsync(2) et fdatasync
(2).
aio_error(
3)
Obtenir l'état d'erreur d'une requête d'E/S mise en file
d'attente.
aio_return(
3)
Obtenir l'état de retour d'une requête d'E/S
terminée.
aio_suspend(
3)
Suspendre l'appelant jusqu'à ce qu'une ou plusieurs requêtes d'E/S d'un ensemble spécifié soient
terminées.
aio_cancel(
3)
Tenter d'annuler les requêtes d'E/S en attente sur un descripteur de fichier
spécifié.
lio_listio(
3)
Mettre en file d'attente plusie

n unique appel de fonction.

La structure aiocb (« asynchronous I/O control block ») définit les paramètres qui contrôlent une opération d'E/S. Un argument de ce type est utilisé avec toutes les fonctions listées ci-dessus. Cette structure a la forme suivante

: 


#include <aiocb.h>
struct aiocb {
    /* L'ordre de ces champs dépend de l'implémentation */
    int             aio_fildes;     /* Descripteur de fichier */
    off_t           aio_offset;     /* Décalage dans le fichier */
    volatile void  *aio_buf;        /* Emplacement du tampon */
    size_t          aio_nbytes;     /* Longueur du transfert */
    int             aio_reqprio;    /* Priorité de la requête */
    struct sigevent aio_sigevent;   /* Méthode de notification */
    int             aio_lio_opcode; /* Opération à effectuer ;
                                       lio_listio() uniquement */
    /* Divers champs internes à l'implémentation non montrés */
};
/* Codes d'opération pour 'aio_lio_opcode' : */
enum { LIO_READ, LIO_WRITE, LIO_NOP }
;

NOTE

S

Il est recommandé de mettre le tampon du bloc de contrôle à zéro avant de l'utiliser (voir memset(3)). Le tampon du bloc de contrôle et le tampon pointé par aio_buf ne doivent pas être modifiés pendant que l'opération d'E/S est en cours. Ces tampons doivent rester valides jusqu'à ce que l'opération d'E/S soit

terminée.

Des opérations de lecture ou d'écriture asynchrones simultanées utilisant la même structure aiocb produisent des résultats

indéfinis.

L'implémentation POSIX AIO actuelle de Linux est fournie en espace utilisateur par glibc. Celle-ci présente un certain nombre de limitations, notamment le fait que le maintien de plusieurs threads pour effectuer des opérations d'E/S est coûteux et ne se scale pas bien. Des travaux sont en cours depuis un certain temps sur une implémentation d'E/S asynchrone basée sur une machine à états dans le noyau (voir io_submit(2), io_setup(2), io_cancel(2), io_destroy(2), io_getevents(2)), mais cette implémentation n'a pas encore atteint la maturité permettant de r

éimplémenter complètement l'implémentation PO
SIX AIO à l'aide des appels système du noyau.

EXEMPLE

S

Le programme ci-dessous ouvre chacun des fichiers nommés dans ses arguments de ligne de commande et met en file d'attente une requête sur le descripteur de fichier résultant en utilisant aio_read(3). Le programme boucle ensuite, surveillant périodiquement chacune des opérations d'E/S encore en cours en utilisant aio_error(3). Chacune des requêtes d'E/S est configurée pour fournir une notification par la livraison d'un signal. Après que toutes les requêtes d'E/S ont été terminées, le programme récupère leur état en utilisant aio_retur

n(3).

Le signal SIGQUIT (généré en tapant control-\) fait demander au programme l'annulation de chacune des requêtes en attente en utilisant aio_cance

l(3).

.

Programme

source 
#include <err.h>
#include <fcntl.h>
#include <stdlib.h>
#include <unistd.h>
#include <stdio.h>
#include <errno.h>
#include <aio.h>
#include <signal.h>
#define BUF_SIZE 20     /* Taille des tampons pour les opérations de lecture */
struct ioRequest {      /* Structure définie par l'application pour suivre
                           les requêtes d'E/S */
    int           reqNum;
    int           status;
    struct aiocb *aiocbp;
};
static volatile sig_atomic_t gotSIGQUIT = 0;
                        /* À la réception de SIGQUIT, nous tentons
                           d'annuler toutes les requêtes d'E/S en attente */
static void             /* Gestionnaire pour SIGQUIT */
quitHandler(int sig)
{
    gotSIGQUIT = 1;
}
#define IO_SIGNAL SIGUSR1   /* Signal utilisé pour notifier la fin des E/S */
static void                 /* Gestionnaire du signal de fin des E/S */
aioSigHandler(int sig, siginfo_t *si, void *ucontext)
{
    if (si->si_code == SI_ASYNCIO) {
        write(STDOUT_FILENO, "I/O completion signal received\n", 31);
        /* La structure ioRequest correspondante serait disponible sous la
               forme :
               struct ioRequest *ioReq = si->si_value.sival_ptr;
           et le descripteur de fichier serait alors accessible via
               ioReq->aiocbp->aio_fildes */
    }
}
int
main(int argc, char *argv[])
{
    struct sigaction sa;
    int s;
    int numReqs;        /* Nombre total de requêtes d'E/S mises en file d'attente */
    int openReqs;       /* Nombre de requêtes d'E/S encore en cours */
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <pathname> <pathname>...\n",
                argv[0]);
        exit(EXIT_FAILURE);
    }
    numReqs = argc - 1;
    /* Allouons nos tableaux.  */
    struct ioRequest *ioList = calloc(numReqs, sizeof(*ioList));
    if (ioList == NULL)
        err(EXIT_FAILURE, "calloc");
    struct aiocb *aiocbList = calloc(numReqs, sizeof(*aiocbList));
    if (aiocbList == NULL)
        err(EXIT_FAILURE, "calloc");
    /* Établissons les gestionnaires pour SIGQUIT et le signal de fin des E/S.  */
    sa.sa_flags = SA_RESTART;
    sigemptyset(&sa.sa_mask);
    sa.sa_handler = quitHandler;
    if (sigaction(SIGQUIT, &sa, NULL) == -1)
        err(EXIT_FAILURE, "sigaction");
    sa.sa_flags = SA_RESTART | SA_SIGINFO;
    sa.sa_sigaction = aioSigHandler;
    if (sigaction(IO_SIGNAL, &sa, NULL) == -1)
        err(EXIT_FAILURE, "sigaction");
    /* Ouvrons chaque fichier spécifié sur la ligne de commande et mettons
       en file d'attente une requête de lecture sur le descripteur de fichier
       résultant.  */
    for (size_t j = 0; j < numReqs; j++) {
        ioList[j].reqNum = j;
        ioList[j].status = EINPROGRESS;
        ioList[j].aiocbp = &aiocbList[j];
        ioList[j].aiocbp->aio_fildes = open(argv[j + 1], O_RDONLY);
        if (ioList[j].aiocbp->aio_fildes == -1)
            err(EXIT_FAILURE, "open");
        printf("opened %s on descriptor %d\n", argv[j + 1],
                ioList[j].aiocbp->aio_fildes);
        ioList[j].aiocbp->aio_buf = malloc(BUF_SIZE);
        if (ioList[j].aiocbp->aio_buf == NULL)
            err(EXIT_FAILURE, "malloc");
        ioList[j].aiocbp->aio_nbytes = BUF_SIZE;
        ioList[j].aiocbp->aio_reqprio = 0;
        ioList[j].aiocbp->aio_offset = 0;
        ioList[j].aiocbp->aio_sigevent.sigev_notify = SIGEV_SIGNAL;
        ioList[j].aiocbp->aio_sigevent.sigev_signo = IO_SIGNAL;
        ioList[j].aiocbp->aio_sigevent.sigev_value.sival_ptr =
                                &ioList[j];
        s = aio_read(ioList[j].aiocbp);
        if (s == -1)
            err(EXIT_FAILURE, "aio_read");
    }
    openReqs = numReqs;
    /* Boucle, surveillant l'état des requêtes d'E/S.  */
    while (openReqs > 0) {
        sleep(3);       /* Délai entre chaque étape de surveillance */
        if (gotSIGQUIT) {
            /* À la réception de SIGQUIT, tentons d'annuler chacune des
               requêtes d'E/S en attente et affichons l'état renvoyé par
               les requêtes d'annulation.  */
            printf("got SIGQUIT; canceling I/O requests: \n");
            for (size_t j = 0; j < numReqs; j++) {
                if (ioList[j].status == EINPROGRESS) {
                    printf("    Request %zu on descriptor %d:", j,
                            ioList[j].aiocbp->aio_fildes);
                    s = aio_cancel(ioList[j].aiocbp->aio_fildes,
                            ioList[j].aiocbp);
                    if (s == AIO_CANCELED)
                        printf("I/O canceled\n");
                    else if (s == AIO_NOTCANCELED)
                        printf("I/O not canceled\n");
                    else if (s == AIO_ALLDONE)
                        printf("I/O all done\n");
                    else
                        perror("aio_cancel");
                }
            }
            gotSIGQUIT = 0;
        }
        /* Vérifions l'état de chaque requête d'E/S encore en cours.  */
        printf("aio_error():\n");
        for (size_t j = 0; j < numReqs; j++) {
            if (ioList[j].status == EINPROGRESS) {
                printf("    for request %zu (descriptor %d): ",
                        j, ioList[j].aiocbp->aio_fildes);
                ioList[j].status = aio_error(ioList[j].aiocbp);
                switch (ioList[j].status) {
                case 0:
                    printf("I/O succeeded\n");
                    break;
                case EINPROGRESS:
                    printf("In progress\n");
                    break;
                case ECANCELED:
                    printf("Canceled\n");
                    break;
                default:
                    perror("aio_error");
                    break;
                }
                if (ioList[j].status != EINPROGRESS)
                    openReqs--;
            }
        }
    }
    printf("All I/O requests completed\n");
    /* Vérifions l'état renvoyé pour toutes les requêtes d'E/S.  */
    printf("aio_return():\n");
    for (size_t j = 0; j < numReqs; j++) {
        ssize_t s;
        s = aio_return(ioList[j].aiocbp);
        printf("    for request %zu (descriptor %d): %zd\n",
                j, ioList[j].aiocbp->aio_fildes, s);
    }
    exit(EXIT_SUCCESS)
;
}
