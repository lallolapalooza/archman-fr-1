RETOUR

Toutes les fonctions de signature renvoient la taille de la signature en octets en cas de succès et zéro en cas

d'échec.

Toutes les fonctions de vérification renvoient 1 si la signature est valide et 0 si la

vérification de la signature échoue. Si la signature n'a pas pu être vérifiée du tout pa
rce qu'elle était mal formée ou qu'une autre erreur s'est produite, alors -1 est renvoyé.

EXEMPLE

S

Dans l'exemple suivant, un objet 'MyObject' est signé en utilisant la clé contenue dans un EVP_MD_CTX. La signature est écrite dans MyObject.signature. L'objet est ensuite produit au format DER puis rechargé et

vérifié. 
 #include <openssl/x509.h>
 #include <openssl/asn1t.h>
 /* Un objet utilisé pour stocker les champs de données ASN1 qui seront signés */
 typedef struct MySignInfoObject_st
 {
     ASN1_INTEGER *version;
     X509_ALGOR sig_alg;
 } MySignInfoObject;
 DECLARE_ASN1_FUNCTIONS(MySignInfoObject)
 /*
  * Un objet de niveau supérieur contenant les champs ASN1, l'algorithme de signature et
  * la signature de sortie.
  */
 typedef struct MyObject_st
 {
     MySignInfoObject info;
     X509_ALGOR sig_alg;
     ASN1_BIT_STRING *signature;
 } MyObject;
 DECLARE_ASN1_FUNCTIONS(MyObject)
 /* La définition ASN1 de MySignInfoObject */
 ASN1_SEQUENCE_cb(MySignInfoObject, NULL) = {
     ASN1_SIMPLE(MySignInfoObject, version, ASN1_INTEGER)
     ASN1_EMBED(MySignInfoObject, sig_alg, X509_ALGOR),
 } ASN1_SEQUENCE_END_cb(MySignInfoObject, MySignInfoObject)
 /* Fonctions new, free, d2i & i2d pour MySignInfoObject */
 IMPLEMENT_ASN1_FUNCTIONS(MySignInfoObject)
 /* La définition ASN1 de MyObject */
 ASN1_SEQUENCE_cb(MyObject, NULL) = {
     ASN1_EMBED(MyObject, info, MySignInfoObject),
     ASN1_EMBED(MyObject, sig_alg, X509_ALGOR),
     ASN1_SIMPLE(MyObject, signature, ASN1_BIT_STRING)
 } ASN1_SEQUENCE_END_cb(MyObject, MyObject)
 /* Fonctions new, free, d2i & i2d pour MyObject */
 IMPLEMENT_ASN1_FUNCTIONS(MyObject)
 int test_asn1_item_sign_verify(const char *mdname, EVP_PKEY *pkey, long version)
 {
    int ret = 0;
    unsigned char *obj_der = NULL;
    const unsigned char *p = NULL;
    MyObject *obj = NULL, *loaded_obj = NULL;
    const ASN1_ITEM *it = ASN1_ITEM_rptr(MySignInfoObject);
    EVP_MD_CTX *sctx = NULL, *vctx = NULL;
    int len;
    /* Créer MyObject et définir sa version */
    obj = MyObject_new();
    if (obj == NULL)
        goto err;
    if (!ASN1_INTEGER_set(obj->info.version, version))
        goto err;
    /* Définir la clé et l'empreinte utilisées pour la signature */
    sctx = EVP_MD_CTX_new();
    if (sctx == NULL
        || !EVP_DigestSignInit_ex(sctx, NULL, mdname, NULL, NULL, pkey))
        goto err;
    /*
     * it contient la correspondance entre les données ASN.1 et un objet MySignInfoObject
     * obj->info est l'objet 'MySignInfoObject' qui sera
     *   converti en données DER puis signé.
     * obj->signature contiendra la signature de sortie.
     * obj->sig_alg est rempli avec l'identifiant de l'algorithme de signature de la clé privée.
     * obj->info.sig_alg est une autre copie de l'identifiant de l'algorithme de signature qui se trouve
     * dans MyObject.
     */
    len = ASN1_item_sign_ctx(it, &obj->sig_alg, &obj->info.sig_alg,
                             obj->signature, &obj->info, sctx);
    if (len <= 0
        || X509_ALGOR_cmp(&obj->sig_alg, &obj->info.sig_alg) != 0)
        goto err;
    /* Produire MyObject sous forme der */
    len = i2d_MyObject(obj, &obj_der);
    if (len <= 0)
        goto err;
    /* Définir la clé et l'empreinte utilisées pour la vérification */
    vctx = EVP_MD_CTX_new();
    if (vctx == NULL
        || !EVP_DigestVerifyInit_ex(vctx, NULL, mdname, NULL, NULL, pkey))
        goto err;
    /* Recharger les données der dans un objet */
    p = obj_der;
    loaded_obj = d2i_MyObject(NULL, &p, len);
    if (loaded_obj == NULL)
        goto err;
    /* Vérifier l'objet chargé */
    ret = ASN1_item_verify_ctx(it, &loaded_obj->sig_alg, loaded_obj->signature,
                               &loaded_obj->info, vctx);
err:
    OPENSSL_free(obj_der);
    MyObject_free(loaded_obj);
    MyObject_free(obj);
    EVP_MD_CTX_free(sctx);
    EVP_MD_CTX_free(vctx);
    return ret;
 }
