template<typename _Alloc = std::allocator<char>>

class __gnu_pbds::lu_move_to_front_policy< _Alloc >"Une politique de mise à jour de liste qui déplace inconditionnellement les éléments au début de la liste. Un type nul signifie que chaque lien dans un conteneur basé sur une liste n'a pas réellement besoin de métadonnées.
