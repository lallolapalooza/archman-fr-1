MAT_OPEN(3) Manuel des fonctions de bibliothèque MAT_OPEN(3)

NOM

Mat_Open — Ouvre un fichier MAT MATLAB.

SYNTAXE

#include <matio.h>

mat_t
Mat_Open(const char *matname, int mode);

DESCRIPTION

La fonction Mat_Open() ouvre un fichier MAT MATLAB en lecture seule ou en lecture-écriture.

EXEMPLES

#include <stdlib.h>
#include <stdio.h>
#include "matio.h"

int main(int argc, char **argv)
{
    mat_t *matfp;

    matfp = Mat_Open(argv[1], MAT_ACC_RDONLY);
    if ( NULL == matfp ) {
        fprintf(stderr, "Erreur lors de l'ouverture du fichier MAT %s0, argv[1]);
        return EXIT_FAILURE;
    }

    Mat_Close(matfp);
    return EXIT_SUCCESS;
}
