Module Style
: sig end

type color =
| Black
| Red
| Green
| Yellow
| Blue
| Magenta
| Cyan
| White

type style =
| FG de color
| BG de color
| Bold
| Reset

type Format.stag +=
| Style de style list

val ansi_of_style_l : style list -> string

type tag_style = {
ansi : style list ;
text_open : string ;
text_close : string ;
}

type styles = {
error : tag_style ;
warning : tag_style ;
loc : tag_style ;
hint : tag_style ;
inline_code : tag_style ;
}
