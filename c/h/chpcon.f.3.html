subroutine chpcon (character uplo, integer n, complex, dimension( * ) ap, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer info)

CHPCON

Objectif:

!>
!> CHPCON estime l'inverse du nombre de condition d'une matrice complexe
!> hermitienne paixée A en utilisant la factorisation A = U*D*U**H ou
!> A = L*D*L**H calculée par CHPTRF.
!>
!> Une estimation est obtenue pour norm(inv(A)), et l'inverse du
!> nombre de condition est calculé comme RCOND = 1 / (ANORM * norm(inv(A))).
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme A = U*D*U**H;
!>          = 'L':  Triangulaire inférieure, forme A = L*D*L**H.
!>

N

!>          N est INTEGER
!>          Ordre de la matrice A.  N >= 0.
!>

AP

!>          AP est tableau COMPLEX, dimension (N*(N+1)/2)
!>          La matrice diagonale par blocs D et les multiplicateurs utilisés pour
!>          obtenir le facteur U ou L calculés par CHPTRF, stockés comme une
!>          matrice triangulaire paixée.
!>

IPIV

!>          IPIV est tableau INTEGER, dimension (N)
!>          Détails des échanges et structure de blocs de D
!>          déterminés par CHPTRF.
!>

ANORM

!>          ANORM est REAL
!>          La norme 1 de la matrice originale A.
!>

RCOND

!>          RCOND est REAL
!>          L'inverse du nombre de condition de la matrice A,
!>          calculé comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée dans cette routine.
!>

WORK

!>          WORK est tableau COMPLEX, dimension (2*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
