subroutine chet21 (integer itype, character uplo, integer n, integer kband, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( * ) tau, complex, dimension( * ) work, real, dimension( * ) rwork, real, dimension( 2 ) result)

CHET21

But:

!>
!> CHET21 vérifie généralement une décomposition de la forme
!>
!>    A = U S U**H
!>
!> où **H signifie transconjugué, A est hermitienne, U est unitaire et
!> S est diagonale (si KBAND=0) ou (réelle) symétrique tridiagonale (si
!> KBAND=1).
!>
!> Si ITYPE=1, alors U est représentée comme une matrice dense ; sinon, U est
!> exprimée comme un produit de transformations de Householder, dont les vecteurs
!> sont stockés dans le tableau  et dont les constantes de mise à l'échelle sont dans .
!> Nous utiliserons la lettre  pour désigner le produit des transformations de Householder
!> (qui devrait être égal à U).
!>
!> Spécifiquement, si ITYPE=1, alors :
!>
!>    RESULT(1) = | A - U S U**H | / ( |A| n ulp ) et
!>    RESULT(2) = | I - U U**H | / ( n ulp )
!>
!> Si ITYPE=2, alors :
!>
!>    RESULT(1) = | A - V S V**H | / ( |A| n ulp )
!>
!> Si ITYPE=3, alors :
!>
!>    RESULT(1) = | I - U V**H | / ( n ulp )
!>
!> Pour ITYPE > 1, la transformation U est exprimée comme un produit
!> V = H(1)...H(n-2), où H(j) = I - tau(j) v(j) v(j)**H et chaque
!> vecteur v(j) a ses premiers j éléments à 0 et les n-j éléments restants
!> stockés dans V(j+1:n,j).
!>

Paramètres

ITYPE 
!>          ITYPE est INTEGER
!>          Spécifie le type de tests à effectuer.
!>          1 : U exprimée comme une matrice unitaire dense :
!>             RESULT(1) = | A - U S U**H | / ( |A| n ulp ) et
!>             RESULT(2) = | I - U U**H | / ( n ulp )
!>
!>          2 : U exprimée comme un produit V de transformations de Householder :
!>             RESULT(1) = | A - V S V**H | / ( |A| n ulp )
!>
!>          3 : U exprimée à la fois comme une matrice unitaire dense et
!>             comme un produit de transformations de Householder :
!>             RESULT(1) = | I - U V**H | / ( n ulp )
!>

UPLO

!>          UPLO est CHARACTER
!>          Si UPLO='U', le triangle supérieur de A et V sera utilisé et
!>          le triangle (strictement) inférieur ne sera pas référencé.
!>          Si UPLO='L', le triangle inférieur de A et V sera utilisé et
!>          le triangle (strictement) supérieur ne sera pas référencé.
!>

N

!>          N est INTEGER
!>          La taille de la matrice. Si elle est nulle, CHET21 ne fait rien.
!>          Elle doit être supérieure ou égale à zéro.
!>

KBAND

!>          KBAND est INTEGER
!>          La largeur de bande de la matrice. Elle peut être zéro ou un.
!>          Si zéro, alors S est diagonale, et E n'est pas référencé. Si
!>          un, alors S est symétrique tri-diagonale.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA, N)
!>          La matrice originale (non factorisée). Elle est supposée
!>          hermitienne, et seul le triangle supérieur (UPLO='U') ou le triangle
!>          inférieur (UPLO='L') sera référencé.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale de A. Elle doit être au moins 1
!>          et au moins N.
!>

D

!>          D est un tableau REAL, dimension (N)
!>          La diagonale de la matrice (tri-) diagonale symétrique.
!>

E

!>          E est un tableau REAL, dimension (N-1)
!>          Les éléments hors diagonale de la matrice (tri-) diagonale symétrique.
!>          E(1) est l'élément (1,2) et (2,1), E(2) est l'élément (2,3) et
!>          (3,2), etc.
!>          Non référencé si KBAND=0.
!>

U

!>          U est un tableau COMPLEX, dimension (LDU, N)
!>          Si ITYPE=1 ou 3, ceci contient la matrice unitaire dans
!>          la décomposition, exprimée comme une matrice dense. Si ITYPE=2,
!>          alors elle n'est pas référencée.
!>

LDU

!>          LDU est INTEGER
!>          La dimension principale de U. LDU doit être au moins N et
!>          au moins 1.
!>

V

!>          V est un tableau COMPLEX, dimension (LDV, N)
!>          Si ITYPE=2 ou 3, les colonnes de ce tableau contiennent les
!>          vecteurs de Householder décrivant la matrice unitaire dans
!>          la décomposition. Si UPLO='L', alors les vecteurs sont dans
!>          le triangle inférieur, si UPLO='U', alors dans le triangle
!>          supérieur.
!>          *NOTE* Si ITYPE=2 ou 3, V est modifié et restauré. La
!>          sous-diagonale (si UPLO='L') ou la super-diagonale (si UPLO='U')
!>          est mise à un, puis restaurée à sa valeur d'origine, pendant
!>          le déroulement du calcul.
!>          Si ITYPE=1, alors il n'est ni référencé ni modifié.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale de V. LDV doit être au moins N et
!>          au moins 1.
!>

TAU

!>          TAU est un tableau COMPLEX, dimension (N)
!>          Si ITYPE >= 2, alors TAU(j) est le facteur scalaire de
!>          v(j) v(j)**H dans la transformation de Householder H(j) du
!>          produit U = H(1)...H(n-2)
!>          Si ITYPE < 2, alors TAU n'est pas référencé.
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (2*N**2)
!>

RWORK

!>          RWORK est un tableau REAL, dimension (N)
!>

RESULT

!>          RESULT est un tableau REAL, dimension (2)
!>          Les valeurs calculées par les deux tests décrits ci-dessus. Les
!>          valeurs sont actuellement limitées à 1/ulp, pour éviter le dépassement.
!>          RESULT(1) est toujours modifié. RESULT(2) est modifié uniquement
!>          si ITYPE=1.
!>
