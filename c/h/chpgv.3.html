subroutine chpgv (integer itype, character jobz, character uplo, integer n, complex, dimension( * ) ap, complex, dimension( * ) bp, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHPGV

Purpose:

!>
!> CHPGV calcule toutes les valeurs propres et, éventuellement, les vecteurs propres
!> d'un problème propre généralisé hermitien-défini complexe, de la forme
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  ou B*A*x=(lambda)*x.
!> Ici, A et B sont supposées hermitiennes, stockées au format compacté,
!> et B est également définie positive.
!>

Paramètres

ITYPE 
!>          ITYPE est de type INTEGER
!>          Spécifie le type de problème à résoudre :
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ est de type CHARACTER*1
!>          = 'N':  Calculer uniquement les valeurs propres ;
!>          = 'V':  Calculer les valeurs propres et les vecteurs propres.
!>

UPLO

!>          UPLO est de type CHARACTER*1
!>          = 'U':  Les triangles supérieurs de A et B sont stockés ;
!>          = 'L':  Les triangles inférieurs de A et B sont stockés.
!>

N

!>          N est de type INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!>

AP

!>          AP est un tableau de type COMPLEX, de dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice hermitienne
!>          A, compacté par colonnes dans un tableau linéaire. La j-ième colonne de A
!>          est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>
!>          À la sortie, le contenu de AP est détruit.
!>

BP

!>          BP est un tableau de type COMPLEX, de dimension (N*(N+1)/2)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice hermitienne
!>          B, compacté par colonnes dans un tableau linéaire. La j-ième colonne de B
!>          est stockée dans le tableau BP comme suit :
!>          si UPLO = 'U', BP(i + (j-1)*j/2) = B(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', BP(i + (j-1)*(2*n-j)/2) = B(i,j) pour j<=i<=n.
!>
!>          À la sortie, le facteur triangulaire U ou L de la factorisation de Cholesky
!>          B = U**H*U ou B = L*L**H, est stocké dans le même format que B.
!>

W

!>          W est un tableau de type REAL, de dimension (N)
!>          Si INFO = 0, les valeurs propres par ordre croissant.
!>

Z

!>          Z est un tableau de type COMPLEX, de dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres. Les vecteurs propres sont normalisés comme suit :
!>          si ITYPE = 1 ou 2, Z**H*B*Z = I ;
!>          si ITYPE = 3, Z**H*inv(B)*Z = I.
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est de type INTEGER
!>          La dimension principale du tableau Z. LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau de type COMPLEX, de dimension (max(1, 2*N-1))
!>

RWORK

!>          RWORK est un tableau de type REAL, de dimension (max(1, 3*N-2))
!>

INFO

!>          INFO est de type INTEGER
!>          = 0:  Terminaison réussie
!>          < 0:  Si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0:  CPPTRF ou CHPEV a retourné un code d'erreur :
!>             <= N:  Si INFO = i, CHPEV a échoué à converger ;
!>                    i éléments hors diagonale d'une forme tridiagonale intermédiaire
!>                    n'ont pas convergé vers zéro ;
!>             > N:   Si INFO = N + i, pour 1 <= i <= n, alors le mineur principal
!>                    principal d'ordre i de B n'est pas positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ou vecteur propre n'a été calculé.
!>
