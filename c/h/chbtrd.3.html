sous-routine chbtrd (caractère vect, caractère uplo, entier n, entier kd, complexe, dimension( ldab, * ) ab, entier ldab, réel, dimension( * ) d, réel, dimension( * ) e, complexe, dimension( ldq, * ) q, entier ldq, complexe, dimension( * ) work, entier info)

CHBTRD

Objectif :

!>
!> CHBTRD réduit une matrice bande complexe hermitienne A en forme tridiagonale
!> réelle symétrique T par une transformation unitaire de similarité :
!> Q**H * A * Q = T.
!>

Paramètres :

VECT 
!>          VECT est CHARACTER*1
!>          = 'N':  ne pas former Q ;
!>          = 'V':  former Q ;
!>          = 'U':  mettre à jour une matrice X en formant X*Q.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  le triangle supérieur de A est stocké ;
!>          = 'L':  le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB est un tableau COMPLEX, dimension (LDAB,N)
!>          En entrée, le triangle supérieur ou inférieur de la matrice bande
!>          hermitienne A, stocké dans les KD+1 premières lignes du tableau. La
!>          j-ième colonne de A est stockée dans la j-ième colonne du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>          En sortie, les éléments diagonaux de AB sont écrasés par les
!>          éléments diagonaux de la matrice tridiagonale T ; si KD > 0, les
!>          éléments de la première superdiagonale (si UPLO = 'U') ou de la
!>          première sous-diagonale (si UPLO = 'L') sont écrasés par les
!>          éléments hors diagonale de T ; le reste de AB est écrasé par
!>          des valeurs générées lors de la réduction.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

D

!>          D est un tableau REAL, dimension (N)
!>          Les éléments diagonaux de la matrice tridiagonale T.
!>

E

!>          E est un tableau REAL, dimension (N-1)
!>          Les éléments hors diagonale de la matrice tridiagonale T :
!>          E(i) = T(i,i+1) si UPLO = 'U' ; E(i) = T(i+1,i) si UPLO = 'L'.
!>

Q

!>          Q est un tableau COMPLEX, dimension (LDQ,N)
!>          En entrée, si VECT = 'U', alors Q doit contenir une matrice X de taille N*N ;
!>          si VECT = 'N' ou 'V', alors Q n'a pas besoin d'être défini.
!>
!>          En sortie :
!>          si VECT = 'V', Q contient la matrice unitaire Q de taille N*N ;
!>          si VECT = 'U', Q contient le produit X*Q ;
!>          si VECT = 'N', le tableau Q n'est pas référencé.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.
!>          LDQ >= 1, et LDQ >= N si VECT = 'V' ou 'U'.
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>
