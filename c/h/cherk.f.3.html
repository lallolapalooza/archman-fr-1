subroutine cherk (character uplo, character trans, integer n, integer k, real alpha, complex, dimension(lda,*) a, integer lda, real beta, complex, dimension(ldc,*) c, integer ldc)

CHERK

Objectif :

!>
!> CHERK  effectue l'une des opérations hermitiennes de rang k
!>
!>    C := alpha*A*A**H + beta*C,
!>
!> ou
!>
!>    C := alpha*A**H*A + beta*C,
!>
!> où  alpha et beta  sont des scalaires réels,  C est une matrice hermitienne
!> n par n et  A  est une matrice n par k  dans le premier cas et une matrice
!> k par n  dans le second cas.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée,  UPLO  spécifie si la partie triangulaire supérieure ou inférieure
!>           du tableau  C  doit être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure de C
!>                                  doit être référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure de C
!>                                  doit être référencée.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée,  TRANS  spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' ou 'n'   C := alpha*A*A**H + beta*C.
!>
!>              TRANS = 'C' ou 'c'   C := alpha*A**H*A + beta*C.
!>

N

!>          N est INTEGER
!>           À l'entrée,  N  spécifie l'ordre de la matrice C.  N doit être
!>           supérieur ou égal à zéro.
!>

K

!>          K est INTEGER
!>           À l'entrée,  si  TRANS = 'N' ou 'n',  K  spécifie le nombre
!>           de colonnes de la matrice A, et à l'entrée  si
!>           TRANS = 'C' ou 'c',  K  spécifie le nombre de lignes de la
!>           matrice A.  K doit être supérieur ou égal à zéro.
!>

ALPHA

!>          ALPHA est REAL
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!>

A

!>          A est un tableau COMPLEXE, de dimension ( LDA, ka ), où ka est
!>           k  quand  TRANS = 'N' ou 'n',  et  n  sinon.
!>           Avant l'entrée avec  TRANS = 'N' ou 'n',  la partie principale n par k
!>           du tableau  A  doit contenir la matrice  A,  sinon
!>           la partie principale k par n  du tableau  A  doit contenir la
!>           matrice A.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle qu'elle est déclarée
!>           dans le programme appelant (subroutine).  Quand  TRANS = 'N' ou 'n'
!>           alors  LDA doit être supérieur ou égal à  max( 1, n ), sinon  LDA doit
!>           être supérieur ou égal à  max( 1, k ).
!>

BETA

!>          BETA est REAL
!>           À l'entrée, BETA spécifie le scalaire beta.
!>

C

!>          C est un tableau COMPLEXE, de dimension ( LDC, N )
!>           Avant l'entrée avec  UPLO = 'U' ou 'u',  la partie triangulaire supérieure
!>           n par n  du tableau C doit contenir la partie triangulaire supérieure
!>           de la matrice hermitienne  et la partie strictement inférieure de C
!>           n'est pas référencée.  À la sortie, la partie triangulaire supérieure
!>           du tableau  C  est remplacée par la partie triangulaire supérieure
!>           de la matrice mise à jour.
!>           Avant l'entrée avec  UPLO = 'L' ou 'l',  la partie triangulaire inférieure
!>           n par n  du tableau C doit contenir la partie triangulaire inférieure
!>           de la matrice hermitienne  et la partie strictement supérieure de C
!>           n'est pas référencée.  À la sortie, la partie triangulaire inférieure
!>           du tableau  C  est remplacée par la partie triangulaire inférieure
!>           de la matrice mise à jour.
!>           Notez que les parties imaginaires des éléments diagonaux n'ont pas
!>           besoin d'être définies, elles sont supposées être zéro, et à la sortie
!>           elles sont mises à zéro.
!>

LDC

!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle qu'elle est déclarée
!>           dans le programme appelant (subroutine).  LDC doit être supérieur ou égal à
!>           max( 1, n ).
!>
