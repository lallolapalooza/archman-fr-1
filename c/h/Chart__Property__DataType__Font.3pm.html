NOM

Chart::Font - objets de tenue de police uniques en lecture seule

SYNTAXE

my $red = Chart::Color->new('red');
say $red->add('blue')->name;              # magenta, mélangé dans l'espace RVB
Chart::Color->new( 0, 0, 255)->hsl        # 240, 100, 50 = bleu
$blue->blend_with({H=> 0, S=> 0, L=> 80}, 0.1);# mélange bleu avec un peu de gris
$red->gradient( '#0000FF', 10);           # 10 couleurs de rouge à bleu  
$red->complementary( 3 );                 # obtenir du rouge, du vert et du bleu assortis
