subroutine chbgvd (character jobz, character uplo, integer n, integer ka, integer kb, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldbb, * ) bb, integer ldbb, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer

info)

CHBGV

D

Objectif

: 
!>
!> CHBGVD calcule toutes les valeurs propres, et éventuellement les vecteurs propres
!> d'un problème aux valeurs propres généralisé hermitien-défini positif par bandes, de
!> la forme A*x=(lambda)*B*x. Ici, A et B sont supposés hermitiens
!> et par bandes, et B est également défini positif. Si des vecteurs propres
!> sont souhaités, un algorithme de diviser pour mieux régner est utilisé.
!>
!
>

Paramètre

s
JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N' :  calcule uniquement les valeurs propres ;
!>          = 'V' :  calcule les valeurs propres et les vecteurs propres.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' :  les triangles supérieurs de A et B sont stockés ;
!>          = 'L' :  les triangles inférieurs de A et B sont stockés.
!
>

N 
!>          N est INTEGER
!>          L'ordre des matrices A et B.  N >= 0.
!
>

K

A 
!>          KA est INTEGER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KA >= 0.
!
>

K

B 
!>          KB est INTEGER
!>          Le nombre de superdiagonales de la matrice B si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'. KB >= 0.
!
>

A

B 
!>          AB est un tableau COMPLEX, dimension (LDAB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          par bandes hermitienne A, stocké dans les premières ka+1 lignes du tableau.  La
!>          j-ième colonne de A est stockée dans la j-ième colonne du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) pour max(1,j-ka)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+ka).
!>
!>          À la sortie, le contenu de AB est détruit.
!
>

LDA

B 
!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KA+1.
!
>

B

B 
!>          BB est un tableau COMPLEX, dimension (LDBB, N)
!>          À l'entrée, le triangle supérieur ou inférieur de la matrice
!>          par bandes hermitienne B, stocké dans les premières kb+1 lignes du tableau.  La
!>          j-ième colonne de B est stockée dans la j-ième colonne du tableau BB
!>          comme suit :
!>          si UPLO = 'U', BB(kb+1+i-j,j) = B(i,j) pour max(1,j-kb)<=i<=j ;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) pour j<=i<=min(n,j+kb).
!>
!>          À la sortie, le facteur S de la factorisation de Cholesky scindée
!>          B = S**H*S, tel que retourné par CPBSTF.
!
>

LDB

B 
!>          LDBB est INTEGER
!>          La dimension principale du tableau BB.  LDBB >= KB+1.
!
>

W 
!>          W est un tableau REAL, dimension (N)
!>          Si INFO = 0, les valeurs propres dans l'ordre croissant.
!
>

Z 
!>          Z est un tableau COMPLEX, dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient la matrice Z des
!>          vecteurs propres, avec la i-ième colonne de Z contenant le
!>          vecteur propre associé à W(i). Les vecteurs propres sont
!>          normalisés de sorte que Z**H*B*Z = I.
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!
>

LD

Z 
!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= N.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO=0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si N <= 1,               LWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LWORK >= 2*N**2.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est supposée ; la routine
!>          calcule uniquement les tailles optimales des tableaux WORK, RWORK et
!>          IWORK, renvoie ces valeurs comme premières entrées des
!>          tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (MAX(1,LRWORK))
!>          À la sortie, si INFO=0, RWORK(1) renvoie le LRWORK optimal.
!
>

LRWOR

K 
!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK >= 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK >= N.
!>          Si JOBZ = 'V' et N > 1, LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, une interrogation d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!
>

IWOR

K 
!>          IWORK est un tableau INTEGER, dimension (MAX(1,LIWORK))
!>          À la sortie, si INFO=0, IWORK(1) renvoie le LIWORK optimal.
!
>

LIWOR

K 
!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ = 'N' ou N <= 1, LIWORK >= 1.
!>          Si JOBZ = 'V' et N > 1, LIWORK >= 3 + 5*N.
!>
!>          Si LIWORK = -1, une interrogation d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées
!>          des tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, et i est :
!>             <= N :  l'algorithme n'a pas convergé :
!>                    i éléments hors diagonale d'une forme
!>                    tridiagonale intermédiaire n'ont pas convergé vers zéro ;
!>             > N :   si INFO = N + i, pour 1 <= i <= N, alors CPBSTF
!>                    a retourné INFO = i : B n'est pas défini positif.
!>                    La factorisation de B n'a pas pu être complétée et
!>                    aucune valeur propre ou vecteur propre n'a été calculé.
!
>
