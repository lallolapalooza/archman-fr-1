subroutine cheequb (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, complex, dimension( * ) work, integer info)

CHEEQUB

Objectif :

!>
!> CHEEQUB calcule les mises à l'échelle des lignes et des colonnes destinées à
!> équilibrer une matrice hermitienne A (par rapport à la norme euclidienne) et
!> à réduire son nombre de conditionnement. Les facteurs d'échelle S sont calculés
!> par l'algorithme BIN (voir les références) de telle sorte que la matrice mise
!> à l'échelle B avec les éléments B(i,j) = S(i)*A(i,j)*S(j) ait un nombre de
!> conditionnement dans un facteur N du plus petit nombre de conditionnement possible
!> sur toutes les mises à l'échelle diagonales possibles.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau COMPLEX, de dimension (LDA,N)
!>          La matrice hermitienne N x N dont les facteurs de mise à l'échelle
!>          doivent être calculés.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

S

!>          S est un tableau REAL, de dimension (N)
!>          Si INFO = 0, S contient les facteurs d'échelle pour A.
!>

SCOND

!>          SCOND est REAL
!>          Si INFO = 0, S contient le rapport du plus petit S(i) au
!>          plus grand S(i). Si SCOND >= 0.1 et AMAX n'est ni trop
!>          grand ni trop petit, il n'est pas utile de mettre à l'échelle avec S.
!>

AMAX

!>          AMAX est REAL
!>          La plus grande valeur absolue d'un élément de matrice. Si AMAX est
!>          très proche du dépassement ou du sous-dépassement, la
!>          matrice doit être mise à l'échelle.
!>

WORK

!>          WORK est un tableau COMPLEX, de dimension (2*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument était invalide
!>          > 0 : si INFO = i, le i-ème élément diagonal est non positif.
!>
