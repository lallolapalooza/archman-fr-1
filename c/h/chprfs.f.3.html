subroutine chprfs (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( * ) afp, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHPRFS

But :

!>
!> CHPRFS améliore la solution calculée d'un système d'équations linéaires
!> lorsque la matrice des coefficients est hermitienne indéfinie
!> et empaquetée, et fournit des bornes d'erreur et des estimations d'erreur en retour
!> pour la solution.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AP

!>          AP est un tableau COMPLEX de dimension (N*(N+1)/2)
!>          Le triangle supérieur ou inférieur de la matrice hermitienne A, empaqueté
!>          colonne par colonne dans un tableau linéaire. La j-ième colonne de A est stockée
!>          dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) pour j<=i<=n.
!>

AFP

!>          AFP est un tableau COMPLEX de dimension (N*(N+1)/2)
!>          La forme factorisée de la matrice A. AFP contient la matrice
!>          diagonale par blocs D et les multiplicateurs utilisés pour obtenir
!>          le facteur U ou L à partir de la factorisation A = U*D*U**H ou
!>          A = L*D*L**H calculée par CHPTRF, stockée sous forme de matrice
!>          triangulaire empaquetée.
!>

IPIV

!>          IPIV est un tableau INTEGER de dimension (N)
!>          Détails des échanges et de la structure de blocs de D
!>          tels que déterminés par CHPTRF.
!>

B

!>          B est un tableau COMPLEX de dimension (LDB,NRHS)
!>          La matrice des membres de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX de dimension (LDX,NRHS)
!>          En entrée, la matrice solution X, telle que calculée par CHPTRS.
!>          En sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X. LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau REAL de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée de la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisé par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau REAL de dimension (NRHS)
!>          L'erreur relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEX de dimension (2*N)
!>

RWORK

!>          RWORK est un tableau REAL de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximum d'étapes de raffinement itératif.
!>
