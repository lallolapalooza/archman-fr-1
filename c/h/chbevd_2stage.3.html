subroutine chbevd_2stage (character jobz, character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

CHBEVD_2STAGE calcule les valeurs propres et, éventuellement, les vecteurs propres gauches et/ou droits pour d'autres matrices

Objectif :

!>
!> CHBEVD_2STAGE calcule toutes les valeurs propres et, éventuellement, les vecteurs propres d'une matrice bande hermitienne complexe A en utilisant la technique 2stage pour la réduction à une matrice tridiagonale. Si des vecteurs propres sont souhaités, elle utilise un algorithme de type diviser pour régner.
!>
!>

Paramètres

JOBZ 
!>          JOBZ est CHARACTER*1
!>          = 'N':  Calcule uniquement les valeurs propres ;
!>          = 'V':  Calcule les valeurs propres et les vecteurs propres.
!>                  Non disponible dans cette version.
!>

UPLO

!>          UPLO est CHARACTER*1
!>          = 'U':  Le triangle supérieur de A est stocké ;
!>          = 'L':  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de superdiagonales de la matrice A si UPLO = 'U',
!>          ou le nombre de sous-diagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB est un tableau COMPLEX de dimension (LDAB, N)
!>          En entrée, le triangle supérieur ou inférieur de la matrice bande hermitienne
!>          A, stocké dans les KD+1 premières lignes du tableau. La
!>          j-ième colonne de A est stockée dans la j-ième colonne du tableau AB
!>          comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>
!>          En sortie, AB est écrasé par les valeurs générées pendant la
!>          réduction à la forme tridiagonale. Si UPLO = 'U', la première
!>          superdiagonale et la diagonale de la matrice tridiagonale T
!>          sont renvoyées dans les lignes KD et KD+1 de AB, et si UPLO = 'L',
!>          la diagonale et la première sous-diagonale de T sont renvoyées dans les
!>          deux premières lignes de AB.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD + 1.
!>

W

!>          W est un tableau REAL de dimension (N)
!>          Si INFO = 0, les valeurs propres par ordre croissant.
!>

Z

!>          Z est un tableau COMPLEX de dimension (LDZ, N)
!>          Si JOBZ = 'V', alors si INFO = 0, Z contient les vecteurs propres
!>          orthonormaux de la matrice A, avec la i-ème colonne de Z
!>          contenant le vecteur propre associé à W(i).
!>          Si JOBZ = 'N', alors Z n'est pas référencé.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEX de dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur du tableau WORK. LWORK >= 1, si N <= 1 ;
!>          sinon
!>          Si JOBZ = 'N' et N > 1, LWORK doit être interrogé.
!>                                   LWORK = MAX(1, dimension) où
!>                                   dimension = (2KD+1)*N + KD*NTHREADS
!>                                   où KD est la taille de la bande.
!>                                   NTHREADS est le nombre de threads utilisés lors
!>                                   de la compilation openMP activée, sinon =1.
!>          Si JOBZ = 'V' et N > 1, LWORK doit être interrogé. Pas encore disponible.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement les tailles optimales des tableaux WORK, RWORK et
!>          IWORK, renvoie ces valeurs comme premières entrées des
!>          tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau REAL,
!>                                         dimension (LRWORK)
!>          En sortie, si INFO = 0, RWORK(1) renvoie le LRWORK optimal.
!>

LRWORK

!>          LRWORK est INTEGER
!>          La dimension du tableau RWORK.
!>          Si N <= 1,               LRWORK doit être au moins 1.
!>          Si JOBZ = 'N' et N > 1, LRWORK doit être au moins N.
!>          Si JOBZ = 'V' et N > 1, LRWORK doit être au moins
!>                        1 + 5*N + 2*N**2.
!>
!>          Si LRWORK = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées des
!>          tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

IWORK

!>          IWORK est un tableau INTEGER de dimension (MAX(1,LIWORK))
!>          En sortie, si INFO = 0, IWORK(1) renvoie le LIWORK optimal.
!>

LIWORK

!>          LIWORK est INTEGER
!>          La dimension du tableau IWORK.
!>          Si JOBZ = 'N' ou N <= 1, LIWORK doit être au moins 1.
!>          Si JOBZ = 'V' et N > 1, LIWORK doit être au moins 3 + 5*N .
!>
!>          Si LIWORK = -1, alors une requête d'espace de travail est supposée ; la
!>          routine calcule uniquement les tailles optimales des tableaux WORK, RWORK
!>          et IWORK, renvoie ces valeurs comme premières entrées des
!>          tableaux WORK, RWORK et IWORK, et aucun message d'erreur
!>          lié à LWORK ou LRWORK ou LIWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0:  fin réussie.
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0:  si INFO = i, l'algorithme n'a pas convergé ; i
!>                éléments hors diagonale d'une forme tridiagonale intermédiaire
!>                n'ont pas convergé vers zéro.
!>
