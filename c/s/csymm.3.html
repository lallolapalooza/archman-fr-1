sous-routine csymm (character side, character uplo, integer m, integer n, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb, complex beta, complex, dimension(ldc,*) c, integer

ldc)

CSYM

M

But

: 
!>
!> CSYMM effectue l'une des opérations matrice-matrice suivantes
!>
!>    C := alpha*A*B + beta*C,
!>
!> ou
!>
!>    C := alpha*B*A + beta*C,
!>
!> où alpha et beta sont des scalaires, A est une matrice symétrique et B et
!> C sont des matrices de dimensions m par n.
!
>

Paramètre

s
SIDE 
!>          SIDE est CHARACTER*1
!>           À l'entrée, SIDE indique si la matrice symétrique A
!>           apparaît à gauche ou à droite dans l'opération comme suit :
!>
!>              SIDE = 'L' ou 'l'   C := alpha*A*B + beta*C,
!>
!>              SIDE = 'R' ou 'r'   C := alpha*B*A + beta*C,
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO indique si la partie triangulaire
!>           supérieure ou inférieure de la matrice symétrique A doit
!>           être référencée comme suit :
!>
!>              UPLO = 'U' ou 'u'   Seule la partie triangulaire supérieure
!>                                  de la matrice symétrique doit être
!>                                  référencée.
!>
!>              UPLO = 'L' ou 'l'   Seule la partie triangulaire inférieure
!>                                  de la matrice symétrique doit être
!>                                  référencée.
!
>

M 
!>          M est INTEGER
!>           À l'entrée, M spécifie le nombre de lignes de la matrice C.
!>           M doit être au moins égal à zéro.
!
>

N 
!>          N est INTEGER
!>           À l'entrée, N spécifie le nombre de colonnes de la matrice C.
!>           N doit être au moins égal à zéro.
!
>

ALPH

A 
!>          ALPHA est COMPLEX
!>           À l'entrée, ALPHA spécifie le scalaire alpha.
!
>

A 
!>          A est un tableau COMPLEX, de dimensions ( LDA, ka ), où ka est
!>           m  quand  SIDE = 'L' ou 'l'  et n  sinon.
!>           Avant l'appel avec  SIDE = 'L' ou 'l',  la partie m par m  du
!>           tableau  A  doit contenir la  matrice symétrique,  de sorte que
!>           quand  UPLO = 'U' ou 'u', la partie triangulaire supérieure
!>           principale m par m  du tableau  A  doit contenir la partie
!>           triangulaire supérieure  de la  matrice symétrique et la  partie
!>           strictement inférieure  de  A  n'est pas référencée,  et quand
!>           UPLO = 'L' ou 'l',  la partie triangulaire inférieure principale
!>           m par m  du  tableau  A  doit  contenir  la  partie triangulaire
!>           inférieure  de la  matrice symétrique et la  partie strictement
!>           supérieure de  A  n'est pas référencée.
!>           Avant l'appel avec  SIDE = 'R' ou 'r',  la partie n par n  du
!>           tableau  A  doit contenir la  matrice symétrique,  de sorte que
!>           quand  UPLO = 'U' ou 'u', la partie triangulaire supérieure
!>           principale n par n  du tableau  A  doit contenir la partie
!>           triangulaire supérieure  de la  matrice symétrique et la  partie
!>           strictement inférieure  de  A  n'est pas référencée,  et quand
!>           UPLO = 'L' ou 'l',  la partie triangulaire inférieure principale
!>           n par n  du  tableau  A  doit  contenir  la  partie triangulaire
!>           inférieure  de la  matrice symétrique et la  partie strictement
!>           supérieure de  A  n'est pas référencée.
!
>

LD

A 
!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que
!>           déclarée dans le (sous-)programme appelant. Quand
!>           SIDE = 'L' ou 'l', LDA doit être au moins  max( 1, m ),
!>           sinon LDA doit être au moins max( 1, n ).
!
>

B 
!>          B est un tableau COMPLEX, de dimensions ( LDB, N )
!>           Avant l'appel, la partie principale m par n du tableau B doit
!>           contenir la matrice B.
!
>

LD

B 
!>          LDB est INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que
!>           déclarée dans le (sous-)programme appelant. LDB doit être au
!>           moins max( 1, m ).
!
>

BET

A 
!>          BETA est COMPLEX
!>           À l'entrée, BETA spécifie le scalaire beta. Quand BETA est
!>           fourni comme zéro, C n'a pas besoin d'être défini en entrée.
!
>

C 
!>          C est un tableau COMPLEX, de dimensions ( LDC, N )
!>           Avant l'appel, la partie principale m par n du tableau C doit
!>           contenir la matrice C, sauf quand beta est zéro, auquel cas
!>           C n'a pas besoin d'être défini à l'entrée.
!>           À la sortie, le tableau C est écrasé par la matrice mise à jour
!>           de dimensions m par n.
!
>

LD

C 
!>          LDC est INTEGER
!>           À l'entrée, LDC spécifie la première dimension de C telle que
!>           déclarée dans le (sous-)programme appelant. LDC doit être au
!>           moins max( 1, m ).
!
>
