subroutine csptri (character uplo, integer n, complex, dimension( * ) ap, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer info)

CSPTRI

Objectif :

!>
!> CSPTRI calcule l'inverse d'une matrice symétrique complexe indéfinie
!> en stockage compacté en utilisant la factorisation A = U*D*U**T ou
!> A = L*D*L**T calculée par CSPTRF.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme est A = U*D*U**T;
!>          = 'L':  Triangulaire inférieure, forme est A = L*D*L**T.
!>

N

!>          N est INTEGER
!>          Ordre de la matrice A.  N >= 0.
!>

AP

!>          AP est un tableau COMPLEX, dimension (N*(N+1)/2)
!>          À l'entrée, la matrice diagonale par blocs D et les multiplicateurs
!>          utilisés pour obtenir le facteur U ou L calculé par CSPTRF,
!>          stockés sous forme de matrice triangulaire compactée.
!>
!>          À la sortie, si INFO = 0, l'inverse (symétrique) de la matrice
!>          originale, stocké sous forme de matrice triangulaire compactée. La j-ème colonne
!>          de inv(A) est stockée dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = inv(A)(i,j) pour 1<=i<=j;
!>          si UPLO = 'L',
!>             AP(i + (j-1)*(2n-j)/2) = inv(A)(i,j) pour j<=i<=n.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure en blocs de D
!>          déterminés par CSPTRF.
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0: sortie réussie
!>          < 0: si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0: si INFO = i, D(i,i) = 0; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!>
