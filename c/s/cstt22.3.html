subroutine cstt22 (n, m, kband, ad, ae, sd, se, u, ldu, work, ldwork, rwork, result)
CSTT22

Documentation de la fonction/sous-programme

sous-programme cstt22 (integer n, integer m, integer kband, real, dimension( * ) ad, real, dimension( * ) ae, real, dimension( * ) sd, real, dimension( * ) se, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real, dimension( 2 ) result)

CSTT22

Objectif :

!>
!> CSTT22  vérifie un ensemble de M valeurs propres et vecteurs propres,
!>
!>     A U = U S
!>
!> où A est tridiagonale hermitienne, les colonnes de U sont unitaires,
!> et S est diagonale (si KBAND=0) ou tridiagonale hermitienne (si KBAND=1).
!> Deux tests sont effectués :
!>
!>    RESULT(1) = | U* A U - S | / ( |A| m ulp )
!>
!>    RESULT(2) = | I - U*U | / ( m ulp )
!>

Paramètres

N 
!>          N est un entier
!>          La taille de la matrice. Si elle est nulle, CSTT22 ne fait rien.
!>          Elle doit être supérieure ou égale à zéro.
!>

M

!>          M est un entier
!>          Le nombre de paires propres à vérifier. Si il est nul, CSTT22 ne fait rien.
!>          Il doit être supérieur ou égal à zéro.
!>

KBAND

!>          KBAND est un entier
!>          La largeur de bande de la matrice S. Elle ne peut être que zéro ou un.
!>          Si elle est nulle, alors S est diagonale, et SE n'est pas référencé. Si
!>          elle est un, alors S est tridiagonale hermitienne.
!>

AD

!>          AD est un tableau REEL de dimension (N)
!>          La diagonale de la matrice originale (non factorisée) A. A
!>          est supposée être tridiagonale hermitienne.
!>

AE

!>          AE est un tableau REEL de dimension (N)
!>          La sous-diagonale de la matrice originale (non factorisée) A. A
!>          est supposée être tridiagonale hermitienne. AE(1) est ignoré,
!>          AE(2) est l'élément (1,2) et (2,1), etc.
!>

SD

!>          SD est un tableau REEL de dimension (N)
!>          La diagonale de la matrice (tridiagonale hermitienne) S.
!>

SE

!>          SE est un tableau REEL de dimension (N)
!>          La sous-diagonale de la matrice (tridiagonale hermitienne) S.
!>          Non référencé si KBSND=0. Si KBAND=1, alors AE(1) est
!>          ignoré, SE(2) est l'élément (1,2) et (2,1), etc.
!>

U

!>          U est un tableau REEL de dimension (LDU, N)
!>          La matrice unitaire dans la décomposition.
!>

LDU

!>          LDU est un entier
!>          La dimension principale de U. LDU doit être au moins N.
!>

WORK

!>          WORK est un tableau COMPLEXE de dimension (LDWORK, M+1)
!>

LDWORK

!>          LDWORK est un entier
!>          La dimension principale de WORK. LDWORK doit être au moins
!>          max(1,M).
!>

RWORK

!>          RWORK est un tableau REEL de dimension (N)
!>

RESULT

!>          RESULT est un tableau REEL de dimension (2)
!>          Les valeurs calculées par les deux tests décrits ci-dessus. Les
!>          valeurs sont actuellement limitées à 1/ulp, pour éviter les dépassements.
!>
