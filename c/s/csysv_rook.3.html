subroutine csysv_rook (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CSYSV_ROOK calcule la solution du système d'équations linéaires A * X = B pour les matrices SY

But :

!>
!> CSYSV_ROOK calcule la solution d'un système d'équations linéaires
!> complexes
!>    A * X = B,
!> où A est une matrice symétrique N×N et X et B sont des matrices
!> N×NRHS.
!>
!> La méthode de pivotement diagonal est utilisée pour factoriser A comme
!>    A = U * D * U**T,  si UPLO = 'U', ou
!>    A = L * D * L**T,  si UPLO = 'L',
!> où U (ou L) est un produit de matrices de permutation et triangulaires
!> supérieures (inférieures) unitaires, et D est symétrique et diagonale par
!> blocs avec des blocs diagonaux 1×1 et 2×2.
!>
!> CSYTRF_ROOK est appelée pour calculer la factorisation d'une matrice
!> symétrique complexe A en utilisant la méthode de pivotement diagonal
!> bornée de Bunch-Kaufman ().
!>
!> La forme factorisée de A est ensuite utilisée pour résoudre le système
!> d'équations A * X = B en appelant CSYTRS_ROOK.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          Le nombre d'équations linéaires, c'est-à-dire l'ordre de la
!>          matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice symétrique A.  Si UPLO = 'U', la partie
!>          triangulaire supérieure N×N de A contient la partie triangulaire
!>          supérieure de la matrice A, et la partie triangulaire
!>          strictement inférieure de A n'est pas référencée.  Si UPLO = 'L',
!>          la partie triangulaire inférieure N×N de A contient la partie
!>          triangulaire inférieure de la matrice A, et la partie
!>          triangulaire strictement supérieure de A n'est pas référencée.
!>
!>          À la sortie, si INFO = 0, la matrice diagonale par blocs D et les
!>          multiplicateurs utilisés pour obtenir le facteur U ou L à partir
!>          de la factorisation A = U*D*U**T ou A = L*D*L**T telle que
!>          calculée par CSYTRF_ROOK.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges et de la structure par blocs de D,
!>          tels que déterminés par CSYTRF_ROOK.
!>
!>          Si UPLO = 'U' :
!>               Si IPIV(k) > 0, alors les lignes et colonnes k et IPIV(k)
!>               ont été échangées et D(k,k) est un bloc diagonal 1×1.
!>
!>               Si IPIV(k) < 0 et IPIV(k-1) < 0, alors les lignes et
!>               colonnes k et -IPIV(k) ont été échangées et les lignes et
!>               colonnes k-1 et -IPIV(k-1) ont été échangées,
!>               D(k-1:k,k-1:k) est un bloc diagonal 2×2.
!>
!>          Si UPLO = 'L' :
!>               Si IPIV(k) > 0, alors les lignes et colonnes k et IPIV(k)
!>               ont été échangées et D(k,k) est un bloc diagonal 1×1.
!>
!>               Si IPIV(k) < 0 et IPIV(k+1) < 0, alors les lignes et
!>               colonnes k et -IPIV(k) ont été échangées et les lignes et
!>               colonnes k+1 et -IPIV(k+1) ont été échangées,
!>               D(k:k+1,k:k+1) est un bloc diagonal 2×2.
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice du second membre B de dimension N×NRHS.
!>          À la sortie, si INFO = 0, la matrice solution X de dimension N×NRHS.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur de WORK.  LWORK >= 1, et pour de meilleures
!>          performances LWORK >= max(1,N*NB), où NB est la taille de bloc
!>          optimale pour CSYTRF_ROOK.
!>
!>          TRS sera effectué avec les BLAS de niveau 2
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille optimale du
!>          tableau WORK, renvoie cette valeur comme première entrée du
!>          tableau WORK, et aucun message d'erreur lié à LWORK n'est émis
!>          par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, D(i,i) est exactement zéro.  La factorisation
!>               a été complétée, mais la matrice diagonale par blocs D est
!>               exactement singulière, la solution n'a donc pas pu être calculée.
!>
