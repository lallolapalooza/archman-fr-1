subroutine csytrf_aa (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CSYTRF_AA

Purpose:

!>
!> CSYTRF_AA calcule la factorisation d'une matrice symétrique complexe A
!> en utilisant l'algorithme d'Aasen. La forme de la factorisation est
!>
!>    A = U**T*T*U  ou  A = L*T*L**T
!>
!> où U (ou L) est un produit de matrices triangulaires unitaires supérieures (inférieures)
!> et de permutations, et T est une matrice symétrique tridiagonale complexe.
!>
!> Ceci est la version bloquée de l'algorithme, appelant le BLAS de niveau 3.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau COMPLEX, de dimension (LDA,N)
!>          À l'entrée, la matrice symétrique A. Si UPLO = 'U', la partie
!>          supérieure triangulaire N par N de A contient la partie
!>          supérieure triangulaire de la matrice A, et la partie strictement
!>          inférieure triangulaire de A n'est pas référencée. Si UPLO = 'L', la partie
!>          inférieure triangulaire N par N de A contient la partie
!>          inférieure triangulaire de la matrice A, et la partie strictement
!>          supérieure triangulaire de A n'est pas référencée.
!>
!>          À la sortie, la matrice tridiagonale est stockée dans les diagonales
!>          et les sous-diagonales de A juste en dessous (ou au-dessus) des diagonales,
!>          et L est stockée en dessous (ou au-dessus) des sous-diagonales, quand UPLO
!>          est 'L' (ou 'U').
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, de dimension (N)
!>          À la sortie, il contient les détails des échanges, c'est-à-dire que
!>          la ligne et la colonne k de A ont été échangées avec la
!>          ligne et la colonne IPIV(k).
!>

WORK

!>          WORK est un tableau COMPLEX, de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur de WORK. LWORK >= MAX(1,2*N). Pour une performance optimale
!>          LWORK >= N*(1+NB), où NB est la taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument était invalide.
!>
