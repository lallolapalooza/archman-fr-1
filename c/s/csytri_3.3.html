subroutine csytri_3 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CSYTRI_3

Purpose:

!> CSYTRI_3 calcule l'inverse d'une matrice complexe symétrique indéfinie
!> A en utilisant la factorisation calculée par CSYTRF_RK ou CSYTRF_BK :
!>
!>     A = P*U*D*(U**T)*(P**T) ou A = P*L*D*(L**T)*(P**T),
!>
!> où U (ou L) est une matrice triangulaire unitaire supérieure (ou inférieure),
!> U**T (ou L**T) est la transposée de U (ou L), P est une matrice de permutation,
!> P**T est la transposée de P, et D est symétrique et bloc diagonale avec des
!> blocs diagonaux de 1x1 et 2x2.
!>
!> CSYTRI_3 définit la dimension principale de l'espace de travail avant d'appeler
!> CSYTRI_3X qui calcule réellement l'inverse. C'est la version bloquée
!> de l'algorithme, appelant le BLAS de niveau 3.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont
!>          stockés sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U' : Le triangle supérieur de A est stocké ;
!>          = 'L' : Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau COMPLEX, de dimension (LDA,N)
!>          À l'entrée, la diagonale de la matrice bloc diagonale D et
!>          les facteurs U ou L calculés par CSYTRF_RK et CSYTRF_BK :
!>            a) UNIQUEMENT les éléments diagonaux de la matrice bloc diagonale
!>               symétrique D sur la diagonale de A, c'est-à-dire D(k,k) = A(k,k) ;
!>               (les éléments superdiagonaux (ou sous-diagonaux) de D
!>                doivent être fournis à l'entrée dans le tableau E), et
!>            b) Si UPLO = 'U' : le facteur U dans la partie superdiagonale de A.
!>               Si UPLO = 'L' : le facteur L dans la partie sous-diagonale de A.
!>
!>          À la sortie, si INFO = 0, l'inverse symétrique de la matrice originale.
!>             Si UPLO = 'U' : la partie triangulaire supérieure de l'inverse
!>             est formée et la partie de A sous la diagonale n'est pas
!>             référencée ;
!>             Si UPLO = 'L' : la partie triangulaire inférieure de l'inverse
!>             est formée et la partie de A au-dessus de la diagonale n'est pas
!>             référencée.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

E

!>          E est un tableau COMPLEX, de dimension (N)
!>          À l'entrée, contient les éléments superdiagonaux (ou sous-diagonaux)
!>          de la matrice bloc diagonale D
!>          avec des blocs diagonaux de 1x1 ou 2x2, où
!>          Si UPLO = 'U' : E(i) = D(i-1,i),i=2:N, E(1) non référencé ;
!>          Si UPLO = 'L' : E(i) = D(i+1,i),i=1:N-1, E(N) non référencé.
!>
!>          NOTE : Pour un bloc diagonal 1x1 D(k), où
!>          1 <= k <= N, l'élément E(k) n'est pas référencé dans les deux
!>          cas UPLO = 'U' ou UPLO = 'L'.
!>

IPIV

!>          IPIV est un tableau INTEGER, de dimension (N)
!>          Détails des échanges et de la structure des blocs de D
!>          tels que déterminés par CSYTRF_RK ou CSYTRF_BK.
!>

WORK

!>          WORK est un tableau COMPLEX, de dimension (N+NB+1)*(NB+3).
!>          À la sortie, si INFO = 0, WORK(1) contient la valeur optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La longueur de WORK. LWORK >= (N+NB+1)*(NB+3).
!>
!>          Si LDWORK = -1, alors une requête d'espace de travail est supposée ;
!>          la routine calcule uniquement la taille optimale de l'espace de travail optimal
!>          optimal de WORK, renvoie cette valeur comme première
!>          entrée du tableau WORK, et aucun message d'erreur lié à
!>          LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 : si INFO = i, D(i,i) = 0 ; la matrice est singulière et son
!>               inverse n'a pas pu être calculé.
!>
