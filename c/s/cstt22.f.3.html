subroutine cstt22 (n, m, kband, ad, ae, sd, se, u, ldu, work, ldwork, rwork, result)
CSTT22

Documentation de la fonction/sous-programme

sous-programme cstt22 (integer n, integer m, integer kband, real, dimension( * ) ad, real, dimension( * ) ae, real, dimension( * ) sd, real, dimension( * ) se, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real, dimension( 2 ) result)

CSTT22

Objectif :

!>
!> CSTT22  vérifie un ensemble de M valeurs propres et vecteurs propres,
!>
!>     A U = U S
!>
!> où A est hermitienne tridiagonale, les colonnes de U sont unitaires,
!> et S est diagonale (si KBAND=0) ou hermitienne tridiagonale (si KBAND=1).
!> Deux tests sont effectués :
!>
!>    RESULT(1) = | U* A U - S | / ( |A| m ulp )
!>
!>    RESULT(2) = | I - U*U | / ( m ulp )
!>

Paramètres

N 
!>          N est un entier
!>          La taille de la matrice. Si elle est nulle, CSTT22 ne fait rien.
!>          Elle doit être au moins nulle.
!>

M

!>          M est un entier
!>          Le nombre de paires de valeurs propres à vérifier. Si elle est nulle, CSTT22 ne fait rien.
!>          Elle doit être au moins nulle.
!>

KBAND

!>          KBAND est un entier
!>          La largeur de bande de la matrice S. Elle ne peut être que zéro ou un.
!>          Si elle est nulle, alors S est diagonale, et SE n'est pas référencée. Si
!>          elle est un, alors S est hermitienne tri-diagonale.
!>

AD

!>          AD est un tableau de REAL, dimension (N)
!>          La diagonale de la matrice A d'origine (non factorisée). A
!>          est supposée être hermitienne tridiagonale.
!>

AE

!>          AE est un tableau de REAL, dimension (N)
!>          La partie hors diagonale de la matrice A d'origine (non factorisée). A
!>          est supposée être hermitienne tridiagonale. AE(1) est ignoré,
!>          AE(2) est l'élément (1,2) et (2,1), etc.
!>

SD

!>          SD est un tableau de REAL, dimension (N)
!>          La diagonale de la matrice S (Hermitienne tri-) diagonale.
!>

SE

!>          SE est un tableau de REAL, dimension (N)
!>          La partie hors diagonale de la matrice S (Hermitienne tri-) diagonale.
!>          Non référencée si KBSND=0. Si KBAND=1, alors AE(1) est
!>          ignoré, SE(2) est l'élément (1,2) et (2,1), etc.
!>

U

!>          U est un tableau de REAL, dimension (LDU, N)
!>          La matrice unitaire dans la décomposition.
!>

LDU

!>          LDU est un entier
!>          La dimension principale de U. LDU doit être au moins N.
!>

WORK

!>          WORK est un tableau de COMPLEX, dimension (LDWORK, M+1)
!>

LDWORK

!>          LDWORK est un entier
!>          La dimension principale de WORK. LDWORK doit être au moins
!>          max(1,M).
!>

RWORK

!>          RWORK est un tableau de REAL, dimension (N)
!>

RESULT

!>          RESULT est un tableau de REAL, dimension (2)
!>          Les valeurs calculées par les deux tests décrits ci-dessus. Les
!>          valeurs sont actuellement limitées à 1/ulp, pour éviter le débordement.
!>
