subroutine csysv_rook (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer

CSYSV_ROOK calcule la solution du système d'équations linéaires A * X = B pour les matrices

SY

But

: 
!>
!> CSYSV_ROOK calcule la solution d'un système d'équations linéaires
!> complexes
!>    A * X = B,
!> où A est une matrice symétrique N par N et X et B sont des matrices
!> N par NRHS.
!>
!> La méthode de pivotage diagonal est utilisée pour factoriser A comme
!>    A = U * D * U**T,  si UPLO = 'U', ou
!>    A = L * D * L**T,  si UPLO = 'L',
!> où U (ou L) est un produit de matrices de permutation et de matrices
!> triangulaires supérieures (inférieures) unitaires, et D est
!> symétrique et bloc-diagonale avec des blocs diagonaux 1 par 1 et
!> 2 par 2.
!>
!> CSYTRF_ROOK est appelée pour calculer la factorisation d'une matrice
!> symétrique complexe A en utilisant la méthode de pivotage diagonal
!> bornée de Bunch-Kaufman ().
!>
!> La forme factorisée de A est ensuite utilisée pour résoudre le
!> système d'équations A * X = B en appelant CSYTRS_ROOK.
!
>

Paramètre

s
UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de A est stocké ;
!>          = 'L' : le triangle inférieur de A est stocké.
!
>

N 
!>          N est INTEGER
!>          Le nombre d'équations linéaires, c'est-à-dire l'ordre de la
!>          matrice A.  N >= 0.
!
>

NRH

S 
!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de
!>          colonnes de la matrice B.  NRHS >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice symétrique A.  Si UPLO = 'U', la
!>          partie triangulaire supérieure N par N de A contient la
!>          partie triangulaire supérieure de la matrice A, et la partie
!>          strictement triangulaire inférieure de A n'est pas
!>          référencée.  Si UPLO = 'L', la partie triangulaire inférieure
!>          N par N de A contient la partie triangulaire inférieure de la
!>          matrice A, et la partie strictement triangulaire supérieure
!>          de A n'est pas référencée.
!>
!>          À la sortie, si INFO = 0, la matrice bloc-diagonale D et les
!>          multiplicateurs utilisés pour obtenir le facteur U ou L à
!>          partir de la factorisation A = U*D*U**T ou A = L*D*L**T tels
!>          que calculés par CSYTRF_ROOK.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!
>

IPI

V 
!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des interchanges et de la structure de blocs de D,
!>          tels que déterminés par CSYTRF_ROOK.
!>
!>          Si UPLO = 'U' :
!>               Si IPIV(k) > 0, alors les lignes et colonnes k et
!>               IPIV(k) ont été interchangées et D(k,k) est un bloc
!>               diagonal 1 par 1.
!>
!>               Si IPIV(k) < 0 et IPIV(k-1) < 0, alors les lignes et
!>               colonnes k et -IPIV(k) ont été interchangées et les
!>               lignes et colonnes k-1 et -IPIV(k-1) ont été
!>               interchangées,
!>               D(k-1:k,k-1:k) est un bloc diagonal 2 par 2.
!>
!>          Si UPLO = 'L' :
!>               Si IPIV(k) > 0, alors les lignes et colonnes k et
!>               IPIV(k) ont été interchangées et D(k,k) est un bloc
!>               diagonal 1 par 1.
!>
!>               Si IPIV(k) < 0 et IPIV(k+1) < 0, alors les lignes et
!>               colonnes k et -IPIV(k) ont été interchangées et les
!>               lignes et colonnes k+1 et -IPIV(k+1) ont été
!>               interchangées,
!>               D(k:k+1,k:k+1) est un bloc diagonal 2 par 2.
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice du second membre B de dimension N
!>          par NRHS.
!>          À la sortie, si INFO = 0, la matrice solution X de dimension
!>          N par NRHS.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La longueur de WORK.  LWORK >= 1, et pour de meilleures
!>          performances LWORK >= max(1,N*NB), où NB est la taille de
!>          bloc optimale pour CSYTRF_ROOK.
!>
!>          TRS sera effectué avec les BLAS de niveau 2
!>
!>          Si LWORK = -1, alors une interrogation d'espace de travail
!>          est supposée ; la routine calcule uniquement la taille
!>          optimale du tableau WORK, renvoie cette valeur comme
!>          première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur
!>               illégale
!>          > 0 : si INFO = i, D(i,i) est exactement zéro.  La
!>               factorisation a été complétée, mais la matrice
!>               bloc-diagonale D est exactement singulière, de sorte
!>               que la solution n'a pas pu être calculée.
!
>
