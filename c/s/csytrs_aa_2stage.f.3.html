sous-routine csytrs_aa_2stage (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, complex, dimension( ldb, * ) b, integer ldb, integer info)

CSYTRS_AA_2STAGE

But:

!>
!> CSYTRS_AA_2STAGE résout un système d'équations linéaires A*X = B avec une matrice
!> complexe symétrique A en utilisant la factorisation A = U**T*T*U ou
!> A = L*T*L**T calculée par CSYTRF_AA_2STAGE.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si les détails de la factorisation sont stockés
!>          sous forme de matrice triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure, forme A = U**T*T*U;
!>          = 'L':  Triangulaire inférieure, forme A = L*T*L**T.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de côtés droits, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          Détails des facteurs calculés par CSYTRF_AA_2STAGE.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

TB

!>          TB est un tableau COMPLEX, dimension (LTB)
!>          Détails des facteurs calculés par CSYTRF_AA_2STAGE.
!>

LTB

!>          LTB est INTEGER
!>          La taille du tableau TB. LTB >= 4*N.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Détails des échanges calculés par
!>          CSYTRF_AA_2STAGE.
!>

IPIV2

!>          IPIV2 est un tableau INTEGER, dimension (N)
!>          Détails des échanges calculés par
!>          CSYTRF_AA_2STAGE.
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice des côtés droits B.
!>          À la sortie, la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0:  terminé avec succès
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur illégale
!>
