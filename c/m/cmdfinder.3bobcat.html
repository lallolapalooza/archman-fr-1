Bien que les associations entre les commandes et les fonctions (membres) soient souvent définies dans un switch, un switch n'est pas la méthode privilégiée pour définir ces associations en raison du fait que la maintenabilité et la clarté des switchs souffrent même pour des ensembles de commandes de taille modérée. De plus, le switch n'est presque jamais autoporteur, puisque généralement un certain traitement des commandes est nécessaire pour déterminer les command/ associations auxvaleurs de case.

L'approche

alternative (et privilégiée), qui est également adoptée par CmdFinder, consiste à définir un tableau de pointeurs vers des fonctions (membres), et à définir les associations entre les commandes et les fonctions membres comme un mappage des commandes vers des indices de tableau. Les associations simples entre des commandes (textuelles) et les fonctions à appeler peuvent également être facilement définies à l'aide d'un std::map ou d'une autre structure de données de type hachage. Cependant, les exigences syntaxiques pour une telle structure std::map ne sont pas triviales, et de plus : les commandes saisies par l'utilisateur nécessitent souvent

#include <iostream>
#include <string>
#include <bobcat/cmdfinder>
using namespace std;
using namespace FBB;
// Définit une classe `Command’ dans laquelle le tableau s_action définit les
// associations commande-fonction. Command est dérivée de CmdFinder,
// spécifiant le prototype des fonctions membres à appeler
class Command: public CmdFinder<bool (Command::*)() const>
{
    static Entry s_action[];
    bool add() const                       // quelques fonctions membres
    {
        cout << "add called: command was `" << cmd() << "’\n";
        if (beyond().length())
            cout << "Beyond " << cmd() << " `" << beyond() << "’\n";
        return true;
    }
    bool error() const
    {
        cout << "unrecognized command: `" << cmd() << "’ called\n" <<
                count() << " matching alternatives found\n";
        return true;
    }
    bool quit() const
    {
        cout << "quit called: quitting this series\n";
        return false;
    }
    public:
        Command();                      // Déclare le constructeur par défaut
        bool run(std::string const &cmd)    // exécute une commande
        {
            return (this->*findCmd(cmd))(); // exécute la commande
                                            // correspondant à ’cmd’
        }
};
// Définit les associations commande-fonction. Notez que la dernière est
// donnée avec un texte de commande vide. Cela n'est pas requis, un texte de
// commande aurait pu être spécifié pour la dernière commande également.
Command::Entry Command::s_action[] =
{
    Entry("add",    &Command::add),
    Entry("quit",   &Command::quit),
    Entry("",       &Command::error),
};
// Définit le constructeur par défaut
Command::Command()                       // Définit le constructeur par défaut
:                               // Notez l'utilisation de `FunctionPtr’
    CmdFinder<FunctionPtr>(s_action, s_action +
                                    sizeof(s_action) / sizeof(Entry))
{}
void run(Command &cmd, char const *descr, size_t mode = 0)
{
    if (mode)
        cmd.setMode(mode);
    cout << "Enter 5 x a command using " << descr << ".\n";
    for (size_t idx = 0; idx++ < 5; )
    {
        cout << "Enter command " << idx << ": ";
        string text;
        getline(cin, text);
        if (!cmd.run(text))     // exécute une commande
            break;
    }
}
int main()
{
    Command cmd;                // définit une commande
                                // entre 5 commandes en utilisant le mode par défaut
    run (cmd, "the default mode");
    run (cmd, "abbreviated commands", Command::UNIQUE);
    run (cmd, "abbreviated case-insensitive commands",
                                   Command::UNIQUE | Command::INSENSITIVE);
    run (cmd, "abbreviated command lines",
                                   Command::USE_FIRST | Command::UNIQUE);
    run (cmd, "abbreviated case-insensitive command lines",
                                   Command::USE_FIRST | Command::UNIQUE |
                                                        Command::INSENSITIVE);
}
