subroutine cla_syrfsx_extended (integer prec_type, character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, logical colequ, real, dimension( * ) c, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldy, * ) y, integer ldy, real, dimension( * ) berr_out, integer n_norms, real, dimension( nrhs, * ) err_bnds_norm, real, dimension( nrhs, * ) err_bnds_comp, complex, dimension( * ) res, real, dimension( * ) ayb, complex, dimension( * ) dy, complex, dimension( * ) y_tail, real rcond, integer ithresh, real rthresh, real dz_ub, logical ignore_cwise, integer info)

CLA_SYRFSX_EXTENDED améliore la solution calculée d'un système d'équations linéaires pour les matrices symétriques indéfinies en effectuant un raffinement itératif de précision supplémentaire et fournit des bornes d'erreur et des estimations d'erreur arrière pour la solution.

But :

!>
!> CLA_SYRFSX_EXTENDED améliore la solution calculée d'un système
!> d'équations linéaires en effectuant un raffinement itératif de
!> précision supplémentaire et fournit des bornes d'erreur et des
!> estimations d'erreur arrière pour la solution.
!> Cette sous-routine est appelée par CSYRFSX pour effectuer le
!> raffinement itératif. En plus de la borne d'erreur en norme, le
!> code fournit, si possible, une borne d'erreur composante par
!> composante. Voir les commentaires pour ERR_BNDS_NORM et
!> ERR_BNDS_COMP pour les détails des bornes d'erreur. Notez que
!> cette sous-routine est uniquement responsable de la définition
!> des deuxième champs de ERR_BNDS_NORM et ERR_BNDS_COMP.
!>

Paramètres

PREC_TYPE 
!>          PREC_TYPE est INTEGER
!>     Spécifie la précision intermédiaire à utiliser pour le raffinement.
!>     La valeur est définie par ILAPREC(P) où P est un CHARACTER et P
!>          = 'S' :  Simple (Single)
!>          = 'D' :  Double
!>          = 'I' :  Indigène (Indigenous)
!>          = 'X' ou 'E' :  Extra
!>

UPLO

!>          UPLO est CHARACTER*1
!>       = 'U' :  Le triangle supérieur de A est stocké ;
!>       = 'L' :  Le triangle inférieur de A est stocké.
!>

N

!>          N est INTEGER
!>     Le nombre d'équations linéaires, c'est-à-dire l'ordre de la
!>     matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>     Le nombre de seconds membres, c'est-à-dire le nombre de colonnes de la
!>     matrice B.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,N)
!>     À l'entrée, la matrice A de taille N par N.
!>

LDA

!>          LDA est INTEGER
!>     La dimension principale du tableau A.  LDA >= max(1,N).
!>

AF

!>          AF est un tableau COMPLEX, dimension (LDAF,N)
!>     La matrice diagonale par blocs D et les multiplicateurs utilisés pour
!>     obtenir le facteur U ou L tels que calculés par CSYTRF.
!>

LDAF

!>          LDAF est INTEGER
!>     La dimension principale du tableau AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>     Détails des échanges et de la structure par blocs de D
!>     tels que déterminés par CSYTRF.
!>

COLEQU

!>          COLEQU est LOGICAL
!>     Si .TRUE., un équilibrage par colonnes a été effectué sur A avant
!>     d'appeler cette routine. Cela est nécessaire pour calculer
!>     correctement la solution et les bornes d'erreur.
!>

C

!>          C est un tableau REAL, dimension (N)
!>     Les facteurs d'échelle par colonne pour A. Si COLEQU = .FALSE., C
!>     n'est pas accédé. Si C est fourni en entrée, chaque élément de C devrait être une
!>     puissance de la base pour garantir une solution et des estimations
!>     d'erreur fiables. La mise à l'échelle par des puissances de la base ne
!>     provoque pas d'erreurs d'arrondi, sauf si le résultat déborde ou
!>     est trop petit. Les erreurs d'arrondi lors de la mise à l'échelle
!>     conduisent à un raffinement avec une matrice qui n'est pas équivalente
!>     à la matrice d'entrée, produisant des estimations d'erreur qui peuvent
!>     ne pas être fiables.
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>     La matrice des seconds membres B.
!>

LDB

!>          LDB est INTEGER
!>     La dimension principale du tableau B.  LDB >= max(1,N).
!>

Y

!>          Y est un tableau COMPLEX, dimension (LDY,NRHS)
!>     À l'entrée, la matrice solution X, telle que calculée par CSYTRS.
!>     À la sortie, la matrice solution améliorée Y.
!>

LDY

!>          LDY est INTEGER
!>     La dimension principale du tableau Y.  LDY >= max(1,N).
!>

BERR_OUT

!>          BERR_OUT est un tableau REAL, dimension (NRHS)
!>     À la sortie, BERR_OUT(j) contient l'erreur arrière relative composante
!>     par composante pour le second membre j donnée par la formule
!>         max(i) ( abs(RES(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>     où abs(Z) est la valeur absolue composante par composante de la
!>     matrice ou du vecteur Z. Ceci est calculé par CLA_LIN_BERR.
!>

N_NORMS

!>          N_NORMS est INTEGER
!>     Détermine quelles bornes d'erreur retourner (voir ERR_BNDS_NORM
!>     et ERR_BNDS_COMP).
!>     Si N_NORMS >= 1, retourne les bornes d'erreur en norme.
!>     Si N_NORMS >= 2, retourne les bornes d'erreur composante par composante.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM est un tableau REAL, dimension (NRHS, N_ERR_BNDS)
!>     Pour chaque second membre, ce tableau contient des informations sur
!>     les diverses bornes d'erreur et nombres de condition correspondant à
!>     l'erreur relative en norme, qui est définie comme suit :
!>
!>     Erreur relative en norme dans le ième vecteur solution :
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     Le tableau est indexé par le type d'information d'erreur comme décrit
!>     ci-dessous. Il y a actuellement jusqu'à trois informations
!>     retournées.
!>
!>     Le premier index dans ERR_BNDS_NORM(i,:) correspond au ième
!>     second membre.
!>
!>     Le deuxième index dans ERR_BNDS_NORM(:,err) contient les trois
!>     champs suivants :
!>     err = 1  booléen. Faire confiance à la réponse si le
!>              nombre de condition réciproque est inférieur au seuil
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  borne d'erreur : L'erreur avant estimée, presque
!>              certainement dans un facteur de 10 de l'erreur réelle
!>              tant que l'entrée suivante est supérieure au seuil
!>              sqrt(n) * slamch('Epsilon'). Cette borne d'erreur ne devrait
!>              être fiable que si le booléen précédent est vrai.
!>
!>     er

Function/Subroutine Documentation


cla_syrfsx_extended performs a detailed error analysis of solutions to
    a system of linear equations A*X = B. It provides more diagnostic
    information than the standard routine cla_syrfs.


This routine computes the componentwise error bounds for the
    solutions to a system of complex linear equations


A*X = B


after which the solutions are refined. The routines compute the
    forward error, a reciprocal condition number and a bound on the forward
    error.


The routine is particularly useful for assessing the accuracy of
    solutions computed by routines such as cla_getrf, cla_getri, cla_getrs.


Arguments:


UPLO is CHARACTER*1


Specifies whether the matrix A is upper or lower triangular as
    follows:



  
'U'

  
A is upper triangular;

  
'L'

  
A is lower triangular.




N is INTEGER


The order of the matrix A. N >= 0.


NRHS is INTEGER


The number of right hand sides, i.e., the number of columns of
    the matrices B, X, and XTRUE. NRHS >= 0.


A is COMPLEX array, dimension (LDA,N)


On entry, the original N x N matrix A.


LDA is INTEGER


The leading dimension of the array A. LDA >= max(1, N).


AF is COMPLEX array, dimension (LDAF,N)


On entry, the factors of A. AF contains the L/U factors of A from
    cla_getrf.


LDAF is INTEGER


The leading dimension of the array AF. LDAF >= max(1, N).


IPIV is INTEGER array, dimension (min(N,N))


The pivot indices from the factorization of A. Column k of the
    factorization was interchanged with column IPIV(k).


B is COMPLEX array, dimension (LDB,NRHS)


On entry, the N x NRHS matrix of right hand side vectors.


LDB is INTEGER


The leading dimension of the array B. LDB >= max(1, N).


X is COMPLEX array, dimension (LDX,NRHS)


On entry, the N x NRHS matrix of approximate solutions. On exit,
    the N x NRHS matrix of solutions computed by this routine.


LDX is INTEGER


The leading dimension of the array X. LDX >= max(1, N).


XTRUE is COMPLEX array, dimension (LDX,NRHS)


On entry, the N x NRHS matrix of the true solutions.


ADJNC is COMPLEX array, dimension (N)


Workspace to hold the intermediate solution.


B_EXT is COMPLEX array, dimension (N)


Workspace to hold the intermediate right hand side.


Y is COMPLEX array, dimension (N)


Workspace to hold the intermediate solution.


LDBY is INTEGER


The leading dimension of the array Y. LDBY >= max(1, N).


ERR_BNDS_NORM


          ERR_BNDS_NORM is REAL array, dimension (NRHS, 3)
     For each right hand side, this array contains information about
     various error bounds and condition numbers corresponding to the
     normwise error, which is defined as follows:

     Normwise relative error in the ith solution vector:
                     norm(XTRUE(i) - X(i), inf)
             -----------------------------------------
                           norm(X(i), inf)

     The array is indexed by the right-hand side i (on which the
     normwise relative error depends), and the type of error
     information as described below. There currently are up to three
     pieces of information returned for each right-hand side. If
     normwise accuracy is not requested (PARAMS(3) = 0.0), then
     ERR_BNDS_NORM is not accessed. If N_ERR_BNDS < 3, then at most
     the first (:,N_ERR_BNDS) entries are returned.

     The first index in ERR_BNDS_NORM(i,:) corresponds to the ith
     right-hand side.

     The second index in ERR_BNDS_NORM(:,err) contains the following
     three fields:
     err = 1  boolean. Trust the answer if the
              reciprocal condition number is less than the threshold
              sqrt(n) * slamch('Epsilon').

     err = 2  error bound: The estimated forward error,
              almost certainly within a factor of 10 of the true error
              so long as the next entry is greater than the threshold
              sqrt(n) * slamch('Epsilon'). This error bound should only
              be trusted if the previous boolean is true.

     err = 3  Reciprocal condition number: Estimated normwise
              reciprocal condition number.  Compared with the threshold
              sqrt(n) * slamch('Epsilon') to determine if the error
              estimate is . These reciprocal condition
              numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
              appropriately scaled matrix Z.
              Let Z = S*A, where S scales each row by a power of the
              radix so all absolute row sums of Z are approximately 1.

     This subroutine is only responsible for setting the second field
     above.
     See Lapack Working Note 165 for further details and extra
     cautions.



ERR_BNDS_CWISE

!>     For each right-hand side, this array contains information about
!>     various error bounds and condition numbers corresponding to the
!>     componentwise relative error, which is defined as follows:
!>
!>     Componentwise relative error in the ith solution vector:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------

!>
!>     The array is indexed by the right-hand side i (on which the
!>     componentwise relative error depends), and the type of error

!>     componentwise accuracy is not requested (PARAMS(3) = 0.0), then
!>     ERR_BNDS_CWISE is not accessed.  If N_ERR_BNDS < 3, then at most
!>     the first (:,N_ERR_BNDS) entries are returned.
!>
!>     The first index in ERR_BNDS_CWISE(i,:) corresponds to the ith
!>     right-hand side.
!>
!>     The second index in ERR_BNDS_CWISE(:,err) contains the following
!>     three fields:
!>     err = 1  boolean. Trust the answer if the
!>              reciprocal condition number is less than the threshold
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  error bound: The estimated forward error,
!>              almost certainly within a factor of 10 of the true error
!>              so long as the next entry is greater than the threshold
!>              sqrt(n) * slamch('Epsilon'). This error bound should only
!>              be trusted if the previous boolean is true.
!>
!>     err = 3  Reciprocal condition number: Estimated componentwise
!>              reciprocal condition number.  Compared with the threshold
!>              sqrt(n) * slamch('Epsilon') to determine if the error
!>              estimate is . These reciprocal condition
!>              numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
!>              appropriately scaled matrix Z.
!>              Let Z = S*(A*diag(x)), where x is the solution for the

!>              A*diag(x) by a power of the radix so all absolute row
!>              sums of Z are approximately 1.
!>
!>     This subroutine is only responsible for setting the second field
!>     above.
!>     See Lapack Working Note 165 for further details and extra
!>     cautions.
!> 


RES


!>          RES is COMPLEX array, dimension (N)
!>     Workspace to hold the intermediate residual.
!> 


AYB


!>          AYB is REAL array, dimension (N)
!>     Workspace.
!> 


DY


!>          DY is COMPLEX array, dimension (N)
!>     Workspace to hold the intermediate solution.
!> 


Y_TAIL


!>          Y_TAIL is COMPLEX array, dimension (N)
!>     Workspace to hold the trailing bits of the intermediate solution.
!> 


RCOND


!>          RCOND is REAL
!>     Reciprocal scaled condition number.  This is an estimate of the
!>     reciprocal Skeel condition number of the matrix A after
!>     equilibration (if done).  If this is less than the machine
!>     precision (in particular, if it is zero), the matrix is singular
!>     to working precision.  Note that the error may still be small even
!>     if this number is very small and the matrix appears ill-
!>     conditioned.
!> 


ITHRESH


!>          ITHRESH is INTEGER
!>     The maximum number of residual computations allowed for
!>     refinement. The default is 10. For 'aggressive' set to 100 to
!>     permit convergence using approximate factorizations or
!>     factorizations other than LU. If the factorization uses a
!>     technique other than Gaussian elimination, the guarantees in
!>     ERR_BNDS_NORM and ERR_BNDS_CWISE may no longer be trustworthy.
!> 


RTHRESH


!>          RTHRESH is REAL
!>     Determines when to stop refinement if the error estimate stops
!>     decreasing. Refinement will stop when the next solution no longer
!>     satisfies norm(dx_{i+1}) < RTHRESH * norm(dx_i) where norm(Z) is
!>     the infinity norm of Z. RTHRESH satisfies 0 < RTHRESH <= 1. The
!>     default value is 0.5. For 'aggressive' set to 0.9 to permit
!>     convergence on extremely ill-conditioned matrices. See LAWN 165
!!>     for more details.
!> 


DZ_UB


!>          DZ_UB is REAL
!>     Determines when to start considering componentwise convergence.
!>     Componentwise convergence is only considered after each component
!>     of the solution Y is stable, which we define as the relative
!>     change in each component being less than DZ_UB. The default value
!>     is 0.25, requiring the first bit to be stable. See LAWN 165 for
!>     more details.
!> 


IGNORE_CWISE


!>          IGNORE_CWISE is LOGICAL
!>     If .TRUE. then ignore componentwise convergence. Default value
!>     is .FALSE..
!> 


INFO


!>          INFO is INTEGER
!>       = 0:  Successful exit.
!>       < 0:  if INFO = -i, the ith argument to CLA_SYRFSX_EXTENDED had an illegal
!>             value
!>
