subroutine clagtm (character trans, integer n, integer nrhs, real alpha, complex, dimension( * ) dl, complex, dimension( * ) d, complex, dimension( * ) du, complex, dimension( ldx, * ) x, integer ldx, real beta, complex, dimension( ldb, * ) b, integer ldb)

CLAGTM effectue un produit matrice-matrice de la forme C = αAB+βC, où A est une matrice tridiagonale, B et C sont des matrices rectangulaires, et α et β sont des scalaires, qui peuvent être 0, 1, ou -1.

Objectif:

!>
!> CLAGTM effectue un produit matrice-matrice de la forme
!>
!>    B := alpha * A * X + beta * B
!>
!> où A est une matrice tridiagonale d'ordre N, B et X sont des matrices
!> N par NRHS, et alpha et beta sont des scalaires réels, chacun pouvant
!> être 0., 1., ou -1.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie l'opération appliquée à A.
!>          = 'N':  Pas de transposée, B := alpha * A * X + beta * B
!>          = 'T':  Transposée,    B := alpha * A**T * X + beta * B
!>          = 'C':  Transposée conjuguée, B := alpha * A**H * X + beta * B
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.
!>

ALPHA

!>          ALPHA est REAL
!>          Le scalaire alpha.  ALPHA doit être 0., 1., ou -1.; sinon,
!>          il est supposé être 0.
!>

DL

!>          DL est tableau COMPLEX, dimension (N-1)
!>          Les éléments sous-diagonaux de T (N-1).
!>

D

!>          D est tableau COMPLEX, dimension (N)
!>          Les éléments diagonaux de T.
!>

DU

!>          DU est tableau COMPLEX, dimension (N-1)
!>          Les éléments super-diagonaux de T (N-1).
!>

X

!>          X est tableau COMPLEX, dimension (LDX,NRHS)
!>          La matrice X de dimensions N par NRHS.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(N,1).
!>

BETA

!>          BETA est REAL
!>          Le scalaire beta.  BETA doit être 0., 1., ou -1.; sinon,
!>          il est supposé être 1.
!>

B

!>          B est tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice B de dimensions N par NRHS.
!>          À la sortie, B est écrasée par l'expression matricielle
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(N,1).
!>
