subroutine clakf2 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( lda, * ) b, complex, dimension( lda, * ) d, complex, dimension( lda, * ) e, complex, dimension( ldz, * ) z, integer ldz)

CLAKF2

But:

!>
!> Former la matrice 2*M*N par 2*M*N
!>
!>        Z = [ kron(In, A)  -kron(B', Im) ]
!>            [ kron(In, D)  -kron(E', Im) ],
!>
!> où In est la matrice identité de taille n et X' est la transposée
!> de X. kron(X, Y) est le produit de Kronecker entre les matrices X
!> et Y.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Taille de la matrice, doit être >= 1.
!>

N

!>          N est un entier (INTEGER)
!>          Taille de la matrice, doit être >= 1.
!>

A

!>          A est un nombre complexe (COMPLEX), dimension ( LDA, M )
!>          La matrice A dans la matrice de sortie Z.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale de A, B, D et E. ( LDA >= M+N )
!>

B

!>          B est un nombre complexe (COMPLEX), dimension ( LDA, N )
!>

D

!>          D est un nombre complexe (COMPLEX), dimension ( LDA, M )
!>

E

!>          E est un nombre complexe (COMPLEX), dimension ( LDA, N )
!>
!>          Les matrices utilisées pour former la matrice de sortie Z.
!>

Z

!>          Z est un nombre complexe (COMPLEX), dimension ( LDZ, 2*M*N )
!>          La matrice résultante de Kronecker M*N*2 par M*N*2 (voir ci-dessus.)
!>

LDZ

!>          LDZ est un entier (INTEGER)
!>          La dimension principale de Z. ( LDZ >= 2*M*N )
!>
