subroutine clamswlq (character side, character trans, integer m, integer n, integer k, integer mb, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CLAMSWLQ

But :

!>
!>    CLAMSWLQ remplace la matrice générale complexe C de taille M par N par
!>
!>
!>                    SIDE = 'L'     SIDE = 'R'
!>    TRANS = 'N':      Q * C          C * Q
!>    TRANS = 'T':      Q**H * C       C * Q**H
!>    où Q est une matrice unitaire complexe définie comme le produit de
!>    réflecteurs élémentaires par blocs calculés par la factorisation
!>    LQ courte et large (CLASWLQ)
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L' : appliquer Q ou Q**H à gauche ;
!>          = 'R' : appliquer Q ou Q**H à droite.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N' : pas de transposition, appliquer Q ;
!>          = 'C' : transposition conjuguée, appliquer Q**H.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C.  M >=0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C. N >= 0.
!>

K

!>          K est INTEGER
!>          Le nombre de réflecteurs élémentaires dont le produit définit
!>          la matrice Q.
!>          M >= K >= 0;
!>
!>

MB

!>          MB est INTEGER
!>          La taille des blocs de lignes à utiliser dans le LQ par blocs.
!>          M >= MB >= 1
!>

NB

!>          NB est INTEGER
!>          La taille des blocs de colonnes à utiliser dans le LQ par blocs.
!>          NB > M.
!>

A

!>          A est un tableau COMPLEX, dimension
!>                               (LDA,M) si SIDE = 'L',
!>                               (LDA,N) si SIDE = 'R'
!>          La i-ième ligne doit contenir le vecteur qui définit le
!>          réflecteur élémentaire par blocs H(i), pour i = 1,2,...,k,
!>          tel que renvoyé par CLASWLQ dans les k premières lignes de son
!>          argument tableau A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA => max(1,K).
!>

T

!>          T est un tableau COMPLEX, dimension
!>          ( M * Nombre de blocs(CEIL(N-K/NB-K)),
!>          Les réflecteurs de blocs triangulaires supérieurs par blocs
!>          stockés sous forme compacte comme une séquence de blocs
!>          triangulaires supérieurs. Voir ci-dessous pour plus de détails.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T.  LDT >= MB.
!>

C

!>          C est un tableau COMPLEX, dimension (LDC,N)
!>          À l'entrée, la matrice C de taille M par N.
!>          À la sortie, C est remplacée par Q*C ou Q**H*C ou C*Q**H ou C*Q.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          (espace de travail) tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK minimal.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK.
!>          Si MIN(M,N,K) = 0, LWORK >= 1.
!>          Si SIDE = 'L', LWORK >= max(1,NB*MB).
!>          Si SIDE = 'R', LWORK >= max(1,M*MB).
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille minimale
!>          du tableau WORK, renvoie cette valeur comme première entrée
!>          du tableau WORK, et aucun message d'erreur lié à LWORK n'est
!>          émis par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ième argument avait une valeur illégale
!>

Détails supplémentaires :

!> Short-Wide LQ (SWLQ) effectue un LQ par une séquence de transformations
!> unitaires, représentant Q comme un produit d'autres matrices unitaires
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> où chaque Q(i) met à zéro les entrées diagonales supérieures d'un bloc de
!> NB lignes de A :
!>   Q(1) met à zéro les entrées diagonales supérieures des lignes 1:NB de A
!>   Q(2) met à zéro les MB-N lignes du bas des lignes [1:M,NB+1:2*NB-M] de A
!>   Q(3) met à zéro les MB-N lignes du bas des lignes [1:M,2*NB-M+1:3*NB-2*M] de A
!>   . . .
!>
!> Q(1) est calculé par GELQT, qui représente Q(1) par des vecteurs de
!> Householder stockés sous la diagonale des lignes 1:MB de A, et par des
!> réflecteurs de blocs triangulaires supérieurs, stockés dans le tableau
!> T(1:LDT,1:N).
!> Pour plus d'informations, voir Détails supplémentaires dans GELQT.
!>
!> Q(i) pour i>1 est calculé par TPLQT, qui représente Q(i) par des vecteurs
!> de Householder stockés dans les colonnes [(i-1)*(NB-M)+M+1:i*(NB-M)+M] de A,
!> et par des réflecteurs de blocs triangulaires supérieurs, stockés dans le
!> tableau T(1:LDT,(i-1)*M+1:i*M).
!> Le dernier Q(k) peut utiliser moins de lignes.
!> Pour plus d'informations, voir Détails supplémentaires dans TPLQT.
!>
!> Pour plus de détails sur l'algorithme global, voir la description du
!> TSQR séquentiel dans la section 2.2 de [1].
!>
!> [1] « Communication-Optimal Parallel and Sequential QR and LU Factorizations, »
!>     J. Demmel, L. Grigori, M. Hoemmen, J. Langou,
!>     SIAM J. Sci. Comput, vol. 34, no. 1, 2012
!>
