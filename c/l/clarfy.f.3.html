subroutine clarfy (character uplo, integer n, complex, dimension( * ) v, integer incv, complex tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work)

CLARFY

Objectif:

!>
!> CLARFY applique un réflecteur élémentaire, ou matrice de Householder, H,
!> à une matrice hermitienne n x n C, à la fois par la gauche et par la droite.
!>
!> H est représentée sous la forme
!>
!>    H = I - tau * v * v'
!>
!> où  tau  est un scalaire et  v  est un vecteur.
!>
!> Si  tau  est  zéro, alors  H  est considérée comme la matrice unité.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la partie triangulaire supérieure ou inférieure de la
!>          matrice hermitienne C est stockée.
!>          = 'U':  Triangle supérieur
!>          = 'L':  Triangle inférieur
!>

N

!>          N est INTEGER
!>          Le nombre de lignes et de colonnes de la matrice C.  N >= 0.
!>

V

!>          V est un tableau COMPLEX, dimension
!>                  (1 + (N-1)*abs(INCV))
!>          Le vecteur v tel que décrit ci-dessus.
!>

INCV

!>          INCV est INTEGER
!>          L'incrément entre les éléments successifs de v.  INCV ne doit
!>          pas être nul.
!>

TAU

!>          TAU est COMPLEX
!>          La valeur tau telle que décrite ci-dessus.
!>

C

!>          C est un tableau COMPLEX, dimension (LDC, N)
!>          À l'entrée, la matrice C.
!>          À la sortie, C est écrasée par H * C * H'.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C.  LDC >= max( 1, N ).
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (N)
!>
