subroutine clals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, real, dimension( ldgnum, * ) givnum, integer ldgnum, real, dimension( ldgnum, * ) poles, real, dimension( * ) difl, real, dimension( ldgnum, * ) difr, real, dimension( * ) z, integer k, real c, real s, real, dimension( * ) rwork, integer

CLALS0 applique les facteurs multiplicatifs en retour dans la résolution du problème des moindres carrés en utilisant l'approche SVD par division et conquête. Utilisé par

sgelsd.

But

: 
!>
!> CLALS0 applique en retour les facteurs multiplicatifs soit de la matrice
!> des vecteurs singuliers gauches soit de la matrice des vecteurs singuliers
!> droites d'une matrice diagonale à laquelle on a ajouté une ligne, à la
!> matrice du membre de droite B dans la résolution du problème des moindres
!> carrés en utilisant l'approche SVD par division et conquête.
!>
!> Pour la matrice des vecteurs singuliers gauches, trois types de matrices
!> orthogonales sont impliqués :
!>
!> (1L) Rotations de Givens : le nombre de telles rotations est GIVPTR ; les
!>      paires de colonnes/lignes auxquelles elles ont été appliquées sont
!>      stockées dans GIVCOL ; et les valeurs C et S de ces rotations sont
!>      stockées dans GIVNUM.
!>
!> (2L) Permutation. La (NL+1)-ème ligne de B doit être déplacée vers la
!>      première ligne, et pour J=2:N, la PERM(J)-ème ligne de B doit être
!>      déplacée vers la J-ème ligne.
!>
!> (3L) La matrice des vecteurs singuliers gauches de la matrice restante.
!>
!> Pour la matrice des vecteurs singuliers droites, quatre types de matrices
!> orthogonales sont impliqués :
!>
!> (1R) La matrice des vecteurs singuliers droites de la matrice restante.
!>
!> (2R) Si SQRE = 1, une rotation de Givens supplémentaire pour générer
!>      l'espace nul droit.
!>
!> (3R) La transformation inverse de (2L).
!>
!> (4R) La transformation inverse de (1L).
!
>

Paramètre

s
ICOMPQ 
!>          ICOMPQ est INTEGER
!>         Spécifie si les vecteurs singuliers doivent être calculés sous
!>         forme factorisée :
!>         = 0 : Matrice des vecteurs singuliers gauches.
!>         = 1 : Matrice des vecteurs singuliers droites.
!
>

N

L 
!>          NL est INTEGER
!>         La dimension de ligne du bloc supérieur. NL >= 1.
!
>

N

R 
!>          NR est INTEGER
!>         La dimension de ligne du bloc inférieur. NR >= 1.
!
>

SQR

E 
!>          SQRE est INTEGER
!>         = 0 : le bloc inférieur est une matrice carrée NR-par-NR.
!>         = 1 : le bloc inférieur est une matrice rectangulaire NR-par-(NR+1).
!>
!>         La matrice bidiagonale a une dimension de ligne N = NL + NR + 1,
!>         et une dimension de colonne M = N + SQRE.
!
>

NRH

S 
!>          NRHS est INTEGER
!>         Le nombre de colonnes de B et BX. NRHS doit être au moins 1.
!
>

B 
!>          B est un tableau COMPLEX, dimension ( LDB, NRHS )
!>         À l'entrée, B contient les membres de droite du problème des
!>         moindres carrés dans les lignes 1 à M. À la sortie, B contient
!>         la solution X dans les lignes 1 à N.
!
>

LD

B 
!>          LDB est INTEGER
!>         La dimension principale de B. LDB doit être au moins
!>         max(1,MAX( M, N ) ).
!
>

B

X 
!>          BX est un tableau COMPLEX, dimension ( LDBX, NRHS )
!
>

LDB

X 
!>          LDBX est INTEGER
!>         La dimension principale de BX.
!
>

PER

M 
!>          PERM est un tableau INTEGER, dimension ( N )
!>         Les permutations (issues du déflation et du tri) appliquées
!>         aux deux blocs.
!
>

GIVPT

R 
!>          GIVPTR est INTEGER
!>         Le nombre de rotations de Givens qui ont eu lieu dans ce
!>         sous-problème.
!
>

GIVCO

L 
!>          GIVCOL est un tableau INTEGER, dimension ( LDGCOL, 2 )
!>         Chaque paire de nombres indique une paire de lignes/colonnes
!>         impliquée dans une rotation de Givens.
!
>

LDGCO

L 
!>          LDGCOL est INTEGER
!>         La dimension principale de GIVCOL, doit être au moins N.
!
>

GIVNU

M 
!>          GIVNUM est un tableau REAL, dimension ( LDGNUM, 2 )
!>         Chaque nombre indique la valeur C ou S utilisée dans la
!>         rotation de Givens correspondante.
!
>

LDGNU

M 
!>          LDGNUM est INTEGER
!>         La dimension principale des tableaux DIFR, POLES et
!>         GIVNUM, doit être au moins K.
!
>

POLE

S 
!>          POLES est un tableau REAL, dimension ( LDGNUM, 2 )
!>         À l'entrée, POLES(1:K, 1) contient les nouvelles valeurs
!>         singulières obtenues en résolvant l'équation séculière, et
!>         POLES(1:K, 2) est un tableau contenant les pôles dans
!>         l'équation séculière.
!
>

DIF

L 
!>          DIFL est un tableau REAL, dimension ( K ).
!>         À l'entrée, DIFL(I) est la distance entre la I-ème valeur
!>         singulière mise à jour (non déflée) et la I-ème (non déflée)
!>         ancienne valeur singulière.
!
>

DIF

R 
!>          DIFR est un tableau REAL, dimension ( LDGNUM, 2 ).
!>         À l'entrée, DIFR(I, 1) contient les distances entre la I-ème
!>         valeur singulière mise à jour (non déflée) et la I+1-ème
!>         (non déflée) ancienne valeur singulière. Et DIFR(I, 2) est le
!>         facteur de normalisation pour le I-ème vecteur singulier droit.
!
>

Z 
!>          Z est un tableau REAL, dimension ( K )
!>         Contient les composantes du vecteur ligne de mise à jour
!>         ajusté par déflation.
!
>

K 
!>          K est INTEGER
!>         Contient la dimension de la matrice non déflée,
!>         C'est l'ordre de l'équation séculière associée. 1 <= K <=N.
!
>

C 
!>          C est REAL
!>         C contient des données inutiles si SQRE =0 et la valeur C d'une
!>         rotation de Givens liée à l'espace nul droit si SQRE = 1.
!
>

S 
!>          S est REAL
!>         S contient des données inutiles si SQRE =0 et la valeur S d'une
!>         rotation de Givens liée à l'espace nul droit si SQRE = 1.
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension
!>         ( K*(1+NRHS) + 2*NRHS )
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie.
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!
>
