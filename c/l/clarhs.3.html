sous-routine clarhs (character*3 path, character xtype, character uplo, character trans, integer m, integer n, integer kl, integer ku, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, integer, dimension( 4 ) iseed, integer

info)

CLARH

S

But

: 
!>
!> CLARHS choisit un ensemble de NRHS vecteurs de solution aléatoires et définit
!> les seconds membres pour le système linéaire
!>    op(A) * X = B,
!> où op(A) = A, A**T, ou A**H, selon TRANS.
!
>

Paramètre

s
PATH 
!>          PATH est CHARACTER*3
!>          Le type de la matrice complexe A.  PATH peut être donné dans n'importe
!>          quelle combinaison de majuscules et de minuscules.  Les chemins valides incluent
!>             xGE :  Matrice générale m x n
!>             xGB :  Matrice générale par bandes
!>             xPO :  Hermittienne définie positive, stockage 2-D
!>             xPP :  Hermittienne définie positive compactée
!>             xPB :  Hermittienne définie positive par bandes
!>             xHE :  Hermittienne indéfinie, stockage 2-D
!>             xHP :  Hermittienne indéfinie compactée
!>             xHB :  Hermittienne indéfinie par bandes
!>             xSY :  Symétrique indéfinie, stockage 2-D
!>             xSP :  Symétrique indéfinie compactée
!>             xSB :  Symétrique indéfinie par bandes
!>             xTR :  Triangulaire
!>             xTP :  Triangulaire compactée
!>             xTB :  Triangulaire par bandes
!>             xQR :  Matrice générale m x n
!>             xLQ :  Matrice générale m x n
!>             xQL :  Matrice générale m x n
!>             xRQ :  Matrice générale m x n
!>          où le caractère initial indique la précision.
!
>

XTYP

E 
!>          XTYPE est CHARACTER*1
!>          Spécifie comment la solution exacte X sera déterminée :
!>          = 'N' :  Nouvelle solution ; générer un X aléatoire.
!>          = 'C' :  Calculée ; utiliser la valeur de X à l'entrée.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          Utilisé uniquement si A est symétrique ou triangulaire ; spécifie si
!>          la partie triangulaire supérieure ou inférieure de la matrice A est stockée.
!>          = 'U' :  Triangulaire supérieure
!>          = 'L' :  Triangulaire inférieure
!
>

TRAN

S 
!>          TRANS est CHARACTER*1
!>          Utilisé uniquement si A est non symétrique ; spécifie l'opération
!>          appliquée à la matrice A.
!>          = 'N' :  B := A    * X  (Pas de transposition)
!>          = 'T' :  B := A**T * X  (Transposition)
!>          = 'C' :  B := A**H * X  (Transposition conjuguée)
!
>

M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!
>

K

L 
!>          KL est INTEGER
!>          Utilisé uniquement si A est une matrice par bandes ; spécifie le nombre de
!>          sous-diagonales de A si A est une matrice générale par bandes ou si A est
!>          symétrique ou triangulaire et UPLO = 'L' ; spécifie le nombre
!>          de sur-diagonales de A si A est symétrique ou triangulaire et
!>          UPLO = 'U'.  0 <= KL <= M-1.
!
>

K

U 
!>          KU est INTEGER
!>          Utilisé uniquement si A est une matrice générale par bandes ou si A est
!>          triangulaire.
!>
!>          Si PATH = xGB, spécifie le nombre de sur-diagonales de A,
!>          et 0 <= KU <= N-1.
!>
!>          Si PATH = xTR, xTP, ou xTB, spécifie si la matrice a ou non une
!>          diagonale unitaire :
!>          = 1 :  la matrice a une diagonale non unitaire (par défaut)
!>          = 2 :  la matrice a une diagonale unitaire
!
>

NRH

S 
!>          NRHS est INTEGER
!>          Le nombre de vecteurs de second membre dans le système A*X = B.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          La matrice de test dont le type est donné par PATH.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.
!>          Si PATH = xGB, LDA >= KL+KU+1.
!>          Si PATH = xPB, xSB, xHB, ou xTB, LDA >= KL+1.
!>          Sinon, LDA >= max(1,M).
!
>

X 
!>          X est un tableau COMPLEX (entrée ou sortie), dimension (LDX,NRHS)
!>          À l'entrée, si XTYPE = 'C' (pour 'Computed'/'Calculée'), alors X contient
!>          la solution exacte du système d'équations linéaires.
!>          À la sortie, si XTYPE = 'N' (pour 'New'/'Nouvelle'), alors X est initialisé
!>          avec des valeurs aléatoires.
!
>

LD

X 
!>          LDX est INTEGER
!>          La dimension principale du tableau X.  Si TRANS = 'N',
!>          LDX >= max(1,N) ; si TRANS = 'T', LDX >= max(1,M).
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          Le ou les vecteurs de second membre pour le système d'équations,
!>          calculés à partir de B = op(A) * X, où op(A) est déterminé par
!>          TRANS.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale du tableau B.  Si TRANS = 'N',
!>          LDB >= max(1,M) ; si TRANS = 'T', LDB >= max(1,N).
!
>

ISEE

D 
!>          ISEED est un tableau INTEGER, dimension (4)
!>          Le vecteur de graine pour le générateur de nombres aléatoires (utilisé dans
!>          CLATMS).  Modifié à la sortie.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!
>
