subroutine clascl (character type, integer kl, integer ku, real cfrom, real cto, integer m, integer n, complex, dimension( lda, * ) a, integer lda, integer info)

CLASCL multiplie la matrice complexe M par N, A, par le scalaire réel CTO/CFROM.

Objectif :

!>
!> CLASCL multiplie la matrice complexe A de taille M par N par le scalaire réel
!> CTO/CFROM. Ceci est fait sans dépassement (sur- ou sous-flux) tant que le résultat
!> final CTO*A(I,J)/CFROM ne dépasse pas (ou ne sous-flue pas). TYPE spécifie si
!> A est pleine, triangulaire supérieure, triangulaire inférieure,
!> upper Hessenberg ou bande.
!>

Paramètres

TYPE 
!>          TYPE est CHARACTER*1
!>          TYPE indexe le type de stockage de la matrice d'entrée.
!>          = 'G':  A est une matrice pleine.
!>          = 'L':  A est une matrice triangulaire inférieure.
!>          = 'U':  A est une matrice triangulaire supérieure.
!>          = 'H':  A est une matrice upper Hessenberg.
!>          = 'B':  A est une matrice bande symétrique avec une bande inférieure KL
!>                  et une bande supérieure KU et avec seulement la moitié inférieure stockée.
!>          = 'Q':  A est une matrice bande symétrique avec une bande inférieure KL
!>                  et une bande supérieure KU et avec seulement la moitié supérieure stockée.
!>          = 'Z':  A est une matrice bande avec une bande inférieure KL et une bande supérieure
!>                  KU. Voir CGBTRF pour les détails de stockage.
!>

KL

!>          KL est INTEGER
!>          La bande inférieure de A.  Référencé uniquement si TYPE = 'B',
!>          'Q' ou 'Z'.
!>

KU

!>          KU est INTEGER
!>          La bande supérieure de A.  Référencé uniquement si TYPE = 'B',
!>          'Q' ou 'Z'.
!>

CFROM

!>          CFROM est REAL
!>

CTO

!>          CTO est REAL
!>
!>          La matrice A est multipliée par CTO/CFROM. A(I,J) est calculé
!>          sans dépassement (sur- ou sous-flux) si le résultat final CTO*A(I,J)/CFROM
!>          peut être représenté sans dépassement (sur- ou sous-flux). CFROM doit être
!>          non nul.
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX, de dimension (LDA,N)
!>          La matrice à multiplier par CTO/CFROM.  Voir TYPE pour le
!>          type de stockage.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.
!>          Si TYPE = 'G', 'L', 'U', 'H', LDA >= max(1,M);
!>             TYPE = 'B', LDA >= KL+1;
!>             TYPE = 'Q', LDA >= KU+1;
!>             TYPE = 'Z', LDA >= 2*KL+KU+1.
!>

INFO

!>          INFO est INTEGER
!>          0  - succès de l'exécution
!>          <0 - si INFO = -i, le i-ème argument avait une valeur illégale.
!>
