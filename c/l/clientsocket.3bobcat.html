DESCRIPTIO

N

Un FBB::ClientSocket peut être construit pour se connecter à un processus serveur sur Internet. Le socket rendu disponible par le FBB:ClientSocket peut être utilisé pour initialiser un std::istream et/ou std::ostream. Le std::istream est utilisé pour lire des informations du processus serveur auquel le FBB::ClientSocket se connecte, Le std::ostream est utilisé pour envoyer des informations au processus serveur auquel le FBB::ClientSocket se connecte. Puisqu'un socket peut être considéré comme un descripteur de fichier, les classes FBB::IFdStream, FBB::IFdStreamBuf, FBB::OFdStream et FBB::OFdStreamBuf disponibles peuvent être utilisées avec profit ici. Notez que disposer d'un socket ne signifie pas que cela définit le protocole de communication. Il est (toujours) de la responsabilité du programmeur de se conformer à un protocole existant ou d'implémenter un protocole sur mesure. Cette dernière situation implique que la séquence des opérations d'entrée et de sortie est définie par le

programmeur.

Un socket client de domaine Unix peut être défini en utilisant FBB::LocalClientSocke

EXEMPL

Voir aussi l'exemple de serversocket(3b

    #include <iostream>
    #include <string>
    #include <bobcat/clientsocket>
    #include <bobcat/ifdstream>
    #include <bobcat/ofdstream>
    using namespace std;
    using namespace FBB;
    int main(int argc, char **argv)
    try
    {
        if (argc == 1)
        {
            cerr << "Provide servername and port number\n";
            return 1;
        }
        size_t     port = stoul(argv[2]);
        ClientSocket client(argv[1], port);
        int fd = client.connect();
        string line;
        cout << "Connecting to socket " << fd << endl <<
                "address = " << client.dottedDecimalAddress() << ", " <<
                                                                 endl <<
                "communication through port " << client.port() << endl;
        IFdStream in(fd);                 // stream to read from
        OFdStream out(fd);                // stream to write to
        while (true)
        {
                                    // Ask for a textline, stop if
            cout << "? ";           // empty / none
            if (!getline(cin, line) || line.length() == 0)
                return 0;
            cout << "Line read: " << line << endl;
                                    // Return the line to the server
            out << line.c_str() << endl;
            cout << "wrote line\n";
            getline(in, line);      // Wait for a reply from the server
            cout << "Answer: " << line << endl;
        }
    }
    catch (Exception const &err)
    {
        cerr << err.what() << "\n" <<
                "Can’t connect to " << argv[1] << ", port " <<
                argv[2] << endl;
        return 1;
}
