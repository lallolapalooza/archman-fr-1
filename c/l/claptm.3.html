subroutine claptm (character uplo, integer n, integer nrhs, real alpha, real, dimension( * ) d, complex, dimension( * ) e, complex, dimension( ldx, * ) x, integer ldx, real beta, complex, dimension( ldb, * ) b, integer ldb)

CLAPTM

But :

!>
!> CLAPTM multiplie une matrice X de taille N par NRHS par une matrice
!> hermitienne tridiagonale A et stocke le résultat dans une matrice B. L'opération
!> est de la forme
!>
!>    B := alpha * A * X + beta * B
!>
!> où alpha peut être soit 1. soit -1. et beta peut être 0., 1. ou -1.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER
!>          Spécifie si la surdiagonale ou la sous-diagonale de la matrice
!>          tridiagonale A est stockée.
!>          = 'U': Supérieure, E est la surdiagonale de A.
!>          = 'L': Inférieure, E est la sous-diagonale de A.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membrres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.
!>

ALPHA

!>          ALPHA est REAL
!>          Le scalaire alpha. ALPHA doit être 1. ou -1. ; sinon,
!>          il est supposé être 0.
!>

D

!>          D est un tableau REAL, dimension (N)
!>          Les n éléments diagonaux de la matrice tridiagonale A.
!>

E

!>          E est un tableau COMPLEX, dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux ou surdiagonaux de A.
!>

X

!>          X est un tableau COMPLEX, dimension (LDX,NRHS)
!>          La matrice X de taille N par NRHS.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X. LDX >= max(N,1).
!>

BETA

!>          BETA est REAL
!>          Le scalaire beta. BETA doit être 0., 1., ou -1. ; sinon,
!>          il est supposé être 1.
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, la matrice B de taille N par NRHS.
!>          À la sortie, B est écrasée par l'expression matricielle
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(N,1).
!>
