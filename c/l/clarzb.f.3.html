subroutine clarzb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( ldwork, * ) work, integer ldwork)

CLARZB applique un réflecteur de bloc complexe H ou sa transposée conjuguée H**H à une matrice générale complexe distribuée C, depuis la gauche ou la droite.

Objectif :

!>
!> CLARZB applique un réflecteur de bloc complexe H ou sa transposée H**H
!> à une matrice complexe distribuée M-par-N C, depuis la gauche ou la droite.
!>
!> Actuellement, seuls STOREV = 'R' et DIRECT = 'B' sont supportés.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>          = 'L': appliquer H ou H**H depuis la gauche
!>          = 'R': appliquer H ou H**H depuis la droite
!>

TRANS

!>          TRANS est CHARACTER*1
!>          = 'N': appliquer H (Pas de transposée)
!>          = 'C': appliquer H**H (Transposée conjuguée)
!>

DIRECT

!>          DIRECT est CHARACTER*1
!>          Indique comment H est formé à partir d'un produit de réflecteurs
!>          élémentaires
!>          = 'F': H = H(1) H(2) . . . H(k) (Avant, non supporté pour l'instant)
!>          = 'B': H = H(k) . . . H(2) H(1) (Arrière)
!>

STOREV

!>          STOREV est CHARACTER*1
!>          Indique comment les vecteurs définissant les réflecteurs élémentaires
!>          sont stockés :
!>          = 'C': par colonne                        (non supporté pour l'instant)
!>          = 'R': par ligne
!>

M

!>          M est INTEGER
!>          Le nombre de lignes de la matrice C.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice C.
!>

K

!>          K est INTEGER
!>          L'ordre de la matrice T (= le nombre de réflecteurs élémentaires
!>          dont le produit définit le réflecteur de bloc).
!>

L

!>          L est INTEGER
!>          Le nombre de colonnes de la matrice V contenant la partie
!>          significative des réflecteurs de Householder.
!>          Si SIDE = 'L', M >= L >= 0, si SIDE = 'R', N >= L >= 0.
!>

V

!>          V est un tableau COMPLEX, dimension (LDV,NV).
!>          Si STOREV = 'C', NV = K ; si STOREV = 'R', NV = L.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V.
!>          Si STOREV = 'C', LDV >= L ; si STOREV = 'R', LDV >= K.
!>

T

!>          T est un tableau COMPLEX, dimension (LDT,K)
!>          La matrice triangulaire K-par-K T dans la représentation du
!>          réflecteur de bloc.
!>

LDT

!>          LDT est INTEGER
!>          La dimension principale du tableau T. LDT >= K.
!>

C

!>          C est un tableau COMPLEX, dimension (LDC,N)
!>          À l'entrée, la matrice C de taille M-par-N.
!>          À la sortie, C est écrasé par H*C ou H**H*C ou C*H ou C*H**H.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (LDWORK,K)
!>

LDWORK

!>          LDWORK est INTEGER
!>          La dimension principale du tableau WORK.
!>          Si SIDE = 'L', LDWORK >= max(1,N) ;
!>          si SIDE = 'R', LDWORK >= max(1,M).
!>
