subroutine clarrv (integer n, real vl, real vu, real, dimension( * ) d, real, dimension( * ) l, real pivmin, integer, dimension( * ) isplit, integer m, integer dol, integer dou, real minrgp, real rtol1, real rtol2, real, dimension( * ) w, real, dimension( * ) werr, real, dimension( * ) wgap, integer, dimension( * ) iblock, integer, dimension( * ) indexw, real, dimension( * ) gers, complex, dimension( ldz, * ) z, integer ldz, integer, dimension( * ) isuppz, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

CLARRV calcule les vecteurs propres de la matrice tridiagonale T = L D LT étant donné L, D et les valeurs propres de L D LT.

But :

!>
!> CLARRV calcule les vecteurs propres de la matrice tridiagonale
!> T = L D L**T étant donné L, D et des APPROXIMATIONS des valeurs propres de L D L**T.
!> Les valeurs propres en entrée doivent avoir été calculées par SLARRE.
!>

Paramètres

N 
!>          N est INTEGER
!>          L'ordre de la matrice.  N >= 0.
!>

VL

!>          VL est REAL
!>          Borne inférieure de l'intervalle contenant les valeurs
!>          propres désirées. VL < VU. Nécessaire pour calculer les écarts à l'extrémité gauche ou droite
!>          des valeurs propres extrêmes dans la plage désirée (RANGE).
!>

VU

!>          VU est REAL
!>          Borne supérieure de l'intervalle contenant les valeurs
!>          propres désirées. VL < VU. Nécessaire pour calculer les écarts à l'extrémité gauche ou droite
!>          des valeurs propres extrêmes dans la plage désirée (RANGE).
!>

D

!>          D est un tableau REAL, dimension (N)
!>          À l'entrée, les N éléments diagonaux de la matrice diagonale D.
!>          À la sortie, D peut être écrasé.
!>

L

!>          L est un tableau REAL, dimension (N)
!>          À l'entrée, les (N-1) éléments sous-diagonaux de la matrice
!>          bidiagonale unitaire L se trouvent dans les éléments 1 à N-1 de L
!>          (si la matrice n'est pas scindée). À la fin de chaque bloc
!>          est stocké le décalage correspondant donné par SLARRE.
!>          À la sortie, L est écrasé.
!>

PIVMIN

!>          PIVMIN est REAL
!>          Le pivot minimal autorisé dans la séquence de Sturm.
!>

ISPLIT

!>          ISPLIT est un tableau INTEGER, dimension (N)
!>          Les points de scission, auxquels T se divise en blocs.
!>          Le premier bloc est constitué des lignes/colonnes 1 à
!>          ISPLIT( 1 ), le deuxième des lignes/colonnes ISPLIT( 1 )+1
!>          jusqu'à ISPLIT( 2 ), etc.
!>

M

!>          M est INTEGER
!>          Le nombre total de valeurs propres en entrée.  0 <= M <= N.
!>

DOL

!>          DOL est INTEGER
!>

DOU

!>          DOU est INTEGER
!>          Si l'utilisateur souhaite calculer uniquement certains vecteurs propres parmi toutes
!>          les valeurs propres fournies, il peut spécifier une plage d'index DOL:DOU.
!>          Sinon, le réglage DOL=1, DOU=M doit être appliqué.
!>          Notez que DOL et DOU se réfèrent à l'ordre dans lequel les valeurs propres
!>          sont stockées dans W.
!>          Si l'utilisateur souhaite calculer uniquement certaines paires propres, alors
!>          les colonnes DOL-1 à DOU+1 de l'espace des vecteurs propres Z contiennent les
!>          vecteurs propres calculés. Toutes les autres colonnes de Z sont mises à zéro.
!>

MINRGP

!>          MINRGP est REAL
!>

RTOL1

!>          RTOL1 est REAL
!>

RTOL2

!>          RTOL2 est REAL
!>           Paramètres pour la bissection.
!>           Un intervalle [LEFT,RIGHT] a convergé si
!>           RIGHT-LEFT < MAX( RTOL1*GAP, RTOL2*MAX(|LEFT|,|RIGHT|) )
!>

W

!>          W est un tableau REAL, dimension (N)
!>          Les M premiers éléments de W contiennent les valeurs propres APPROXIMÉES pour
!>          lesquelles les vecteurs propres doivent être calculés. Les valeurs propres
!>          doivent être regroupées par bloc détaché et ordonnées de
!>          la plus petite à la plus grande à l'intérieur du bloc ( Le tableau de sortie
!>          W de SLARRE est attendu ici ). De plus, elles sont par
!>          rapport au décalage de la représentation par racines correspondante
!>          pour leur bloc. À la sortie, W contient les valeurs propres de la
!>          matrice SANS décalage.
!>

WERR

!>          WERR est un tableau REAL, dimension (N)
!>          Les M premiers éléments contiennent la demi-largeur de l'intervalle
!>          d'incertitude de la valeur propre correspondante dans W
!>

WGAP

!>          WGAP est un tableau REAL, dimension (N)
!>          La séparation avec la valeur propre voisine de droite dans W.
!>

IBLOCK

!>          IBLOCK est un tableau INTEGER, dimension (N)
!>          Les index des blocs (sous-matrices) associés aux
!>          valeurs propres correspondantes dans W ; IBLOCK(i)=1 si la valeur propre
!>          W(i) appartient au premier bloc en partant du haut, =2 si W(i)
!>          appartient au deuxième bloc, etc.
!>

INDEXW

!>          INDEXW est un tableau INTEGER, dimension (N)
!>          Les index des valeurs propres à l'intérieur de chaque bloc (sous-matrice) ;
!>          par exemple, INDEXW(i)= 10 et IBLOCK(i)=2 impliquent que la
!>          i-ème valeur propre W(i) est la 10-ème valeur propre dans le deuxième bloc.
!>

GERS

!>          GERS est un tableau REAL, dimension (2*N)
!>          Les N intervalles de Gerschgorin (le i-ème intervalle de Gerschgorin
!>          est (GERS(2*i-1), GERS(2*i)). Les intervalles de Gerschgorin doivent
!>          être calculés à partir de la matrice originale SANS décalage.
!>

Z

!>          Z est un tableau COMPLEX, dimension (LDZ, max(1,M) )
!>          Si INFO = 0, les M premières colonnes de Z contiennent les
!>          vecteurs propres orthonormaux de la matrice T
!>          correspondant aux valeurs propres en entrée, avec la i-ème
!>          colonne de Z contenant le vecteur propre associé à W(i).
!>          Note : l'utilisateur doit s'assurer qu'au moins max(1,M) colonnes sont
!>          fournies dans le tableau Z.
!>

LDZ

!>          LDZ est INTEGER
!>          La dimension principale du tableau Z.  LDZ >= 1, et si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

ISUPPZ

!>          ISUPPZ est un tableau INTEGER, dimension ( 2*max(1,M) )
!>          Le support des vecteurs propres dans Z, c'est-à-dire les index
!>          indiquant les éléments non nuls dans Z. Le I-ème vecteur propre
!>          est non nul uniquement dans les éléments ISUPPZ( 2*I-1 ) jusqu'à
!>          ISUPPZ( 2*I ).
!>

WORK

!>          WORK est un tableau REAL, dimension (12*N)
!>

IWORK

!>          IWORK est un tableau INTEGER, dimension (7*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>
!>          > 0 :  Un problème est survenu dans CLARRV.
!>          < 0 :  L'une des sous-routines appelées a signalé un problème interne.
!>                Nécessite l'inspection du paramètre correspondant IINFO
!>                pour plus de détails.nformation.
!>
!>          =-1:  Problème dans SLARRB lors de l'affinage des valeurs propres d'un enfant.
!>          =-2:  Problème dans SLARRF lors du calcul du RRR d'un enfant.
!>                Lorsqu'un enfant est à l'intérieur d'un cluster serré, il peut être difficile
!>                de trouver un RRR. Un remède partiel du point de vue de l'utilisateur
!>                est de rendre le paramètre MINRGP plus petit et de recompiler.
!>                Cependant, comme l'orthogonalité des vecteurs calculés est
!>                proportionnelle à 1/MINRGP, l'utilisateur doit être conscient
!>                qu'il pourrait sacrifier de la précision lorsqu'il diminue MINRGP.
!>          =-3:  Problème dans SLARRB lors de l'affinage d'une seule valeur propre
!>                après que la correction de Rayleigh a été rejetée.
!>          = 5:  L'itération du quotient de Rayleigh n'a pas réussi à converger vers
!>                une pleine précision en MAXITR étapes.
!>
