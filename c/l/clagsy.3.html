subroutine clagsy (integer n, integer k, real, dimension( * ) d, complex, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex, dimension( * ) work, integer info)

CLAGSY

Objectif:

!>
!> CLAGSY génère une matrice complexe symétrique A, en pré- et post-
!> multipliant une matrice diagonale réelle D par une matrice unitaire aléatoire :
!> A = U*D*U**T. La bande semi-continue peut ensuite être réduite à k par
!> des transformations unitaires supplémentaires.
!>

Paramètres

N 
!>          N est un entier
!>          L'ordre de la matrice A.  N >= 0.
!>

K

!>          K est un entier
!>          Le nombre de sous-diagonales non nulles dans la bande de A.
!>          0 <= K <= N-1.
!>

D

!>          D est un tableau réel, dimension (N)
!>          Les éléments diagonaux de la matrice diagonale D.
!>

A

!>          A est un tableau complexe, dimension (LDA,N)
!>          La matrice symétrique générée n par n A (la matrice complète est
!>          stockée).
!>

LDA

!>          LDA est un entier
!>          La dimension principale du tableau A.  LDA >= N.
!>

ISEED

!>          ISEED est un tableau entier, dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires ; les éléments
!>          du tableau doivent être compris entre 0 et 4095, et ISEED(4) doit
!>          être impair.
!>          À la sortie, la graine est mise à jour.
!>

WORK

!>          WORK est un tableau complexe, dimension (2*N)
!>

INFO

!>          INFO est un entier
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
