coin_shaders_page - Les shaders dans Coin

Coin 2.5 a ajouté la prise en charge des shaders. Les nœuds principaux utilisés sont SoShaderProgram, SoVertexShader, SoFragmentShader et SoGeometryShader. Un graphe de scène typique avec shaders ressemblera à ceci :

Separator {
  ShaderProgram {
    shaderObject [
      VertexShader {
        sourceProgram "myvertexshader.glsl"
        parameter [
          ShaderParameter1f { name "myvertexparam" value 1.0 }
        ]
      }
      FragmentShader {
        sourceProgram "myfragmentshader.glsl"
        parameter [
          ShaderParameter1f { name "myfragmentparam" value 2.0 }
        ]
      }
    ]
  }
  Cube { }
}

Ceci rendra le cube avec les shaders de vertex et de fragment spécifiés dans myvertexshader.glsl et myfragmentshader.glsl. Coin prend également en charge les shaders ARB et les shaders Cg (si la bibliothèque Cg est installée). Cependant, nous recommandons l'utilisation de GLSL car nous nous concentrerons principalement sur la prise en charge de ce langage de shader.

Coin définit certains paramètres nommés qui peuvent être ajoutés par le programmeur d'application, et qui seront automatiquement mis à jour par Coin lors du parcours du graphe de scène.

•
coin_texunit[n]_model - Défini à 0 lorsque la texturation est désactivée, et à SoTextureImageElement::Model s'il y a une texture actuelle sur l'état pour l'unité n.
•
coin_light_model - Défini à 1 pour PHONG, 0 pour l'éclairage BASE_COLOR.
•
coin_two_sided_lighting - Défini à 1 pour bilatéral, 0 pour normal

Exemple de graphe de scène qui rend l'éclairage Phong OpenGL par fragment pour une source lumineuse. Les shaders supposent que la première source lumineuse est une lumière directionnelle. C'est le cas si vous ouvrez le fichier dans une visionneuse d'examen standard.

Le fichier iv :

Separator {
  ShaderProgram {
    shaderObject [
      VertexShader {
        sourceProgram "perpixel_vertex.glsl"
      }
      FragmentShader {
        sourceProgram "perpixel_fragment.glsl"
      }
    ]
  }
  Complexity { value 1.0 }
  Material { diffuseColor 1 0 0 specularColor 1 1 1 shininess 0.9 }
  Sphere { }
  Translation { translation 3 0 0 }
  Material { diffuseColor 0 1 0 specularColor 1 1 1 shininess 0.9 }
  Cone { }
  Translation { translation 3 0 0 }
  Material { diffuseColor 0.8 0.4 0.1 specularColor 1 1 1 shininess 0.9 }
  Cylinder { }
}

Le shader de vertex (perpixel_vertex.glsl) :

varying vec3 ecPosition3;
varying vec3 fragmentNormal;
void main(void)
{
  vec4 ecPosition = gl_ModelViewMatrix * gl_Vertex;
  ecPosition3 = ecPosition.xyz / ecPosition.w;
  fragmentNormal = normalize(gl_NormalMatrix * gl_Normal);
  gl_Position = ftransform();
  gl_FrontColor = gl_Color;
}

Le shader de fragment (perpixel_fragment.glsl) :

varying vec3 ecPosition3;
varying vec3 fragmentNormal;
void DirectionalLight(in int i,
                      in vec3 normal,
                      inout vec4 ambient,
                      inout vec4 diffuse,
                      inout vec4 specular)
{
  float nDotVP; // normal . direction de la lumière
  float nDotHV; // normal . vecteur demi-lumière
  float pf;     // facteur de puissance
  nDotVP = max(0.0, dot(normal, normalize(vec3(gl_LightSource[i].position))));
  nDotHV = max(0.0, dot(normal, vec3(gl_LightSource[i].halfVector)));
  if (nDotVP == 0.0)
    pf = 0.0;
  else
    pf = pow(nDotHV, gl_FrontMaterial.shininess);
  ambient += gl_LightSource[i].ambient;
  diffuse += gl_LightSource[i].diffuse * nDotVP;
  specular += gl_LightSource[i].specular * pf;
}
void main(void)
{
  vec3 eye = -normalize(ecPosition3);
  vec4 ambient = vec4(0.0);
  vec4 diffuse = vec4(0.0);
  vec4 specular = vec4(0.0);
  vec3 color;
  DirectionalLight(0, normalize(fragmentNormal), ambient, diffuse, specular);
  color =
    gl_FrontLightModelProduct.sceneColor.rgb +
    ambient.rgb * gl_FrontMaterial.ambient.rgb +
    diffuse.rgb * gl_Color.rgb +
    specular.rgb * gl_FrontMaterial.specular.rgb;
  gl_FragColor = vec4(color, gl_Color.a);
}
