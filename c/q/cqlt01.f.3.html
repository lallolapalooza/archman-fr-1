subroutine cqlt01 (integer m, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) l, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CQLT01

But :

!>
!> CQLT01 teste CGEQLF, qui calcule la factorisation QL d'une matrice m-par-n
!> A, et teste partiellement CUNGQL qui forme la matrice orthogonale m-par-m
!> Q.
!>
!> CQLT01 compare L avec Q'*A, et vérifie que Q est orthogonale.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau complexe (COMPLEX), dimension (LDA,N)
!>          La matrice m-par-n A.
!>

AF

!>          AF est un tableau complexe (COMPLEX), dimension (LDA,N)
!>          Détails de la factorisation QL de A, tel que retourné par CGEQLF.
!>          Voir CGEQLF pour plus de détails.
!>

Q

!>          Q est un tableau complexe (COMPLEX), dimension (LDA,M)
!>          La matrice orthogonale m-par-m Q.
!>

L

!>          L est un tableau complexe (COMPLEX), dimension (LDA,max(M,N))
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La dimension principale des tableaux A, AF, Q et R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU est un tableau complexe (COMPLEX), dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires, tel que retourné
!>          par CGEQLF.
!>

WORK

!>          WORK est un tableau complexe (COMPLEX), dimension (LWORK)
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau réel (REAL), dimension (M)
!>

RESULT

!>          RESULT est un tableau réel (REAL), dimension (2)
!>          Les rapports de test :
!>          RESULT(1) = norm( L - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
