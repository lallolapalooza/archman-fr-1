subroutine cqrt05 (integer m, integer n, integer l, integer nb, real, dimension(6) result)

CQRT05

But :

!>
!> CQRT05 teste CTPQRT et CTPMQRT.
!>

Paramètres

M 
!>          M est un entier
!>          Nombre de lignes dans la partie inférieure de la matrice de test.
!>

N

!>          N est un entier
!>          Nombre de colonnes dans la matrice de test.
!>

L

!>          L est un entier
!>          Le nombre de lignes de la partie trapézoïdale supérieure de la
!>          matrice de test inférieure.  0 <= L <= M.
!>

NB

!>          NB est un entier
!>          Taille des blocs de la matrice de test.  NB <= N.
!>

RESULT

!>          RESULT est un tableau REEL, de dimension (6)
!>          Résultats de chacun des six tests ci-dessous.
!>
!>          RESULT(1) = | A - Q R |
!>          RESULT(2) = | I - Q^H Q |
!>          RESULT(3) = | Q C - Q C |
!>          RESULT(4) = | Q^H C - Q^H C |
!>          RESULT(5) = | C Q - C Q |
!>          RESULT(6) = | C Q^H - C Q^H |
!>
