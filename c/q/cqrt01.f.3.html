subroutine cqrt01 (m, n, a, af, q, r, lda, tau, work, lwork, rwork, result)
CQRT01

Documentation de la fonction/sous-programme

sous-programme cqrt01 (entier m, entier n, complexe, dimension( lda, * ) a, complexe, dimension( lda, * ) af, complexe, dimension( lda, * ) q, complexe, dimension( lda, * ) r, entier lda, complexe, dimension( * ) tau, complexe, dimension( lwork ) work, entier lwork, réel, dimension( * ) rwork, réel, dimension( * ) result)

CQRT01

Objectif :

!>
!> CQRT01 teste CGEQRF, qui calcule la factorisation QR d'une matrice m-par-n
!> A, et teste partiellement CUNGQR qui forme la matrice orthogonale m-par-m
!> Q.
!>
!> CQRT01 compare R avec Q'*A, et vérifie que Q est orthogonale.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEXE, dimension (LDA,N)
!>          La matrice A m-par-n.
!>

AF

!>          AF est un tableau COMPLEXE, dimension (LDA,N)
!>          Détails de la factorisation QR de A, tels que retournés par CGEQRF.
!>          Voir CGEQRF pour plus de détails.
!>

Q

!>          Q est un tableau COMPLEXE, dimension (LDA,M)
!>          La matrice orthogonale m-par-m Q.
!>

R

!>          R est un tableau COMPLEXE, dimension (LDA,max(M,N))
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale des tableaux A, AF, Q et R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU est un tableau COMPLEXE, dimension (min(M,N))
!>          Les facteurs scalaires des réflexions élémentaires, tels que retournés
!>          par CGEQRF.
!>

WORK

!>          WORK est un tableau COMPLEXE, dimension (LWORK)
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau RÉEL, dimension (M)
!>

RESULT

!>          RESULT est un tableau RÉEL, dimension (2)
!>          Les rapports de test :
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
