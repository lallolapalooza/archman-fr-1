subroutine cpttrs (character uplo, integer n, integer nrhs, real, dimension( * ) d, complex, dimension( * ) e, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPTTRS

Objectif:

!>
!> CPTTRS résout un système tridiagonal de la forme
!>    A * X = B
!> en utilisant la factorisation A = U**H*D*U ou A = L*D*L**H calculée par CPTTRF.
!> D est une matrice diagonale spécifiée dans le vecteur D, U (ou L) est une matrice
!> bidiagonale unitaire dont la superdiagonale (ou sous-diagonale) est spécifiée
!> dans le vecteur E, et X et B sont des matrices de N x NRHS.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie la forme de la factorisation et si le vecteur E est la
!>          superdiagonale du facteur bidiagonal supérieur U ou la sous-diagonale
!>          du facteur bidiagonal inférieur L.
!>          = 'U' :  A = U**H*D*U, E est la superdiagonale de U
!>          = 'L' :  A = L*D*L**H, E est la sous-diagonale de L
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice tridiagonale A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          de la matrice B.  NRHS >= 0.
!>

D

!>          D est un tableau REAL, dimension (N)
!>          Les n éléments diagonaux de la matrice diagonale D de la
!>          factorisation A = U**H*D*U ou A = L*D*L**H.
!>

E

!>          E est un tableau COMPLEX, dimension (N-1)
!>          Si UPLO = 'U', les (n-1) éléments de la superdiagonale du facteur
!>          bidiagonal unitaire U de la factorisation A = U**H*D*U.
!>          Si UPLO = 'L', les (n-1) éléments de la sous-diagonale du facteur
!>          bidiagonal unitaire L de la factorisation A = L*D*L**H.
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          À l'entrée, les vecteurs membres de droite B pour le système d'équations
!>          linéaires.
!>          À la sortie, les vecteurs solution, X.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0: sortie réussie
!>          < 0: si INFO = -k, le k-ème argument avait une valeur illégale
!>
