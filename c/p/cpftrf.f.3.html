subroutine cpftrf (character transr, character uplo, integer n, complex, dimension( 0: * ) a, integer

info)

CPFTR

F

But

: 
!>
!> CPFTRF calcule la factorisation de Cholesky d'une matrice hermitienne
!> définie positive A.
!>
!> La factorisation prend la forme
!>    A = U**H * U,  si UPLO = 'U', ou
!>    A = L  * L**H,  si UPLO = 'L',
!> où U est une matrice triangulaire supérieure et L est triangulaire inférieure.
!>
!> Il s'agit de la version par blocs de l'algorithme, appelant les BLAS de niveau 3.
!
>

Paramètre

s
TRANSR 
!>          TRANSR est CHARACTER*1
!>          = 'N' : le TRANSR normal de RFP A est stocké ;
!>          = 'C' : le TRANSR transposé-conjugué de RFP A est stocké.
!
>

UPL

O 
!>          UPLO est CHARACTER*1
!>          = 'U' : le triangle supérieur de RFP A est stocké ;
!>          = 'L' : le triangle inférieur de RFP A est stocké.
!
>

N 
!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension ( N*(N+1)/2 ) ;
!>          À l'entrée, la matrice hermitienne A au format RFP. Le format RFP est
!>          décrit par TRANSR, UPLO et N comme suit : si TRANSR = 'N'
!>          alors RFP A est (0:N,0:k-1) quand N est pair ; k=N/2. RFP A est
!>          (0:N-1,0:k) quand N est impair ; k=N/2. Si TRANSR = 'C' alors RFP est
!>          le transposé-conjugué de RFP A tel que défini lorsque
!>          TRANSR = 'N'. Le contenu de RFP A est défini par UPLO comme
!>          suit : si UPLO = 'U', le RFP A contient les nt éléments de
!>          A empaqueté supérieur. Si UPLO = 'L', le RFP A contient les éléments
!>          de A empaqueté inférieur. Le LDA de RFP A est (N+1)/2 quand TRANSR =
!>          'C'. Quand TRANSR vaut 'N', le LDA est N+1 quand N est pair et N
!>          est impair. Voir la Note ci-dessous pour plus de détails.
!>
!>          À la sortie, si INFO = 0, le facteur U ou L de la factorisation
!>          de Cholesky RFP A = U**H*U ou RFP A = L*L**H.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>          > 0 :  si INFO = i, le mineur principal de tête d'ordre i
!>                n'est pas positif, et la factorisation n'a pas pu être
!>                complétée.
!>
!>  Notes supplémentaires sur le format RFP :
!>  ==========================================
!>
!>  Nous considérons d'abord le format empaqueté standard quand N est pair.
!>  Nous donnons un exemple où N = 6.
!>
!>     AP est supérieur             AP est inférieur
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>  Soit TRANSR = 'N'. RFP contient AP comme suit :
!>  Pour UPLO = 'U', le trapèze supérieur A(0:5,0:2) est constitué des trois
!>  dernières colonnes de AP supérieur. Le triangle inférieur A(4:6,0:2) est constitué du
!>  transposé-conjugué des trois premières colonnes de AP supérieur.
!>  Pour UPLO = 'L', le trapèze inférieur A(1:6,0:2) est constitué des trois
!>  premières colonnes de AP inférieur. Le triangle supérieur A(0:2,0:2) est constitué du
!>  transposé-conjugué des trois dernières colonnes de AP inférieur.
!>  Pour indiquer le conjugué, nous plaçons -- au-dessus de l'élément. Cela couvre le
!>  cas N pair et TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  Maintenant soit TRANSR = 'C'. RFP A dans les deux cas UPLO est simplement le transposé-
!>  conjugué de RFP A ci-dessus. On obtient donc :
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     -- -- -- -- -- --                -- -- -- --
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>  Nous considérons ensuite le format empaqueté standard quand N est impair.
!>  Nous donnons un exemple où N = 5.
!>
!>     AP est supérieur                 AP est inférieur
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>  Soit TRANSR = 'N'. RFP contient AP comme suit :
!>  Pour UPLO = 'U', le trapèze supérieur A(0:4,0:2) est constitué des trois
!>  dernières colonnes de AP supérieur. Le triangle inférieur A(3:4,0:1) est constitué du
!>  transposé-conjugué des deux premières colonnes de AP supérieur.
!>  Pour UPLO = 'L', le trapèze inférieur A(0:4,0:2) est constitué des trois
!>  premières colonnes de AP inférieur. Le triangle supérieur A(0:1,1:2) est constitué du
!>  transposé-conjugué des deux dernières colonnes de AP inférieur.
!>  Pour indiquer le conjugué, nous plaçons -- au-dessus de l'élément. Cela couvre le
!>  cas N impair et TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                   -- --
!>        02 03 04                00 33 43
!>                                      --
!>        12 13 14                10 11 44
!>
!>        22 23 24                20 21 22
!>        --
!>        00 33 34                30 31 32
!>        -- --
!>        01 11 44                40 41 42
!>
!>  Maintenant soit TRANSR = 'C'. RFP A dans les deux cas UPLO est simplement le transposé-
!>  conjugué de RFP A ci-dessus. On obtient donc :
!>
!>           RFP A                   RFP A
!>
!>     -- -- --                   -- -- -- -- -- --
!>     02 12 22 00 01             00 10 20 30 40 50
!>     -- -- -- --                   -- -- -- -- --
!>     03 13 23 33 11             33 11 21 31 41 51
!>     -- -- -- -- --                   -- -- -- --
!>     04 14 24 34 44             43 44 22 32 42 52
!
>
