sous-routine

subroutine cgebrd (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( * ) tauq, complex, dimension( * ) taup, complex, dimension( * ) work, integer lwork, integer

info)

CGEBR

D

But

: 
!>
!> CGEBRD réduit une matrice générale complexe M par N A à une forme
!> bidiagonale supérieure ou inférieure B par une transformation unitaire :
!> Q**H * A * P = B.
!>
!> Si m >= n, B est bidiagonale supérieure ; si m < n, B est bidiagonale
!> inférieure.
!
>

Paramètre

s
M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!
>

N 
!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          À l'entrée, la matrice générale M par N à réduire.
!>          À la sortie,
!>          si m >= n, la diagonale et la première surdiagonale sont
!>            écrasées par la matrice bidiagonale supérieure B ; les
!>            éléments sous la diagonale, avec le tableau TAUQ, représentent
!>            la matrice unitaire Q sous forme de produit de réflecteurs
!>            élémentaires, et les éléments au-dessus de la première
!>            surdiagonale, avec le tableau TAUP, représentent la matrice
!>            unitaire P sous forme de produit de réflecteurs élémentaires ;
!>          si m < n, la diagonale et la première sous-diagonale sont
!>            écrasées par la matrice bidiagonale inférieure B ; les
!>            éléments sous la première sous-diagonale, avec le tableau
!>            TAUQ, représentent la matrice unitaire Q sous forme de
!>            produit de réflecteurs élémentaires, et les éléments au-dessus
!>            de la diagonale, avec le tableau TAUP, représentent la matrice
!>            unitaire P sous forme de produit de réflecteurs élémentaires.
!>          Voir Détails supplémentaires.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!
>

D 
!>          D est un tableau REAL, dimension (min(M,N))
!>          Les éléments diagonaux de la matrice bidiagonale B :
!>          D(i) = A(i,i).
!
>

E 
!>          E est un tableau REAL, dimension (min(M,N)-1)
!>          Les éléments hors diagonale de la matrice bidiagonale B :
!>          si m >= n, E(i) = A(i,i+1) pour i = 1,2,...,n-1 ;
!>          si m < n, E(i) = A(i+1,i) pour i = 1,2,...,m-1.
!
>

TAU

Q 
!>          TAUQ est un tableau COMPLEX, dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires qui
!>          représentent la matrice unitaire Q. Voir Détails supplémentaires.
!
>

TAU

P 
!>          TAUP est un tableau COMPLEX, dimension (min(M,N))
!>          Les facteurs scalaires des réflecteurs élémentaires qui
!>          représentent la matrice unitaire P. Voir Détails supplémentaires.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La longueur du tableau WORK.
!>          LWORK >= 1, si MIN(M,N) = 0, et LWORK >= MAX(M,N), sinon.
!>          Pour des performances optimales, LWORK >= (M+N)*NB, où NB
!>          est la taille de bloc optimale.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille optimale
!>          du tableau WORK, renvoie cette valeur comme première entrée
!>          du tableau WORK, et aucun message d'erreur lié à LWORK n'est
!>          émis par XERBLA.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie.
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale.
!
>

Détails supplémentaires

: 
!>
!>  Les matrices Q et P sont représentées comme des produits de réflecteurs
!>  élémentaires :
!>
!>  Si m >= n,
!>
!>     Q = H(1) H(2) . . . H(n)  et  P = G(1) G(2) . . . G(n-1)
!>
!>  Chaque H(i) et G(i) a la forme :
!>
!>     H(i) = I - tauq * v * v**H  et G(i) = I - taup * u * u**H
!>
!>  où tauq et taup sont des scalaires complexes, et v et u sont des
!>  vecteurs complexes ; v(1:i-1) = 0, v(i) = 1, et v(i+1:m) est stocké
!>  à la sortie dans A(i+1:m,i) ; u(1:i) = 0, u(i+1) = 1, et u(i+2:n) est
!>  stocké à la sortie dans A(i,i+2:n) ; tauq est stocké dans TAUQ(i) et
!>  taup dans TAUP(i).
!>
!>  Si m < n,
!>
!>     Q = H(1) H(2) . . . H(m-1)  et  P = G(1) G(2) . . . G(m)
!>
!>  Chaque H(i) et G(i) a la forme :
!>
!>     H(i) = I - tauq * v * v**H  et G(i) = I - taup * u * u**H
!>
!>  où tauq et taup sont des scalaires complexes, et v et u sont des
!>  vecteurs complexes ; v(1:i) = 0, v(i+1) = 1, et v(i+2:m) est stocké
!>  à la sortie dans A(i+2:m,i) ; u(1:i-1) = 0, u(i) = 1, et u(i+1:n) est
!>  stocké à la sortie dans A(i,i+1:n) ; tauq est stocké dans TAUQ(i) et
!>  taup dans TAUP(i).
!>
!>  Le contenu de A à la sortie est illustré par les exemples suivants :
!>
!>  m = 6 et n = 5 (m > n) :          m = 5 et n = 6 (m < n) :
!>
!>    (  d   e   u1  u1  u1 )           (  d   u1  u1  u1  u1  u1 )
!>    (  v1  d   e   u2  u2 )           (  e   d   u2  u2  u2  u2 )
!>    (  v1  v2  d   e   u3 )           (  v1  e   d   u3  u3  u3 )
!>    (  v1  v2  v3  d   e  )           (  v1  v2  e   d   u4  u4 )
!>    (  v1  v2  v3  v4  d  )           (  v1  v2  v3  e   d   u5 )
!>    (  v1  v2  v3  v4  v5 )
!>
!>  où d et e désignent les éléments diagonaux et hors diagonale de B, vi
!>  désigne un élément du vecteur définissant H(i), et ui un élément du
!>  vecteur définissant G(i).
!
>
