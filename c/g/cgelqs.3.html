sous-routine cgelqs (entier m, entier n, entier nrhs, complexe, dimension( lda, * ) a, entier lda, complexe, dimension( * ) tau, complexe, dimension( ldb, * ) b, entier ldb, complexe, dimension( lwork ) work, entier lwork, entier info)

CGELQS

But :

!>
!> Calcule une solution de norme minimale
!>     min || A*X - B ||
!> en utilisant la factorisation LQ
!>     A = L*Q
!> calculée par CGELQF.
!>

Paramètres

M 
!>          M est un ENTIER
!>          Le nombre de lignes de la matrice A.  M >= 0.
!>

N

!>          N est un ENTIER
!>          Le nombre de colonnes de la matrice A.  N >= M >= 0.
!>

NRHS

!>          NRHS est un ENTIER
!>          Le nombre de colonnes de B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEXE, de dimension (LDA,N)
!>          Détails de la factorisation LQ de la matrice originale A retournés
!>          par CGELQF.
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale du tableau A.  LDA >= M.
!>

TAU

!>          TAU est un tableau COMPLEXE, de dimension (M)
!>          Détails de la matrice orthogonale Q.
!>

B

!>          B est un tableau COMPLEXE, de dimension (LDB,NRHS)
!>          À l'entrée, la matrice membre de droite B de dimension m-par-nrhs.
!>          À la sortie, la matrice solution X de dimension n-par-nrhs.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale du tableau B. LDB >= N.
!>

WORK

!>          WORK est un tableau COMPLEXE, de dimension (LWORK)
!>

LWORK

!>          LWORK est un ENTIER
!>          La longueur du tableau WORK.  LWORK doit être au moins NRHS,
!>          et devrait être au moins NRHS*NB, où NB est la taille de bloc
!>          pour cet environnement.
!>

INFO

!>          INFO est un ENTIER
!>          = 0 : succès
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
