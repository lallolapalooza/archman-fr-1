subroutine cgges3 (character jobvsl, character jobvsr, character sort, external selctg, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, integer sdim, complex, dimension( * ) alpha, complex, dimension( * ) beta, complex, dimension( ldvsl, * ) vsl, integer ldvsl, complex, dimension( ldvsr, * ) vsr, integer ldvsr, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, logical, dimension( * ) bwork, integer

CGGES3 calcule les valeurs propres, la forme de Schur et, en option, la matrice des vecteurs de Schur pour les matrices GE (algorithme par

blocs)

But

: 
!>
!> CGGES3 calcule pour une paire de matrices complexes non symétriques
!> N par N (A,B), les valeurs propres généralisées, la forme de Schur
!> complexe généralisée (S, T), et en option les vecteurs de Schur
!> gauche et/ou droit (VSL et VSR). Cela donne la factorisation de
!> Schur généralisée
!>
!>         (A,B) = ( (VSL)*S*(VSR)**H, (VSL)*T*(VSR)**H )
!>
!> où (VSR)**H est le conjugué-transposé de VSR.
!>
!> En option, elle ordonne également les valeurs propres de sorte
!> qu'un cluster sélectionné de valeurs propres apparaisse dans les
!> blocs diagonaux supérieurs de la matrice triangulaire supérieure S
!> et de la matrice triangulaire supérieure T. Les colonnes
!> principales de VSL et VSR forment alors une base unitaire pour les
!> espaces propres gauche et droit correspondants (sous-espaces
!> déflateurs).
!>
!> (Si seules les valeurs propres généralisées sont nécessaires,
!> utilisez plutôt le programme pilote CGGEV, qui est plus rapide.)
!>
!> Une valeur propre généralisée pour une paire de matrices (A,B) est
!> un scalaire w ou un rapport alpha/beta = w, tel que A - w*B soit
!> singulière. Elle est généralement représentée sous la forme de la
!> paire (alpha,beta), car il existe une interprétation raisonnable
!> pour beta=0, et même pour les deux étant nuls.
!>
!> Une paire de matrices (S,T) est en forme de Schur complexe
!> généralisée si S et T sont triangulaires supérieures et, de plus,
!> les éléments diagonaux de T sont des nombres réels non négatifs.
!
>

Paramètre

s
JOBVSL 
!>          JOBVSL est CHARACTER*1
!>          = 'N' :  ne pas calculer les vecteurs de Schur gauche ;
!>          = 'V' :  calculer les vecteurs de Schur gauche.
!
>

JOBVS

R 
!>          JOBVSR est CHARACTER*1
!>          = 'N' :  ne pas calculer les vecteurs de Schur droit ;
!>          = 'V' :  calculer les vecteurs de Schur droit.
!
>

SOR

T 
!>          SORT est CHARACTER*1
!>          Indique si les valeurs propres doivent être ordonnées sur
!>          la diagonale de la forme de Schur généralisée.
!>          = 'N' :  les valeurs propres ne sont pas ordonnées ;
!>          = 'S' :  les valeurs propres sont ordonnées (voir SELCTG).
!
>

SELCT

G 
!>          SELCTG est une FUNCTION LOGICAL de deux arguments COMPLEX
!>          SELCTG doit être déclarée EXTERNAL dans la sous-routine
!>          appelante.
!>          Si SORT = 'N', SELCTG n'est pas référencée.
!>          Si SORT = 'S', SELCTG est utilisée pour sélectionner les
!>          valeurs propres à trier en haut à gauche de la forme de
!>          Schur.
!>          Une valeur propre ALPHA(j)/BETA(j) est sélectionnée si
!>          SELCTG(ALPHA(j),BETA(j)) est vraie.
!>
!>          Notez qu'une valeur propre complexe sélectionnée peut ne
!>          plus satisfaire SELCTG(ALPHA(j),BETA(j)) = .TRUE. après
!>          l'ordre, car l'ordre peut changer la valeur des valeurs
!>          propres complexes (surtout si la valeur propre est mal
!>          conditionnée) ; dans ce cas, INFO est défini à N+2 (voir
!>          INFO ci-dessous).
!
>

N 
!>          N est INTEGER
!>          L'ordre des matrices A, B, VSL et VSR.  N >= 0.
!
>

A 
!>          A est un tableau COMPLEX, dimension (LDA, N)
!>          À l'entrée, la première de la paire de matrices.
!>          À la sortie, A a été écrasée par sa forme de Schur
!>          généralisée S.
!
>

LD

A 
!>          LDA est INTEGER
!>          La dimension principale de A.  LDA >= max(1,N).
!
>

B 
!>          B est un tableau COMPLEX, dimension (LDB, N)
!>          À l'entrée, la seconde de la paire de matrices.
!>          À la sortie, B a été écrasée par sa forme de Schur
!>          généralisée T.
!
>

LD

B 
!>          LDB est INTEGER
!>          La dimension principale de B.  LDB >= max(1,N).
!
>

SDI

M 
!>          SDIM est INTEGER
!>          Si SORT = 'N', SDIM = 0.
!>          Si SORT = 'S', SDIM = le nombre de valeurs propres (après
!>          tri) pour lesquelles SELCTG est vraie.
!
>

ALPH

A 
!>          ALPHA est un tableau COMPLEX, dimension (N)
!
>

BET

A 
!>          BETA est un tableau COMPLEX, dimension (N)
!>          À la sortie, ALPHA(j)/BETA(j), j=1,...,N, seront les
!>          valeurs propres généralisées. ALPHA(j), j=1,...,N et
!>          BETA(j), j=1,...,N sont les diagonales de la forme de Schur
!>          complexe (A,B) produite par CGGES3. Les BETA(j) seront des
!>          réels non négatifs.
!>
!>          Note : les quotients ALPHA(j)/BETA(j) peuvent facilement
!>          déborder ou être trop petits, et BETA(j) peut même être
!>          nul. Ainsi, l'utilisateur devrait éviter de calculer
!>          naïvement le rapport alpha/beta. Cependant, ALPHA sera
!>          toujours inférieur et généralement comparable en
!>          magnitude avec norm(A), et BETA toujours inférieur et
!>          généralement comparable avec norm(B).
!
>

VS

L 
!>          VSL est un tableau COMPLEX, dimension (LDVSL,N)
!>          Si JOBVSL = 'V', VSL contiendra les vecteurs de Schur
!>          gauche. Non référencée si JOBVSL = 'N'.
!
>

LDVS

L 
!>          LDVSL est INTEGER
!>          La dimension principale de la matrice VSL. LDVSL >= 1, et
!>          si JOBVSL = 'V', LDVSL >= N.
!
>

VS

R 
!>          VSR est un tableau COMPLEX, dimension (LDVSR,N)
!>          Si JOBVSR = 'V', VSR contiendra les vecteurs de Schur
!>          droit. Non référencée si JOBVSR = 'N'.
!
>

LDVS

R 
!>          LDVSR est INTEGER
!>          La dimension principale de la matrice VSR. LDVSR >= 1, et
!>          si JOBVSR = 'V', LDVSR >= N.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) renvoie le LWORK optimal.
!
>

LWOR

K 
!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= MAX(1,2*N).
!>          Pour de bonnes performances, LWORK doit généralement être
!>          plus grand.
!>
!>          Si LWORK = -1, une interrogation d'espace de travail est
!>          supposée ; la routine calcule uniquement la taille
!>          optimale du tableau WORK, renvoie cette valeur comme
!>          première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (8*N)
!
>

BWOR

K 
!>          BWORK est un tableau LOGICAL, dimension (N)
!>          Non référencée si SORT = 'N'.
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 :  sortie réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur
!>          illégale.
!>          =1,...,N :
!>                L'itération QZ a échoué. (A,B) ne sont pas en former
!>                form, mais ALPHA(j) et BETA(j) devraient être corrects pour
!>                j=INFO+1,...,N.
!>          > N:  =N+1: autre échec que l'itération QZ dans CLAQZ0
!>                =N+2: après réordonnancement, l'arrondi a modifié les valeurs de
!>                      certaines valeurs propres complexes de sorte que les
!>                      valeurs propres dominantes dans la forme de Schur généralisée
!>                      ne satisfassent plus SELCTG=.TRUE.. Cela peut aussi
!>                      être dû à une mise à l'échelle.
!>                =N+3: échec du réordonnancement dans CTGSEN.
!
>
