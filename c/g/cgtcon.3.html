subroutine cgtcon (norm, n, dl, d, du, du2, ipiv, anorm, rcond, work, info)
CGTCON

Documentation de la fonction/sous-routine

sous-routine cgtcon (caractère norm, entier n, complexe, dimension( * ) dl, complexe, dimension( * ) d, complexe, dimension( * ) du, complexe, dimension( * ) du2, entier, dimension( * ) ipiv, réel anorm, réel rcond, complexe, dimension( * ) work, entier info)

CGTCON

Objectif :

!>
!> CGTCON estime l'inverse du nombre de condition d'une matrice complexe
!> tridiagonale A en utilisant la factorisation LU calculée par
!> CGTTRF.
!>
!> Une estimation de norm(inv(A)) est obtenue, et l'inverse du nombre de
!> condition est calculé comme RCOND = 1 / (ANORM * norm(inv(A))).
!>

Paramètres

NORM 
!>          NORM est CHARACTER*1
!>          Spécifie si le nombre de condition en norme 1 ou en norme
!>          inférieure est requis :
!>          = '1' ou 'O' :  norme 1 ;
!>          = 'I' :         norme inférieure.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

DL

!>          DL est un tableau COMPLEX de dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L de la
!>          factorisation LU de A calculée par CGTTRF.
!>

D

!>          D est un tableau COMPLEX de dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire supérieure U
!>          issue de la factorisation LU de A.
!>

DU

!>          DU est un tableau COMPLEX de dimension (N-1)
!>          Les (n-1) éléments de la première superdiagonale de U.
!>

DU2

!>          DU2 est un tableau COMPLEX de dimension (N-2)
!>          Les (n-2) éléments de la deuxième superdiagonale de U.
!>

IPIV

!>          IPIV est un tableau INTEGER de dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la matrice a
!>          été échangée avec la ligne IPIV(i).  IPIV(i) sera toujours égal à
!>          i ou i+1 ; IPIV(i) = i indique qu'un échange de ligne n'était pas
!>          nécessaire.
!>

ANORM

!>          ANORM est REAL
!>          Si NORM = '1' ou 'O', la norme 1 de la matrice originale A.
!>          Si NORM = 'I', la norme inférieure de la matrice originale A.
!>

RCOND

!>          RCOND est REAL
!>          L'inverse du nombre de condition de la matrice A,
!>          calculé comme RCOND = 1/(ANORM * AINVNM), où AINVNM est une
!>          estimation de la norme 1 de inv(A) calculée dans cette routine.
!>

WORK

!>          WORK est un tableau COMPLEX de dimension (2*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 :  exécution réussie
!>          < 0 :  si INFO = -i, le i-ème argument avait une valeur illégale
!>
