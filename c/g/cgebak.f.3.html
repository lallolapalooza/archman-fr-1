subroutine cgebak (character job, character side, integer n, integer ilo, integer ihi, real, dimension( * ) scale, integer m, complex, dimension( ldv, * ) v, integer ldv, integer info)

CGEBAK

But :

!>
!> CGEBAK forme les vecteurs propres à droite ou à gauche d'une matrice générale complexe
!> par transformation arrière sur les vecteurs propres calculés de la matrice équilibrée
!> générée par CGEBAL.
!>

Paramètres

JOB 
!>          JOB est CHARACTER*1
!>          Spécifie le type de transformation arrière requise :
!>          = 'N' : ne rien faire, retourner immédiatement ;
!>          = 'P' : effectuer la transformation arrière pour la permutation uniquement ;
!>          = 'S' : effectuer la transformation arrière pour la mise à l'échelle uniquement ;
!>          = 'B' : effectuer les transformations arrière pour la permutation et la
!>                 mise à l'échelle.
!>          JOB doit être le même que l'argument JOB fourni à CGEBAL.
!>

SIDE

!>          SIDE est CHARACTER*1
!>          = 'R' : V contient les vecteurs propres à droite ;
!>          = 'L' : V contient les vecteurs propres à gauche.
!>

N

!>          N est INTEGER
!>          Le nombre de lignes de la matrice V. N >= 0.
!>

ILO

!>          ILO est INTEGER
!>

IHI

!>          IHI est INTEGER
!>          Les entiers ILO et IHI déterminés par CGEBAL.
!>          1 <= ILO <= IHI <= N, si N > 0 ; ILO=1 et IHI=0, si N=0.
!>

SCALE

!>          SCALE est un tableau REAL, de dimension (N)
!>          Détails des facteurs de permutation et de mise à l'échelle, tels que renvoyés
!>          par CGEBAL.
!>

M

!>          M est INTEGER
!>          Le nombre de colonnes de la matrice V. M >= 0.
!>

V

!>          V est un tableau COMPLEX, de dimension (LDV,M)
!>          En entrée, la matrice des vecteurs propres à droite ou à gauche à transformer,
!>          telle que renvoyée par CHSEIN ou CTREVC.
!>          En sortie, V est écrasé par les vecteurs propres transformés.
!>

LDV

!>          LDV est INTEGER
!>          La dimension principale du tableau V. LDV >= max(1,N).
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>
