subroutine cgelss (integer m, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) s, real rcond, integer rank, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer info)

CGELSS résout des systèmes surdéterminés ou sousdéterminés pour des matrices GE

But:

!>
!> CGELSS calcule la solution de norme minimale pour un problème de
!> moindres carrés linéaires complexes :
!>
!> Minimiser la norme 2 de | b - A*x |.
!>
!> en utilisant la décomposition en valeurs singulières (SVD) de A. A est une
!> matrice M par N qui peut être de rang déficient.
!>
!> Plusieurs vecteurs de second membre b et vecteurs solution x peuvent être
!> gérés en un seul appel ; ils sont stockés comme colonnes de la matrice de
!> second membre B de taille M x NRHS et de la matrice solution X de taille
!> N x NRHS.
!>
!> Le rang effectif de A est déterminé en traitant comme nulles les valeurs
!> singulières inférieures à RCOND fois la plus grande valeur singulière.
!>

Paramètres

M 
!>          M est INTEGER
!>          Le nombre de lignes de la matrice A. M >= 0.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes de la matrice A. N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX de dimension (LDA,N)
!>          À l'entrée, la matrice A de taille M x N.
!>          À la sortie, les min(m,n) premières lignes de A sont écrasées par
!>          ses vecteurs singuliers droits, stockés ligne par ligne.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX de dimension (LDB,NRHS)
!>          À l'entrée, la matrice de second membre B de taille M x NRHS.
!>          À la sortie, B est écrasée par la matrice solution X de taille
!>          N x NRHS.
!>          Si m >= n et RANK = n, la somme des carrés des résidus pour
!>          la solution dans la i-ème colonne est donnée par la somme des
!>          carrés des modules des éléments n+1:m de cette colonne.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,M,N).
!>

S

!>          S est un tableau REAL de dimension (min(M,N))
!>          Les valeurs singulières de A par ordre décroissant.
!>          Le nombre de condition de A en norme 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND est REAL
!>          RCOND est utilisé pour déterminer le rang effectif de A.
!>          Les valeurs singulières S(i) <= RCOND*S(1) sont traitées comme nulles.
!>          Si RCOND < 0, la précision machine est utilisée à la place.
!>

RANK

!>          RANK est INTEGER
!>          Le rang effectif de A, c'est-à-dire le nombre de valeurs singulières
!>          supérieures à RCOND*S(1).
!>

WORK

!>          WORK est un tableau COMPLEX de dimension (MAX(1,LWORK))
!>          À la sortie, si INFO = 0, WORK(1) contient la valeur LWORK optimale.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= 1, et aussi :
!>          LWORK >=  2*min(M,N) + max(M,N,NRHS)
!>          Pour de bonnes performances, LWORK devrait généralement être plus grand.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, renvoie
!>          cette valeur comme première entrée du tableau WORK, et aucun message d'erreur
!>          lié à LWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau REAL de dimension (5*min(M,N))
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 : l'algorithme de calcul de la SVD n'a pas convergé ;
!>                si INFO = i, i éléments hors diagonale d'une forme
!>                bidiagonale intermédiaire n'ont pas convergé vers zéro.
!>
