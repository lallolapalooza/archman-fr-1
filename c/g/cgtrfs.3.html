subroutine cgtrfs (character trans, integer n, integer nrhs, complex, dimension( * ) dl, complex, dimension( * ) d, complex, dimension( * ) du, complex, dimension( * ) dlf, complex, dimension( * ) df, complex, dimension( * ) duf, complex, dimension( * ) du2, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CGTRFS

But :

!>
!> CGTRFS améliore la solution calculée d'un système d'équations
!> linéaires lorsque la matrice des coefficients est tridiagonale, et
!> fournit des bornes d'erreur et des estimations d'erreur arrière
!> pour la solution.
!>

Paramètres

TRANS 
!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposition)
!>          = 'T':  A**T * X = B  (Transposition)
!>          = 'C':  A**H * X = B  (Transposition conjuguée)
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de seconds membres, c'est-à-dire le nombre de
!>          colonnes de la matrice B.  NRHS >= 0.
!>

DL

!>          DL est un tableau COMPLEX, dimension (N-1)
!>          Les (n-1) éléments sous-diagonaux de A.
!>

D

!>          D est un tableau COMPLEX, dimension (N)
!>          Les éléments diagonaux de A.
!>

DU

!>          DU est un tableau COMPLEX, dimension (N-1)
!>          Les (n-1) éléments sur-diagonaux de A.
!>

DLF

!>          DLF est un tableau COMPLEX, dimension (N-1)
!>          Les (n-1) multiplicateurs qui définissent la matrice L à
!>          partir de la factorisation LU de A telle que calculée par
!>          CGTTRF.
!>

DF

!>          DF est un tableau COMPLEX, dimension (N)
!>          Les n éléments diagonaux de la matrice triangulaire
!>          supérieure U issue de la factorisation LU de A.
!>

DUF

!>          DUF est un tableau COMPLEX, dimension (N-1)
!>          Les (n-1) éléments de la première sur-diagonale de U.
!>

DU2

!>          DU2 est un tableau COMPLEX, dimension (N-2)
!>          Les (n-2) éléments de la deuxième sur-diagonale de U.
!>

IPIV

!>          IPIV est un tableau INTEGER, dimension (N)
!>          Les indices de pivot ; pour 1 <= i <= n, la ligne i de la
!>          matrice a été interchangée avec la ligne IPIV(i).
!>          IPIV(i) sera toujours soit i soit i+1 ; IPIV(i) = i
!>          indique qu'un échange de lignes n'a pas été nécessaire.
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          La matrice du second membre B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX, dimension (LDX,NRHS)
!>          À l'entrée, la matrice solution X, telle que calculée par
!>          CGTTRS.
!>          À la sortie, la matrice solution améliorée X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau REAL, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur
!>          solution X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la solution réelle correspondant à X(j),
!>          FERR(j) est une borne supérieure estimée pour la grandeur
!>          du plus grand élément de (X(j) - XTRUE) divisée par la
!>          grandeur du plus grand élément de X(j).  L'estimation est
!>          aussi fiable que celle pour RCOND, et est presque toujours
!>          une légère surestimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau REAL, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de
!>          chaque vecteur solution X(j) (c'est-à-dire le plus petit
!>          changement relatif dans n'importe quel élément de A ou B
!>          qui rend X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (2*N)
!>

RWORK

!>          RWORK est un tableau REAL, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0:  sortie réussie
!>          < 0:  si INFO = -i, le i-ème argument avait une valeur
!>          illégale
!>

Paramètres internes :

!>  ITMAX est le nombre maximal d'étapes de raffinement itératif.
!>
