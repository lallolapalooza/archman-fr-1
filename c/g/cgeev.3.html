subroutine cgeev (character jobvl, character jobvr, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) w, complex, dimension( ldvl, * ) vl, integer ldvl, complex, dimension( ldvr, * ) vr, integer ldvr, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer info)

CGEEV calcule les valeurs propres et, éventuellement, les vecteurs propres gauches et/ou droits pour les matrices GE

Objectif :

!>
!> CGEEV calcule pour une matrice complexe non symétrique A de taille N-par-N,
!> les valeurs propres et, éventuellement, les vecteurs propres gauches et/ou droits.
!>
!> Le vecteur propre droit v(j) de A satisfait
!>                  A * v(j) = lambda(j) * v(j)
!> où lambda(j) est sa valeur propre.
!> Le vecteur propre gauche u(j) de A satisfait
!>               u(j)**H * A = lambda(j) * u(j)**H
!> où u(j)**H dénote la transposée conjuguée de u(j).
!>
!> Les vecteurs propres calculés sont normalisés pour avoir une norme euclidienne
!> égale à 1 et leur composante réelle la plus grande.
!>

Paramètres

JOBVL 
!>          JOBVL est CHARACTER*1
!>          = 'N': les vecteurs propres gauches de A ne sont pas calculés ;
!>          = 'V': les vecteurs propres gauches de A sont calculés.
!>

JOBVR

!>          JOBVR est CHARACTER*1
!>          = 'N': les vecteurs propres droits de A ne sont pas calculés ;
!>          = 'V': les vecteurs propres droits de A sont calculés.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          En entrée, la matrice A de taille N-par-N.
!>          En sortie, A a été écrasé.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,N).
!>

W

!>          W est un tableau COMPLEX, dimension (N)
!>          W contient les valeurs propres calculées.
!>

VL

!>          VL est un tableau COMPLEX, dimension (LDVL,N)
!>          Si JOBVL = 'V', les vecteurs propres gauches u(j) sont stockés les uns
!>          après les autres dans les colonnes de VL, dans le même ordre
!>          que leurs valeurs propres.
!>          Si JOBVL = 'N', VL n'est pas référencé.
!>          u(j) = VL(:,j), la j-ème colonne de VL.
!>

LDVL

!>          LDVL est INTEGER
!>          La dimension principale du tableau VL. LDVL >= 1 ; si
!>          JOBVL = 'V', LDVL >= N.
!>

VR

!>          VR est un tableau COMPLEX, dimension (LDVR,N)
!>          Si JOBVR = 'V', les vecteurs propres droits v(j) sont stockés les uns
!>          après les autres dans les colonnes de VR, dans le même ordre
!>          que leurs valeurs propres.
!>          Si JOBVR = 'N', VR n'est pas référencé.
!>          v(j) = VR(:,j), la j-ème colonne de VR.
!>

LDVR

!>          LDVR est INTEGER
!>          La dimension principale du tableau VR. LDVR >= 1 ; si
!>          JOBVR = 'V', LDVR >= N.
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) contient la taille optimale de LWORK.
!>

LWORK

!>          LWORK est INTEGER
!>          La dimension du tableau WORK. LWORK >= max(1,2*N).
!>          Pour de bonnes performances, LWORK doit généralement être plus grand.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ; la routine
!>          calcule uniquement la taille optimale du tableau WORK, la retourne
!>          comme première entrée du tableau WORK, et aucun message d'erreur
!>          relatif à LWORK n'est émis par XERBLA.
!>

RWORK

!>          RWORK est un tableau REAL, dimension (2*N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!>          > 0 : si INFO = i, l'algorithme QR n'a pas réussi à calculer toutes les
!>                valeurs propres, et aucun vecteur propre n'a été calculé ;
!>                les éléments i+1:N de W contiennent les valeurs propres qui ont
!>                convergé.
!>
