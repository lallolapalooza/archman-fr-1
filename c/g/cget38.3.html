subroutine cget38 (real, dimension( 3 ) rmax, integer, dimension( 3 ) lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

CGET38

Objectif :

!>
!> CGET38 teste CTRSEN, une routine pour estimer les nombres de condition
!> d'un cluster de valeurs propres et/ou de son sous-espace invariant droit associé.
!>
!> Les matrices de test sont lues à partir d'un fichier avec l'unité logique NIN.
!>

Paramètres

RMAX 
!>          RMAX est un tableau REEL, dimension (3)
!>          Valeurs des plus grands rapports de test.
!>          RMAX(1) = plus grands résidus de CHST01 ou comparaison
!>                    entre différents appels à CTRSEN
!>          RMAX(2) = plus grande erreur dans l'inverse des nombres
!>                    de condition en tenant compte de leur conditionnement
!>          RMAX(3) = plus grande erreur dans l'inverse des nombres
!>                    de condition sans tenir compte de leur conditionnement
!>                    (peut être plus grande que RMAX(2))
!>

LMAX

!>          LMAX est un tableau ENTIER, dimension (3)
!>          LMAX(i) est le numéro de l'exemple où le plus grand rapport de test
!>          RMAX(i) est atteint. Également :
!>          Si CGEHRD retourne INFO non nul pour l'exemple i, LMAX(1)=i
!>          Si CHSEQR retourne INFO non nul pour l'exemple i, LMAX(2)=i
!>          Si CTRSEN retourne INFO non nul pour l'exemple i, LMAX(3)=i
!>

NINFO

!>          NINFO est un tableau ENTIER, dimension (3)
!>          NINFO(1) = Nombre de fois où CGEHRD a retourné INFO non nul
!>          NINFO(2) = Nombre de fois où CHSEQR a retourné INFO non nul
!>          NINFO(3) = Nombre de fois où CTRSEN a retourné INFO non nul
!>

KNT

!>          KNT est un ENTIER
!>          Nombre total d'exemples testés.
!>

NIN

!>          NIN est un ENTIER
!>          Numéro de l'unité logique d'entrée.
!>
