Documentation de la fonction/sous-routine

subroutine cglmts (integer n, integer m, integer p, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, integer lda, complex, dimension( ldb, * ) b, complex, dimension( ldb, * ) bf, integer ldb, complex, dimension( * ) d, complex, dimension( * ) df, complex, dimension( * ) x, complex, dimension( * ) u, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real result)

CGLMTS

Objectif:

!>
!> CGLMTS teste CGGGLM - une sous-routine pour résoudre le problème du
!> modèle linéaire généralisé.
!>

Paramètres

N 
!>          N est un ENTIER
!>          Le nombre de lignes des matrices A et B.  N >= 0.
!>

M

!>          M est un ENTIER
!>          Le nombre de colonnes de la matrice A.  M >= 0.
!>

P

!>          P est un ENTIER
!>          Le nombre de colonnes de la matrice B.  P >= 0.
!>

A

!>          A est un tableau COMPLEXE, dimension (LDA,M)
!>          La matrice A de taille N-par-M.
!>

AF

!>          AF est un tableau COMPLEXE, dimension (LDA,M)
!>

LDA

!>          LDA est un ENTIER
!>          La dimension principale des tableaux A, AF. LDA >= max(M,N).
!>

B

!>          B est un tableau COMPLEXE, dimension (LDB,P)
!>          La matrice B de taille N-par-P.
!>

BF

!>          BF est un tableau COMPLEXE, dimension (LDB,P)
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale des tableaux B, BF. LDB >= max(P,N).
!>

D

!>          D est un tableau COMPLEXE, dimension( N )
!>          En entrée, le membre de gauche de la GLM.
!>

DF

!>          DF est un tableau COMPLEXE, dimension( N )
!>

X

!>          X est un tableau COMPLEXE, dimension( M )
!>          vecteur solution X dans le problème GLM.
!>

U

!>          U est un tableau COMPLEXE, dimension( P )
!>          vecteur solution U dans le problème GLM.
!>

WORK

!>          WORK est un tableau COMPLEXE, dimension (LWORK)
!>

LWORK

!>          LWORK est un ENTIER
!>          La dimension du tableau WORK.
!>

RWORK

!>          RWORK est un tableau REEL, dimension (M)
!>

RESULT

!>          RESULT est REEL
!>          Le ratio de test:
!>                           norme( d - A*x - B*u )
!>            RESULT = -----------------------------------------
!>                     (norme(A)+norme(B))*(norme(x)+norme(u))*EPS
!>
