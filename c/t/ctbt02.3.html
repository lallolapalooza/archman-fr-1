sous-programme ctbt02 (caractère uplo, caractère trans, caractère diag, entier n, entier kd, entier nrhs, complexe, dimension( ldab, * ) ab, entier ldab, complexe, dimension( ldx, * ) x, entier ldx, complexe, dimension( ldb, * ) b, entier ldb, complexe, dimension( * ) work, réel, dimension( * ) rwork, réel resid)

CTBT02

Objectif :

!>
!> CTBT02 calcule le résidu pour la solution calculée d'un système
!> d'équations linéaires triangulaires op(A)*X = B, où A est une
!> matrice triangulaire bande. Le rapport de test est le maximum sur
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS ),
!> où op(A) = A, A**T, ou A**H, b est la colonne de B, x est le
!> vecteur solution, et EPS est l'epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie l'opération appliquée à A.
!>          = 'N':  A    * X = B  (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire unitaire ou non.
!>          = 'N':  Triangulaire non unitaire
!>          = 'U':  Triangulaire unitaire
!>

N

!>          N est un ENTIER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est un ENTIER
!>          Le nombre de super-diagonales ou sous-diagonales de la
!>          matrice triangulaire bande A.  KD >= 0.
!>

NRHS

!>          NRHS est un ENTIER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.  NRHS >= 0.
!>

AB

!>          AB est un tableau COMPLEXE, de dimension (LDA,N)
!>          La matrice triangulaire bande supérieure ou inférieure A, stockée dans les
!>          kd+1 premières lignes du tableau. La j-ième colonne de A est stockée
!>          dans la j-ième colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB est un ENTIER
!>          La dimension principale du tableau AB.  LDAB >= max(1,KD+1).
!>

X

!>          X est un tableau COMPLEXE, de dimension (LDX,NRHS)
!>          Les vecteurs solution calculée pour le système d'équations
!>          linéaires.
!>

LDX

!>          LDX est un ENTIER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est un tableau COMPLEXE, de dimension (LDB,NRHS)
!>          Les vecteurs membres de droite pour le système d'équations
!>          linéaires.
!>

LDB

!>          LDB est un ENTIER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEXE, de dimension (N)
!>

RWORK

!>          RWORK est un tableau RÉEL, de dimension (N)
!>

RESID

!>          RESID est RÉEL
!>          Le maximum sur le nombre de membres de droite de
!>          norm(op(A)*x - b) / ( norm(op(A)) * norm(x) * EPS ).
!>
