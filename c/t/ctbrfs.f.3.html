sous-routine ctbrfs (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CTBRFS

But :

!>
!> CTBRFS fournit des bornes d'erreur et des estimations d'erreur arrière pour la
!> solution d'un système d'équations linéaires avec une matrice de bande triangulaire
!> coefficient.
!>
!> La matrice solution X doit être calculée par CTBTRS ou par un autre
!> moyen avant d'entrer dans cette routine. CTBRFS n'effectue pas de raffinement itératif
!> car cela ne peut pas améliorer l'erreur arrière.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  A est triangulaire supérieure ;
!>          = 'L':  A est triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          = 'N':  A est triangulaire non unitaire ;
!>          = 'U':  A est triangulaire unitaire.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

KD

!>          KD est INTEGER
!>          Le nombre de super-diagonales ou de sous-diagonales de la
!>          matrice de bande triangulaire A.  KD >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X.  NRHS >= 0.
!>

AB

!>          AB est un tableau COMPLEX, dimension (LDAB,N)
!>          La matrice de bande triangulaire supérieure ou inférieure A, stockée dans les
!>          kd+1 premières lignes du tableau. La j-ème colonne de A est stockée
!>          dans la j-ème colonne du tableau AB comme suit :
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) pour max(1,j-kd) <= i <= j ;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) pour j <= i <= min(n,j+kd).
!>          Si DIAG = 'U', les éléments diagonaux de A ne sont pas référencés
!>          et sont supposés être 1.
!>

LDAB

!>          LDAB est INTEGER
!>          La dimension principale du tableau AB.  LDAB >= KD+1.
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,NRHS)
!>          La matrice des membres de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEX, dimension (LDX,NRHS)
!>          La matrice solution X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau REAL, dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ème colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée pour la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisé par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation pour RCOND, et est presque toujours une légère
!>          sur-estimation de l'erreur vraie.
!>

BERR

!>          BERR est un tableau REAL, dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEX, dimension (2*N)
!>

RWORK

!>          RWORK est un tableau REAL, dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
