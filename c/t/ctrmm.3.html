subroutine ctrmm (side, uplo, transa, diag, m, n, alpha, a, lda, b, ldb)

Documentation de la fonction/sous-programme

sous-programme ctrmm (caractère side, caractère uplo, caractère transa, caractère diag, entier m, entier n, complexe alpha, complexe, dimension(lda,*) a, entier lda, complexe, dimension(ldb,*) b, entier ldb)

CTRMM

Objectif :

!>
!> CTRMM  effectue l'une des opérations matricielles
!>
!>    B := alpha*op( A )*B,   ou   B := alpha*B*op( A )
!>
!> où  alpha  est un scalaire,  B  est une matrice m par n,  A  est une matrice
!> triangulaire unitaire, ou non unitaire, supérieure ou inférieure, et  op( A )
!> est l'une des suivantes :
!>
!>    op( A ) = A   ou   op( A ) = A**T   ou   op( A ) = A**H.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>           À l'entrée, SIDE spécifie si op( A ) multiplie B par la gauche ou
!>           par la droite comme suit :
!>
!>              SIDE = 'L' ou 'l'   B := alpha*op( A )*B.
!>
!>              SIDE = 'R' ou 'r'   B := alpha*B*op( A ).
!>

UPLO

!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la matrice A est une matrice
!>           triangulaire supérieure ou inférieure comme suit :
!>
!>              UPLO = 'U' ou 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' ou 'l'   A est une matrice triangulaire inférieure.
!>

TRANSA

!>          TRANSA est CHARACTER*1
!>           À l'entrée, TRANSA spécifie la forme de op( A ) à utiliser dans la
!>           multiplication matricielle comme suit :
!>
!>              TRANSA = 'N' ou 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' ou 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' ou 'c'   op( A ) = A**H.
!>

DIAG

!>          DIAG est CHARACTER*1
!>           À l'entrée, DIAG spécifie si A est unitaire ou non, et triangulaire
!>           comme suit :
!>
!>              DIAG = 'U' ou 'u'   A est supposée être unitaire triangulaire.
!>
!>              DIAG = 'N' ou 'n'   A n'est pas supposée être unitaire
!>                                  triangulaire.
!>

M

!>          M est INTEGER
!>           À l'entrée, M spécifie le nombre de lignes de B. M doit être
!>           supérieur ou égal à zéro.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie le nombre de colonnes de B. N doit être
!>           supérieur ou égal à zéro.
!>

ALPHA

!>          ALPHA est COMPLEX
!>           À l'entrée, ALPHA spécifie le scalaire alpha. Lorsque alpha est
!>           zéro, A n'est pas référencé et B n'a pas besoin d'être défini avant
!>           l'entrée.
!>

A

!>          A est un tableau COMPLEX, de dimension ( LDA, k ), où k est m
!>           lorsque SIDE = 'L' ou 'l' et n lorsque SIDE = 'R' ou 'r'.
!>           Avant l'entrée, avec UPLO = 'U' ou 'u', la partie supérieure
!>           triangulaire principale de k par k du tableau A doit contenir la
!>           matrice triangulaire supérieure et la partie strictement inférieure
!>           triangulaire de A n'est pas référencée.
!>           Avant l'entrée, avec UPLO = 'L' ou 'l', la partie inférieure
!>           triangulaire principale de k par k du tableau A doit contenir la
!>           matrice triangulaire inférieure et la partie strictement supérieure
!>           triangulaire de A n'est pas référencée.
!>           Notez que lorsque DIAG = 'U' ou 'u', les éléments diagonaux de
!>           A ne sont pas non plus référencés, mais sont supposés être unitaires.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (subroutine). Lorsque SIDE = 'L' ou 'l',
!>           LDA doit être au moins max( 1, m ), lorsque SIDE = 'R' ou 'r',
!>           LDA doit être au moins max( 1, n ).
!>

B

!>          B est un tableau COMPLEX, de dimension ( LDB, N ).
!>           Avant l'entrée, la partie principale m par n du tableau B doit contenir
!>           la matrice B, et à la sortie, elle est écrasée par la matrice transformée.
!>

LDB

!>          LDB est INTEGER
!>           À l'entrée, LDB spécifie la première dimension de B telle que déclarée
!>           dans le programme appelant (subroutine). LDB doit être au moins
!>           max( 1, m ).
!>
