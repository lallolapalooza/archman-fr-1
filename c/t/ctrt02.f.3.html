sous-routine ctrt02 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CTRT02

But:

!>
!> CTRT02 calcule le résidu pour la solution calculée d'un système d'équations linéaires triangulaire op(A)*X = B, où A est une matrice triangulaire. Le ratio de test est le maximum sur
!>    norme(b - op(A)*x) / ( ||op(A)||_1 * norme(x) * EPS ),
!> où op(A) = A, A**T, ou A**H, b est la colonne de B, x est le
!> vecteur solution, et EPS est l'epsilon machine.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie l'opération appliquée à A.
!>          = 'N':  A    * X = B  (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Transposée conjuguée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          Spécifie si la matrice A est unitaire triangulaire ou non.
!>          = 'N':  Triangulaire non unitaire
!>          = 'U':  Triangulaire unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices X et B.  NRHS >= 0.
!>

A

!>          A est un tableau COMPLEX, de dimension (LDA,N)
!>          La matrice triangulaire A.  Si UPLO = 'U', la partie triangulaire supérieure
!>          supérieure n par n du tableau A contient la matrice triangulaire
!>          supérieure, et la partie strictement triangulaire inférieure de
!>          A n'est pas référencée.  Si UPLO = 'L', la partie triangulaire inférieure
!>          inférieure n par n du tableau A contient la matrice triangulaire
!>          inférieure, et la partie strictement triangulaire supérieure de A n'est pas
!>          référencée.  Si DIAG = 'U', les éléments diagonaux de A sont
!>          également non référencés et sont supposés être 1.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

X

!>          X est un tableau COMPLEX, de dimension (LDX,NRHS)
!>          Les vecteurs solution calculés pour le système d'équations linéaires.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X.  LDX >= max(1,N).
!>

B

!>          B est un tableau COMPLEX, de dimension (LDB,NRHS)
!>          Les vecteurs membres de droite pour le système d'équations linéaires.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B.  LDB >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEX, de dimension (N)
!>

RWORK

!>          RWORK est un tableau REAL, de dimension (N)
!>

RESID

!>          RESID est REAL
!>          Le maximum sur le nombre de membres de droite de
!>          norme(op(A)*X - B) / ( norme(op(A)) * norme(X) * EPS ).
!>
