subroutine ctbsv (character uplo, character trans, character diag, integer n, integer k, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx)

CTBSV

Objectif :

!>
!> CTBSV résout l'un des systèmes d'équations
!>
!>    A*x = b,   ou   A**T*x = b,   ou   A**H*x = b,
!>
!> où b et x sont des vecteurs de n éléments et A est une matrice bande
!> triangulaire supérieure ou inférieure de taille n par n, unitaire ou non,
!> avec ( k + 1 ) diagonales.
!>
!> Ce sous-programme ne contient aucun test de singularité ou de
!> quasi-singularité. De tels tests doivent être effectués avant d'appeler
!> ce sous-programme.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           En entrée, UPLO spécifie si la matrice est une matrice
!>           triangulaire supérieure ou inférieure comme suit :
!>
!>              UPLO = 'U' ou 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' ou 'l'   A est une matrice triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           En entrée, TRANS spécifie les équations à résoudre comme suit :
!>
!>              TRANS = 'N' ou 'n'   A*x = b.
!>
!>              TRANS = 'T' ou 't'   A**T*x = b.
!>
!>              TRANS = 'C' ou 'c'   A**H*x = b.
!>

DIAG

!>          DIAG est CHARACTER*1
!>           En entrée, DIAG spécifie si A est unitaire ou non
!>           triangulaire comme suit :
!>
!>              DIAG = 'U' ou 'u'   A est supposée être unitaire
!>                                  triangulaire.
!>
!>              DIAG = 'N' ou 'n'   A n'est pas supposée être unitaire
!>                                  triangulaire.
!>

N

!>          N est INTEGER
!>           En entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

K

!>          K est INTEGER
!>           En entrée avec UPLO = 'U' ou 'u', K spécifie le nombre de
!>           super-diagonales de la matrice A.
!>           En entrée avec UPLO = 'L' ou 'l', K spécifie le nombre de
!>           sous-diagonales de la matrice A.
!>           K doit satisfaire  0 .le. K.
!>

A

!>          A est un tableau de type COMPLEX, de dimension ( LDA, N )
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie bande triangulaire
!>           supérieure de la matrice des coefficients, fournie colonne par
!>           colonne, avec la diagonale principale de la matrice dans la ligne
!>           ( k + 1 ) du tableau, la première super-diagonale commençant à la
!>           position 2 dans la ligne k, et ainsi de suite. Le triangle supérieur
!>           gauche de taille k par k du tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera une matrice bande
!>           triangulaire supérieure depuis un stockage de matrice complète
!>           conventionnelle vers un stockage bande :
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie bande triangulaire
!>           inférieure de la matrice des coefficients, fournie colonne par
!>           colonne, avec la diagonale principale de la matrice dans la ligne 1
!>           du tableau, la première sous-diagonale commençant à la position 1
!>           dans la ligne 2, et ainsi de suite. Le triangle inférieur droit
!>           de taille k par k du tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera une matrice bande
!>           triangulaire inférieure depuis un stockage de matrice complète
!>           conventionnelle vers un stockage bande :
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Notez que lorsque DIAG = 'U' ou 'u', les éléments du tableau A
!>           correspondant aux éléments diagonaux de la matrice ne sont pas
!>           référencés, mais sont supposés être unitaires.
!>

LDA

!>          LDA est INTEGER
!>           En entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme). LDA doit être au moins
!>           ( k + 1 ).
!>

X

!>          X est un tableau de type COMPLEX, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le vecteur
!>           colonne b de n éléments. En sortie, X est écrasé par le vecteur
!>           solution x.
!>

INCX

!>          INCX est INTEGER
!>           En entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être nul.
!>
