subroutine ctgevc (character side, character howmny, logical, dimension( * ) select, integer n, complex, dimension( lds, * ) s, integer lds, complex, dimension( ldp, * ) p, integer ldp, complex, dimension( ldvl, * ) vl, integer ldvl, complex, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, complex, dimension( * ) work, real, dimension( * ) rwork, integer

CTGEV

C

But

: 
!>
!> CTGEVC calcule certaines ou toutes les vecteurs propres à droite et/ou à gauche
!> d'une paire de matrices complexes (S,P), où S et P sont triangulaires supérieures.
!> Les paires de matrices de ce type sont produites par la factorisation
!> de Schur généralisée d'une paire de matrices complexes (A,B) :
!>
!>    A = Q*S*Z**H,  B = Q*P*Z**H
!>
!> telle que calculée par CGGHRD + CHGEQZ.
!>
!> Le vecteur propre à droite x et le vecteur propre à gauche y de (S,P)
!> correspondant à une valeur propre w sont définis par :
!>
!>    S*x = w*P*x,  (y**H)*S = w*(y**H)*P,
!>
!> où y**H désigne la transposée conjuguée de y.
!> Les valeurs propres ne sont pas fournies en entrée à cette routine, mais sont calculées
!> directement à partir des éléments diagonaux de S et P.
!>
!> Cette routine renvoie les matrices X et/ou Y des vecteurs propres à droite et à gauche
!> de (S,P), ou les produits Z*X et/ou Q*Y,
!> où Z et Q sont les matrices d'entrée.
!> Si Q et Z sont les facteurs unitaires de la factorisation de Schur
!> généralisée d'une paire de matrices (A,B), alors Z*X et Q*Y
!> sont les matrices des vecteurs propres à droite et à gauche de (A,B).
!
>

Paramètre

s
SIDE 
!>          SIDE est CHARACTER*1
!>          = 'R' : ne calcule que les vecteurs propres à droite ;
!>          = 'L' : ne calcule que les vecteurs propres à gauche ;
!>          = 'B' : calcule les vecteurs propres à droite et à gauche.
!
>

HOWMN

Y 
!>          HOWMNY est CHARACTER*1
!>          = 'A' : calcule tous les vecteurs propres à droite et/ou à gauche ;
!>          = 'B' : calcule tous les vecteurs propres à droite et/ou à gauche,
!>                 rétrotransformés par les matrices dans VR et/ou VL ;
!>          = 'S' : calcule les vecteurs propres à droite et/ou à gauche sélectionnés,
!>                 spécifiés par le tableau logique SELECT.
!
>

SELEC

T 
!>          SELECT est un tableau LOGICAL, dimension (N)
!>          Si HOWMNY='S', SELECT spécifie les vecteurs propres à
!>          calculer. Le vecteur propre correspondant à la j-ième
!>          valeur propre est calculé si SELECT(j) = .TRUE..
!>          Non référencé si HOWMNY = 'A' ou 'B'.
!
>

N 
!>          N est INTEGER
!>          L'ordre des matrices S et P. N >= 0.
!
>

S 
!>          S est un tableau COMPLEX, dimension (LDS,N)
!>          La matrice triangulaire supérieure S issue d'une factorisation
!>          de Schur généralisée, telle que calculée par CHGEQZ.
!
>

LD

S 
!>          LDS est INTEGER
!>          La dimension principale du tableau S. LDS >= max(1,N).
!
>

P 
!>          P est un tableau COMPLEX, dimension (LDP,N)
!>          La matrice triangulaire supérieure P issue d'une factorisation
!>          de Schur généralisée, telle que calculée par CHGEQZ. P doit avoir des
!>          éléments diagonaux réels.
!
>

LD

P 
!>          LDP est INTEGER
!>          La dimension principale du tableau P. LDP >= max(1,N).
!
>

V

L 
!>          VL est un tableau COMPLEX, dimension (LDVL,MM)
!>          À l'entrée, si SIDE = 'L' ou 'B' et HOWMNY = 'B', VL doit
!>          contenir une matrice N x N Q (généralement la matrice unitaire Q
!>          des vecteurs de Schur gauche renvoyée par CHGEQZ).
!>          À la sortie, si SIDE = 'L' ou 'B', VL contient :
!>          si HOWMNY = 'A', la matrice Y des vecteurs propres à gauche de (S,P) ;
!>          si HOWMNY = 'B', la matrice Q*Y ;
!>          si HOWMNY = 'S', les vecteurs propres à gauche de (S,P) spécifiés par
!>                      SELECT, stockés consécutivement dans les colonnes de
!>                      VL, dans le même ordre que leurs valeurs propres.
!>          Non référencé si SIDE = 'R'.
!
>

LDV

L 
!>          LDVL est INTEGER
!>          La dimension principale du tableau VL. LDVL >= 1, et si
!>          SIDE = 'L' ou 'l' ou 'B' ou 'b', LDVL >= N.
!
>

V

R 
!>          VR est un tableau COMPLEX, dimension (LDVR,MM)
!>          À l'entrée, si SIDE = 'R' ou 'B' et HOWMNY = 'B', VR doit
!>          contenir une matrice N x N Z (généralement la matrice unitaire Z
!>          des vecteurs de Schur droit renvoyée par CHGEQZ).
!>          À la sortie, si SIDE = 'R' ou 'B', VR contient :
!>          si HOWMNY = 'A', la matrice X des vecteurs propres à droite de (S,P) ;
!>          si HOWMNY = 'B', la matrice Z*X ;
!>          si HOWMNY = 'S', les vecteurs propres à droite de (S,P) spécifiés par
!>                      SELECT, stockés consécutivement dans les colonnes de
!>                      VR, dans le même ordre que leurs valeurs propres.
!>          Non référencé si SIDE = 'L'.
!
>

LDV

R 
!>          LDVR est INTEGER
!>          La dimension principale du tableau VR. LDVR >= 1, et si
!>          SIDE = 'R' ou 'B', LDVR >= N.
!
>

M

M 
!>          MM est INTEGER
!>          Le nombre de colonnes dans les tableaux VL et/ou VR. MM >= M.
!
>

M 
!>          M est INTEGER
!>          Le nombre de colonnes dans les tableaux VL et/ou VR effectivement
!>          utilisées pour stocker les vecteurs propres. Si HOWMNY = 'A' ou 'B', M
!>          est fixé à N. Chaque vecteur propre sélectionné occupe une colonne.
!
>

WOR

K 
!>          WORK est un tableau COMPLEX, dimension (2*N)
!
>

RWOR

K 
!>          RWORK est un tableau REAL, dimension (2*N)
!
>

INF

O 
!>          INFO est INTEGER
!>          = 0 : sortie réussie.
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale.
!
>
