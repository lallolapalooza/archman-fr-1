sous-routine ctrsyl3 (caractere trana, caractere tranb, entier isgn, entier m, entier n, complexe, dimension( lda, * ) a, entier lda, complexe, dimension( ldb, * ) b, entier ldb, complexe, dimension( ldc, * ) c, entier ldc, reel scale, reel, dimension( ldswork, * ) swork, entier ldswork, entier info)

CTRSYL3

Objectif :

!>
!>  CTRSYL3 résout l'équation matricielle de Sylvester complexe :
!>
!>     op(A)*X + X*op(B) = scale*C ou
!>     op(A)*X - X*op(B) = scale*C,
!>
!>  où op(A) = A ou A**H, et A et B sont toutes deux triangulaires supérieures. A
!>  est de taille M par M et B est de taille N par N ; C est le membre de droite
!>  et X la solution, toutes deux de taille M par N ; et scale est un facteur
!>  d'échelle en sortie, défini à <= 1 pour éviter le débordement dans X.
!>
!>  Il s'agit de la version bloc de l'algorithme.
!>

Paramètres

TRANA 
!>          TRANA est CHARACTER*1
!>          Spécifie l'option op(A) :
!>          = 'N' : op(A) = A (Pas de transposée)
!>          = 'C' : op(A) = A**H (Transposée conjuguée)
!>

TRANB

!>          TRANB est CHARACTER*1
!>          Spécifie l'option op(B) :
!>          = 'N' : op(B) = B (Pas de transposée)
!>          = 'C' : op(B) = B**H (Transposée conjuguée)
!>

ISGN

!>          ISGN est INTEGER
!>          Spécifie le signe dans l'équation :
!>          = +1 : résout op(A)*X + X*op(B) = scale*C
!>          = -1 : résout op(A)*X - X*op(B) = scale*C
!>

M

!>          M est INTEGER
!>          L'ordre de la matrice A, et le nombre de lignes des matrices X et C.
!>          M >= 0.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice B, et le nombre de colonnes des matrices X et C.
!>          N >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,M)
!>          La matrice triangulaire supérieure A.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A. LDA >= max(1,M).
!>

B

!>          B est un tableau COMPLEX, dimension (LDB,N)
!>          La matrice triangulaire supérieure B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

C

!>          C est un tableau COMPLEX, dimension (LDC,N)
!>          À l'entrée, la matrice C du membre de droite de taille M par N.
!>          À la sortie, C est écrasé par la matrice solution X.
!>

LDC

!>          LDC est INTEGER
!>          La dimension principale du tableau C. LDC >= max(1,M).
!>

SCALE

!>          SCALE est REAL
!>          Le facteur d'échelle, scale, défini à <= 1 pour éviter le débordement
!>          dans X.
!>

SWORK

!>          SWORK est un tableau REAL, dimension (MAX(2, ROWS), MAX(1,COLS)).
!>          À la sortie, si INFO = 0, SWORK(1) renvoie la valeur optimale ROWS
!>          et SWORK(2) renvoie la valeur optimale COLS.
!>

LDSWORK

!>          LDSWORK est INTEGER
!>          LDSWORK >= MAX(2,ROWS), où ROWS = ((M + NB - 1) / NB + 1)
!>          et NB est la taille de bloc optimale.
!>
!>          Si LDSWORK = -1, alors une requête d'espace de travail est supposée ;
!>          la routine calcule uniquement les dimensions optimales du tableau
!>          SWORK, renvoie ces valeurs comme première et deuxième entrées du
!>          tableau SWORK, et aucun message d'erreur lié à LWORK n'est émis
!>          par XERBLA.
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>          = 1 : A et B ont des valeurs propres communes ou très proches ; des
!>               valeurs perturbées ont été utilisées pour résoudre l'équation
!>               (mais les matrices A et B ne sont pas modifiées).
!>
