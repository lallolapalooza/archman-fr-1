subroutine ctrt06 (real rcond, real rcondc, character uplo, character diag, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) rwork, real rat)

CTRT06

But :

!>
!> CTRT06 calcule un rapport de test comparant RCOND (le nombre de condition réciproque
!> d'une matrice triangulaire A) et RCONDC, l'estimation calculée par CTRCON.
!> Les informations sur la matrice triangulaire A sont utilisées si une estimation est nulle
!> et l'autre non nulle pour décider si un sous-dépassement dans l'estimation est justifié.
!>

Paramètres

RCOND 
!>          RCOND est REAL
!>          L'estimation du nombre de condition réciproque obtenue en formant
!>          l'inverse explicite de la matrice A et en calculant
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC est REAL
!>          L'estimation du nombre de condition réciproque calculée par
!>          CTRCON.
!>

UPLO

!>          UPLO est CHARACTER
!>          Spécifie si la matrice A est triangulaire supérieure ou inférieure.
!>          = 'U':  Triangulaire supérieure
!>          = 'L':  Triangulaire inférieure
!>

DIAG

!>          DIAG est CHARACTER
!>          Spécifie si la matrice A est unitaire ou non.
!>          = 'N':  Non unitaire
!>          = 'U':  Unitaire
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A.  N >= 0.
!>

A

!>          A est un tableau COMPLEX, dimension (LDA,N)
!>          La matrice triangulaire A.  Si UPLO = 'U', la partie triangulaire
!>          supérieure gauche de taille n par n du tableau A contient la matrice
!>          triangulaire supérieure, et la partie strictement triangulaire inférieure
!>          de A n'est pas référencée.  Si UPLO = 'L', la partie triangulaire
!>          inférieure gauche de taille n par n du tableau A contient la matrice
!>          triangulaire inférieure, et la partie strictement triangulaire supérieure
!>          de A n'est pas référencée.  Si DIAG = 'U', les éléments diagonaux
!>          de A ne sont pas non plus référencés et sont supposés être 1.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,N).
!>

RWORK

!>          RWORK est un tableau REAL, dimension (N)
!>

RAT

!>          RAT est REAL
!>          Le rapport de test.  Si RCOND et RCONDC sont tous deux non nuls,
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          Si RAT = 0, les deux estimations sont exactement identiques.
!>
