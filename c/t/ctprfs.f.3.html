subroutine ctprfs (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CTPRFS

Objectif :

!>
!> CTPRFS fournit des bornes d'erreur et des estimations d'erreurs arrière pour
!> la solution d'un système d'équations linéaires avec une matrice
!> triangulaire factorisée comme coefficient.
!>
!> La matrice solution X doit être calculée par CTPTRS ou par un autre
!> moyen avant d'entrer dans cette routine. CTPRFS n'effectue pas de
!> raffinement itératif car cela ne peut pas améliorer l'erreur arrière.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U':  A est triangulaire supérieure ;
!>          = 'L':  A est triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>          Spécifie la forme du système d'équations :
!>          = 'N':  A * X = B     (Pas de transposée)
!>          = 'T':  A**T * X = B  (Transposée)
!>          = 'C':  A**H * X = B  (Conjuguée transposée)
!>

DIAG

!>          DIAG est CHARACTER*1
!>          = 'N':  A est triangulaire non unitaire ;
!>          = 'U':  A est triangulaire unitaire.
!>

N

!>          N est INTEGER
!>          L'ordre de la matrice A. N >= 0.
!>

NRHS

!>          NRHS est INTEGER
!>          Le nombre de membres de droite, c'est-à-dire le nombre de colonnes
!>          des matrices B et X. NRHS >= 0.
!>

AP

!>          AP est un tableau COMPLEXE, de dimension (N*(N+1)/2)
!>          La matrice triangulaire supérieure ou inférieure A, factorisée colonne
!>          par colonne dans un tableau linéaire. La j-ième colonne de A est stockée
!>          dans le tableau AP comme suit :
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) pour 1<=i<=j ;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) pour j<=i<=n.
!>          Si DIAG = 'U', les éléments diagonaux de A ne sont pas référencés
!>          et sont supposés être 1.
!>

B

!>          B est un tableau COMPLEXE, de dimension (LDB,NRHS)
!>          La matrice membre de droite B.
!>

LDB

!>          LDB est INTEGER
!>          La dimension principale du tableau B. LDB >= max(1,N).
!>

X

!>          X est un tableau COMPLEXE, de dimension (LDX,NRHS)
!>          La matrice solution X.
!>

LDX

!>          LDX est INTEGER
!>          La dimension principale du tableau X. LDX >= max(1,N).
!>

FERR

!>          FERR est un tableau RÉEL, de dimension (NRHS)
!>          La borne d'erreur avant estimée pour chaque vecteur solution
!>          X(j) (la j-ième colonne de la matrice solution X).
!>          Si XTRUE est la vraie solution correspondant à X(j), FERR(j)
!>          est une borne supérieure estimée de la magnitude du plus grand
!>          élément de (X(j) - XTRUE) divisée par la magnitude du plus grand
!>          élément de X(j). L'estimation est aussi fiable que
!>          l'estimation de RCOND, et est presque toujours une légère
!>          sur-estimation de l'erreur réelle.
!>

BERR

!>          BERR est un tableau RÉEL, de dimension (NRHS)
!>          L'erreur arrière relative composante par composante de chaque vecteur solution
!>          X(j) (c'est-à-dire le plus petit changement relatif dans
!>          n'importe quel élément de A ou B qui fait de X(j) une solution exacte).
!>

WORK

!>          WORK est un tableau COMPLEXE, de dimension (2*N)
!>

RWORK

!>          RWORK est un tableau RÉEL, de dimension (N)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
