subroutine ctbmv (character uplo, character trans, character diag, integer n, integer k, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx)

CTBMV

Objectif :

!>
!> CTBMV  effectue l'une des opérations matricielles-vectorielles suivantes :
!>
!>    x := A*x,   ou   x := A**T*x,   ou   x := A**H*x,
!>
!> où x est un vecteur de n éléments et  A est une matrice bande triangulaire
!> de n par n, unitaire ou non, supérieure ou inférieure, avec ( k + 1 ) diagonales.
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>           À l'entrée, UPLO spécifie si la matrice est une matrice supérieure
!>           ou inférieure comme suit :
!>
!>              UPLO = 'U' or 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' or 'l'   A est une matrice triangulaire inférieure.
!>

TRANS

!>          TRANS est CHARACTER*1
!>           À l'entrée, TRANS spécifie l'opération à effectuer comme suit :
!>
!>              TRANS = 'N' or 'n'   x := A*x.
!>
!>              TRANS = 'T' or 't'   x := A**T*x.
!>
!>              TRANS = 'C' or 'c'   x := A**H*x.
!>

DIAG

!>          DIAG est CHARACTER*1
!>           À l'entrée, DIAG spécifie si A est unitaire ou non
!>           triangulaire comme suit :
!>
!>              DIAG = 'U' or 'u'   A est supposée être unitaire triangulaire.
!>
!>              DIAG = 'N' or 'n'   A n'est pas supposée être unitaire
!>                                  triangulaire.
!>

N

!>          N est INTEGER
!>           À l'entrée, N spécifie l'ordre de la matrice A.
!>           N doit être supérieur ou égal à zéro.
!>

K

!>          K est INTEGER
!>           À l'entrée avec UPLO = 'U' or 'u', K spécifie le nombre de
!>           super-diagonales de la matrice A.
!>           À l'entrée avec UPLO = 'L' or 'l', K spécifie le nombre de
!>           sous-diagonales de la matrice A.
!>           K doit satisfaire  0 .le. K.
!>

A

!>          A est un tableau de type COMPLEX, de dimension ( LDA, N ).
!>           À l'entrée avec UPLO = 'U' or 'u', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie bande triangulaire supérieure
!>           de la matrice des coefficients, fournie colonne par colonne,
!>           avec la diagonale principale de la matrice dans la ligne
!>           ( k + 1 ) du tableau, la première super-diagonale commençant en
!>           position 2 sur la ligne k, et ainsi de suite. Le triangle supérieur
!>           gauche k par k du tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera une matrice bande triangulaire
!>           supérieure depuis un stockage matriciel complet conventionnel vers un stockage bande :
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           À l'entrée avec UPLO = 'L' or 'l', la partie principale ( k + 1 )
!>           par n du tableau A doit contenir la partie bande triangulaire inférieure
!>           de la matrice des coefficients, fournie colonne par colonne,
!>           avec la diagonale principale de la matrice dans la ligne 1 du
!>           tableau, la première sous-diagonale commençant en position 1 sur
!>           la ligne 2, et ainsi de suite. Le triangle inférieur droit k par k du
!>           tableau A n'est pas référencé.
!>           Le segment de programme suivant transférera une matrice bande triangulaire
!>           inférieure depuis un stockage matriciel complet conventionnel vers un stockage bande :
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           Notez que lorsque DIAG = 'U' or 'u', les éléments du tableau A
!>           correspondant aux éléments diagonaux de la matrice ne sont pas
!>           référencés, mais sont supposés être unitaires.
!>

LDA

!>          LDA est INTEGER
!>           À l'entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sous-programme). LDA doit être au moins
!>           ( k + 1 ).
!>

X

!>          X est un tableau de type COMPLEX, de dimension au moins
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Avant l'entrée, le tableau incrémenté X doit contenir le vecteur
!>           x de n éléments. À la sortie, X est écrasé par le vecteur
!>           transformé x.
!>

INCX

!>          INCX est INTEGER
!>           À l'entrée, INCX spécifie l'incrément pour les éléments de
!>           X. INCX ne doit pas être nul.
!>
