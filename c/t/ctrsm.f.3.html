subroutine ctrsm (character side, character uplo, character transa, character diag, integer m, integer n, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb)

CTRSM

Objectif:

!>
!> CTRSM  résout une des équations matricielles
!>
!>    op( A )*X = alpha*B,   ou   X*op( A ) = alpha*B,
!>
!> où alpha est un scalaire, X et B sont des matrices m par n, A est une matrice
!> triangulaire unitaire, ou non unitaire, supérieure ou inférieure et op( A )
!> est l'une des suivantes :
!>
!>    op( A ) = A   ou   op( A ) = A**T   ou   op( A ) = A**H.
!>
!> La matrice X est écrasée sur B.
!>

Paramètres

SIDE 
!>          SIDE est CHARACTER*1
!>           En entrée, SIDE spécifie si op( A ) apparaît à gauche
!>           ou à droite de X comme suit :
!>
!>              SIDE = 'L' ou 'l'   op( A )*X = alpha*B.
!>
!>              SIDE = 'R' ou 'r'   X*op( A ) = alpha*B.
!>

UPLO

!>          UPLO est CHARACTER*1
!>           En entrée, UPLO spécifie si la matrice A est une matrice triangulaire
!>           supérieure ou inférieure comme suit :
!>
!>              UPLO = 'U' ou 'u'   A est une matrice triangulaire supérieure.
!>
!>              UPLO = 'L' ou 'l'   A est une matrice triangulaire inférieure.
!>

TRANSA

!>          TRANSA est CHARACTER*1
!>           En entrée, TRANSA spécifie la forme de op( A ) à utiliser dans
!>           la multiplication matricielle comme suit :
!>
!>              TRANSA = 'N' ou 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' ou 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' ou 'c'   op( A ) = A**H.
!>

DIAG

!>          DIAG est CHARACTER*1
!>           En entrée, DIAG spécifie si A est unitaire ou non, et triangulaire
!>           comme suit :
!>
!>              DIAG = 'U' ou 'u'   A est supposée être unitaire et triangulaire.
!>
!>              DIAG = 'N' ou 'n'   A n'est pas supposée être unitaire et
!>                                  triangulaire.
!>

M

!>          M est INTEGER
!>           En entrée, M spécifie le nombre de lignes de B. M doit être
!>           au moins zéro.
!>

N

!>          N est INTEGER
!>           En entrée, N spécifie le nombre de colonnes de B. N doit être
!>           au moins zéro.
!>

ALPHA

!>          ALPHA est COMPLEX
!>           En entrée, ALPHA spécifie le scalaire alpha. Quand alpha est
!>           zéro, A n'est pas référencé et B n'a pas besoin d'être défini avant
!>           l'entrée.
!>

A

!>          A est un tableau COMPLEX, de dimension ( LDA, k ),
!>           où k est m quand SIDE = 'L' ou 'l'
!>             et k est n quand SIDE = 'R' ou 'r'.
!>           Avant l'entrée avec UPLO = 'U' ou 'u', la partie triangulaire
!>           supérieure de k par k du tableau A doit contenir la matrice
!>           triangulaire supérieure et la partie strictement triangulaire
!>           inférieure de A n'est pas référencée.
!>           Avant l'entrée avec UPLO = 'L' ou 'l', la partie triangulaire
!>           inférieure de k par k du tableau A doit contenir la matrice
!>           triangulaire inférieure et la partie strictement triangulaire
!>           supérieure de A n'est pas référencée.
!>           Notez que quand DIAG = 'U' ou 'u', les éléments diagonaux de
!>           A ne sont pas non plus référencés, mais sont supposés être
!>           l'unité.
!>

LDA

!>          LDA est INTEGER
!>           En entrée, LDA spécifie la première dimension de A telle que déclarée
!>           dans le programme appelant (sub)routine. Quand SIDE = 'L' ou 'l',
!>           LDA doit être au moins max( 1, m ), quand SIDE = 'R' ou 'r',
!>           LDA doit être au moins max( 1, n ).
!>

B

!>          B est un tableau COMPLEX, de dimension ( LDB, N )
!>           Avant l'entrée, la partie m par n du tableau B doit contenir
!>           la matrice B du membre de droite, et à la sortie, elle est
!>           écrasée par la matrice solution X.
!>

LDB

!>          LDB est INTEGER
!>           En entrée, LDB spécifie la première dimension de B telle que déclarée
!>           dans le programme appelant (sub)routine. LDB doit être
!>           au moins max( 1, m ).
!>
