subroutine cungrq (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CUNGRQ

But:

!>
!> CUNGRQ génère une matrice complexe Q de taille M par N avec des lignes
!> orthonormales, qui est définie comme le produit des K dernières lignes
!> d'un produit de K transformées élémentaires d'ordre N
!>
!>       Q  =  H(1)**H H(2)**H . . . H(k)**H
!>
!> tel que retourné par CGERQF.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes de la matrice Q. M >= 0.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes de la matrice Q. N >= M.
!>

K

!>          K est un entier (INTEGER)
!>          Le nombre de transformées élémentaires dont le produit définit
!>          la matrice Q. M >= K >= 0.
!>

A

!>          A est un tableau de nombres complexes (COMPLEX), dimension (LDA,N)
!>          En entrée, la ligne (m-k+i) doit contenir le vecteur qui
!>          définit la transformée élémentaire H(i), pour i = 1,2,...,k, tel
!>          que retourné par CGERQF dans les k dernières lignes de son argument
!>          tableau A.
!>          En sortie, la matrice Q de taille M par N.
!>

LDA

!>          LDA est un entier (INTEGER)
!>          La première dimension du tableau A. LDA >= max(1,M).
!>

TAU

!>          TAU est un tableau de nombres complexes (COMPLEX), dimension (K)
!>          TAU(i) doit contenir le facteur scalaire de la transformée
!>          élémentaire H(i), tel que retourné par CGERQF.
!>

WORK

!>          WORK est un tableau de nombres complexes (COMPLEX), dimension (MAX(1,LWORK))
!>          En sortie, si INFO = 0, WORK(1) retourne le LWORK optimal.
!>

LWORK

!>          LWORK est un entier (INTEGER)
!>          La dimension du tableau WORK. LWORK >= max(1,M).
!>          Pour une performance optimale, LWORK >= M*NB, où NB est la
!>          taille de bloc optimale.
!>
!>          Si LWORK = -1, alors une requête d'espace de travail est supposée ;
!>          la routine calcule uniquement la taille optimale du tableau WORK,
!>          retourne cette valeur comme première entrée du tableau WORK, et
!>          aucun message d'erreur lié à LWORK n'est émis par XERBLA.
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0:  succès de l'exécution
!>          < 0:  si INFO = -i, le i-ème argument a une valeur illégale
!>
