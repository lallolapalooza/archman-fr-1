subroutine cupgtr (character uplo, integer n, complex, dimension( * ) ap, complex, dimension( * ) tau, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( * ) work, integer info)

CUPGTR

Objectif:

!>
!> CUPGTR génère une matrice unitaire complexe Q qui est définie comme le
!> produit de n-1 réflecteurs élémentaires H(i) d'ordre n, tels que retournés par
!> CHPTRD en utilisant un stockage groupé :
!>
!> si UPLO = 'U', Q = H(n-1) . . . H(2) H(1),
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(n-1).
!>

Paramètres

UPLO 
!>          UPLO est CHARACTER*1
!>          = 'U': stockage de la partie triangulaire supérieure groupée utilisé dans l'appel précédent
!>                 à CHPTRD ;
!>          = 'L': stockage de la partie triangulaire inférieure groupée utilisé dans l'appel précédent
!>                 à CHPTRD.
!>

N

!>          N est INTEGER
!>          Ordre de la matrice Q. N >= 0.
!>

AP

!>          AP est un tableau COMPLEX de dimension (N*(N+1)/2)
!>          Les vecteurs qui définissent les réflecteurs élémentaires, tels que retournés
!>          par CHPTRD.
!>

TAU

!>          TAU est un tableau COMPLEX de dimension (N-1)
!>          TAU(i) doit contenir le facteur scalaire du réflecteur élémentaire
!>          H(i), tel que retourné par CHPTRD.
!>

Q

!>          Q est un tableau COMPLEX de dimension (LDQ,N)
!>          La matrice unitaire complexe N x N.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q. LDQ >= max(1,N).
!>

WORK

!>          WORK est un tableau COMPLEX de dimension (N-1)
!>

INFO

!>          INFO est INTEGER
!>          = 0 : sortie réussie
!>          < 0 : si INFO = -i, le i-ème argument avait une valeur illégale
!>
