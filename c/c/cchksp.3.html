sous-programme cchksp (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nns, entier, dimension( * ) nsval, relatif thresh, logique tsterr, entier nmax, complexe, dimension( * ) a, complexe, dimension( * ) afac, complexe, dimension( * ) ainv, complexe, dimension( * ) b, complexe, dimension( * ) x, complexe, dimension( * ) xact, complexe, dimension( * ) work, relatif, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

CCHKSP

But :

!>
!> CCHKSP teste CSPTRF, -TRI, -TRS, -RFS, et -CON
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE. ; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'ENTIERS, dimension (NN)
!>          Les valeurs de la dimension de la matrice N.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'ENTIERS, dimension (NNS)
!>          Les valeurs du nombre de seconds membres NRHS.
!>

THRESH

!>          THRESH est un RELATIF
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour
!>          afficher chaque rapport de test, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est LOGIQUE
!>          Indique si les sorties d'erreur doivent être testées.
!>

NMAX

!>          NMAX est un ENTIER
!>          La valeur maximale autorisée pour N, utilisée pour dimensionner les
!>          tableaux de travail.
!>

A

!>          A est un tableau COMPLEXE, dimension
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC est un tableau COMPLEXE, dimension
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV est un tableau COMPLEXE, dimension
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B est un tableau COMPLEXE, dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau COMPLEXE, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau COMPLEXE, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau COMPLEXE, dimension
!>                      (NMAX*max(2,NSMAX))
!>

RWORK

!>          RWORK est un tableau RELATIF,
!>                                 dimension (NMAX+2*NSMAX)
!>

IWORK

!>          IWORK est un tableau d'ENTIERS, dimension (NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro de l'unité pour la sortie.
!>
