subroutine cckgsv (integer nm, integer, dimension( * ) mval, integer, dimension( * ) pval, integer, dimension( * ) nval, integer nmats, integer, dimension( 4 ) iseed, real thresh, integer nmax, complex, dimension( * ) a, complex, dimension( * ) af, complex, dimension( * ) b, complex, dimension( * ) bf, complex, dimension( * ) u, complex, dimension( * ) v, complex, dimension( * ) q, real, dimension( * ) alpha, real, dimension( * ) beta, complex, dimension( * ) r, integer, dimension( * ) iwork, complex, dimension( * ) work, real, dimension( * ) rwork, integer nin, integer nout, integer info)

CCKGSV

Objectif :

!>
!> CCKGSV teste CGGSVD :
!>        la GSVD pour la matrice A de taille M par N et la matrice B de taille P par N.
!>

Paramètres

NM 
!>          NM est un entier (INTEGER)
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau d'entiers (INTEGER), dimension (NM)
!>          Les valeurs de la dimension des lignes M de la matrice.
!>

PVAL

!>          PVAL est un tableau d'entiers (INTEGER), dimension (NP)
!>          Les valeurs de la dimension des lignes P de la matrice.
!>

NVAL

!>          NVAL est un tableau d'entiers (INTEGER), dimension (NN)
!>          Les valeurs de la dimension des colonnes N de la matrice.
!>

NMATS

!>          NMATS est un entier (INTEGER)
!>          Le nombre de types de matrices à tester pour chaque combinaison
!>          de dimensions matricielles. Si NMATS >= NTYPES (le nombre maximum
!>          de types de matrices), alors tous les différents types sont
!>          générés pour le test. Si NMATS < NTYPES, une autre ligne d'entrée
!>          est lue pour obtenir les numéros des types de matrices à utiliser.
!>

ISEED

!>          ISEED est un tableau d'entiers (INTEGER), dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires. Les éléments
!>          du tableau doivent être compris entre 0 et 4095, sinon ils seront
!>          réduits modulo 4096, et ISEED(4) doit être impair.
!>          À la sortie, la graine suivante dans la séquence de nombres aléatoires
!>          après la génération de toutes les matrices de test.
!>

THRESH

!>          THRESH est un nombre réel (REAL)
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

NMAX

!>          NMAX est un entier (INTEGER)
!>          La valeur maximale permise pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!>

A

!>          A est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

AF

!>          AF est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

B

!>          B est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

BF

!>          BF est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

U

!>          U est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

V

!>          V est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

Q

!>          Q est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

ALPHA

!>          ALPHA est un tableau de nombres réels (REAL), dimension (NMAX)
!>

BETA

!>          BETA est un tableau de nombres réels (REAL), dimension (NMAX)
!>

R

!>          R est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

IWORK

!>          IWORK est un tableau d'entiers (INTEGER), dimension (NMAX)
!>

WORK

!>          WORK est un tableau de nombres complexes (COMPLEX), dimension (NMAX*NMAX)
!>

RWORK

!>          RWORK est un tableau de nombres réels (REAL), dimension (NMAX)
!>

NIN

!>          NIN est un entier (INTEGER)
!>          Le numéro d'unité pour l'entrée.
!>

NOUT

!>          NOUT est un entier (INTEGER)
!>          Le numéro d'unité pour la sortie.
!>

INFO

!>          INFO est un entier (INTEGER)
!>          = 0 :  fin réussie
!>          > 0 :  Si CLATMS renvoie un code d'erreur, sa valeur absolue est
!>                 renvoyée.
!>
