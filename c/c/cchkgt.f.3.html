sous-routine cchkgt (logique, dimension( * ) dotype, entier nn, entier, dimension( * ) nval, entier nns, entier, dimension( * ) nsval, réel thresh, logique tsterr, complexe, dimension( * ) a, complexe, dimension( * ) af, complexe, dimension( * ) b, complexe, dimension( * ) x, complexe, dimension( * ) xact, complexe, dimension( * ) work, réel, dimension( * ) rwork, entier, dimension( * ) iwork, entier nout)

CCHKGT

!>
!> CCHKGT teste CGTTRF, -TRS, -RFS, et -CON
!>

Paramètres

DOTYPE 
!>          DOTYPE est un tableau LOGIQUE, dimension (NTYPES)
!>          Les types de matrices à utiliser pour les tests. Les matrices de type j
!>          (pour 1 <= j <= NTYPES) sont utilisées pour les tests si DOTYPE(j) =
!>          .TRUE. ; si DOTYPE(j) = .FALSE., alors le type j n'est pas utilisé.
!>

NN

!>          NN est un ENTIER
!>          Le nombre de valeurs de N contenues dans le vecteur NVAL.
!>

NVAL

!>          NVAL est un tableau d'ENTIERS, dimension (NN)
!>          Les valeurs de la dimension N de la matrice.
!>

NNS

!>          NNS est un ENTIER
!>          Le nombre de valeurs de NRHS contenues dans le vecteur NSVAL.
!>

NSVAL

!>          NSVAL est un tableau d'ENTIERS, dimension (NNS)
!>          Les valeurs du nombre de membres de droite NRHS.
!>

THRESH

!>          THRESH est un RÉEL
!>          La valeur seuil pour les rapports de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque rapport de test soit imprimé, utilisez THRESH = 0.
!>

TSTERR

!>          TSTERR est un LOGIQUE
!>          Indicateur qui spécifie si les sorties d'erreur doivent être testées.
!>

A

!>          A est un tableau COMPLEXE, dimension (NMAX*4)
!>

AF

!>          AF est un tableau COMPLEXE, dimension (NMAX*4)
!>

B

!>          B est un tableau COMPLEXE, dimension (NMAX*NSMAX)
!>          où NSMAX est la plus grande entrée dans NSVAL.
!>

X

!>          X est un tableau COMPLEXE, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT est un tableau COMPLEXE, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK est un tableau COMPLEXE, dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK est un tableau RÉEL, dimension
!>                      (max(NMAX)+2*NSMAX)
!>

IWORK

!>          IWORK est un tableau d'ENTIERS, dimension (NMAX)
!>

NOUT

!>          NOUT est un ENTIER
!>          Le numéro d'unité pour la sortie.
!>
