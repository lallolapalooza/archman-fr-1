subroutine cckgsv (integer nm, integer, dimension( * ) mval, integer, dimension( * ) pval, integer, dimension( * ) nval, integer nmats, integer, dimension( 4 ) iseed, real thresh, integer nmax, complex, dimension( * ) a, complex, dimension( * ) af, complex, dimension( * ) b, complex, dimension( * ) bf, complex, dimension( * ) u, complex, dimension( * ) v, complex, dimension( * ) q, real, dimension( * ) alpha, real, dimension( * ) beta, complex, dimension( * ) r, integer, dimension( * ) iwork, complex, dimension( * ) work, real, dimension( * ) rwork, integer nin, integer nout, integer info)

CCKGSV

But :

!>
!> CCKGSV teste CGGSVD :
!>        le GSVD pour les matrices A de taille M-par-N et B de taille P-par-N.
!>

Paramètres

NM 
!>          NM est un entier
!>          Le nombre de valeurs de M contenues dans le vecteur MVAL.
!>

MVAL

!>          MVAL est un tableau d'entiers, dimension (NM)
!>          Les valeurs de la dimension ligne M de la matrice.
!>

PVAL

!>          PVAL est un tableau d'entiers, dimension (NP)
!>          Les valeurs de la dimension ligne P de la matrice.
!>

NVAL

!>          NVAL est un tableau d'entiers, dimension (NN)
!>          Les valeurs de la dimension colonne N de la matrice.
!>

NMATS

!>          NMATS est un entier
!>          Le nombre de types de matrices à tester pour chaque combinaison
!>          de dimensions de matrices. Si NMATS >= NTYPES (le nombre maximum
!>          de types de matrices), alors tous les différents types sont
!>          générés pour les tests. Si NMATS < NTYPES, une autre ligne
!>          d'entrée est lue pour obtenir les numéros des types de matrices à utiliser.
!>

ISEED

!>          ISEED est un tableau d'entiers, dimension (4)
!>          À l'entrée, la graine du générateur de nombres aléatoires. Les éléments
!>          du tableau doivent être compris entre 0 et 4095, sinon ils seront
!>          réduits modulo 4096, et ISEED(4) doit être impair.
!>          À la sortie, la prochaine graine dans la séquence de nombres aléatoires après
!>          que toutes les matrices de test aient été générées.
!>

THRESH

!>          THRESH est un nombre réel
!>          La valeur seuil pour les ratios de test. Un résultat est
!>          inclus dans le fichier de sortie si RESULT >= THRESH. Pour que
!>          chaque ratio de test soit imprimé, utilisez THRESH = 0.
!>

NMAX

!>          NMAX est un entier
!>          La valeur maximale autorisée pour M ou N, utilisée pour dimensionner
!>          les tableaux de travail.
!>

A

!>          A est un tableau complexe, dimension (NMAX*NMAX)
!>

AF

!>          AF est un tableau complexe, dimension (NMAX*NMAX)
!>

B

!>          B est un tableau complexe, dimension (NMAX*NMAX)
!>

BF

!>          BF est un tableau complexe, dimension (NMAX*NMAX)
!>

U

!>          U est un tableau complexe, dimension (NMAX*NMAX)
!>

V

!>          V est un tableau complexe, dimension (NMAX*NMAX)
!>

Q

!>          Q est un tableau complexe, dimension (NMAX*NMAX)
!>

ALPHA

!>          ALPHA est un tableau réel, dimension (NMAX)
!>

BETA

!>          BETA est un tableau réel, dimension (NMAX)
!>

R

!>          R est un tableau complexe, dimension (NMAX*NMAX)
!>

IWORK

!>          IWORK est un tableau d'entiers, dimension (NMAX)
!>

WORK

!>          WORK est un tableau complexe, dimension (NMAX*NMAX)
!>

RWORK

!>          RWORK est un tableau réel, dimension (NMAX)
!>

NIN

!>          NIN est un entier
!>          Le numéro de l'unité pour l'entrée.
!>

NOUT

!>          NOUT est un entier
!>          Le numéro de l'unité pour la sortie.
!>

INFO

!>          INFO est un entier
!>          = 0 :  succès de l'exécution
!>          > 0 :  Si CLATMS renvoie un code d'erreur, sa valeur absolue est retournée.
!>
