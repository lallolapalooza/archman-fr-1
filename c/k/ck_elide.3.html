DESCRIPTIO

N

Ces macros implémentent des enveloppes d'élimination de verrou pour une interface de verrou à argument unique spécifiée par l'utilisateur. Les enveloppes tenteront d'éliminer l'acquisition du verrou, permettant l'exécution concurrente de sections critiques qui n'émettent pas d'opérations mémoire conflictuelles. Si des threads ont réussi à éliminer une acquisition de verrou, des opérations mémoire conflictuelles annuleront les effets de bord de la section critique et forceront chaque thread à réessayer l'acquisition du verrou

régulièrement.

CK_ELIDE_LOCK(), CK_ELIDE_UNLOCK(), CK_ELIDE_LOCK_ADAPTIVE(), et CK_ELIDE_UNLOCK_ADAPTIVE() nécessitent un CK_ELIDE_PROTOTYPE() précédent avec le même NAME. L'élimination est tentée si la fonction LOCK_PREDICATE retourne false. Si LOCK_PREDICATE retourne true, alors l'élimination est annulée et LOCK_FUNCTION est exécutée à la place. Si des threads sont dans une section critique éliminée, LOCK_FUNCTION doit les forcer à annuler via une opération mémoire conflictuelle. La fonction UNLOCK_PREDICATE doit retourner true si le verrou est acquis par l'appelant, ce qui signifie que le verrou n'a pas été éliminé avec succès. Si UNLOCK_PREDICATE retourne true, alors UNLOCK_FUNCTION est exécutée. Si RTM n'est pas pris en charge (pas de macro CK_F_PR_RTM) alors CK_ELIDE_LOCK() et CK_ELIDE_LOCK_ADAPTIVE() appelleront immédiatement LOCK_FUNCTION(). CK_ELIDE_UNLOCK() et CK_ELIDE_UNLOCK_ADAPTIVE() appelleront immédiatement UNLOCK_FUNCT

ION().

CK_ELIDE_TRYLOCK() nécessite un CK_ELIDE_TRYLOCK_PROTOTYPE() précédent avec le même nom. L'élimination est tentée si la fonction LOCK_PREDICATE retourne false. Si LOCK_PREDICATE retourne true ou si l'élimination échoue, alors l'opération est annulée. Si RTM n'est pas pris en charge (pas de macro CK_F_PR_RTM) alors CK_ELIDE_TRYLOCK() appelleront immédiatement TRYLOCK_FUNCT

ION().

CK_ELIDE_LOCK_ADAPTIVE() et CK_ELIDE_UNLOCK_ADAPTIVE() adapteront le comportement d'élimination associé aux opérations de verrou en fonction du comportement à l'exécution du programme. Ce comportement est défini par le pointeur de structure ck_elide_config passé à CK_ELIDE_LOCK_ADAPTIVE(). Une structure ck_elide_stat locale au thread doit être passée à la fois à CK_ELIDE_LOCK_ADAPTIVE() et CK_ELIDE_UNLOCK_ADAPTIVE(). Cette structure est censée être unique pour différentes charges de travail, ne peut pas être réutilisée dans des acquisitions récursives et doit correspondre à la durée de vie du verrou avec lequel elle est associée. Il est sûr de mélanger les appels adaptatifs avec les appels de meilleur

effort.

EXEMPLE

S

Cet exemple utilise les fonctionnalités d'élimination de verrou intégrées dans ck_rwlock et

ck_spinlock. 
#include <ck_rwlock.h>
#include <ck_spinlock.h>

static ck_rwlock_t rw = CK_RWLOCK_INITIALIZER;
static struct ck_elide_config rw_config =
    CK_ELIDE_CONFIG_DEFAULT_INITIALIZER;
static __thread ck_elide_stat_t rw_stat =
    CK_ELIDE_STAT_INITIALIZER;

static ck_spinlock_t spinlock = CK_SPINLOCK_INITIALIZER;
static struct ck_elide_config spinlock_config =
    CK_ELIDE_CONFIG_DEFAULT_INITIALIZER;
static __thread ck_elide_stat_t spinlock_stat =
    CK_ELIDE_STAT_INITIALIZER;

void
function(void)
{

	/* Verrouiller-déverrouiller le verrou côté écriture de manière faible et de meilleur effort. */
	CK_ELIDE_LOCK(ck_rwlock_write, &rw);
	CK_ELIDE_UNLOCK(ck_rwlock_write, &rw);

	/* Tenter d'acquérir le verrou côté écriture. */
	if (CK_ELIDE_TRYLOCK(ck_rwlock_write, &rw) == true)
		CK_ELIDE_UNLOCK(ck_rwlock_write, &rw);

	/* Verrouiller-déverrouiller le verrou côté lecture de manière faible et de meilleur effort. */
	CK_ELIDE_LOCK(ck_rwlock_read, &rw);
	CK_ELIDE_UNLOCK(ck_rwlock_read, &rw);

	/* Tenter d'acquérir le verrou côté lecture. */
	if (CK_ELIDE_TRYLOCK(ck_rwlock_read, &rw) == true)
		CK_ELIDE_UNLOCK(ck_rwlock_read, &rw);

	/* Verrouiller-déverrouiller le verrou côté écriture de manière adaptative. */
	CK_ELIDE_LOCK_ADAPTIVE(ck_rwlock_write, &rw_stat,
	    &rw_config, &rw);
	CK_ELIDE_UNLOCK_ADAPTIVE(ck_rwlock_write, &rw_stat,
	    &rw_config, &rw);

	/* Verrouiller-déverrouiller le verrou côté lecture de manière adaptative. */
	CK_ELIDE_LOCK_ADAPTIVE(ck_rwlock_read, &rw_stat,
	    &rw_config, &rw);
	CK_ELIDE_UNLOCK_ADAPTIVE(ck_rwlock_read, &rw_stat,
	    &rw_config, &rw);

	/* Verrouiller-déverrouiller le verrou tournant de manière faible et de meilleur effort. */
	CK_ELIDE_LOCK(ck_spinlock, &spinlock);
	CK_ELIDE_UNLOCK(ck_spinlock, &spinlock);

	/* Tenter d'acquérir le verrou. */
	if (CK_ELIDE_TRYLOCK(ck_spinlock, &lock) == true)
		CK_ELIDE_UNLOCK(ck_spinlock, &spinlock);

	/* Verrouiller-déverrouiller le verrou tournant de manière adaptative. */
	CK_ELIDE_LOCK_ADAPTIVE(ck_spinlock, &spinlock_stat,
	    &spinlock_config, &spinlock);
	CK_ELIDE_UNLOCK_ADAPTIVE(ck_spinlock, &spinlock_stat,
	    &spinlock_config, &spinlock);
}

Dans cet exemple, des fonctions de verrouillage définies par l'utilisateur sont fournies à une implémentation d'élision.

/* Supposons que lock_t a été défini précédemment. */
#include <ck_elide.h>

/*
 * Cette fonction renvoie vrai si le verrou est indisponible au moment
 * où elle a été appelée, ou faux si le verrou est disponible.
 */
bool is_locked(lock_t *);

/*
 * Cette fonction acquiert le verrou fourni.
 */
void lock(lock_t *);

/*
 * Cette fonction libère le verrou.
 */
void unlock(lock_t *);

CK_ELIDE_PROTOTYPE(my_lock, lock_t, is_locked, lock, is_locked, unlock)

static lock_t lock;

void
function(void)
{

	CK_ELIDE_LOCK(my_lock, &lock);
	CK_ELIDE_UNLOCK(my_lock, &lock);
}
