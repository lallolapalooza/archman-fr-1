DESCRIPTIO

N

ck_rwcohort.h fournit une interface pour définir des verrous lecteur-écrivain qui utilisent des verrous de cohorte en interne pour améliorer les performances sur les architectures NUMA. Voir ck_cohort(3) pour plus d'informations sur les verrous de

cohorte.

Avant d'utiliser un verrou lecteur-écrivain de cohorte, l'utilisateur doit définir un type de cohorte à l'aide soit de la macro CK_COHORT_PROTOTYPE(3) soit de la macro CK_COHORT_TRYLOCK_PROTOTYPE(3) , et définir un type de verrou lecteur-écrivain à l'aide de la macro CK_RWCOHORT_PROTOTYPE(3)

.

EXEMPL

E 
#include <stdlib.h>
#include <pthread.h>

#include <ck_pr.h>
#include <ck_cohort.h>
#include <ck_rwcohort.h>
#include <ck_spinlock.h>

/* Créer des méthodes de cohorte avec des signatures correspondant à la signature requise */

static void
ck_spinlock_lock_with_context(ck_spinlock_t *lock, void *context)
{
	(void)context;
	ck_spinlock_lock(lock);
	return;
}

static void
ck_spinlock_unlock_with_context(ck_spinlock_t *lock, void *context)
{
	(void)context;
	ck_spinlock_unlock(lock);
	return;
}

static bool
ck_spinlock_locked_with_context(ck_spinlock_t *lock, void *context)
{
	(void)context;
	return ck_spinlock_locked(lock);
}

/*
 * définir un type de cohorte nommé "test_cohort" qui utilisera
 * les méthodes ci-dessus pour ses verrous global et local
 */
CK_COHORT_PROTOTYPE(test_cohort,
	ck_spinlock_lock_with_context, ck_spinlock_unlock_with_context, ck_spinlock_locked_with_context,
	ck_spinlock_lock_with_context, ck_spinlock_unlock_with_context, ck_spinlock_locked_with_context)

/* définir un type lecteur-écrivain utilisant le même type de cohorte */
CK_RWCOHORT_WP_PROTOTYPE(test_cohort)

static ck_spinlock_t global_lock = CK_SPINLOCK_INITIALIZER;
static CK_COHORT_INSTANCE(test_cohort) *cohorts;
static CK_RWCOHORT_WP_INSTANCE(test_cohort) rw_cohort = CK_RWCOHORT_WP_INITIALIZER;
static unsigned int ready;

static void *
function(void *context)
{
	CK_COHORT_INSTANCE(test_cohort) *cohort = context;

	while (ck_pr_load_uint(&ready) == 0);

	while (ck_pr_load_uint(&ready) > 0) {
		/*
		 * acquérir le verrou de cohorte avant d'effectuer la section critique.
		 * noter que nous passons NULL pour les arguments de contexte global et local
		 * car ni les fonctions de verrouillage ni de déverrouillage
		 * ne les utiliseront.
		 */
		CK_COHORT_LOCK(test_cohort, cohort, NULL, NULL);

		/* effectuer la section critique */

		/* libérer le verrou de cohorte */
		CK_COHORT_UNLOCK(test_cohort, cohort, NULL, NULL);
	}

	return NULL;
}

int
main(void)
{
	unsigned int nthr = 4;
	unsigned int n_cohorts = 2;
	unsigned int i;

	/* allouer 2 cohortes du type défini */
	CK_COHORT_INSTANCE(test_cohort) *cohorts =
	    calloc(n_cohorts, sizeof(CK_COHORT_INSTANCE(test_cohort)));

	/* créer des verrous locaux à utiliser avec chaque cohorte */
	ck_spinlock_t *local_locks =
		calloc(n_cohorts, sizeof(ck_spinlock_t));

	pthread_t *threads =
		calloc(nthr, sizeof(pthread_t));

	/* initialiser chaque cohorte avant de les utiliser */
	for (i = 0 ; i < n_cohorts ; ++i) {
		CK_COHORT_INIT(test_cohort, cohorts + i, &global_lock, local_locks + i,
			CK_COHORT_DEFAULT_LOCAL_PASS_LIMIT);
	}

	/* démarrer chaque thread et attribuer les cohortes également */
	for (i = 0 ; i < nthr ; ++i) {
		pthread_create(threads + i, NULL, function, cohorts + (i % n_cohorts));
	}

	ck_pr_store_uint(&ready, 1);
	sleep(10);
	ck_pr_store_uint(&ready, 0);

	for (i = 0 ; i < nthr ; ++i) {
		pthread_join(threads[i], NULL);
	}

	return
0;
}
