DESCRIPTION

Les verrous à gros lecteurs sont des verrous de type lecteur-rédacteur distribués avec une acquisition de lecteur en temps constant de faible latence (par rapport au nombre de lecteurs concurrents). D'autre part, les acquisitions par les rédacteurs sont une opération relativement coûteuse en O(n). Il s'agit d'un verrou biaisé en écriture.

EXEMPLE

static ck_brlock_t lock = CK_BRLOCK_INITIALIZER;
static __thread ck_brlock_reader_t reader;

static void
reader(void)
{

	/* Ajoutez notre fil d'exécution comme participant au verrou. */
	ck_brlock_read_register(&lock, &reader);

	for (;;) {
		ck_brlock_read_lock(&lock, &reader);
		/* Section critique côté lecture. */
		ck_brlock_read_unlock(&reader);

		if (ck_brlock_read_trylock(&lock, &reader, 1) == true) {
			/* Section critique côté lecture. */
			ck_brlock_read_unlock(&reader);
		}
	}

	return;
}

static void
writer(void)
{

	for (;;) {
		ck_brlock_write_lock(&lock);
		/* Section critique côté écriture. */
		ck_brlock_write_unlock(&lock);

		if (ck_brlock_write_trylock(&lock, 1) == true) {
			/* Section critique côté écriture. */
			ck_brlock_write_unlock(&lock);
		}
	}

	return;
}
