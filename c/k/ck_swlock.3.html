DESCRIPTION

Il s'agit d'un verrou de lecture-écriture centralisé, biaisé en écriture, à écrivain unique. Il nécessite deux fois moins d'espace que ck_rwlock et offre une latence faible sur le chemin rapide. Le verrou prend en charge les opérations de verrouillage (latch) et de déverrouillage (unlatch) qui lui permettent d'être utilisé d'une manière sûre en copie (les bits de lecteur peuvent être écrasés en toute sécurité).

EXEMPLE

#include <ck_swlock.h>

static ck_swlock_t lock = CK_SWLOCK_INITIALIZER;

static void
reader(void)
{

	for (;;) {
		ck_swlock_read_lock(&lock);
		/* Section critique côté lecture. */
		ck_swlock_read_unlock(&lock);

		if (ck_swlock_read_trylock(&lock) == true) {
			/* Section critique côté lecture. */
			ck_swlock_read_unlock(&lock);
		}
	}

	return;
}

static void
writer(void)
{
	ck_swlock_t contrived;

	for (;;) {
		ck_swlock_write_lock(&lock);
		/* Section critique côté écriture. */
		ck_swlock_write_unlock(&lock);

		if (ck_swlock_write_trylock(&lock) == true) {
			/* Section critique côté écriture. */
			ck_swlock_write_unlock(&lock);
		}

		ck_swlock_write_latch(&lock);
		/* Section critique côté écriture. */

		/* Ceci est sûr à faire dans un verrouillage (latch). */
		contrived = lock;
		lock = contrived;
		ck_swlock_write_unlatch(&lock);
	}

	return;
}
