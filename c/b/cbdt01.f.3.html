Documentation de fonction/sous-programme

subroutine cbdt01 (integer m, integer n, integer kd, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldq, * ) q, integer ldq, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldpt, * ) pt, integer ldpt, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CBDT01

Objectif:

!>
!> CBDT01 reconstruit une matrice générale A à partir de sa forme
!>    bidiagonale A = Q * B * P**H
!> où Q (m par min(m,n)) et P**H (min(m,n) par n) sont des matrices unitaires
!> et B est bidiagonale.
!>
!> Le ratio de test pour tester la réduction est
!>    RESID = norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!> où EPS est la précision de la machine.
!>

Paramètres

M 
!>          M est INTEGER
!>          Le nombre de lignes des matrices A et Q.
!>

N

!>          N est INTEGER
!>          Le nombre de colonnes des matrices A et P**H.
!>

KD

!>          KD est INTEGER
!>          Si KD = 0, B est diagonale et le tableau E n'est pas référencé.
!>          Si KD = 1, la réduction a été effectuée par xGEBRD; B est
!>          bidiagonale supérieure si m >= n, et bidiagonale inférieure si m < n.
!>          Si KD = -1, la réduction a été effectuée par xGBBRD; B est
!>          toujours bidiagonale supérieure.
!>

A

!>          A est un tableau COMPLEXE, dimension (LDA,N)
!>          La matrice A de taille m par n.
!>

LDA

!>          LDA est INTEGER
!>          La dimension principale du tableau A.  LDA >= max(1,M).
!>

Q

!>          Q est un tableau COMPLEXE, dimension (LDQ,N)
!>          La matrice unitaire Q de taille m par min(m,n) dans la réduction
!>          A = Q * B * P**H.
!>

LDQ

!>          LDQ est INTEGER
!>          La dimension principale du tableau Q.  LDQ >= max(1,M).
!>

D

!>          D est un tableau REAL, dimension (min(M,N))
!>          Les éléments diagonaux de la matrice bidiagonale B.
!>

E

!>          E est un tableau REAL, dimension (min(M,N)-1)
!>          Les éléments supradiagonaux de la matrice bidiagonale B si
!>          m >= n, ou les éléments sous-diagonaux de B si m < n.
!>

PT

!>          PT est un tableau COMPLEXE, dimension (LDPT,N)
!>          La matrice unitaire P**H de taille min(m,n) par n dans la réduction
!>          A = Q * B * P**H.
!>

LDPT

!>          LDPT est INTEGER
!>          La dimension principale du tableau PT.
!>          LDPT >= max(1,min(M,N)).
!>

WORK

!>          WORK est un tableau COMPLEXE, dimension (M+N)
!>

RWORK

!>          RWORK est un tableau REAL, dimension (M)
!>

RESID

!>          RESID est REAL
!>          Le ratio de test :
!>          norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!>
