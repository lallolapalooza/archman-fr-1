subroutine cbdt02 (integer m, integer n, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CBDT02

But:

!>
!> CBDT02 teste le changement de base C = U**H * B en calculant le
!> résidu
!>
!>    RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
!> où B et C sont des matrices M par N, U est une matrice orthogonale M par M,
!> et EPS est la précision machine.
!>

Paramètres

M 
!>          M est un entier (INTEGER)
!>          Le nombre de lignes des matrices B et C et l'ordre de
!>          la matrice Q.
!>

N

!>          N est un entier (INTEGER)
!>          Le nombre de colonnes des matrices B et C.
!>

B

!>          B est un tableau de nombres complexes (COMPLEX), dimension (LDB,N)
!>          La matrice B de dimensions m x n.
!>

LDB

!>          LDB est un entier (INTEGER)
!>          La dimension principale du tableau B.  LDB >= max(1,M).
!>

C

!>          C est un tableau de nombres complexes (COMPLEX), dimension (LDC,N)
!>          La matrice C de dimensions m x n, supposée contenir U**H * B.
!>

LDC

!>          LDC est un entier (INTEGER)
!>          La dimension principale du tableau C.  LDC >= max(1,M).
!>

U

!>          U est un tableau de nombres complexes (COMPLEX), dimension (LDU,M)
!>          La matrice orthogonale U de dimensions m x m.
!>

LDU

!>          LDU est un entier (INTEGER)
!>          La dimension principale du tableau U.  LDU >= max(1,M).
!>

WORK

!>          WORK est un tableau de nombres complexes (COMPLEX), dimension (M)
!>

RWORK

!>          RWORK est un tableau de nombres réels (REAL), dimension (M)
!>

RESID

!>          RESID est un nombre réel (REAL)
!>          RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
