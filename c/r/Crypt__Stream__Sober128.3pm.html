NOM

Crypt::Stream::Sober128 - Chiffre de flux Sober128

SYNOPSIS

use Crypt::Stream::Sober128;
# chiffrement
$cle = "1234567890123456";
$iv  = "123456789012";
$flux = Crypt::Stream::Sober128->new($cle, $iv);
$ct = $flux->crypt("message clair");
# déchiffrement
$cle = "1234567890123456";
$iv  = "123456789012";
$flux = Crypt::Stream::Sober128->new($cle, $iv);
$pt = $flux->crypt($ct);

DESCRIPTION

Fournit une interface au chiffre de flux Sober128.

MÉTHODES

new

$flux = Crypt::Stream::Sober128->new($cle, $iv);
# $cle .. la longueur de la clé doit être un multiple de 4 octets
# $iv  .. la longueur de l'IV doit être un multiple de 4 octets

crypt

$chiffré = $flux->crypt($clair);
#ou
$clair = $flux->crypt($chiffré);

keystream

$cle_aleatoire = $flux->keystream($longueur);

clone

$flux->clone();
