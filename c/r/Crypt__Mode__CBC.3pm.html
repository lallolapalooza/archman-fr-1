NOM

Crypt::Mode::CBC - Mode de chiffrement par blocs CBC [Chaînage de blocs de chiffrement]

SYNOPSIS

use Crypt::Mode::CBC;
my $m = Crypt::Mode::CBC->new('AES');
#(déchiffrer|chiffrer) en une seule fois
my $ciphertext = $m->encrypt($plaintext, $key, $iv);
my $plaintext = $m->decrypt($ciphertext, $key, $iv);
#chiffrer plusieurs blocs
$m->start_encrypt($key, $iv);
my $ciphertext = $m->add('certaines données');
$ciphertext .= $m->add('plus de données');
$ciphertext .= $m->finish;
#déchiffrer plusieurs blocs
$m->start_decrypt($key, $iv);
my $plaintext = $m->add($some_ciphertext);
$plaintext .= $m->add($more_ciphertext);
$plaintext .= $m->finish;

DESCRIPTION

Ce module implémente le mode de chiffrement CBC. NOTE: il ne fonctionne qu'avec les chiffrements de CryptX (Crypt::Cipher::NNNN).

MÉTHODES

new

my $m = Crypt::Mode::CBC->new($name);
#ou
my $m = Crypt::Mode::CBC->new($name, $padding);
#ou
my $m = Crypt::Mode::CBC->new($name, $padding, $cipher_rounds);
# $name ....... un parmi 'AES', 'Anubis', 'Blowfish', 'CAST5', 'Camellia', 'DES', 'DES_EDE',
#               'KASUMI', 'Khazad', 'MULTI2', 'Noekeon', 'RC2', 'RC5', 'RC6',
#               'SAFERP', 'SAFER_K128', 'SAFER_K64', 'SAFER_SK128', 'SAFER_SK64',
#               'SEED', 'Skipjack', 'Twofish', 'XTEA', 'IDEA', 'Serpent'
#               simplement n'importe quel <NOM> pour lequel il existe Crypt::Cipher::<NOM>
# $padding .... 0 pas de remplissage (la taille du texte en clair doit être un multiple de la longueur du bloc)
#               1 remplissage PKCS5, le "standard" de Crypt::CBC - PAR DÉFAUT
#               2 "oneandzeroes" de Crypt::CBC
#               3 remplissage ANSI X.923
#               4 remplissage par zéros
#               5 remplissage par zéros (+ un bloc de zéros si la longueur de sortie est divisible par la taille du bloc)
# $cipher_rounds ... nombre facultatif de tours pour le chiffrement donné

encrypt

my $ciphertext = $m->encrypt($plaintext, $key, $iv);

decrypt

my $plaintext = $m->decrypt($ciphertext, $key, $iv);

start_encrypt

$m->start_encrypt($key, $iv);

start_decrypt

$m->start_decrypt($key, $iv);

add

# en mode chiffrement
my $plaintext = $m->add($ciphertext);
# en mode déchiffrement
my $ciphertext = $m->add($plaintext);

finish

#chiffrer plusieurs blocs
$m->start_encrypt($key, $iv);
my $ciphertext = '';
$ciphertext .= $m->add('certaines données');
$ciphertext .= $m->add('plus de données');
$ciphertext .= $m->finish;
#déchiffrer plusieurs blocs
$m->start_decrypt($key, $iv);
my $plaintext = '';
$plaintext .= $m->add($some_ciphertext);
$plaintext .= $m->add($more_ciphertext);
$plaintext .= $m->finish;
