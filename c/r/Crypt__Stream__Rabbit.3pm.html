NOM

Crypt::Stream::Rabbit - Chiffrement de flux Rabbit

SYNTAXE

use Crypt::Stream::Rabbit;
# chiffrement
$clé = "1234567890123456";
$iv  = "12345678";
$stream = Crypt::Stream::Rabbit->new($clé, $iv);
$ct = $stream->crypt("message en clair");
# déchiffrement
$clé = "1234567890123456";
$iv  = "12345678";
$stream = Crypt::Stream::Rabbit->new($clé, $iv);
$pt = $stream->crypt($ct);

DESCRIPTION

Fournit une interface pour le chiffrement de flux Rabbit.

MÉTHODES

new

$stream = Crypt::Stream::Rabbit->new($clé, $iv);
# $clé .. la longueur de la clé doit être jusqu'à 16 octets
# $iv  .. la longueur de l'IV doit être jusqu'à 8 octets
$stream = Crypt::Stream::Rabbit->new($clé);
# ATTENTION: ceci est différent de new($clé, "")

crypt

$chiffré = $stream->crypt($clair);
# ou
$clair = $stream->crypt($chiffré);

keystream

$clé_aléatoire = $stream->keystream($longueur);

clone

$stream->clone();

  • Crypt::Stream::RC4, Crypt::Stream::ChaCha, Crypt::Stream::Salsa20, Crypt::Stream::Sober128
