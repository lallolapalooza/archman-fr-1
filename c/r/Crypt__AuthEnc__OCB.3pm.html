Crypt::AuthEnc::OCB(3) Documentation Perl fournie par l'utilisateur Crypt::AuthEnc::OCB(3)

NOM

Crypt::AuthEnc::OCB - Chiffrement authentifié en mode OCBv3

SYNTAXE

### Interface OO
use Crypt::AuthEnc::OCB;
# chiffrer et authentifier
my $ae = Crypt::AuthEnc::OCB->new("AES", $key, $nonce, $tag_len);
$ae->adata_add('donnees_authentifiees_supplementaires1');
$ae->adata_add('donnees_authentifiees_supplementaires2');
my $ct = $ae->encrypt_add('donnees1');
$ct .= $ae->encrypt_add('donnees2');
$ct .= $ae->encrypt_add('donnees3');
$ct .= $ae->encrypt_last('reste des donnees');
my $tag = $ae->encrypt_done();
# déchiffrer et vérifier
my $ae = Crypt::AuthEnc::OCB->new("AES", $key, $nonce, $tag_len);
$ae->adata_add('donnees_authentifiees_supplementaires1');
$ae->adata_add('donnees_authentifiees_supplementaires2');
my $pt = $ae->decrypt_add('chiffre1');
$pt .= $ae->decrypt_add('chiffre2');
$pt .= $ae->decrypt_add('chiffre3');
$pt .= $ae->decrypt_last('reste des donnees');
my $tag = $ae->decrypt_done();
die "echec du dechiffrement" unless $tag eq $tag_attendu;
#ou
my $result = $ae->decrypt_done($tag_attendu); # 0 ou 1
### Interface fonctionnelle
use Crypt::AuthEnc::OCB qw(ocb_encrypt_authenticate ocb_decrypt_verify);
my ($ciphertext, $tag) = ocb_encrypt_authenticate('AES', $key, $nonce, $adata, $tag_len, $plaintext);
my $plaintext = ocb_decrypt_verify('AES', $key, $nonce, $adata, $ciphertext, $tag);

DESCRIPTION

Ce module implémente OCB v3 conformément à la norme https://tools.ietf.org/html/rfc7253

EXPORT

Rien n'est exporté par défaut.

Vous pouvez exporter des fonctions sélectionnées :

use Crypt::AuthEnc::OCB qw(ocb_encrypt_authenticate ocb_decrypt_verify);

FONCTIONS

ocb_encrypt_authenticate

my ($ciphertext, $tag) = ocb_encrypt_authenticate($cipher, $key, $nonce, $adata, $tag_len, $plaintext);
# $cipher .. 'AES' ou le nom de tout autre chiffrement avec une longueur de bloc de 16 octets
# $key ..... clé AES de longueur appropriée (128/192/256 bits)
# $nonce ... nonce/sel unique (pas besoin de le garder secret)
# $adata ... données authentifiées supplémentaires
# $tag_len . longueur requise de la balise de sortie

ocb_decrypt_verify

my $plaintext = ocb_decrypt_verify($cipher, $key, $nonce, $adata, $ciphertext, $tag);
# en cas d'erreur retourne undef

MÉTHODES

new

my $ae = Crypt::AuthEnc::OCB->new($cipher, $key, $nonce, $tag_len);
# $cipher .. 'AES' ou le nom de tout autre chiffrement avec une longueur de bloc de 16 octets
# $key ..... clé AES de longueur appropriée (128/192/256 bits)
# $nonce ... nonce/sel unique (pas besoin de le garder secret)
# $tag_len . longueur requise de la balise de sortie

adata_add

$ae->adata_add($adata);                        # peut être appelé plusieurs fois

encrypt_add

$ciphertext = $ae->encrypt_add($data);         # peut être appelé plusieurs fois
# ATTENTION : la taille de $data doit être un multiple de la longueur du bloc (16 pour AES)

encrypt_last

$ciphertext = $ae->encrypt_last($data);

encrypt_done

$tag = $ae->encrypt_done();                    # retourne la valeur de $tag

decrypt_add

$plaintext = $ae->decrypt_add($ciphertext);    # peut être appelé plusieurs fois
# ATTENTION : la taille de $ciphertext doit être un multiple de la longueur du bloc (16 pour AES)

decrypt_last

$plaintext = $ae->decrypt_last($data);

decrypt_done

my $tag = $ae->decrypt_done;           # retourne la valeur de $tag
#ou
my $result = $ae->decrypt_done($tag);  # retourne 1 (succès) ou 0 (échec)

clone

my $ae_new = $ae->clone;
