SYNOPSI

S 
### Interface fonctionnelle :
use Crypt::Digest::SHA512 qw( sha512 sha512_hex sha512_b64 sha512_b64u
                             sha512_file sha512_file_hex sha512_file_b64 sha512_file_b64u );
# calculer l'empreinte à partir d'une chaîne/d'un tampon
$sha512_raw  = sha512('data string');
$sha512_hex  = sha512_hex('data string');
$sha512_b64  = sha512_b64('data string');
$sha512_b64u = sha512_b64u('data string');
# calculer l'empreinte à partir d'un fichier $sha512_raw = sha512_file('filename.dat'); $sha512_hex = sha512_file_hex('filename.dat'); $sha512_b64 = sha512_file_b64('filename.dat'); $sha512_b64u = sha512_file_b64u('filename.dat'); # calculer l'empreinte à partir d'un descripteur de fichier $sha512_raw = sha512_file(*FILEHANDLE); $sha512_hex = sha512_file_hex(*FILEHANDLE); $sha512_b64 = sha512_file_b64(*FIL
EHANDLE); $sha512_b64u = sha512_file_b64u(*FILEHANDLE); ### Interface OO : use Crypt::Digest::SHA512; $d = Crypt::Digest::SHA512->new; $d->add('any data'); $d->addfile('filename.dat'); $d->addfile(*FILEHANDLE); $result_raw = $d->digest; # octets bruts $result_hex = $d->hexdigest; # forme hexadécimale $result_b64 = $d->b64digest; # forme Base64 $result_b64u = $d->b64udigest; # forme Base64 compatible URL
