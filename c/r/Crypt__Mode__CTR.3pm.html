NOM

Crypt::Mode::CTR - Mode de chiffrement par bloc CTR [mode compteur]

SYNOPIS

use Crypt::Mode::CTR;
my $m = Crypt::Mode::CTR->new('AES');
#(en|de)chiffrement en une seule fois
my $ciphertext = $m->encrypt($plaintext, $key, $iv);
my $plaintext = $m->decrypt($ciphertext, $key, $iv);
#chiffrement de plusieurs blocs
$m->start_encrypt($key, $iv);
my $ciphertext = $m->add('quelques données');
$ciphertext .= $m->add('plus de données');
#déchiffrement de plusieurs blocs
$m->start_decrypt($key, $iv);
my $plaintext = $m->add($some_ciphertext);
$plaintext .= $m->add($more_ciphertext);

DESCRIPTION

Ce module implémente le mode de chiffrement CTR. NOTE : il ne fonctionne qu'avec les chiffrements de CryptX (Crypt::Cipher::NNNN).

MÉTHODES

new

my $m = Crypt::Mode::CTR->new($cipher_name);
#ou
my $m = Crypt::Mode::CTR->new($cipher_name, $ctr_mode, $ctr_width);
#ou
my $m = Crypt::Mode::CTR->new($cipher_name, $ctr_mode, $ctr_width, $cipher_rounds);
# $cipher_name .. l'un des 'AES', 'Anubis', 'Blowfish', 'CAST5', 'Camellia', 'DES', 'DES_EDE',
#                 'KASUMI', 'Khazad', 'MULTI2', 'Noekeon', 'RC2', 'RC5', 'RC6',
#                 'SAFERP', 'SAFER_K128', 'SAFER_K64', 'SAFER_SK128', 'SAFER_SK64',
#                 'SEED', 'Skipjack', 'Twofish', 'XTEA', 'IDEA', 'Serpent'
#                 simplement tout <NOM> pour lequel il existe Crypt::Cipher::<NOM>
# $ctr_mode ..... compteur little-endian (DÉFAUT)
#                 1 compteur big-endian
#                 2 little-endian + incrémentation RFC3686
#                 3 big-endian + incrémentation RFC3686
# $ctr_width .... largeur du compteur en octets (DÉFAUT = largeur de bloc complète)
# $cipher_rounds ... nombre optionnel de rondes pour le chiffrement donné

encrypt

my $ciphertext = $m->encrypt($plaintext, $key, $iv);

decrypt

my $plaintext = $m->decrypt($ciphertext, $key, $iv);

start_encrypt

$m->start_encrypt($key, $iv);

start_decrypt

$m->start_decrypt($key, $iv);

add

# en mode chiffrement
my $plaintext = $m->add($ciphertext);
# en mode déchiffrement
my $ciphertext = $m->add($plaintext);
