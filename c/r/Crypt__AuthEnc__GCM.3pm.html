NOM

Crypt::AuthEnc::GCM - Chiffrement authentifié en mode GCM

SYNOPIS

### Interface orientée objet
use Crypt::AuthEnc::GCM;
# chiffrer et authentifier
my $ae = Crypt::AuthEnc::GCM->new("AES", $key, $iv);
$ae->adata_add('donnees_authentifiees_supplementaires1');
$ae->adata_add('donnees_authentifiees_supplementaires2');
my $ct = $ae->encrypt_add('donnees1');
$ct .= $ae->encrypt_add('donnees2');
$ct .= $ae->encrypt_add('donnees3');
my $tag = $ae->encrypt_done();
# déchiffrer et vérifier
my $ae = Crypt::AuthEnc::GCM->new("AES", $key, $iv);
$ae->adata_add('donnees_authentifiees_supplementaires1');
$ae->adata_add('donnees_authentifiees_supplementaires2');
my $pt = $ae->decrypt_add('chiffrement1');
$pt .= $ae->decrypt_add('chiffrement2');
$pt .= $ae->decrypt_add('chiffrement3');
my $tag = $ae->decrypt_done();
die "déchiffrement échoué" unless $tag eq $tag_attendu;
#ou
my $result = $ae->decrypt_done($tag_attendu); # 0 ou 1
### Interface fonctionnelle
use Crypt::AuthEnc::GCM qw(gcm_encrypt_authenticate gcm_decrypt_verify);
my ($chiffrement, $tag) = gcm_encrypt_authenticate('AES', $key, $iv, $adata, $texte_clair);
my $texte_clair = gcm_decrypt_verify('AES', $key, $iv, $adata, $chiffrement, $tag);

DESCRIPTION

Mode Galois/Compteur (GCM) - fournit chiffrement et authentification.

EXPORT

Rien n'est exporté par défaut.

Vous pouvez exporter des fonctions sélectionnées :

use Crypt::AuthEnc::GCM qw(gcm_encrypt_authenticate gcm_decrypt_verify);

FONCTIONS

gcm_encrypt_authenticate

my ($chiffrement, $tag) = gcm_encrypt_authenticate($cipher, $key, $iv, $adata, $texte_clair);
# $cipher .. 'AES' ou nom de tout autre chiffrement avec une longueur de bloc de 16 octets
# $key ..... clé AES de longueur appropriée (128/192/256 bits)
# $iv ...... vecteur d'initialisation
# $adata ... données authentifiées supplémentaires

gcm_decrypt_verify

my $texte_clair = gcm_decrypt_verify($cipher, $key, $iv, $adata, $chiffrement, $tag);
# en cas d'erreur renvoie undef

MÉTHODES

new

my $ae = Crypt::AuthEnc::GCM->new($cipher, $key);
#ou
my $ae = Crypt::AuthEnc::GCM->new($cipher, $key, $iv);
# $cipher .. 'AES' ou nom de tout autre chiffrement
# $key ..... clé de chiffrement de longueur appropriée
# $iv ...... vecteur d'initialisation (facultatif, vous pouvez le définir plus tard via la méthode iv_add)

iv_add

Définit le vecteur d'initialisation (IV).

$ae->iv_add($iv_data);                        #peut être appelé plusieurs fois

adata_add

Ajoute des données authentifiées supplémentaires. Peut être appelé après tous les appels à "iv_add" mais avant le premier appel à "encrypt_add" ou "decrypt_add".

$ae->adata_add($aad_data);                    # peut être appelé plusieurs fois

encrypt_add

$ciphertext = $ae->encrypt_add($data);        # peut être appelé plusieurs fois

encrypt_done

$tag = $ae->encrypt_done();                   # renvoie la valeur de $tag

decrypt_add

$plaintext = $ae->decrypt_add($ciphertext);   # peut être appelé plusieurs fois

decrypt_done

my $tag = $ae->decrypt_done;           # renvoie la valeur de $tag
#ou
my $result = $ae->decrypt_done($tag);  # renvoie 1 (succès) ou 0 (échec)

reset

$ae->reset;

clone

my $ae_new = $ae->clone;
