NOM

Crypt::Stream::Salsa20 - Chiffrement par flot Salsa20

SYNOPSIS

use Crypt::Stream::Salsa20;
# chiffrement
$cle = "1234567890123456";
$iv  = "12345678";
$flux = Crypt::Stream::Salsa20->new($cle, $iv);
$ct = $flux->crypt("message clair");
# déchiffrement
$cle = "1234567890123456";
$iv  = "12345678";
$flux = Crypt::Stream::Salsa20->new($cle, $iv);
$pt = $flux->crypt($ct);

DESCRIPTION

Fournit une interface au chiffrement par flot Salsa20.

MÉTHODES

new

$flux = Crypt::Stream::Salsa20->new($cle, $iv);
#ou
$flux = Crypt::Stream::Salsa20->new($cle, $iv, $compteur, $tours);
# $cle     .. 32 ou 16 octets
# $iv      .. 8 octets
# $compteur .. valeur initiale du compteur (DÉFAUT: 0)
# $tours  .. nombre de tours (DÉFAUT: 20)

crypt

$chiffré = $flux->crypt($clair);
#ou
$clair = $flux->crypt($chiffré);

keystream

$cle_aléatoire = $flux->keystream($longueur);

clone

$flux->clone();
