NOM

Crypt::AuthEnc::ChaCha20Poly1305 - Chiffrement authentifié en mode ChaCha20-Poly1305

SYNOPIS

### Interface OO
use Crypt::AuthEnc::ChaCha20Poly1305;
# chiffrer et authentifier
my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key, $iv);
$ae->adata_add('donnees_authentifiees_supplementaires1');
$ae->adata_add('donnees_authentifiees_supplementaires2');
my $ct = $ae->encrypt_add('donnees1');
$ct .= $ae->encrypt_add('donnees2');
$ct .= $ae->encrypt_add('donnees3');
my $tag = $ae->encrypt_done();
# déchiffrer et vérifier
my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key, $iv);
$ae->adata_add('donnees_authentifiees_supplementaires1');
$ae->adata_add('donnees_authentifiees_supplementaires2');
my $pt = $ae->decrypt_add('chiffrement1');
$pt .= $ae->decrypt_add('chiffrement2');
$pt .= $ae->decrypt_add('chiffrement3');
my $tag = $ae->decrypt_done();
die "échec du déchiffrement" unless $tag eq $expected_tag;
#ou
my $result = $ae->decrypt_done($expected_tag); # 0 ou 1
### Interface fonctionnelle
use Crypt::AuthEnc::ChaCha20Poly1305 qw(chacha20poly1305_encrypt_authenticate chacha20poly1305_decrypt_verify);
my ($ciphertext, $tag) = chacha20poly1305_encrypt_authenticate($key, $iv, $adata, $plaintext);
my $plaintext = chacha20poly1305_decrypt_verify($key, $iv, $adata, $ciphertext, $tag);

DESCRIPTION

Fournit le chiffrement et l'authentification basés sur ChaCha20 + Poly1305 tels que définis dans la RFC 7539 - https://tools.ietf.org/html/rfc7539

EXPORT

Rien n'est exporté par défaut.

Vous pouvez exporter des fonctions sélectionnées :

use Crypt::AuthEnc::ChaCha20Poly1305 qw(chacha20poly1305_encrypt_authenticate chacha20poly1305_decrypt_verify);

FONCTIONS

chacha20poly1305_encrypt_authenticate

my ($ciphertext, $tag) = chacha20poly1305_encrypt_authenticate($key, $iv, $adata, $plaintext);
# $key ..... clé de longueur appropriée (128 ou 256 bits / 16 ou 32 octets)
# $iv ...... vecteur d'initialisation (64 ou 96 bits / 8 ou 12 octets)
# $adata ... données authentifiées supplémentaires (optionnel)

chacha20poly1305_decrypt_verify

my $plaintext = chacha20poly1305_decrypt_verify($key, $iv, $adata, $ciphertext, $tag);
# en cas d'erreur retourne undef

MÉTHODES

new

my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key, $iv);
# $key ..... clé de chiffrement de longueur appropriée (128 ou 256 bits / 16 ou 32 octets)
# $iv ...... vecteur d'initialisation (64 ou 96 bits / 8 ou 12 octets)

adata_add

Ajoute des données authentifiées supplémentaires. Peut être appelée avant la première appel à "encrypt_add" ou "decrypt_add";

$ae->adata_add($aad_data);                     # peut être appelée plusieurs fois

encrypt_add

$ciphertext = $ae->encrypt_add($data);         # peut être appelée plusieurs fois

encrypt_done

$tag = $ae->encrypt_done();                    # retourne la valeur $tag

decrypt_add

$plaintext = $ae->decrypt_add($ciphertext);    # peut être appelée plusieurs fois

decrypt_done

my $tag = $ae->decrypt_done;           # retourne la valeur $tag
#ou
my $result = $ae->decrypt_done($tag);  # retourne 1 (succès) ou 0 (échec)

set_iv

my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key)->set_iv($iv);
# $iv ...... vecteur d'initialisation (64 ou 96 bits / 8 ou 12 octets)

set_iv_rfc7905

Voir https://tools.ietf.org/html/rfc7905

my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key)->set_iv_rfc7905($iv, $seqnum);
# $iv ...... vecteur d'initialisation (96 bits / 12 octets)
# $seqnum .. entier 64 bits (numéro de séquence)

clone

my $ae_new = $ae->clone;
