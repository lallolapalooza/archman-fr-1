Le squelette BPF est une interface alternative aux API libbpf existantes pour travailler avec des objets BPF. Le code du squelette est destiné à raccourcir et simplifier considérablement le code pour charger et travailler avec des programmes BPF côté espace utilisateur. Le code généré est adapté à l'objet BPF d'entrée spécifique FICHIER [[109]], reflétant sa structure en listant les cartes, programmes, variables, etc. disponibles. Le squelette élimine le besoin de rechercher les composants mentionnés par leur nom. Au lieu de cela, si l'instanciation du squelette réussit, ils sont remplis dans la structure du squelette en tant que types libbpf valides (par exemple, struct bpf_mappointeur) et peuvent être passés aux API génériques libbpf
existantes.
En plus d'un accès simple et fiable aux cartes et aux programmes, le squelette fournit un stockage pour les liens BPF ( struct bpf_link ) pour chaque programme BPF au sein de l'objet BPF. Lorsque demandé, les programmes BPF pris en charge seront automatiquement attachés et les liens BPF résultants stockés pour une utilisation ultérieure par l'utilisateur dans des champs pré-alloués dans la structure du squelette. Pour les programmes BPF qui ne peuvent pas être automatiquement attachés par libbpf, l'utilisateur peut les attacher manuellement, mais stocker le lien BPF résultant dans le champ de lien par programme. Tous ces liens configurés seront automatiquement détruits lors de la destruction du squelette BPF. Cela élimine le besoin pour les utilisateurs de gérer les liens manuellement et de s'appuyer sur la prise en charge de libbpf pour détacher les programmes et libérer les ressources.

Une autre fonctionnalité fournie par le squelette BPF est une interface pour les variables globales de tous les types pris en charge : modifiables, en lecture seule, ainsi que externes. Cette interface permet de préconfigurer les valeurs initiales des variables avant que l'objet BPF ne soit chargé et vérifié par le noyau. Pour les variables non en lecture seule, la même interface peut être utilisée pour récupérer les valeurs des variables globales côté espace utilisateur, même si elles sont modifiées par le code BPF. Habituellement, un fichier BTF externe est construit à partir des données DWARF du noyau existantes à l'aide de pahole. Il contient tous les types utilisés par l'image de son noyau respectif et, en raison de cela, est

volumineux.

La fonctionnalité min_core_btfe construit des fichiers BTF plus petits, personnalisés pour un ou plusieurs objets eBPF, afin qu'ils puissent être distribués avec une application eBPF basée sur CO-RE, rendant l'application portable entre différentes versions du noyau.

bpftool gen help
Affiche un court message d'aide.

OPTIONS

-h, --help
Affiche un court message d'aide (similaire à bpftool help).
-V, --version
Affiche le numéro de version de bpftool (similaire à bpftool version), le numéro de la version de libbpf utilisée, et les fonctionnalités optionnelles qui ont été incluses lorsque bpftool a été compilé. Les fonctionnalités optionnelles incluent la liaison avec LLVM ou libbfd pour fournir le désassembleur pour les programmes JIT (bpftool prog dump jited) et l'utilisation des squelettes BPF (certaines fonctionnalités comme bpftool prog profile ou l'affichage des pids associés aux objets BPF peuvent en dépendre).
-j, --json
Génère une sortie JSON. Pour les commandes qui ne peuvent pas produire de JSON, cette option n'a aucun effet.
-p, --pretty
Génère une sortie JSON lisible par l'homme. Implique -j.
-d, --debug
Affiche tous les journaux disponibles, y compris les informations de niveau debug. Cela inclut les journaux de libbpf ainsi que ceux du vérificateur, lors de la tentative de chargement des programmes.
-L, --use-loader
Pour les squelettes, génère un squelette « léger » (également connu sous le nom de squelette « loader »). Un squelette léger contient un programme eBPF loader. Il n'utilise pas la majorité de l'infrastructure de libbpf et n'a pas besoin de libelf.

$ cat example1.bpf.c

#include <stdbool.h>
#include <linux/ptrace.h>
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
const volatile int param1 = 42;
bool global_flag = true;
struct { int x; } data = {};
SEC("raw_tp/sys_enter")
int handle_sys_enter(struct pt_regs *ctx)
{
      static long my_static_var;
      if (global_flag)
              my_static_var++;
      else
              data.x += param1;
      return 0;
}

$ cat example2.bpf.c

#include <linux/ptrace.h>
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
struct {
      __uint(type, BPF_MAP_TYPE_HASH);
      __uint(max_entries, 128);
      __type(key, int);
      __type(value, long);
} my_map SEC(".maps");
SEC("raw_tp/sys_exit")
int handle_sys_exit(struct pt_regs *ctx)
{
      int zero = 0;
      bpf_map_lookup_elem(&my_map, &zero);
      return 0;
}

$ cat example3.bpf.c

#include <linux/ptrace.h>
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
/* Ce fichier d'en-tête est fourni par le module bpf_testmod. */
#include "bpf_testmod.h"
int test_2_result = 0;
/* bpf_Testmod.ko appelle cette fonction, en passant un "4"
 * et testmod_map->data.
 */
SEC("struct_ops/test_2")
void BPF_PROG(test_2, int a, int b)
{
      test_2_result = a + b;
}
SEC(".struct_ops")
struct bpf_testmod_ops testmod_map = {
      .test_2 = (void *)test_2,
      .data = 0x1,
};

Ceci est un exemple d'application BPF avec trois programmes BPF et un mélange de maps BPF et de variables globales. Le code source est réparti sur trois fichiers de code source.

$ clang --target=bpf -g example1.bpf.c -o example1.bpf.o

$ clang --target=bpf -g example2.bpf.c -o example2.bpf.o

$ clang --target=bpf -g example3.bpf.c -o example3.bpf.o

$ bpftool gen object example.bpf.o example1.bpf.o example2.bpf.o example3.bpf.o

Cette série de commandes compile example1.bpf.c, example2.bpf.c et example3.bpf.c individuellement puis lie statiquement les fichiers objet respectifs dans le fichier objet ELF BPF final example.bpf.o.

$ cat example.c

#include "example.skel.h"
int main()
{
      struct example *skel;
      int err = 0;
      skel = example__open();
      if (!skel)
              goto cleanup;
      skel->rodata->param1 = 128;
      /* Change la valeur à travers le pointeur du type shadow */
      skel->struct_ops.testmod_map->data = 13;
      err = example__load(skel);
      if (err)
              goto cleanup;
      /* Le résultat de la fonction test_2() */
      printf("test_2_result: %d\n", skel->bss->test_2_result);
      err = example__attach(skel);
      if (err)
              goto cleanup;
      /* toutes les API de libbpf sont utilisables */
      printf("my_map name: %s\n", bpf_map__name(skel->maps.my_map));
      printf("sys_enter prog FD: %d\n",
             bpf_program__fd(skel->progs.handle_sys_enter));
      /* détache et réattache le programme sys_exit */
      bpf_link__destroy(skel->links.handle_sys_exit);
      skel->links.handle_sys_exit =
              bpf_program__attach(skel->progs.handle_sys_exit);
      printf("my_static_var: %ld\n",
             skel->bss->handle_sys_enter_my_static_var);
cleanup:
      example__destroy(skel);
      return err;
}
